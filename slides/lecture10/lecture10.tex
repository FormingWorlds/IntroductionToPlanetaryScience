% ── 3:2 resonance ─────────────────────────────────────────
\begin{frame}{The 3:2 Spin-Orbit Resonance}
  Mercury rotates exactly three times for every two orbital revolutions, locked into a stable 3:2 spin-orbit resonance.
  \vskip8pt
  \begin{itemize}
    \item Discovered by radar (Pettengill \& Dyce 1965); stabilised by high eccentricity ($e = 0.206$)
    \item Solar day lasts $176$ Earth days, exactly $2$ Mercury orbits
    \item Solar tidal torque on a permanent quadrupole moment locks the resonance
  \end{itemize}
\end{frame}

% ── Lobate scarps ─────────────────────────────────────────
\begin{frame}{Lobate Scarps: Mercury Has Shrunk}
  Global secular cooling caused Mercury to contract, forming compressional thrust faults across a single unbroken lithospheric lid.
  \vskip6pt
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig[0.62\textheight]{mercury_lobate_scarp}
    \source{NASA/JHUAPL (MESSENGER)}

    \column{\recaptextcol}
    \begin{itemize}
      \item Total radial shrinkage $\Delta R \approx 7$ km since late bombardment
      \item Area loss $\Delta A/A = 2\Delta R/R \approx 0.6\%$, taken up by thrust scarps
      \item Scarp relief reaches $3$ km with lengths exceeding $1000$ km
    \end{itemize}
    \vskip4pt
    \sourcenote{Byrne et al.\ (2014)}
  \end{columns}
\end{frame}

% ── Polar volatiles ───────────────────────────────────────

  \begin{columns}[c]
    \column{\recapfigcol}

    \source{Neumann et al.\ (2013)}

    \column{\recaptextcol}
    \begin{itemize}
      \item Permanently shadowed polar craters never see sunlight
      \item $T < 100$ K creates an efficient cold trap for water ice
      \item Topography, low insolation, and temperature maps all coincide
      \item Neutron spectrometry confirms $\mathrm{H_2O}$ ice; source: comet and asteroid impacts
    \end{itemize}
  \end{columns}
\end{frame}

% ── Surface chemistry ────────────────────────────────────
\begin{frame}{Mercury's Surface Chemistry}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{nittler2020_mercury_chemistry}
    \source{Nittler et al.\ (2020)}

    \column{\recaptextcol}
    Surprises from MESSENGER X-ray and gamma-ray spectrometers:
    \begin{itemize}
      \item High volatile abundances (S, K, Cl, Na) despite $700$ K daytime
      \item Low FeO ($< 2\%$): mantle is highly reducing
      \item Mg/Si ratio higher than Earth's
      \item Constrains formation under low oxygen fugacity
    \end{itemize}
  \end{columns}
\end{frame}

% ── Exosphere and magnetosphere ──────────────────────────
\begin{frame}{Exosphere and Magnetosphere}
  Mercury possesses a surface-bounded exosphere sustained by sputtering, photon desorption, and micrometeoroid impact vaporisation.
  \vskip8pt
  \begin{itemize}
    \item Surface pressure $\sim 10^{-12}$ bar; detected species include Na, K, Ca, Mg, He, H
    \item Sodium tail extends $\sim 10^6$ km anti-sunward via radiation pressure
    \item Miniature magnetosphere stands off the solar wind at subsolar distance $\sim 1.5\,R_{Hg}$
  \end{itemize}
\end{frame}

% ── Mercury interior layers ──────────────────────────────
\begin{frame}{A Giant Iron Core}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{margot2018_mercury_layers}
    \source{Margot et al.\ (2018)}

    \column{\recaptextcol}
    Margot et al.\ libration measurements pin down the layers:
    \begin{itemize}
      \item Crust: $\sim 35$ km
      \item Mantle (silicate): $\sim 400$ km
      \item Liquid outer core below $\sim 2020$ km radius, $\sim 83\%$ of the planetary radius
      \item Inner core (solid, possibly Fe-S)
      \item Core mass fraction $\sim 74\%$
    \end{itemize}
  \end{columns}
\end{frame}

% ── Margot libration ─────────────────────────────────────
\begin{frame}{How We Know: Forced Libration}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{margot2007_libration}
    \source{Margot et al.\ (2007), Fig.~3}

    \column{\recaptextcol}
    \begin{itemize}
      \item Radar speckle interferometry measures an 88-day forced libration amplitude of $35.8 \pm 2$ arcseconds
      \item Decoupling ratio $C_m/C \approx 0.5$: the silicate shell carries half the moment of inertia, so the mantle floats on a liquid core
      \item Only the solid silicate mantle librates, decoupled from the fluid core
    \end{itemize}
  \end{columns}
\end{frame}

% ── Why iron-rich ───────────────────────────────────────
\begin{frame}{Why is Mercury so Iron-Rich?}
  Mercury's anomalous core fraction likely reflects giant-impact mantle stripping rather than high-temperature solar evaporation.
  \vskip6pt
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{terrestrial_interiors_comparison}
    \source{NASA Solar System Exploration}

    \column{\recaptextcol}
    \begin{itemize}
      \item \textbf{Giant impact hypothesis} (leading): hit-and-run collisions strip the silicate mantle
      \item \textbf{Thermal vaporization}: disfavoured by MESSENGER retention of volatiles (Na, S, K)
      \item \textbf{Selective condensation}: Fe-Ni metal condenses before silicates in the hot inner nebula (no longer favoured)
    \end{itemize}
  \end{columns}
\end{frame}

% ── Franco mercury impact outcomes ──────────────────────
\begin{frame}{Impact-Stripping Outcomes}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{franco2022_mercury_outcomes}
    \source{Franco et al.\ (2022), Fig.~A1}

    \column{\recaptextcol}
    \begin{itemize}
      \item N-body terrestrial-formation simulations: remnant mass vs.\ semimajor axis
      \item Six panels: surface-density slope $x = 0.5$--$5.5$; triangles mark real planets
      \item Mercury-mass and iron-rich remnant at $0.39$ AU occurs in $\ll 1\%$ of trials
      \item Single-impact origin is dynamically rare
    \end{itemize}
  \end{columns}
\end{frame}

% ── Wicht offset dipole ─────────────────────────────────
\begin{frame}{A Weak, Offset Dynamo}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{wicht_offset_dipole}
    \source{Wicht \& Heyner (2017)}

    \column{\recaptextcol}
    \begin{itemize}
      \item Mercury's dipole is offset $\sim 0.2\, R_{Hg}$ northward
      \item Surface field strength: $\sim 200$--$400$ nT (Earth: $\sim 50,000$ nT)
      \item Dynamo generated by convection in a thin outer-core shell
      \item Stably stratified liquid layer suppresses high-degree field harmonics
    \end{itemize}
  \end{columns}
\end{frame}

\section{Blackboard Derivation: Jeans Escape Flux}
% ════════════════════════════════════════════════════════════

% ── Setup ──────────────────────────────────────────────────
\begin{frame}{Setup: The Exobase}
  \textbf{Goal:} derive the rate at which thermal energy alone allows molecules to escape from a planetary atmosphere; apply it to Mars.
  \vskip6pt
  \textbf{The exobase:} Altitude where mean free path equals atmospheric scale height; molecules move ballistically above it.
  \vskip6pt
  Most-probable speed: $v_{\mathrm{th}} = \sqrt{2 \kB T / m}$
  \vskip4pt
  Escape velocity: $v_{\mathrm{esc}} = \sqrt{2 G M / r_{\mathrm{exo}}}$
\end{frame}

\begin{frame}{The Escape Parameter}
  \keyresult{$$\lambda \equiv \left(\frac{v_{\mathrm{esc}}}{v_{\mathrm{th}}}\right)^{\!2} = \frac{G\,M\,m}{\kB\,T\,r_{\mathrm{exo}}}$$}
  \vskip6pt
  \begin{itemize}
    \item $\lambda \gg 1$: molecules gravitationally bound; escape rare
    \item $\lambda \lesssim 1$: thermal energy comparable to escape energy; rapid loss
    \item Depends strongly on molecular mass $m$; independent of magnetic field, ionisation and solar wind
  \end{itemize}
\end{frame}

% ── Jeans flux derivation ────────────────────────────────
\begin{frame}{Result: The Jeans Formula}
  Integrating upward velocity over the upper hemisphere with cutoff $v \geq v_{\mathrm{esc}}$:
  $$\Phi_J = n\,\langle v\rangle\,\frac{1}{4}\,(1+\lambda)\,e^{-\lambda}$$
  \vskip4pt
  where $\langle v\rangle = \sqrt{8 \kB T / (\pi m)}$ is the mean thermal speed.
  \vskip6pt
  Substituting and simplifying:
  \keyresult{$$\Phi_J \;=\; n\,\sqrt{\frac{\kB T}{2\pi m}}\,(1+\lambda)\,e^{-\lambda}$$}
  \vskip4pt
  Jeans (1925). Limits: $\lambda \to 0$ gives the effusion flux $n\sqrt{\kB T/(2\pi m)}$; for $\lambda \gg 1$ the flux falls as $\lambda e^{-\lambda}$, so $\lambda$ from $5$ to $10$ cuts it by $\sim 80$.
\end{frame}

  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.72\paperheight, keepaspectratio]{maxwell_boltzmann_jeans}}\\
  {\footnotesize Maxwell-Boltzmann speed distribution for atomic hydrogen at Earth (1000 K) and Mars (270 K) exobases: only molecules in the high-speed tail above escape velocity contribute to Jeans escape. Credit: Course-original figure.}
\end{frame}

% ── Mars application ─────────────────────────────────────
\begin{frame}{Apply to Mars}
  $M = 6.4\times 10^{23}$ kg, $r_{\mathrm{exo}} \approx 3.6\times 10^6$ m, $T_{\mathrm{exo}} \approx 270$ K.
  \vskip2pt
  Escape velocity at the exobase: $v_{\mathrm{esc}} \approx 4.9\,\mathrm{km/s}$
  \vskip4pt
  \centering
  \small
  \begin{tabular}{l c c}
    \toprule
    \textbf{Species} & $\lambda$ & $e^{-\lambda}$ \\
    \midrule
    H & $5.3$ & $5\times 10^{-3}$ \\
    $\mathrm{H_2}$ & $10.6$ & $2$--$3\times 10^{-5}$ \\
    He & $21.2$ & $\sim 10^{-9}$ \\
    O & $84.8$ & $10^{-37}$ \\
    $\mathrm{CO_2}$ & $230$ & $10^{-100}$ \\
    \bottomrule
  \end{tabular}
  \vskip4pt
  \keyresult{Mars cannot lose $\mathrm{CO_2}$ thermally. Only H and $\mathrm{H_2}$ escape via Jeans; heavier species need non-thermal channels.}
\end{frame}

\section{Comparative Payoff: Mercury vs Mars}
% ════════════════════════════════════════════════════════════

% ── BepiColombo hero ────────────────────────────────────────
\begin{frame}
  \centering

  {\footnotesize Mercury's magnetosphere, the smallest in the solar system, with its subsolar magnetopause at $\sim 1.5\,R_{Hg}$. BepiColombo (ESA/JAXA) enters Mercury orbit in late 2026 with two orbiters to map this offset field, the core and the exosphere. Credit: Wicht \& Heyner (2017).}

% ── Summary ────────────────────────────────────────────────
\begin{frame}{Size and Distance Set the Trajectory}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig[0.55\textheight]{magnetosphere_anatomy_esa}
    \source{ESA}

    \column{\recaptextcol}
    \footnotesize
    \begin{itemize}
      \item \textbf{Small bodies cool fast}: surface-area-to-volume ratio $\propto 1/R$; Mars cooled faster than Earth, so its dynamo shut off early
      \item \textbf{Mercury keeps a weak dynamo}: light elements (S, Si) keep a thin outer-core shell liquid; thermal stratification gives the weak, offset field
      \item \textbf{Lesson}: dynamo persistence needs (a) core size, (b) a slow-cooling mantle, (c) chemical buoyancy from inner-core growth; Mars failed (b), Mercury keeps (a), Earth has all three
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Summary}
  \begin{enumerate}
    \item \textbf{Resonance and core}: 3:2 spin-orbit resonance; $\sim 74\%$ iron core from mantle stripping
    \item \textbf{Contraction}: Secular cooling shrank Mercury by $\Delta R \approx 7$ km forming lobate scarps
    \item \textbf{Mars interior}: a liquid core, $\sim 1650$--$1675$ km below a molten silicate layer (2023 reanalyses)
    \item \textbf{Jeans escape}: Flux $\Phi_J \propto (1+\lambda)e^{-\lambda}$ allows H loss but retains $\mathrm{CO_2}$
    \item \textbf{Non-thermal loss}: sputtering and ion pickup remove the heavy species
    \item \textbf{Dynamo death}: between $\sim 4.1$ and $\sim 3.7$ Ga
    \item \textbf{Opposite limits}: Size and distance govern core cooling, dynamos, and volatile retention
  \end{enumerate}

\end{frame}
