% ── 3:2 resonance ─────────────────────────────────────────
\begin{frame}{The 3:2 Spin-Orbit Resonance}
  Mercury rotates exactly $3$ times for every $2$ orbits.
  \vskip6pt
  \begin{itemize}
    \item Discovered by radar, Pettengill \& Dyce (1965)
    \item Solar tidal torque on a permanent quadrupole moment (a bulge along the long axis) locks Mercury into the resonance
    \item Stable because of high orbital eccentricity ($e = 0.206$)
    \item Solar day on Mercury: $176$ Earth days = $2$ orbits
    \item Hot pole and cold pole alternate with orbital phase
  \end{itemize}
  \vskip8pt
  \keyresult{Mercury is the only known planet in a non-1:1 spin-orbit resonance. The capture is dynamically robust because of eccentricity-driven tidal evolution.}
\end{frame}

% ── Reading the surface map ──────────────────────────────
\begin{frame}{Reading Mercury's Topography}
  MESSENGER's Mercury Laser Altimeter mapped the planet at $\sim 250$ m vertical resolution.
  \vskip6pt
  \begin{itemize}
    \item Total topographic range: $\sim 10$ km (less than Mars or the Moon)
    \item Highlands: ancient cratered terrain, $> 4$ Gyr old
    \item Smooth plains: volcanic infill, $\sim 3.7$--$3.9$ Ga
    \item \textbf{Caloris Basin} ($\sim 1550$ km diameter): one of the largest impacts in the inner solar system
    \item \textbf{Lobate scarps}: thrust faults (crust overlapping under compression) from global contraction
  \end{itemize}
  \vskip6pt
  \textbf{Two geological epochs:} pre-tectonic + heavy bombardment, then a long stagnant cooling.
\end{frame}

% ── Exosphere ────────────────────────────────────────────
\begin{frame}{Exosphere and Magnetosphere}
  Mercury has no real atmosphere; instead, a tenuous \textbf{exosphere} of atoms knocked off the surface.
  \vskip6pt
  \begin{itemize}
    \item Surface pressure $\sim 10^{-12}$ bar (effectively vacuum)
    \item Species detected: Na, K, Ca, Mg, He, H
    \item Sources: micrometeorite impact, photon-stimulated desorption, sputtering, thermal release
    \item Sodium tail extends $\sim 10^6$ km anti-sunward
  \end{itemize}
  \vskip4pt
  Magnetosphere: small, asymmetric, driven by the planet's weak global dipole + solar-wind interaction.
\end{frame}

% ── Mercury mission history ──────────────────────────────
\begin{frame}{Missions to Mercury}
  \begin{itemize}
    \item \textbf{Mariner 10} (1974--1975): three flybys, $\sim 45\%$ of surface imaged
    \item \textbf{MESSENGER} (2011--2015): orbiter, full surface mapping, MLA, MAG, X/GRS
    \item \textbf{BepiColombo} (ESA/JAXA, launched 2018): two orbiters, Mercury orbit insertion late 2026
    \begin{itemize}
      \item MPO: surface, interior, exosphere
      \item Mio (MMO): magnetosphere, plasma environment
    \end{itemize}
  \end{itemize}
  \vskip6pt
  \keyresult{In a year and a half, BepiColombo's two orbiters will revolutionise our view of Mercury's interior, magnetosphere, and surface composition.}
\end{frame}

\section{Mercury Interior and Dynamo}
% ════════════════════════════════════════════════════════════

% ── Mercury interior layers ──────────────────────────────
\begin{frame}{A Giant Iron Core}
  \begin{columns}[c]
    \column{0.42\textwidth}
    \centering
    \includegraphics[width=\linewidth, height=0.65\textheight, keepaspectratio]{margot2018_mercury_layers}
    \source{Margot et al.\ (2018)}

    \column{0.55\textwidth}
    Margot et al.\ libration measurements pin down the layers:
    \begin{itemize}
      \item Crust: $\sim 35$ km
      \item Mantle (silicate): $\sim 400$ km
      \item Outer core (liquid): below $\sim 2440 - 400 - 35 = 2005$ km radius
      \item Inner core (solid, possibly Fe-S)
      \item Core radius $\sim 2020$ km, $\sim 83\%$ of planetary radius
      \item Core mass fraction $\sim 65$--$70\%$
    \end{itemize}
  \end{columns}
\end{frame}

% ── Margot libration ─────────────────────────────────────
\begin{frame}{How We Know: Forced Libration}
  \begin{columns}[c]
    \column{0.50\textwidth}
    \centering
    \includegraphics[width=\linewidth, height=0.62\textheight, keepaspectratio]{margot2007_libration}
    \source{Margot et al.\ (2007)}

    \column{0.47\textwidth}
    \begin{itemize}
      \item Mercury rocks slightly back and forth as it orbits
      \item Libration amplitude depends on whether the mantle is decoupled from the core
      \item Earth-based radar speckle interferometry: amplitude $35.8 \pm 2$ arcseconds
      \item Implies a liquid outer core: only the silicate mantle librates freely
    \end{itemize}
  \end{columns}
\end{frame}

% ── Margot data plot ─────────────────────────────────────
\begin{frame}{The Libration Measurement}
  \begin{columns}[c]
    \column{0.55\textwidth}
    \centering
    \includegraphics[width=\linewidth, height=0.65\textheight, keepaspectratio]{margot2007_libdata}
    \source{Margot et al.\ (2007)}

    \column{0.42\textwidth}
    \begin{itemize}
      \item Posterior on $C_m/C$ (mantle moment / total moment) from Monte Carlo realisations
      \item Peaks at $\sim 0.5$, inconsistent with a fully solid Mercury (which would peak near $1$)
      \item Direct evidence the silicate mantle floats on a fluid core
      \item Combined with $C_{20}$ and $C_{22}$ from gravity, fully constrains $C/MR^2$ and core radius
    \end{itemize}
  \end{columns}
\end{frame}

% ── Why iron-rich ───────────────────────────────────────
\begin{frame}{Why is Mercury so Iron-Rich?}
  Three competing hypotheses for Mercury's anomalous core fraction:
  \vskip6pt
  \begin{enumerate}
    \item \textbf{One or more giant impacts} (leading hypothesis; Benz, Slattery \& Cameron 1988; Chau et al.\ 2018): late hit-and-run collisions strip a silicate mantle from a proto-Mercury
    \item \textbf{Vapour and aerosol stripping by an early luminous Sun}: thermal evaporation / wind-driven loss of part of an originally silicate-rich proto-Mercury
    \item \textbf{Selective condensation in the inner nebula}: hot inner-disk midplane condenses Fe-Ni metal preferentially over silicates (no longer favoured)
  \end{enumerate}
  \vskip6pt
  \textit{MESSENGER} surface volatiles (Na, S, K) disfavour thermal stripping; the giant-impact family is the modern leading candidate.
\end{frame}

\begin{frame}{Impact-Stripping Outcomes}
  \begin{columns}[c]

% ── Wicht offset dipole ─────────────────────────────────
\begin{frame}{A Weak, Offset Dynamo}
  \begin{columns}[c]
    \column{0.50\textwidth}
    \centering
    \includegraphics[width=\linewidth, height=0.62\textheight, keepaspectratio]{wicht_offset_dipole}
    \source{Wicht \& Heyner (2017)}

    \column{0.47\textwidth}
    \begin{itemize}
      \item Mercury's dipole offset $\sim 0.2\, R_{Hg}$ northward
      \item Surface field strength: $\sim 200$--$400$ nT (Earth's: $\sim 50,000$ nT)
      \item Dynamo (self-sustained field from convection) in a thin outer-core shell
      \item Stably stratified upper liquid-core layer suppresses high-degree harmonics; inner-core freezing asymmetry produces the offset
    \end{itemize}
  \end{columns}
\end{frame}

% ── Phobos and Deimos ─────────────────────────────────────
\begin{frame}{Phobos and Deimos}
  Mars' two tiny moons:
  \vskip6pt
  \begin{itemize}
    \item \textbf{Phobos}: $22$ km mean diameter, orbit at $9376$ km, period $7.7$ h (faster than Mars rotation)
    \item \textbf{Deimos}: $12$ km, orbit at $23463$ km, period $30.3$ h
    \item Both have very low density ($\sim 1900$ kg\,m$^{-3}$), consistent with rubble-pile internal structure
    \item Two competing origin scenarios:
    \begin{itemize}
      \item Captured asteroids
      \item In situ formation from a giant-impact debris disk
    \end{itemize}
  \end{itemize}
  \vskip4pt
  Phobos is below synchronous orbit and will tidally decay onto Mars in $\sim 30$--$50$ Myr.
\end{frame}

\section{Blackboard Derivation: Jeans Escape Flux}
% ════════════════════════════════════════════════════════════

% ── Setup ──────────────────────────────────────────────────
\begin{frame}{Setup: The Exobase}
  \textbf{Goal:} derive the rate at which thermal energy alone allows molecules to escape from a planetary atmosphere; apply it to Mars.
  \vskip6pt
  \textbf{The exobase.} Altitude where the mean free path equals the atmospheric scale height. Above it, molecules move ballistically.
  \vskip6pt
  Most-probable speed in Maxwell-Boltzmann distribution:
  $$v_{\mathrm{th}} = \sqrt{2 \kB T / m}$$
  \vskip2pt
  Escape velocity at the exobase:
  $$v_{\mathrm{esc}} = \sqrt{2 G M / r_{\mathrm{exo}}}$$
\end{frame}

% ── Escape parameter ──────────────────────────────────────
\begin{frame}{Step 1: The Escape Parameter}
  Define dimensionless ratio:
  \keyresult{$$\lambda \equiv \left(\frac{v_{\mathrm{esc}}}{v_{\mathrm{th}}}\right)^{\!2} = \frac{G\,M\,m}{\kB\,T\,r_{\mathrm{exo}}}$$}
  \vskip6pt
  \begin{itemize}
    \item $\lambda \gg 1$: molecules gravitationally bound; escape rare
    \item $\lambda \lesssim 1$: thermal energy comparable to escape energy; rapid loss
    \item Depends strongly on molecular mass $m$
  \end{itemize}
\end{frame}

% ── Maxwell distribution ─────────────────────────────────
\begin{frame}{Step 2: Maxwell-Boltzmann Distribution}
  At the exobase, gas in local thermal equilibrium:
  $$f(v)\,\dd v = n\,\left(\frac{m}{2\pi \kB T}\right)^{3/2}\,4\pi v^2 \exp\!\left(-\frac{m v^2}{2 \kB T}\right) \dd v$$
  \vskip6pt
  Number flux of upward-moving molecules with $v_z > 0$:
  $$\Phi = \int_{\mathrm{up}} v_z\, f(\mathbf{v})\,\dd^3 v$$
  \vskip4pt
  We need $v_z$ such that the speed exceeds $v_{\mathrm{esc}}$.
\end{frame}

% ── Jeans flux derivation ────────────────────────────────
\begin{frame}{Step 3: The Jeans Formula}
  Integrating over the upper hemisphere of velocity space, with the speed cutoff $v \geq v_{\mathrm{esc}}$:
  $$\Phi_J = n\,\langle v\rangle\,\frac{1}{4}\,(1+\lambda)\,e^{-\lambda}$$
  \vskip4pt
  where $\langle v\rangle = \sqrt{8 \kB T / (\pi m)}$ is the mean speed.
  \vskip4pt
  Substituting and simplifying:
  \keyresult{$$\Phi_J \;=\; n\,\sqrt{\frac{\kB T}{2\pi m}}\,(1+\lambda)\,e^{-\lambda}$$}
  \vskip4pt
  This is the Jeans escape flux (Jeans 1925).
\end{frame}

% ── Limits ────────────────────────────────────────────────
\begin{frame}{Step 4: Checking the Limits}
  \begin{itemize}
    \item $\lambda \to 0$: $(1+\lambda)e^{-\lambda} \to 1$, formula reduces to standard kinetic-theory effusion flux $n \sqrt{\kB T / (2 \pi m)}$
    \item $\lambda \gg 1$: exponential decay dominates; flux $\propto \lambda\, e^{-\lambda}$
    \item Steepness: a change in $\lambda$ from $5$ to $10$ reduces escape by a factor of $\sim 80$
  \end{itemize}
  \vskip8pt
  \keyresult{Jeans escape is exponentially sensitive to $\lambda$: small changes in temperature or molecular mass change the loss rate by orders of magnitude.}
\end{frame}

% ── Mars application ─────────────────────────────────────
\begin{frame}{Step 5: Apply to Mars}
  $M = 6.4\times 10^{23}$ kg, $r_{\mathrm{exo}} \approx 3.6\times 10^6$ m, $T_{\mathrm{exo}} \approx 270$ K.
  \vskip4pt
  Escape velocity at the exobase:
  $$v_{\mathrm{esc}} \approx 4.9\,\mathrm{km/s}$$
  \vskip4pt
  \begin{tabular}{l c c}
    \toprule
    \textbf{Species} & $\lambda$ & $e^{-\lambda}$ \\
    \midrule
    H & $5.3$ & $5\times 10^{-3}$ \\
    $\mathrm{H_2}$ & $10.6$ & $2$--$3\times 10^{-5}$ \\
    He & $21.2$ & $\sim 10^{-9}$ \\
    O & $84.8$ & $10^{-37}$ \\
    $\mathrm{CO_2}$ & $230$ & $10^{-100}$ \\
    \bottomrule
  \end{tabular}
  \vskip2pt
  \keyresult{Mars cannot lose $\mathrm{CO_2}$ thermally. Only H and $\mathrm{H_2}$ escape via Jeans; heavier species need non-thermal channels.}
\end{frame}

% ── Why robust ───────────────────────────────────────────
\begin{frame}{Why Jeans Escape is the Foundation}
  \begin{itemize}
    \item Underpins atmospheric escape theory in general
    \item Always present at any temperature above absolute zero
    \item Sets the lower bound on escape rate
    \item Independent of magnetic field, ionisation state, or solar wind
  \end{itemize}
  \vskip8pt
  Other mechanisms (photochemical, sputtering, ion pickup, hydrodynamic) operate \textbf{in addition} to Jeans, with different mass selectivities and intensities.
\end{frame}

\section{Comparative Payoff: Mercury vs Mars}
% ════════════════════════════════════════════════════════════

% ── Opposite limits ─────────────────────────────────────
\begin{frame}{Mercury and Mars as Opposite Limits}
  \centering
  \small
  \begin{tabular}{l c c}
    \toprule
    \textbf{Property} & \textbf{Mercury} & \textbf{Mars} \\
    \midrule
    Distance from Sun & $0.39$ AU & $1.52$ AU \\
    Mass ($\Mearth$) & $0.055$ & $0.107$ \\
    Core fraction & $\sim 65$--$70\%$ & $\sim 25\%$ \\
    Mantle thickness & $\sim 400$ km & $\sim 1560$ km \\
    Surface volatiles & cold trap ice only & once-flowing water \\
    Magnetic field today & weak global & none global \\
    Atmosphere & exosphere & $6$ mbar $\mathrm{CO_2}$ \\
    Tectonics & shrinking, lobate scarps & stagnant lid \\
    \bottomrule
  \end{tabular}
  \vskip8pt
  \keyresult{Mercury sits at the maximum-iron, no-volatile end. Mars sits at the maximum-volatile, low-iron end. Earth and Venus fall in between.}
\end{frame}

% ── Size and distance set the trajectory ────────────────
\begin{frame}{Size and Distance Set the Trajectory}
  \begin{itemize}
    \item \textbf{Small bodies cool fast}: surface-area-to-volume ratio $\propto 1/R$
    \item Mars cools faster than Earth $\Rightarrow$ early dynamo shutoff
    \item Mercury sustains a weak dynamo because light elements (S, Si) keep the outer core liquid; a thin convecting shell + thermal stratification produce the offset, weak field
    \item \textbf{Distance from the Sun} sets atmospheric escape and irradiation history
  \end{itemize}
  \vskip6pt
  Combined: small + warm $\Rightarrow$ Mercury (volatile-poor, iron-dominant). Small + cold $\Rightarrow$ Mars (water-rich early, escapes to space later).
\end{frame}

% ── Dynamo lifetimes ───────────────────────────────────
\begin{frame}{The Timing Problem: Dynamo Lifetimes}
  \begin{itemize}
    \item Earth: dynamo active for $\geq 3.5$ Gyr, still going
    \item Mars: dynamo active in the first $\sim 500$--$800$ Myr; cessation $\sim 4.1$ Ga (Acu\~na 1999), possibly as late as $\sim 3.7$ Ga (Mittelholz 2020, Steele 2024)
    \item Mercury: dynamo weak but still active
    \item Venus: dynamo unclear, possibly dead
  \end{itemize}
  \vskip6pt
  \textbf{Lesson:} dynamo persistence depends on (a) core size, (b) cooling rate of the mantle, (c) chemical buoyancy from inner-core nucleation. Mars failed (b); Mercury preserves (a); Earth has all three.
\end{frame}

% ── What makes a planet habitable ────────────────────────
\begin{frame}{Comparative View: What Makes a Rocky Planet Habitable?}
  \begin{itemize}
    \item Stellar flux inside the runaway-greenhouse limit (rules out Venus)
    \item Long-lasting atmosphere (rules out Mercury, late Mars)
    \item Liquid water reservoir at the surface (rules out Mercury entirely)
    \item Internal heat sufficient for tectonic recycling (rules out Mars after $\sim 1$ Ga)
    \item Magnetic shield against solar-wind erosion (helps but not strictly required)
  \end{itemize}
  \vskip6pt
  \keyresult{Habitability is a multi-dimensional constraint. Each of Mercury, Venus, Mars fails on different axes; Earth happens to satisfy all of them simultaneously.}
\end{frame}


% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}
    \item Mercury: 3:2 spin-orbit resonance, $65$--$70\%$ iron core, weak offset dynamo, polar ice in cold traps
    \item Mars: half Earth's radius, one tenth its mass, lost its dynamo at $\sim 4.1$ Ga and most of its atmosphere thereafter
    \item Jeans escape: $\Phi_J \propto (1+\lambda)e^{-\lambda}$, mass-selective; H and $\mathrm{H_2}$ escape Mars, $\mathrm{CO_2}$ does not
    \item Non-thermal escape (photochemical, ion pickup, sputtering) drives the heavy-species loss
    \item InSight delivered the first marsquake catalogue and confirmed a liquid Mars core
    \item Mercury and Mars sit at opposite limits of terrestrial-planet evolution: maximum core fraction vs.\ maximum atmospheric volatile loss
    \item BepiColombo (2026) and MMX (2026) will revolutionise our view of both planets in the next few years
  \end{enumerate}
\end{frame}
