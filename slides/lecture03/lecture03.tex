\section{Energy sources}
% ════════════════════════════════════════════════════════════

% ── Four energy sources ────────────────────────────────────
\begin{frame}{What heats a planet?}
  \begin{enumerate}
    \item \textbf{Accretional heating:} gravitational energy of assembly
    \item \textbf{Gravitational differentiation:} core formation, about 10\% more
    \item \textbf{Radioactive decay:} sustained for billions of years
    \item \textbf{Tidal heating:} orbital energy dissipated as friction
  \end{enumerate}
  \vskip10pt
  \keyresult{No internal heat means no volcanism, no magnetic field, no plate tectonics, no outgassing.}
\end{frame}

% ── Accretional heating ───────────────────────────────────
\begin{frame}{Accretional heating}
  Gravitational energy of assembling a uniform sphere:
  $$E_{\mathrm{acc}} \sim \frac{3}{5}\frac{GM^2}{R} \approx 2.2 \times 10^{32}~\text{J for Earth}$$
  \vskip6pt
  If all of it is retained as heat:
  $$\Delta T = \frac{E_{\mathrm{acc}}}{M c_p} \approx \frac{2.2 \times 10^{32}}{5.97 \times 10^{24} \times 1200} \approx 30{,}000\;\text{K}$$
  \vskip6pt
  \keyresult{Enough to melt Earth many times over: accretion produces a \textbf{magma ocean}.}
\end{frame}

\begin{frame}
  \ipsherokey[0.65]{accretional_heating}{Accretional energy and retained temperature rise against body size for a uniform sphere at Earth's density. Course figure.}{Only 10 to 50\% of $E_{\mathrm{acc}}$ is retained (radiation to space, thermal blanketing, depth of deposition), still enough to melt a Mars-sized body.}
\end{frame}

% ── Lithophile vs siderophile preview ───────────────────
\begin{frame}{Where do heat-producing elements live?}
  \begin{itemize}
    \item \textbf{Lithophile} (``rock-loving''): U, Th, K stay in the silicate mantle and crust
    \item \textbf{Siderophile} (``iron-loving''): Fe, Ni, Co, Pt, Ir, Au go to the core
  \end{itemize}
  \vskip8pt
  \keyresult{Radiogenic heat is produced in the \textbf{mantle}, not the core, and that sustains mantle convection.}
  \vskip4pt
  \sourcenote{Goldschmidt (1937); McDonough (2003)}
\end{frame}

% ── Gravitational differentiation ─────────────────────────
\begin{frame}
  \ipsherokey[0.66]{differentiation_energy}{Metal droplets dispersed after accretion (a) sink to form a core (b). Course schematic.}{$E_{\mathrm{diff}} \sim \frac{3}{10}\,\frac{GM_{\mathrm{core}}^2}{R_{\mathrm{core}}}\left(\frac{R}{R_{\mathrm{core}}} - 1\right) \approx 2 \times 10^{31}$~J for Earth, about 10\% of accretion, released \textbf{once} during core formation.}

\section{Heat transport mechanisms}
% ════════════════════════════════════════════════════════════

\begin{frame}
  \ipsherokey{heat_transport_modes}{Three ways to move heat: conduction down a temperature gradient, convection by bulk flow, radiation by photons. \begin{frame}{Conduction: Fourier's law and the diffusion timescale}
  \begin{columns}[c]
    \column{0.5\textwidth}
    \begin{itemize}\setlength\itemsep{2pt}
      \item Heat flux $\vec{q} = -k\nabla T$
      \item Thermal diffusivity $\kappa = k/(\rho c_p) \approx 10^{-6}$ m$^2$\,s$^{-1}$ for rock
      \item Metals: high $k$ and high $\rho c_p$; rocks: low and low; $\kappa$ similar
    \end{itemize}
    \keyresult{$$\tau_{\mathrm{cond}} \sim \frac{L^2}{\kappa}$$}
    \column{0.5\textwidth}
    \small
    \begin{tabular}{l c c}
      \toprule
      \textbf{Material} & $\boldsymbol{k}$ (W\,m$^{-1}$\,K$^{-1}$) & $\boldsymbol{\kappa}$ (m$^2$\,s$^{-1}$) \\
      \midrule
      Silicate rock & 3--5 & $\sim$10$^{-6}$ \\
      Bridgmanite & 5--10 & $\sim$2$\times 10^{-6}$ \\
      Water ice & 2.2 & 1.1$\times 10^{-6}$ \\
      Iron (solid) & 80 & $\sim$2$\times 10^{-5}$ \\
      Air & 0.025 & $\sim$2$\times 10^{-5}$ \\
      \bottomrule
    \end{tabular}
  \end{columns}
\end{frame}

\section{Blackboard derivation: cooling timescale}
% ════════════════════════════════════════════════════════════

% ── Setup: goal and strategy ──────────────────────────────
\begin{frame}{Goal and strategy}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig[0.62\textheight]{board_sketch_l03}
    \column{\recaptextcolwide}
    \textbf{Goal:} how long does heat take to conduct out of a body of size $L$?
    \vskip8pt
    \begin{itemize}\setlength\itemsep{2pt}
      \item Write the heat equation
      \item Replace derivatives by scales: $\tau_{\mathrm{cond}}(L, \kappa)$
      \item Put in rock values and compare with 4.6~Gyr
    \end{itemize}
    \vskip8pt
    \keyresult{Bigger bodies have a longer path for heat: we derive \textit{how much} longer.}
  \end{columns}
\end{frame}

% ── Step 1: heat equation ─────────────────────────────────
\begin{frame}{Step 1: the heat diffusion equation}
  Fourier's law $\vec{q} = -k\nabla T$ and energy conservation give:
  $$\rho c_p\,\pdv{T}{t} = \nabla\!\cdot\!(k\nabla T)$$
  \vskip6pt
  For constant $k$, with $\kappa = k/(\rho c_p)$, in one dimension:
  $$\pdv{T}{t} = \kappa \nabla^2 T = \kappa \pdv{^2 T}{x^2}$$
  \vskip6pt
  \keyresult{The temperature changes at a rate set by the \textbf{curvature} of the profile: flat profiles do not evolve.}
\end{frame}

% ── Step 2: dimensional analysis ──────────────────────────
\begin{frame}{Step 2: dimensional analysis}
  Replace derivatives with characteristic scales:
  $$\pdv{T}{t} \sim \frac{T}{\tau}, \qquad \pdv{^2 T}{x^2} \sim \frac{T}{L^2}
  \quad\Rightarrow\quad \frac{T}{\tau} \sim \kappa\,\frac{T}{L^2}$$
  \vskip4pt
  Temperature cancels: the timescale is intrinsic to the body.
  \vskip4pt
  \keyresult{$$\boxed{\tau_{\mathrm{cond}} \sim \frac{L^2}{\kappa}}$$}
  \vskip4pt
  \textbf{Key feature:} the $L^2$ scaling. Doubling the size gives 4$\times$ the cooling time.
\end{frame}

% ── Application table ──────────────────────────────────────
\begin{frame}{Cooling timescales across the solar system}
  Using $\kappa \approx 10^{-6}$~m$^2$\,s$^{-1}$ (silicate rock):
  \vskip8pt
  \centering
  \begin{tabular}{l r r r}
    \toprule
    \textbf{Body} & \textbf{Size $L$} & $\boldsymbol{\tau = L^2/\kappa}$ & \textbf{Familiar units} \\
    \midrule
    1~m boulder  & 1~m               & $10^6$~s              & $\sim$12 days \\
    100~km asteroid & $10^5$~m        & $10^{16}$~s           & $\sim$300~Myr \\
    Moon         & $1.7 \times 10^6$~m & $3 \times 10^{18}$~s & $\sim$100~Gyr \\
    Earth        & $6.4 \times 10^6$~m & $4 \times 10^{19}$~s & $\sim$1300~Gyr \\
    \bottomrule
  \end{tabular}
  \vskip10pt
  \begin{alertblock}{}
    Earth's conductive cooling time $\gg$ age of the universe.\\
    Yet Earth loses 47~TW $\Rightarrow$ there \textbf{must} be convection.
  \end{alertblock}
\end{frame}

  \ipsherokey{tau_cond_vs_size}{Conductive cooling time $\tau = L^2/\kappa$ against body size, slope 2 on log axes. Course figure.}{A chondrule cools in under a second, a 1~km asteroid in 30\,000~yr; any body larger than a few hundred km must convect, or stay hot.}
\end{frame}

\section{Rayleigh number \& convective vigour}
% ════════════════════════════════════════════════════════════

% ── Rayleigh number ───────────────────────────────────────
\begin{frame}{The Rayleigh number}
  Does a fluid layer convect or merely conduct?
  \vskip8pt
  $$\mathrm{Ra} = \frac{\alpha \rho g \Delta T\, d^3}{\kappa \eta}
  \qquad \frac{\text{buoyancy}}{\text{dissipation}}$$
  \vskip4pt
  $\alpha$ thermal expansion, $\rho$ density, $g$ gravity, $\Delta T$ temperature contrast, $d$ layer depth; $\kappa$ thermal diffusivity, $\eta$ dynamic viscosity.
  \vskip10pt
  \keyresult{Convection starts above $\mathrm{Ra}_c \approx 1708$ (rigid boundaries; 658 for free-slip).}
\end{frame}

\begin{frame}
  \ipshero{marginal_stability}{Marginal stability: conduction below the curve, convection above; the minimum $\mathrm{Ra}_c \approx 1708$ sits at $k_c \approx 3.12$. Course figure, after Turcotte \& Schubert (2002).}
\end{frame}

% ── Earth's mantle Ra ─────────────────────────────────────
\begin{frame}{Earth's mantle: vigorous convection, and how efficient}
  $$\mathrm{Ra}_\oplus \sim \frac{2 \times 10^{-5} \times 4000 \times 10 \times 2500 \times (3 \times 10^6)^3}{10^{-6} \times 10^{21}} \sim 10^7\text{--}10^8 \gg \mathrm{Ra}_c$$
  \vskip4pt
  \begin{itemize}\setlength\itemsep{2pt}
    \item Nusselt number $\mathrm{Nu} = q_{\mathrm{total}}/q_{\mathrm{cond}}$: efficiency, $\mathrm{Nu} = 1$ is pure conduction
    \item Boundary-layer theory: $\mathrm{Nu} \propto \mathrm{Ra}^{1/3}$, a factor $10^3$ in Ra gives 10 in Nu
  \end{itemize}
  \vskip4pt
  \keyresult{For Earth, $\mathrm{Nu} = (\mathrm{Ra}/\mathrm{Ra}_c)^{1/3} \approx 18$--$39$: convection moves heat tens of times faster than conduction.}
  \sourcenote{Schubert et al.\ (2001), \textit{Mantle Convection in the Earth and Planets}}
\end{frame}

% ── Rayleigh number comparison ────────────────────────────
\begin{frame}{Rayleigh number across the solar system}
  \centering
  \small
  \begin{tabular}{l c l}
    \toprule
    \textbf{Body} & \textbf{Ra} & \textbf{Regime} \\
    \midrule
    Earth's mantle      & $10^7$--$10^8$ & Vigorous convection \\
    Venus' mantle       & $10^7$--$10^8$ & Vigorous (stagnant lid) \\
    Mars' mantle        & $10^6$--$10^7$ & Moderate convection \\
    Moon's mantle       & $10^4$--$10^5$ & Weak; cooling by conduction \\
    Mercury's mantle    & $\sim$$10^4$    & Marginal; largely cooled \\
    Magma ocean         & $\gtrsim 10^{25}$ & Turbulent convection \\
    Earth's outer core  & $\gtrsim 10^{25}$ & Turbulent (dynamo) \\
    \bottomrule
  \end{tabular}
  \vskip8pt
  \keyresult{Ra scales as $d^3/\eta$: small bodies and stiff interiors drop below $\mathrm{Ra}_c$ quickly.}
\end{frame}

\end{frame}

% ── Adiabatic temperature gradient ───────────────────────
\begin{frame}{The adiabatic gradient}
  In the well-mixed convecting interior, temperature follows an \textbf{adiabat}:
  \vskip6pt
  $$\left(\dv{T}{r}\right)_{\mathrm{adiabat}} = -\frac{\alpha g T}{c_p}$$
  \vskip6pt
  \begin{itemize}
    \item Earth's upper mantle: about 0.3 to 0.5~K\,km$^{-1}$
    \item Conductive gradient in the lithosphere: about 25~K\,km$^{-1}$
  \end{itemize}
  \vskip6pt
  \keyresult{Only a few hundred K across the whole convecting mantle: efficient mixing keeps the interior nearly adiabatic.}
\end{frame}

% ── Core cooling and the dynamo ──────────────────────────
\begin{frame}{Core cooling and the dynamo}
  The mantle is the thermostat of the core.
  \vskip6pt
  \begin{itemize}
    \item A convecting mantle extracts core heat and lets the core convect
    \item Core cooling crystallises the inner core: latent heat and light elements
    \item This compositional buoyancy drives today's geodynamo (Lecture~4)
  \end{itemize}
  \vskip6pt
  \keyresult{Earth's field exists because mantle convection cools the core; stagnant-lid Venus likely lacks this.}
\end{frame}

% ── Secular cooling ──────────────────────────────────────
\begin{frame}{Secular cooling of planets}
  \begin{itemize}
    \item Stored heat $\sim E_{\mathrm{acc}} + E_{\mathrm{diff}} + \int L_{\mathrm{rad}}(t)\,dt$; lost heat $\sim \int L_{\mathrm{surface}}(t)\,dt$
    \item Earth cools at about 50 to 100~K per Gyr
    \item Smaller bodies cool faster: lower thermal inertia, thinner boundary layer
  \end{itemize}
  \vskip8pt
  \keyresult{All terrestrial planets are in net secular cooling; convection slows Earth's cooling but does not stop it.}
\end{frame}

% ── Tidal heating rate formula ────────────────────────────
\begin{frame}{Tidal heating: eccentric orbits and the dissipation rate}
  \keyresult{$$\dot{E}_{\mathrm{tidal}} = \frac{21}{2}\,\frac{k_2}{Q}\,\frac{G M_p^2 R^5 n e^2}{a^6}$$}
  \vskip4pt
  \begin{itemize}\setlength\itemsep{3pt}
    \item $k_2$: Love number (about 0.3 for rock); $Q$: tidal quality factor, lower means more dissipative; $n = \sqrt{GM_p/a^3}$: mean motion
    \item A circular, tidally locked orbit has a static bulge and no heating; eccentricity flexes the body every orbit
    \item At fixed $e$ the heating scales as $a^{-15/2}$: dramatic for close-in moons
  \end{itemize}
\end{frame}

\begin{frame}{Where in Enceladus is tidal heat dissipated?}

    \item \textbf{Ice shell:} viscous dissipation at the warm shell base, 10 to 20~GW, focused at the south pole

\begin{frame}{Summary}
  \begin{enumerate}\setlength\itemsep{6pt}
    \item Four heat sources; Earth loses 47~TW, 40\% radiogenic and 60\% primordial
    \item Conduction ($\tau \sim L^2/\kappa$) is too slow for planets: they convect, $\mathrm{Ra} \gg \mathrm{Ra}_c$, Earth as a mobile lid, the others as stagnant lids
    \item Tidal heating powers Io and keeps the oceans of Enceladus, Europa and Titan liquid
  \end{enumerate}
\end{frame}
