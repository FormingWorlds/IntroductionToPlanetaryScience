\section{Energy sources}
% ════════════════════════════════════════════════════════════

% ── Four energy sources ────────────────────────────────────
\begin{frame}{What heats a planet?}
  Four principal sources of internal heat:
  \vskip8pt
  \begin{enumerate}
    \item \textbf{Accretional heating:} gravitational energy from assembly
    \item \textbf{Gravitational differentiation:} core formation releases $\sim$10\% more
    \item \textbf{Radioactive decay:} sustained for billions of years
    \item \textbf{Tidal heating:} orbital energy dissipated as friction
  \end{enumerate}
  \vskip8pt
  Without internal heat $\rightarrow$ no volcanism, no magnetic field, no plate tectonics, no atmospheric outgassing.
\end{frame}

% ── Accretional heating ───────────────────────────────────
\begin{frame}{Accretional heating}
  Gravitational energy of assembling a uniform sphere:
  $$E_{\mathrm{acc}} \sim \frac{3}{5}\frac{GM^2}{R}$$
  \vskip6pt
  For Earth: $E_{\mathrm{acc}} \approx 2.2 \times 10^{32}$~J
  \vskip6pt
  If all retained as heat:
  $$\Delta T = \frac{E_{\mathrm{acc}}}{M c_p} \approx \frac{2.2 \times 10^{32}}{5.97 \times 10^{24} \times 1200} \approx 30{,}000\;\text{K}$$
  \vskip6pt
  \keyresult{Enough to melt Earth many times over. Accretion inevitably produces a \textbf{magma ocean}.}
\end{frame}

% ── Accretional energy partitioning ───────────────────────
\begin{frame}{Where does the accretional energy go?}
  Not all $E_{\mathrm{acc}}$ is retained as interior heat.
  \vskip6pt
  \begin{itemize}
    \item \textbf{Radiation to space:} hot surface cools efficiently during late accretion
    \item \textbf{Thermal blanketing:} a thick primordial atmosphere traps heat, raising retention
    \item \textbf{Depth of deposition:} energy from giant impacts is buried deep, while small impacts deposit near the surface
    \item Typical retained fraction: 10--50\% (depends on impact size distribution)
  \end{itemize}
  \vskip6pt
  \keyresult{Even 10\% retention is enough to fully melt a terrestrial planet: the magma ocean is effectively guaranteed for bodies $\gtrsim$ Mars-sized.}
\end{frame}

% ── Lithophile vs siderophile preview ───────────────────
\begin{frame}{Where do heat-producing elements live?}
  \textbf{Preview for Lecture 4}: elements partition differently between metal and silicate phases.
  \vskip2pt
  \begin{itemize}
    \item \textbf{Lithophile} (``rock-loving''): partition into silicate phase
    \begin{itemize}
      \item U, Th, K: the long-lived heat producers $\Rightarrow$ \textbf{mantle and crust}
    \end{itemize}
    \vskip4pt
    \item \textbf{Siderophile} (``iron-loving''): partition into metallic iron
    \begin{itemize}
      \item Fe, Ni, Co, Pt, Ir, Au $\Rightarrow$ \textbf{core}; very little radiogenic heating there today
    \end{itemize}
  \end{itemize}
  \vskip6pt
  \keyresult{Because U, Th, K are lithophile, radiogenic heat is produced in the \textbf{mantle}, not the core, and that sustains mantle convection.}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Goldschmidt (1937); McDonough (2003)}
\end{frame}

% ── Gravitational differentiation ─────────────────────────
\begin{frame}{Gravitational differentiation}
  A second, one-time burst of heat accompanies \textbf{core formation}.
  \vskip6pt
  \begin{itemize}
    \item Dense iron droplets sink from the magma ocean to the centre
    \item Gravitational potential energy $\rightarrow$ heat via viscous friction
    \item Energy release $\sim$10\% of the accretional budget
  \end{itemize}
  \vskip6pt
  Order-of-magnitude estimate for Earth:
  $$E_{\mathrm{diff}} \sim \frac{3}{10}\,\frac{GM_{\mathrm{core}}^2}{R_{\mathrm{core}}}\left(\frac{R}{R_{\mathrm{core}}} - 1\right) \approx 2 \times 10^{31}~\text{J}$$
  \vskip4pt
  \keyresult{This heat is released \textbf{once}, during core formation, and sustains the magma ocean long enough for chemical differentiation (Lecture~4).}
\end{frame}

  \end{itemize}
\end{frame}

% ── Short-lived isotopes ──────────────────────────────────
\begin{frame}{Short-lived isotopes: ${}^{26}\mathrm{Al}$}
  \begin{itemize}
    \item Half-life: only 0.72~Myr (now extinct)
    \item Produced by nearby stellar nucleosynthesis, delivered into the protoplanetary disk
    \item Initial ${}^{26}\mathrm{Al}/^{27}\mathrm{Al}$ ratio $\sim 5 \times 10^{-5}$ (canonical)
    \item In the first few Myr: extraordinarily potent heat source ($10^{-7}$~W\,kg$^{-1}$ initially)
    \item Could melt planetesimals as small as a few km radius
    \item Initiated core formation and volatile loss \textbf{before} planets fully assembled
  \end{itemize}
  \vskip8pt
  \keyresult{${}^{26}\mathrm{Al}$ is why even small asteroids show evidence of melting and differentiation in the meteorite record.}
\end{frame}

% ── Nucleosynthesis origins ──────────────────────────────
\begin{frame}{Where did ${}^{26}\mathrm{Al}$ come from?}
  Short-lived isotopes require a nearby stellar source just before solar system formation.
  \vskip2pt
  \begin{itemize}
    \item ${}^{26}\mathrm{Al}$ is produced in:
    \begin{itemize}
      \item Massive star winds (Wolf-Rayet stars)
      \item Core-collapse supernovae
    \end{itemize}
    \item With a 0.72~Myr half-life, it must reach the nascent solar nebula within $\sim$1--2 Myr of production
    \item Implies the Sun formed in a \textbf{massive star cluster}, an Orion-like star-forming region
    \item ${}^{60}\mathrm{Fe}$: produced primarily by supernovae; detection confirms contribution from a nearby SN
    \item Isotopic abundances in meteorites constrain: one nearby massive star, or multiple injection events
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Dauphas \& Chaussidon (2011); Desch et al.\ (2023)}
\end{frame}

% ── Heat budget over time ─────────────────────────────────
\begin{frame}{How heat production has changed over time}
  \begin{itemize}
    \item $t = 0$ (solar system formation): ${}^{26}\mathrm{Al}$ dominates ($\sim$10$^{-7}$ W\,kg$^{-1}$)
    \item $t = 5$~Myr: ${}^{26}\mathrm{Al}$ mostly gone
    \item $t = 10$--100~Myr: transition to long-lived isotopes
    \item $t = 0.5$~Gyr: ${}^{235}\mathrm{U}$ and ${}^{40}\mathrm{K}$ dominate
    \item $t = 4.5$~Gyr (today): ${}^{238}\mathrm{U}$ and ${}^{232}\mathrm{Th}$ dominate
    \item Total radiogenic heat today is $\sim$20\% of its value at formation (1/4.7)
  \end{itemize}
  \vskip6pt
  \keyresult{Small bodies melted early thanks to ${}^{26}\mathrm{Al}$, even if their long-lived budget was modest.}
\end{frame}

\section{Heat transport mechanisms}
% ════════════════════════════════════════════════════════════

% ── Three mechanisms ──────────────────────────────────────
\begin{frame}{Three ways to move heat}
  \begin{description}
    \item[Conduction:] Heat transferred by molecular/atomic collisions\\
    $\vec{q} = -k\nabla T$ (Fourier's law)\\
    Slow, diffusive; dominant in lithosphere, small bodies

    \vskip6pt
    \item[Convection:] Heat transported by bulk fluid motion\\
    Hot material rises, cool material sinks\\
    Dominant in mantles, outer cores, giant planet interiors

    \vskip6pt
    \item[Radiation:] Energy carried by electromagnetic waves\\
    $F = \epsilon\sigma T^4$ (Stefan--Boltzmann)\\
    Dominant at surfaces and in atmospheres
  \end{description}
\end{frame}

% ── Conduction details ────────────────────────────────────
\begin{frame}{Conduction: Fourier's law}
  \begin{itemize}
    \item Heat flux: $\vec{q} = -k\nabla T$
    \item Thermal diffusivity: $\kappa = k/(\rho c_p) \approx 10^{-6}$~m$^2$\,s$^{-1}$ (rock)
    \item Inherently \textbf{diffusive}, slow over large distances
    \item The time for heat to conduct through distance $L$:
  \end{itemize}
  \vskip6pt
  \keyresult{$$\tau_{\mathrm{cond}} \sim \frac{L^2}{\kappa}$$}
  \vskip6pt
  We will derive this next.
\end{frame}

% ── Thermal conductivity values ───────────────────────────
\begin{frame}{Thermal conductivity: typical values}
  \centering
  \small
  \begin{tabular}{l c c}
    \toprule
    \textbf{Material} & $\boldsymbol{k}$ \textbf{(W\,m$^{-1}$\,K$^{-1}$)} & $\boldsymbol{\kappa}$ \textbf{(m$^2$\,s$^{-1}$)} \\
    \midrule
    Silicate rock (upper mantle) & 3--5 & $\sim$10$^{-6}$ \\
    Bridgmanite (lower mantle) & 5--10 & $\sim$2$\times 10^{-6}$ \\
    Water ice & 2.2 & 1.1$\times 10^{-6}$ \\
    Iron (solid) & 80 & $\sim$2$\times 10^{-5}$ \\
    Air (atmosphere) & 0.025 & $\sim$2$\times 10^{-5}$ \\
    \bottomrule
  \end{tabular}
  \vskip6pt
  \flushleft
  \begin{itemize}
    \item Metals have high $k$ but also high $\rho c_p$: intermediate $\kappa$
    \item Rocks have low $k$ and low $\rho c_p$: similar $\kappa$
    \item This is why the $L^2/\kappa$ scaling matters more than $k$ alone
  \end{itemize}
\end{frame}

% ── Radiation dominance ───────────────────────────────────
\begin{frame}{When does radiation dominate?}
  The Stefan--Boltzmann law $F = \epsilon\sigma T^4$ gives the heat flux from a surface at temperature $T$.
  \vskip6pt
  \textbf{Radiation matters in:}
  \begin{itemize}
    \item \textbf{Planetary surfaces:} heat loss to space (Lecture~5 energy balance)
    \item \textbf{Optically thin atmospheres:} e.g.\ Mars' upper atmosphere
    \item \textbf{Magma oceans:} molten silicate at $T \gtrsim 2000$~K radiates efficiently
  \end{itemize}
  \vskip6pt
  \textbf{Why not in solid interiors?}
  \begin{itemize}
    \item Silicate rocks are opaque at infrared wavelengths
    \item Photon mean free paths are much smaller than conductive length scales
    \item Exception: partially transparent silicates above $\sim$2000~K become radiatively conductive
  \end{itemize}
\end{frame}

% ── Convection overview ──────────────────────────────────
\begin{frame}{Convection: nature's heat elevator}
  \begin{columns}[c]

    \centering

      \item Cool material at top: dense, sinks

\end{frame}

\section{Blackboard derivation: cooling timescale}
% ════════════════════════════════════════════════════════════

% ── Setup: goal and strategy ──────────────────────────────
\begin{frame}{Goal and strategy}
  \textbf{Goal:} Estimate how long it takes heat to conduct out of a body of size $L$.
  \vskip8pt
  \textbf{Strategy:} Start from the heat equation, then use dimensional analysis instead of solving the PDE.
  \vskip10pt
  \begin{itemize}
    \item We want a characteristic timescale $\tau_{\mathrm{cond}}(L, \kappa)$
    \item Plug in representative values for rock
    \item Compare to the age of the solar system ($4.6$~Gyr)
  \end{itemize}
  \vskip8pt
  \textbf{Physical intuition:} bigger bodies have a longer path for heat to diffuse, so they stay hot for longer. We will derive exactly \textit{how much} longer.
\end{frame}

% ── Step 1: heat equation ─────────────────────────────────
\begin{frame}{Step 1: the heat diffusion equation}
  Fourier's law $\vec{q} = -k\nabla T$ combined with energy conservation gives:
  $$\rho c_p\,\pdv{T}{t} = \nabla\!\cdot\!(k\nabla T)$$
  \vskip6pt
  For constant thermal conductivity $k$, define $\kappa = k/(\rho c_p)$:
  $$\pdv{T}{t} = \kappa \nabla^2 T$$
  \vskip6pt
  In one dimension:
  $$\pdv{T}{t} = \kappa \pdv{^2 T}{x^2}$$
  \vskip6pt
  The rate of temperature change is proportional to the \textbf{curvature} of the temperature profile; flat profiles do not evolve, curved ones relax.
\end{frame}

% ── Step 2: dimensional analysis ──────────────────────────
\begin{frame}{Step 2: dimensional analysis}
  Replace derivatives with characteristic scales:
  \begin{itemize}\setlength\itemsep{1pt}
    \item Time derivative: $\pdv{T}{t} \sim \dfrac{T}{\tau}$
    \item Spatial derivative: $\pdv{^2 T}{x^2} \sim \dfrac{T}{L^2}$
  \end{itemize}
  Substitute into the heat equation:
  $$\frac{T}{\tau} \sim \kappa\,\frac{T}{L^2}$$
  \vskip2pt
  Temperature cancels: the timescale is intrinsic, not tied to any particular $T$.
  \vskip2pt
  \keyresult{$$\boxed{\tau_{\mathrm{cond}} \sim \frac{L^2}{\kappa}}$$}
  \vskip4pt
  \textbf{Key feature:} the $L^2$ scaling. Doubling the body size $\rightarrow$ 4$\times$ the cooling time.
\end{frame}

% ── Application table ──────────────────────────────────────
\begin{frame}{Cooling timescales across the solar system}
  Using $\kappa \approx 10^{-6}$~m$^2$\,s$^{-1}$ (silicate rock):
  \vskip8pt
  \centering
  \begin{tabular}{l r r r}
    \toprule
    \textbf{Body} & \textbf{Size $L$} & $\boldsymbol{\tau = L^2/\kappa}$ & \textbf{Familiar units} \\
    \midrule
    1~m boulder  & 1~m               & $10^6$~s              & $\sim$12 days \\
    100~km asteroid & $10^5$~m        & $10^{16}$~s           & $\sim$300~Myr \\
    Moon         & $1.7 \times 10^6$~m & $3 \times 10^{18}$~s & $\sim$100~Gyr \\
    Earth        & $6.4 \times 10^6$~m & $4 \times 10^{19}$~s & $\sim$1300~Gyr \\
    \bottomrule
  \end{tabular}
  \vskip10pt
  \begin{alertblock}{}
    Earth's conductive cooling time $\gg$ age of the universe.\\
    Yet Earth loses 47~TW $\Rightarrow$ there \textbf{must} be convection.
  \end{alertblock}
\end{frame}

\begin{frame}{Reading the cooling-timescale plot}

    \begin{itemize}\setlength\itemsep{1pt}
      \item The line is $\tau = L^2/\kappa$ on log-log axes: the $L^2$ scaling appears as slope 2.
      \item Horizontal dashes mark the ages of the solar system (4.57 Gyr) and the universe (13.8 Gyr).
      \item A chondrule (0.5~mm) cools in under a second; a 1~km asteroid needs $\sim$30\,000 yr; the Moon, Mars, and Earth plot far above both age lines.
    \end{itemize}
    \vskip2pt\relax
    \keyresult{Any body larger than a few hundred km cannot have cooled by conduction: it must convect, or stay hot.}
  \end{columns}
\end{frame}

\section{Rayleigh number \& convective vigour}
% ════════════════════════════════════════════════════════════

% ── Rayleigh number ───────────────────────────────────────
\begin{frame}{The Rayleigh number}
  Does a fluid layer convect or merely conduct?
  \vskip8pt
  $$\mathrm{Ra} = \frac{\alpha \rho g \Delta T\, d^3}{\kappa \eta}$$
  \vskip6pt
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Numerator} (buoyancy):
    \begin{itemize}
      \item $\alpha$ = thermal expansion
      \item $\rho$ = density
      \item $g$ = gravity
      \item $\Delta T$ = temperature contrast
      \item $d$ = layer depth
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Denominator} (dissipation):
    \begin{itemize}
      \item $\kappa$ = thermal diffusivity
      \item $\eta$ = dynamic viscosity
    \end{itemize}
    \vskip10pt
    Critical value: $\mathrm{Ra}_c \approx 1708$\\
    {\small (rigid boundaries: fluid sticks to wall; $658$ for free-slip: fluid slides without friction)}
  \end{columns}
\end{frame}

% ── Marginal stability pair ───────────────────────────────
\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{marginal_stability}}\\
  {\footnotesize The marginal stability curve for Rayleigh-B\'enard convection. Course figure, after Turcotte \& Schubert (2002).}
\end{frame}

\begin{frame}{Reading the marginal stability curve}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \makebox[\linewidth][c]{\includegraphics[width=\linewidth, height=0.62\textheight, keepaspectratio]{marginal_stability}}
    \column{0.50\textwidth}
    \small
    \begin{itemize}\setlength\itemsep{1pt}
      \item Below the curve, every perturbation decays: conduction is the only steady state.
      \item Above it, convection grows; the \textbf{minimum} of the curve is $\mathrm{Ra}_c \approx 1708$, reached at wavenumber $k_c \approx 3.12$.
      \item The curve rises on both sides: very wide and very narrow cells are both harder to drive than the preferred cell size.
    \end{itemize}
    \vskip2pt\relax
    \keyresult{Convection is a threshold phenomenon with a preferred wavelength, set by the layer depth.}
  \end{columns}
\end{frame}


% ── Earth's mantle Ra ─────────────────────────────────────
\begin{frame}{Earth's mantle: vigorous convection}
  For Earth's mantle:
  $$\mathrm{Ra}_\oplus \sim \frac{2 \times 10^{-5} \times 4000 \times 10 \times 2500 \times (3 \times 10^6)^3}{10^{-6} \times 10^{21}} \sim 10^7\text{--}10^8$$
  \vskip8pt
  \begin{itemize}
    \item $\mathrm{Ra}_\oplus \gg \mathrm{Ra}_c \approx 1708$: \textbf{vigorous convection confirmed}
    \item Explains why Earth's interior cools efficiently despite conduction being inadequate
  \end{itemize}
  \vskip6pt
  \keyresult{The Nusselt number $\mathrm{Nu} = (\mathrm{Ra}/\mathrm{Ra}_c)^{1/3}$ measures convection efficiency. For Earth: $\mathrm{Nu} \approx 18$--$39$, so convection moves heat tens of times faster than conduction.}
\end{frame}

% ── Rayleigh number comparison ────────────────────────────
\begin{frame}{Rayleigh number across the solar system}
  \centering
  \small
  \begin{tabular}{l c l}
    \toprule
    \textbf{Body} & \textbf{Ra} & \textbf{Regime} \\
    \midrule
    Earth's mantle      & $10^7$--$10^8$ & Vigorous convection \\
    Venus' mantle       & $10^7$--$10^8$ & Vigorous (stagnant lid) \\
    Mars' mantle        & $10^6$--$10^7$ & Moderate convection \\
    Moon's mantle       & $10^4$--$10^5$ & Weak; cooling by conduction \\
    Mercury's mantle    & $\sim$$10^4$    & Marginal; largely cooled \\
    Magma ocean         & $\gtrsim 10^{25}$ & Turbulent convection \\
    Earth's outer core  & $\gtrsim 10^{25}$ & Turbulent (dynamo) \\
    \bottomrule
  \end{tabular}
  \vskip6pt
  \flushleft
  {\small Ra is hyper-sensitive to size ($\propto d^3$) and viscosity ($\propto 1/\eta$). Small bodies and very viscous interiors quickly drop below Ra$_c$.}
\end{frame}

% ── Nusselt scaling ───────────────────────────────────────
\begin{frame}{Nusselt number: efficiency of convection}
  \begin{itemize}
    \item $\mathrm{Nu} = q_{\mathrm{total}}/q_{\mathrm{cond}}$ = actual flux over conductive flux
    \item Nu~$=$~1: pure conduction
    \item Nu~$\gg$~1: vigorous convection
  \end{itemize}
  \vskip6pt
  Classical scaling from boundary-layer theory:
  $$\mathrm{Nu} \propto \mathrm{Ra}^\beta, \quad \beta \approx 1/3$$
  \vskip6pt
  Consequences:
  \begin{itemize}
    \item Doubling Ra only $\sim$26\% more efficient (weak sensitivity)
    \item Increasing Ra by $10^3$: Nu increases by 10$\times$
    \item Earth, with the scaling normalised to vanish at $\mathrm{Ra}_c$: $\mathrm{Nu} = (\mathrm{Ra}/\mathrm{Ra}_c)^{1/3} \approx 18$--$39$
  \end{itemize}
  \vskip6pt
  {\tiny\color{ipsText!60} Source: Schubert et al.\ (2001), \textit{Mantle Convection in the Earth and Planets}}
\end{frame}

  {\footnotesize Heat-transport efficiency: the Nusselt number against the Rayleigh number. Course figure.}
\end{frame}

\end{frame}


% ── Boundary layers ───────────────────────────────────────
\begin{frame}{Thermal boundary layers}
  A convecting interior develops three layers:
  \vskip6pt
  \begin{enumerate}
    \item \textbf{Cold boundary layer (top):} conductive, steep $T$ drop\\
    $\rightarrow$ Earth's \textbf{lithosphere} ($\sim$100--200~km)
    \vskip4pt
    \item \textbf{Well-mixed interior:} nearly adiabatic, slow $T$ variation\\
    $\rightarrow$ bulk of the mantle
    \vskip4pt
    \item \textbf{Hot boundary layer (bottom):} steep $T$ rise into the core\\
    $\rightarrow$ Earth's \textbf{D'' layer} (pronounced ``D double prime''; $\sim$200~km)
  \end{enumerate}
  \vskip8pt
  Nearly all the temperature drop occurs in the two thin boundary layers.
\end{frame}

% ── Adiabatic temperature gradient ───────────────────────
\begin{frame}{The adiabatic gradient}
  In the well-mixed convecting interior, temperature follows an \textbf{adiabat}:
  \vskip6pt
  $$\left(\dv{T}{r}\right)_{\mathrm{adiabat}} = -\frac{\alpha g T}{c_p}$$
  \vskip6pt
  \begin{itemize}
    \item $\alpha$ = thermal expansion, $g$ = gravity, $c_p$ = specific heat
    \item For Earth's upper mantle: $\sim$0.3~K/km (a few K per km)
    \item Much shallower than the conductive gradient in the lithosphere ($\sim$25~K/km)
    \item Adiabatic profiles are nearly isothermal on the scale of the mantle
  \end{itemize}
  \vskip6pt
  \keyresult{Temperature changes only a few hundred K over the entire convecting mantle: efficient mixing keeps the interior nearly adiabatic.}
\end{frame}

% ── Tectonic regimes ──────────────────────────────────────
\begin{frame}{Two tectonic regimes}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Mobile lid (plate tectonics):}
    \begin{itemize}
      \item Cold boundary layer \textbf{breaks} into plates
      \item Plates subduct, recycle material
      \item Most efficient heat loss
      \item \textbf{Only Earth}
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Stagnant lid:}
    \begin{itemize}
      \item Single thick, immobile shell
      \item Heat escapes by conduction + volcanism
      \item Less efficient cooling
      \item Venus, Mars, Mercury, Moon
    \end{itemize}
  \end{columns}
  \vskip8pt
  Why Earth is unique: likely a combination of water, size, and thermal history.
\end{frame}

% ── Why Earth has plate tectonics ────────────────────────
\begin{frame}{Why does Earth have plate tectonics?}
  One of the biggest open questions in planetary geophysics.
  \vskip2pt
  Proposed factors:
  \begin{itemize}
    \item \textbf{Water:} weakens rocks at fault zones; lowers melting point in subducting slabs
    \item \textbf{Planet size:} higher convective stresses in a larger mantle
    \item \textbf{Thermal state:} must be cold enough for a rigid lid but hot enough to sustain plume-driven rifting
    \item \textbf{Prior history:} once plate tectonics starts, it may be self-sustaining (recycling weakens the lid)
    \item Venus may have had plate tectonics in the past (tessera highlands); lost it after drying out
  \end{itemize}
  \vskip2pt
  \keyresult{Habitability-oriented view: plate tectonics is linked to liquid water, and both are linked to long-term climate stability.}
\end{frame}

% ── Heat-pipe volcanism ──────────────────────────────────
\begin{frame}{Heat-pipe volcanism}
  An alternative heat-loss regime for very hot bodies.
  \vskip6pt
  \begin{itemize}
    \item Magma rises through conduits (``heat pipes'')
    \item Erupts on the surface, cools, and is buried by new lava
    \item Old crust subsides into the hot interior as new lava covers it
    \item \textbf{Io today:} global resurfacing by volcanism every $\sim$10$^6$~yr
    \item \textbf{Early Earth:} may have been a heat-pipe regime (Hadean, $>$4~Ga)
    \item Allows efficient cooling of a very hot mantle without plate tectonics
  \end{itemize}
  \vskip6pt
  {\tiny\color{ipsText!60} Source: Turcotte (1989); Moore \& Webb (2013), \textit{Nature}}
\end{frame}

% ── Mantle plumes and hotspots ────────────────────────────
\begin{frame}{Mantle plumes and hotspots}
  Whole-mantle convection is dominated by sheet-like downwellings (subducting slabs), but the hot bottom boundary layer is also unstable.
  \vskip6pt
  \begin{itemize}
    \item Hot, buoyant material collects at the base of the mantle (D$''$)
    \item Periodically detaches as narrow, rising \textbf{plumes} ($\sim$100~km diameter)
    \item Reach the lithosphere and generate \textbf{intraplate volcanism}
    \item Plumes are long-lived ($10^7$--$10^8$~yr) and approximately fixed in the deep mantle
  \end{itemize}
  \vskip6pt
  \textbf{Examples (Earth):} Hawaii, Iceland, Yellowstone, Réunion.
  \vskip6pt
  \keyresult{Plumes move heat from the core--mantle boundary to the surface independently of plate tectonics.}
\end{frame}

% ── Core cooling and the dynamo ──────────────────────────
\begin{frame}{Core cooling and the dynamo}
  The mantle acts as a thermostat on the core.
  \vskip6pt
  \begin{itemize}
    \item Heat flux from core to mantle sets the rate of core cooling
    \item A vigorously convecting mantle extracts heat efficiently, enabling core convection
    \item Core cooling drives the inner core's gradual crystallisation
    \item Inner core growth releases latent heat + light elements $\rightarrow$ compositional buoyancy
    \item This compositional convection is the \textbf{dominant} driver of today's geodynamo (Lecture~4)
  \end{itemize}
  \vskip6pt
  \keyresult{Earth's magnetic field ultimately exists because mantle convection efficiently cools the core; stagnant-lid Venus likely lacks this.}
\end{frame}

% ── Whole-mantle vs layered convection ──────────────────
\begin{frame}{Whole-mantle vs.\ layered convection}
  \begin{columns}[c]
    \column{0.48\textwidth}
    \centering
    \includegraphics[width=\linewidth,height=0.42\textheight,keepaspectratio]{seismic_tomography}
    \source{Wikimedia (USGS/Caltech), CC BY-SA}

    \column{0.50\textwidth}
    \textbf{The debate:} does Earth's mantle convect as one layer or two?
    \vskip6pt
    \begin{itemize}
      \item Seismic tomography images the Farallon slab descending to the core--mantle boundary
      \item Plumes rise from the hot bottom boundary layer through the full mantle
      \item Strong evidence for \textbf{whole-mantle convection}
      \item A partial barrier at 660~km slows, but does not stop, slabs; the mantle convects as one layer
    \end{itemize}
  \end{columns}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Grand (2002); Ritsema et al.\ (2011)}
\end{frame}

% ── Mantle plume detail ──────────────────────────────────
\begin{frame}{Plumes vs.\ plates: two modes of upwelling}
  \begin{itemize}
    \item \textbf{Mid-ocean ridges:} passive upwelling where plates pull apart
    \item Temperatures close to the ambient mantle
    \item \textbf{Mantle plumes:} active upwelling from the deep mantle
    \item $\sim$200--300~K hotter than the ambient mantle (from olivine thermobarometry)
    \item Plumes deliver primordial (3He-rich) material from the lower mantle
    \item Large Igneous Provinces (LIPs) may be plume-heads: Deccan Traps, Siberian Traps, Ontong-Java Plateau
    \item LIPs correlate with mass extinction events in geological history
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Courtillot \& Renne (2003), \textit{Comptes Rendus Geoscience}}
\end{frame}

      \item BUT concentration of U, Th, K in the crust adds local contribution
      \item $\sim$40\% of the continental surface flux is crustal-radiogenic
    \end{itemize}
  \end{itemize}
  \vskip4pt

    \column{0.50\textwidth}
    \footnotesize
    \begin{itemize}\setlength\itemsep{1pt}
      \item The blue curve is the half-space prediction $q \propto t^{-1/2}$: the same erf-solution as before, differentiated at the surface.
      \item At old ages the data level off at the plate-cooling asymptote ($\sim$48 mW\,m$^{-2}$): the lithosphere reaches an equilibrium thickness.
      \item Over young crust, measured values fall \emph{below} the curve because hydrothermal circulation carries part of the heat.
    \end{itemize}
    \vskip2pt\relax
    \keyresult{One curve connects the heat equation to ship-borne heat-flow surveys, and its deviations are physics too.}
  \end{columns}
\end{frame}


% ── Tidal heating mechanism ───────────────────────────────
\begin{frame}{Tidal heating: how it works}
  \begin{itemize}
    \item Circular orbit + tidal lock $\rightarrow$ static bulge, \textbf{no heating}
    \item \textbf{Eccentric orbit:} distance varies each orbit
    \begin{itemize}
      \item Closer: tidal force stronger, moon moves faster
      \item Farther: tidal force weaker, moon moves slower
    \end{itemize}
    \item Continuously varying deformation $\rightarrow$ \textbf{flexing dissipates heat}
  \end{itemize}
  \vskip6pt
  Heating depends on:
  \begin{itemize}
    \item Eccentricity $e$ (zero $e$ $\rightarrow$ zero heating)
    \item Distance $a$: heating $\propto a^{-15/2}$ at fixed $e$ (the explicit $a^{-6}$ times $n \propto a^{-3/2}$)
    \item Internal dissipation (tidal quality factor $Q$)
  \end{itemize}
\end{frame}

% ── Tidal flexing pair ────────────────────────────────────
\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{tidal_flexing}}\\
  {\footnotesize Tidal flexing on an eccentric orbit: the origin of tidal heating. Course schematic.}
\end{frame}

\begin{frame}{Reading the tidal-flexing schematic}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \makebox[\linewidth][c]{\includegraphics[width=\linewidth, height=0.62\textheight, keepaspectratio]{tidal_flexing}}
    \column{0.50\textwidth}
    \small
    \begin{itemize}\setlength\itemsep{1pt}
      \item \textbf{Left, periapsis} ($r = a(1-e)$): the moon is closest and its tidal bulge is largest.
      \item \textbf{Right, apoapsis} ($r = a(1+e)$): the bulge is smallest.
      \item Each orbit, the body is squeezed and stretched between the two states; internal friction turns that cyclic work into heat at a rate $\propto e^2/Q$.
    \end{itemize}
    \vskip2pt\relax
    \keyresult{Eccentricity is the on-off switch of tidal heating: no $e$, no flexing, no heat.}
  \end{columns}
\end{frame}


% ── Secular cooling ──────────────────────────────────────
\begin{frame}{Secular cooling of planets}
  Planets are thermally non-stationary: they started hot and are gradually cooling.
  \vskip6pt
  \begin{itemize}
    \item Stored heat $\sim E_{\mathrm{acc}} + E_{\mathrm{diff}} + \int L_{\mathrm{rad}}(t)\,dt$
    \item Lost heat $\sim \int L_{\mathrm{surface}}(t)\,dt$
    \item The difference is the present-day thermal state
    \item Earth: still cooling at $\sim$50--100~K/Gyr
    \item Smaller bodies cool faster (lower thermal inertia, thinner conductive boundary layer)
  \end{itemize}
  \vskip6pt
  \keyresult{All terrestrial planets are in net secular cooling; Earth's is slowed by convective resurfacing but not stopped.}
\end{frame}

% ── Tidal heating rate formula ────────────────────────────
\begin{frame}{Tidal dissipation rate}
  For a synchronously rotating body on an eccentric orbit, the tidal heating power is (to leading order in $e$):
  \vskip2pt
  \keyresult{$$\dot{E}_{\mathrm{tidal}} = \frac{21}{2}\,\frac{k_2}{Q}\,\frac{G M_*^2 R^5 n e^2}{a^6}$$}
  \vskip2pt
  where:
  \begin{itemize}
    \item $k_2$ = Love number (dimensionless, $\sim$0.3 for rocky bodies)
    \item $Q$ = tidal quality factor (lower = more dissipative)
    \item $M_*$ = mass of the primary (Jupiter, Saturn, etc.)
    \item $R$ = radius of the moon
    \item $n$ = mean motion ($= \sqrt{GM_*/a^3}$); $a$, $e$ = semi-major axis and eccentricity
  \end{itemize}
  \vskip2pt
  \flushleft
  {\small Note the steep $R^5/a^6$ dependence: heating is dramatic for close-in moons.}
\end{frame}

% ── Laplace resonance ─────────────────────────────────────
\begin{frame}{The Laplace resonance: sustaining eccentricity}
  Io's orbit would circularise in $\sim$10~Myr if left alone. Why is its eccentricity maintained?
  \vskip6pt
  \begin{itemize}
    \item Io--Europa--Ganymede are locked in a 4:2:1 \textbf{mean-motion resonance}
    \item Every time Io laps Europa, the two meet at the same orbital phase
    \item Periodic gravitational kicks continually \textbf{pump} Io's eccentricity
    \item Without this, tidal dissipation would damp $e$ and shut off the heating
    \item The resonance was first identified by Laplace in 1805
  \end{itemize}
  \vskip6pt
  \keyresult{The three-body resonance is what makes Io's tidal heating indefinite, not transient.}
\end{frame}

% ── Enceladus ──────────────────────────────────────────────
\begin{frame}{Enceladus: a tiny moon with a big secret}
  \begin{columns}[c]
    \column{0.45\textwidth}
    \centering
    \includegraphics[width=\textwidth,height=0.44\textheight,keepaspectratio]{enceladus_plumes}

      \item Radius only 252~km

      \item Thermal power: $\sim$16~GW (tidal dissipation)
      \item Global subsurface ocean beneath $\sim$15--25~km ice
      \item Plumes carry salts, silica, organics, and $\mathrm{H_2}$: hydrothermal activity on the ocean floor
    \end{itemize}
  \end{columns}
  \vskip2pt
  \keyresult{Tidal heating can sustain liquid-water oceans far outside the habitable zone.}
\end{frame}

% ── Enceladus dissipation location ──────────────────────
\begin{frame}{Where in Enceladus is tidal heat dissipated?}
  Two competing scenarios explain the observed $\sim$16~GW of thermal power:
  \vskip6pt
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Rocky core dissipation:}
    \begin{itemize}
      \item Tidal flexing of porous silicate core
      \item Heat conducted up through the ocean
      \item Matches hydrothermal signatures (silica nanoparticles at $\sim$90\textdegree C)
      \item Difficulty: plume power exceeds simple core-heating estimates
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Ice-shell dissipation:}
    \begin{itemize}
      \item Viscous dissipation concentrated in the warm ice shell base
      \item Positive feedback: softer ice $\rightarrow$ more dissipation
      \item Can reach 10--20~GW with realistic rheology
      \item Concentrates heat near the south pole tiger stripes
    \end{itemize}
  \end{columns}
  \vskip6pt
  Likely: both contribute, with the ice shell dominating the plume heat flux.
  \vskip4pt
  {\tiny\color{ipsText!60} Source: \v{C}adek et al.\ (2016); Hemingway \& Mittal (2019)}
\end{frame}

% ── Enceladus biosignatures ──────────────────────────────
\begin{frame}{Enceladus plumes: a habitability laboratory}
  Cassini's multiple plume flythroughs revealed:
  \vskip6pt
  \begin{itemize}
    \item \textbf{Salts:} NaCl, NaHCO$_3$: direct samples of the subsurface ocean
    \item \textbf{Silica nanoparticles:} imply ocean-floor temperature $>$90\textdegree C (hydrothermal)
    \item \textbf{Molecular hydrogen} (H$_2$): produced by water--rock serpentinisation
    \item \textbf{Organic molecules:} CO$_2$, CH$_4$, NH$_3$, HCN, complex species up to C$_6$
    \item Phosphate detected in 2023 (Postberg et al.), last major missing bioessential element
  \end{itemize}
  \vskip6pt
  \keyresult{Enceladus' ocean has all the ingredients (liquid water, energy, organics, phosphorus) for chemoautotrophic life, if it exists there.}
\end{frame}

% ── More recent advances ──────────────────────────────────
\begin{frame}{InSight: Mars heat flow and interior}
  InSight's seismometer (SEIS) and heat flow probe (HP$^3$) transformed our view of Mars.
  \vskip6pt
  \begin{itemize}
    \item Over 1300 marsquakes detected during the mission
    \item Seismic core radius: $1830 \pm 40$~km, larger than pre-mission estimates
    \item Reanalyses (2023): a $\sim$1675~km metal core beneath a $\sim$150~km molten silicate layer
    \item Crustal thickness: 24--72~km (two-layer model)
    \item Heat flow probe stuck at 40~cm depth in unexpectedly cohesive regolith
    \item Upper limit on near-surface heat flux: $\sim$22~mW\,m$^{-2}$
  \end{itemize}
  \vskip6pt
  {\tiny\color{ipsText!60} Source: Stähler et al.\ (2021); Khan et al.\ (2021, 2023); Samuel et al.\ (2023); Spohn et al.\ (2022)}
\end{frame}

% ── Europa Clipper ────────────────────────────────────────
\begin{frame}{Europa Clipper: targeting the ice shell}
  NASA Europa Clipper launched October 2024, arrives at Jupiter in 2030.
  \vskip6pt
  \begin{itemize}
    \item Will make $\sim$50 close flybys of Europa over $\sim$4 years
    \item \textbf{Ice Penetrating Radar (REASON):} maps ice shell thickness and near-surface layers
    \item \textbf{Magnetometer:} constrains ocean salinity from induced response
    \item \textbf{MASPEX + SUDA:} samples the composition of possible plume material
    \item Thermal mapping will locate regions of thinner ice and recent activity
    \item Will directly test whether tidal dissipation is concentrated in the shell or in the rocky core
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Howell \& Pappalardo (2020), \textit{Nature Communications}}
\end{frame}

% ── Mars interior heat models ─────────────────────────────
\begin{frame}{Modelling Mars' thermal evolution}
  Mars' thermal history explains key observations:
  \vskip6pt
  \begin{itemize}
    \item Early intense volcanism (Tharsis) $\rightarrow$ hot magma ocean + dynamo (4.1~Ga)
    \item Dynamo shutdown $\sim$4.1~Ga $\rightarrow$ core cooling slowed, crustal magnetisation frozen
    \item Olympus Mons still eruptive until $\sim$100--200~Ma
    \item Present-day heat flow $\sim$15--25~mW/m$^2$ (InSight constraints)
    \item Partially molten upper core retains heat; slow mantle convection prolongs cooling
  \end{itemize}
  \vskip6pt
  \keyresult{Mars' thermal history ties directly to its atmospheric evolution: the loss of the dynamo caused the loss of the field, and eventually the atmosphere.}
\end{frame}

% ── Io heat flow ──────────────────────────────────────────
\begin{frame}{Io in the decade of Juno}
  Juno's extended mission (2021--) has delivered the closest-ever Io observations:
  \vskip6pt
  \begin{itemize}
    \item Global heat flow map: polar regions are hotter than expected, contradicting simple tidal models
    \item Thermal emission imaging from JIRAM instrument
    \item Gravity data consistent with a global magma ocean just beneath the crust (Park et al.\ 2024)
    \item New implications for the structure of Io's interior: is the ``mantle'' largely molten?
    \item Future: IVO (Io Volcano Observer) concept proposal for dedicated Io mission
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Park et al.\ (2024); Keszthelyi et al.\ (2022)}
\end{frame}

% ── JUICE ────────────────────────────────────────────────
\begin{frame}{JUICE: The Jupiter Icy Moons Explorer}
  ESA's JUICE launched in April 2023, arrives Jupiter 2031.
  \vskip6pt
  \begin{itemize}
    \item Multiple flybys of Europa, Callisto, and Ganymede
    \item Orbit insertion at \textbf{Ganymede} in 2034: first orbiter of a moon other than our own
    \item Will determine the thickness of Ganymede's ice shell and ocean via laser altimetry and induced magnetic field
    \item Direct measurement of tidal heating through tidal Love number $k_2$
    \item Complementary to Europa Clipper: two flagship ocean world missions in the 2030s
  \end{itemize}

% ── Ocean world habitability ─────────────────────────────
\begin{frame}{Are Ocean Worlds Habitable?}
  \begin{itemize}
    \item Require \textbf{three things}: liquid water, energy, chemical nutrients
    \item \textbf{Liquid water:} tidal heating + ice shell insulation
    \item \textbf{Energy:} chemical disequilibria (serpentinisation produces H$_2$)
    \item \textbf{Nutrients:} rock--water reactions deliver P, S, trace metals
    \item \textbf{Open questions:} redox state, temperature range, ocean pH, mixing timescales
    \item Ganymede, Europa, Enceladus, Titan, Triton, Pluto and maybe Ceres all have possible subsurface oceans
  \end{itemize}
  \vskip6pt
  {\tiny\color{ipsText!60} Source: Nimmo \& Pappalardo (2016), \textit{J.\ Geophys.\ Res.\ Planets}}
\end{frame}

% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}
    \item Four heat sources: accretional, gravitational differentiation, radioactive decay, tidal heating
    \item Conduction is diffusive ($\tau \sim L^2/\kappa$), too slow for planet-sized bodies
    \item Convection dominates in large bodies: $\mathrm{Ra} \gg \mathrm{Ra}_c$
    \item Earth: mobile lid (plate tectonics); other rocky bodies: stagnant lid
    \item Tidal heating powers Io's volcanism and maintains subsurface oceans on Enceladus, Europa, and Titan
    \item Earth's total heat loss: 47~TW (40\% radiogenic, 60\% primordial)
  \end{enumerate}
\end{frame}
