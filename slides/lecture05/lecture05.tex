\section{Atmospheric composition}
% ════════════════════════════════════════════════════════════

% ── Primary atmospheres ──────────────────────────────────────
\begin{frame}{Primary atmospheres}
  \textbf{Captured directly from the protoplanetary disk.}
  \vskip8pt
  \begin{itemize}
    \item Dominated by $\mathrm{H_2}$ and He (solar composition)
    \item Requires mass $\gtrsim 5$--$10\;\Mearth$ to capture nebular gas
    \item Disk disperses in $\sim$3--10~Myr (Lecture~2)
    \item \textbf{Gas giants:} Jupiter and Saturn, massive $\mathrm{H_2}$/He envelopes
    \item \textbf{Ice giants:} Uranus and Neptune, envelopes only 10--20\% of total mass
    \item Terrestrial planets too small $\rightarrow$ any primordial H quickly lost
  \end{itemize}
\end{frame}

% ── Secondary atmospheres ────────────────────────────────────
\begin{frame}{Secondary atmospheres}
  \textbf{Produced by outgassing from the interior.}
  \vskip8pt
  \begin{itemize}
    \item Volcanism and magma ocean degassing release dissolved volatiles
    \item Speciation depends on \textbf{oxygen fugacity} (melt's O$_2$ redox state; Lecture~4):
    \begin{itemize}
      \item Oxidising: $\mathrm{CO_2}$, $\mathrm{H_2O}$, $\mathrm{N_2}$
      \item Reducing: $\mathrm{H_2}$, CO, $\mathrm{N_2}$
    \end{itemize}
    \vskip4pt
    \item \textbf{Venus:} $\mathrm{CO_2}$-dominated (96.5\%), 92~bar
    \item \textbf{Mars:} $\mathrm{CO_2}$-dominated (95\%), but only 6~mbar
    \item \textbf{Titan:} thick $\mathrm{N_2}$ (from $\mathrm{NH_3}$ conversion), 1.5~bar
  \end{itemize}
\end{frame}

% ── Atmospheric origin summary ──────────────────────────
\begin{frame}{Where do atmospheres come from?}
  A pictorial summary of the three classes.
  \vskip6pt
  \begin{description}
    \item[Primary] Directly captured nebular gas; H$_2$/He dominated.\\
    Requires $M \gtrsim 5$--$10\,\Mearth$ before disk dispersal.
    \vskip4pt
    \item[Secondary] Outgassed from the interior during accretion + cooling.\\
    Composition set by $f_{\mathrm{O}_2}$ of the mantle.
    \vskip4pt
    \item[Tertiary] Modified by surface processes, photochemistry, and (for Earth) biology.
  \end{description}
  \vskip6pt
  Over time, a planet's atmospheric inventory can be \textbf{replaced} or \textbf{re-equilibrated} entirely: the present composition may not reflect the initial inventory.
\end{frame}

% ── Tertiary atmospheres ─────────────────────────────────────
\begin{frame}{Tertiary atmospheres}
  \textbf{Modified by surface processes, photochemistry, or biology.}
  \vskip8pt
  \begin{itemize}
    \item Earth's original outgassed atmosphere: likely $\mathrm{CO_2}$ + $\mathrm{N_2}$ (similar to Venus)
    \item Oxygenic photosynthesis transformed it over billions of years
    \item Present atmosphere: $\mathrm{N_2}$ (78\%) + $\mathrm{O_2}$ (21\%)
    \item \textbf{$\mathrm{O_2}$ is entirely biogenic}, and would disappear in $\sim$Myr without life
  \end{itemize}
  \vskip6pt
  \keyresult{Earth's atmosphere is a \textbf{tertiary atmosphere}: the product of billions of years of biological modification.}
\end{frame}

\section{Hydrostatic equilibrium}
% ════════════════════════════════════════════════════════════

% ── Slab force balance ───────────────────────────────────────
\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{hydrostatic_slab}}\\
  {\footnotesize Force balance on a thin slab of area $A$ and thickness $\dd z$: pressure on the lower face pushes up, pressure on the upper face and the slab's own weight push down. Course figure.}
\end{frame}

% ── Hydrostatic equilibrium derivation ───────────────────────
\begin{frame}{Hydrostatic equilibrium}
  Consider a thin slab of atmosphere at height $z$, thickness $\dd z$, area $A$:
  \vskip6pt
  \begin{itemize}
    \item Pressure from below (upward): $P(z) \cdot A$
    \item Pressure from above (downward): $P(z + \dd z) \cdot A$
    \item Weight (downward): $\rho(z)\,g\,A\,\dd z$
  \end{itemize}
  \vskip6pt
  Force balance:
  $$P(z)\,A - P(z + \dd z)\,A = \rho\,g\,A\,\dd z$$
  \vskip4pt
  \keyresult{$$\boxed{\dv{P}{z} = -\rho\,g}$$}
  \vskip4pt
  Pressure decreases with altitude: each layer supports the weight above it.
\end{frame}

% ── Ideal gas law ────────────────────────────────────────────
\begin{frame}{The ideal gas law (atmospheric form)}
  To solve hydrostatic equilibrium, we need $\rho(P)$:
  \vskip8pt
  $$P = n\,\kB T = \frac{\rho\,\kB T}{\mu\,m_u}$$
  \vskip6pt
  \begin{tabular}{l l}
    $n$    & number density \\
    $\kB$  & Boltzmann constant ($1.381 \times 10^{-23}$~J\,K$^{-1}$) \\
    $\mu$  & mean molecular weight (dimensionless, in amu) \\
    $m_u$  & atomic mass unit ($1.661 \times 10^{-27}$~kg) \\
  \end{tabular}
  \vskip8pt
  Rearranging: $\rho = P\,\mu\,m_u / (\kB T)$
\end{frame}

% ── Barometric formula ───────────────────────────────────────
\begin{frame}{The barometric formula}
  Substitute $\rho = P\mu m_u/(\kB T)$ into hydrostatic equilibrium:
  $$\dv{P}{z} = -\frac{\mu\,m_u\,g}{\kB T}\,P$$
  \vskip4pt
  For an \textbf{isothermal} atmosphere (constant $T$, constant $g$), this separates:
  \vskip4pt
  \keyresult{$$\boxed{P(z) = P_0\,\exp\!\left(-\frac{z}{H}\right)}$$}
  \vskip6pt
  where $H = \kB T / (\mu\,m_u\,g)$ is the \textbf{pressure scale height}.
  \vskip6pt
  \begin{itemize}
    \item Pressure drops by $e \approx 2.718$ every scale height
    \item 99\% of the atmospheric mass lies below $4.6\,H$ for an isothermal column
  \end{itemize}
\end{frame}

\section{Blackboard derivation: scale height}
% ════════════════════════════════════════════════════════════

% ── Setup ────────────────────────────────────────────────────
\begin{frame}{Deriving the scale height}
  \textbf{Goal:} Derive $H = \kB T/(\mu\,m_u\,g)$ from hydrostatic equilibrium + ideal gas law, and compute $H$ for solar system bodies.
  \vskip10pt
  Start from:
  $$\dv{P}{z} = -\rho\,g \qquad\text{and}\qquad P = \frac{\rho\,\kB T}{\mu\,m_u}$$
  \vskip6pt
  Express $\rho$ in terms of $P$:
  $$\rho = \frac{P\,\mu\,m_u}{\kB T}$$
  \vskip4pt
  Substitute:
  $$\dv{P}{z} = -\frac{\mu\,m_u\,g}{\kB T}\,P \equiv -\frac{P}{H}$$
\end{frame}

% ── Result ───────────────────────────────────────────────────
\begin{frame}{The atmospheric scale height}
  \keyresult{$$\boxed{H = \frac{\kB T}{\mu\,m_u\,g}}$$}
  \vskip8pt
  Physical interpretation:
  \vskip4pt
  \begin{itemize}
    \item \textbf{Higher $T$} $\rightarrow$ larger $H$: hotter gas extends further
    \item \textbf{Heavier molecules} (larger $\mu$) $\rightarrow$ smaller $H$: harder to loft
    \item \textbf{Stronger gravity} $g$ $\rightarrow$ smaller $H$: more compressed
  \end{itemize}
  \vskip8pt
  \textbf{Key insight:} $H$ encodes the competition between thermal energy and gravitational binding.
\end{frame}

% ── Application: Earth ───────────────────────────────────────
\begin{frame}{Worked example: Earth's scale height}
  $$H_\oplus = \frac{\kB T}{\mu\,m_u\,g} = \frac{1.381 \times 10^{-23} \times 288}{28.97 \times 1.661 \times 10^{-27} \times 9.81}$$
  \vskip6pt
  $$= \frac{3.98 \times 10^{-21}}{4.72 \times 10^{-25}} \approx 8400\;\text{m} \approx 8.4\;\text{km}$$
  \vskip8pt
  \begin{itemize}
    \item Commercial aircraft: 10--12~km ($\sim$1.2--1.4~$H$), pressure $\sim$0.2--0.3~atm
    \item Top of Everest: 8.8~km ($\sim$1~$H$), pressure $\sim$0.3~atm
    \item 99\% of mass below $4.6\,H$; colder layers aloft compress this to $\sim$30~km
  \end{itemize}
\end{frame}

% ── Scale heights across the solar system ────────────────────
\begin{frame}{Scale heights across the solar system}
  \centering
  \begin{tabular}{l c c c c}
    \toprule
    \textbf{Body} & $\boldsymbol{T}$~\textbf{(K)} & $\boldsymbol{\mu}$ & $\boldsymbol{g}$~\textbf{(m\,s$^{-2}$)} & $\boldsymbol{H}$~\textbf{(km)} \\
    \midrule
    Venus   & 737  & 43.4 & 8.87  & 15.9 \\
    Earth   & 288  & 29.0 & 9.81  & 8.4  \\
    Mars    & 215  & 43.3 & 3.72  & 11.1 \\
    Jupiter & 165  & 2.2  & 24.8  & 25   \\
    Titan   & 94   & 28.6 & 1.35  & 20   \\
    \bottomrule
  \end{tabular}
  \vskip10pt
  \begin{itemize}
    \item Jupiter: large $H$ despite strong gravity, because $\mathrm{H_2}$ is very light ($\mu = 2.2$)
    \item Titan: large $H$ despite low $T$, thanks to very weak gravity ($g = 1.35$~m\,s$^{-2}$)
    \item Venus: hot but heavy molecules + decent gravity $\rightarrow$ moderate $H$
  \end{itemize}
\end{frame}

% ── Layers overview ──────────────────────────────────────────
\begin{frame}{Atmospheric layers}
  \begin{columns}[c]
    \column{0.35\textwidth}
    \centering
    \includegraphics[width=\linewidth,height=0.65\textheight,keepaspectratio]{earth_tz_layers}
    \source{Course figure; US Standard Atmosphere 1976}

    \column{0.62\textwidth}
    Layers defined by the sign of $\dv{T}{z}$:
    \vskip6pt
    \begin{enumerate}
      \item \textbf{Troposphere} (0--12~km): $T$ decreases with $z$\\
      Heated from below; convective
      \item \textbf{Stratosphere} (12--50~km): $T$ increases\\
      UV absorbed by ozone layer
      \item \textbf{Mesosphere} (50--85~km): $T$ decreases\\
      Coldest point at mesopause ($\sim$190~K)
      \item \textbf{Thermosphere} ($>$85~km): $T$ increases\\
      EUV (extreme-ultraviolet) absorption; $T > 1000$~K but gas is rarefied
    \end{enumerate}
  \end{columns}
\end{frame}

% ── Lapse rate ───────────────────────────────────────────────
\begin{frame}{The adiabatic lapse rate}
  In the troposphere, convection sets the temperature gradient:
  \vskip6pt
  \keyresult{$$\Gamma_d = -\dv{T}{z} = \frac{g}{c_p}$$}
  \vskip6pt
  \begin{itemize}
    \item \textbf{Dry adiabatic lapse rate} for Earth: $\Gamma_d = 9.81/1004 \approx 9.8$~K\,km$^{-1}$
    \item Observed average: $\sim$6.5~K\,km$^{-1}$ (lower due to latent heat from water condensation)
    \item Physical picture: rising air expands, does work on surroundings, cools
  \end{itemize}
  \vskip6pt
  \textbf{Tropopause height:} $\sim$8~km (poles) to $\sim$17~km (equator).
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{dry_moist_adiabat}}\\
  {\footnotesize Dry and moist adiabats against the US Standard Atmosphere 1976 reference profile. Course figure.}
\end{frame}

\begin{frame}{Reading the lapse-rate plot}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{dry_moist_adiabat}
    \column{\recaptextcolwide}
    \footnotesize
    \begin{itemize}\setlength\itemsep{1pt}
      \item Dashed: the dry adiabat $\Gamma_d = g/c_p \approx 9.8$ K\,km$^{-1}$; dotted: a representative moist adiabat near 5 K\,km$^{-1}$.
      \item Solid: the US Standard Atmosphere 1976 reference profile; its 6.5 K\,km$^{-1}$ tropospheric lapse rate lies between the two limits because real convection is partly saturated.
      \item Above the tropopause ($\sim$11 km) the adiabats no longer apply: the profile turns nearly isothermal.
    \end{itemize}
    \vskip2pt\relax
    \keyresult{Latent heat makes the moist adiabat shallower; the real troposphere sits between the dry and moist limits.}
  \end{columns}
\end{frame}

% ── Moist adiabat ────────────────────────────────────────────
\begin{frame}{The moist adiabatic lapse rate}
  \small
  When a parcel rises and condensation occurs, latent heat release warms it relative to the dry adiabat.
  \vskip1pt
  {\small $$\Gamma_{\mathrm{moist}} = \Gamma_{\mathrm{dry}} \cdot \frac{1 + \dfrac{L_v\,r_s}{R_d\,T}}{1 + \dfrac{L_v^2\,r_s}{c_p\,R_v\,T^2}}$$}
  \vskip2pt
  \begin{itemize}\small\setlength\itemsep{1pt}
    \item $r_s$: saturation mixing ratio; $L_v$: latent heat of vaporisation
    \item Earth's warm lower troposphere: $\Gamma_{\mathrm{moist}} \approx 5$~K/km; the observed mean (6.5) lies between moist and dry
    \item Moist convection is \textbf{deeper and more vigorous} than dry: latent heat supplies the buoyancy that lifts tropical convection to the tropopause, and drives thunderstorms and hurricanes
  \end{itemize}
  \vskip2pt
  {\tiny\color{ipsText!60} Source: Pierrehumbert (2010); Catling \& Kasting (2017)}
\end{frame}

% ── Comparative vertical structure ───────────────────────────
\begin{frame}{Comparative vertical structures}
  \small
  \begin{description}
    \item[Venus:] Massive troposphere ($\sim$65~km). No ozone layer $\rightarrow$ no stratospheric inversion. Cloud deck at 48--70~km.

    \vskip4pt
    \item[Mars:] Thin troposphere ($\sim$40~km), directly into thermosphere. No significant ozone. Dust dominates atmospheric heating.

    \vskip4pt
    \item[Jupiter:] Troposphere extends hundreds of km deep. Stratosphere heated by $\mathrm{CH_4}$. No solid surface; pressure increases continuously.

    \vskip4pt
    \item[Titan:] Thick troposphere ($\sim$40~km). Stratosphere heated by organic haze. Extended thermosphere reaches $\sim$1400~km (weak gravity).
  \end{description}
\end{frame}

\section{Radiative transfer basics}
% ════════════════════════════════════════════════════════════

% ── Absorption, emission, scattering ─────────────────────────
\begin{frame}{Three interactions: radiation and matter}
  When radiation passes through an atmosphere:
  \vskip8pt
  \begin{description}
    \item[Absorption:] Photon energy $\rightarrow$ internal energy (vibrational, rotational)\\
    Key absorbers: $\mathrm{CO_2}$, $\mathrm{H_2O}$, $\mathrm{CH_4}$, $\mathrm{O_3}$, $\mathrm{N_2O}$

    \vskip4pt
    \item[Emission:] Warm gas radiates at wavelengths where it absorbs (Kirchhoff's law)\\
    This is how the atmosphere radiates energy to space

    \vskip4pt
    \item[Scattering:] Photons redirected, not absorbed\\
    Rayleigh ($\propto \lambda^{-4}$): blue sky, red sunsets\\
    Mie (larger particles): less wavelength-dependent
  \end{description}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{rayleigh_scattering}}\\
  {\footnotesize Rayleigh scattering: the long sunset path removes blue light from the beam, and the scattering strength scales as $\lambda^{-4}$. Course figure.}
\end{frame}

% ── Blackbody spectrum figure ────────────────────────────────
\begin{frame}{Stellar vs.\ planetary emission}
  \begin{columns}[c]

    \begin{itemize}
      \item Sun ($\sim$5800~K): peak at $\sim$0.5~$\mu$m (visible)
      \item Planet ($\sim$300~K): peak at $\sim$10~$\mu$m (thermal IR)
    \end{itemize}
    \vskip6pt
    This \textbf{wavelength separation} is the physical basis of the greenhouse effect: the atmosphere can be transparent at visible wavelengths but opaque in the infrared.
  \end{columns}
\end{frame}

% ── Optical depth ────────────────────────────────────────────
\begin{frame}{Optical depth}
  The cumulative absorption along a path through the atmosphere:
  \vskip6pt
  $$\tau = \int_0^s \kappa\,\rho\,\dd s'$$
  \vskip4pt
  \begin{itemize}
    \item $\kappa$ = mass absorption coefficient (m$^2$\,kg$^{-1}$)
    \item $\tau$ is dimensionless: counts ``e-folding lengths'' of absorption
  \end{itemize}
  \vskip6pt
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Optically thin} ($\tau \ll 1$):\\
    Most radiation passes through.

    \column{0.48\textwidth}
    \textbf{Optically thick} ($\tau \gg 1$):\\
    Radiation strongly absorbed; only photons from near the ``surface'' escape.
  \end{columns}
\end{frame}

% ── Beer-Lambert law ─────────────────────────────────────────
\begin{frame}{The Beer--Lambert law}
  For a beam of initial intensity $I_0$ through a medium of optical depth $\tau$:
  \vskip6pt
  \keyresult{$$\boxed{I = I_0\,e^{-\tau}}$$}
  \vskip8pt
  \begin{itemize}
    \item Intensity drops by factor $e$ per unit optical depth
    \item \textbf{Atmospheric photosphere:} at $\tau \approx 1$, photons escape to space
    \item Below $\tau = 1$: photons reabsorbed before escaping
    \item Above $\tau = 1$: photons escape freely
    \item Temperature at the $\tau \approx 1$ level sets $T_{\mathrm{eff}}$
  \end{itemize}
\end{frame}

  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{tau_one_schematic}}\\
  {\footnotesize The atmospheric photosphere: Beer-Lambert attenuation and the $\tau \approx 1$ emission level. Course figure.}
\end{frame}

    \column{\recapfigcol}
    \ipsrecapfig{tau_one_schematic}
    \column{\recaptextcol}

    \begin{itemize}\setlength\itemsep{1pt}

      \item Panel (b): IR photons emitted deep down ($\tau \gg 1$) are reabsorbed; only photons from $\tau \lesssim 1$ reach space.
      \item The temperature at the $\tau \approx 1$ level is what an observer from space measures: the planet's effective temperature.
    \end{itemize}
    \vskip2pt\relax
    \keyresult{A planet, like a star, has a photosphere: the $\tau \approx 1$ surface from which its photons escape.}
  \end{columns}
\end{frame}

\section{Energy balance \& greenhouse effect}

% ── Radiative-convective equilibrium ────────────────────
\begin{frame}{Radiative-convective equilibrium}
  A realistic atmosphere combines two regimes:
  \vskip6pt
  \begin{itemize}
    \item \textbf{Troposphere:} strong greenhouse absorption $\rightarrow$ unstable to convection
    \begin{itemize}
      \item Temperature follows the adiabatic lapse rate
      \item Most of the atmospheric mass
      \item Clouds and weather occur here
    \end{itemize}
    \vskip4pt
    \item \textbf{Stratosphere:} optically thin $\rightarrow$ radiative equilibrium
    \begin{itemize}
      \item Temperature set by the balance of absorbed vs.\ emitted radiation
      \item Stable against convection
      \item On Earth, warmed by ozone UV absorption
    \end{itemize}
  \end{itemize}
  \vskip6pt
  The boundary (the tropopause) occurs where the radiative gradient becomes less steep than the adiabat.
\end{frame}

% ── Manabe-Strickler ─────────────────────────────────────
\begin{frame}{Manabe and Strickler (1964)}
  The first realistic 1D radiative-convective model of Earth's atmosphere.
  \vskip6pt
  \begin{itemize}
    \item Solved for the vertical temperature profile with absorbing gases (H$_2$O, CO$_2$, O$_3$)
    \item Reproduced the surface temperature and the tropopause self-consistently
    \item Extended in 1967 to a fixed distribution of relative humidity, giving the first estimate of the warming from doubled CO$_2$
    \item Foundation of modern climate modelling
    \item Manabe shared the 2021 Nobel Prize in Physics for the physical modelling of Earth's climate
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Manabe \& Strickler (1964); Manabe \& Wetherald (1967), \textit{J.\ Atmos.\ Sci.}}
\end{frame}

% ── Energy balance derivation ────────────────────────────────
\begin{frame}{Planetary energy balance}
  Absorbed stellar power = emitted thermal radiation:
  \vskip6pt
  $$(1-A)\frac{L_\star}{4\pi d^2}\,\pi R_p^2 = 4\pi R_p^2\,\sigma\,T_{\mathrm{eff}}^4$$
  \vskip4pt
  \begin{itemize}
    \item $A$ = Bond albedo (fraction of light reflected)
    \item $\pi R_p^2$ = cross-sectional area (intercepts light)
    \item $4\pi R_p^2$ = total surface area (emits isotropically)
  \end{itemize}
  \vskip6pt
  Solving:
  \vskip4pt
  \keyresult{$$\boxed{T_{\mathrm{eff}} = \left[\frac{(1-A)\,L_\star}{16\pi\sigma\,d^2}\right]^{1/4}}$$}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{trenberth_energy_budget}}\\
  {\footnotesize Earth's globally averaged energy budget in W\,m$^{-2}$. After Trenberth et al.\ (2009).}
\end{frame}

    \begin{itemize}\setlength\itemsep{1pt}
      \item Of 340 W\,m$^{-2}$ of sunlight, 100 is reflected ($A = 0.30$), 80 is absorbed in the atmosphere, 160 at the surface.
      \item The surface emits 396 W\,m$^{-2}$ in the infrared; the atmosphere returns 333 W\,m$^{-2}$ as back-radiation.

    \end{itemize}
    \vskip0pt\relax
    \keyresult{Back-radiation, not direct sunlight, is the largest flux the surface receives: 333 W\,m$^{-2}$ of recycled solar energy versus 160 direct.}
  \end{columns}
\end{frame}

  $\Delta T = T_{\mathrm{surf}} - T_{\mathrm{eff}}$ reveals the \textbf{greenhouse effect}.\\
  Venus: 510~K. Earth: 33~K. Mars: $\sim$5~K (only 6~mbar, no water-vapour amplifier).
\end{frame}

% ── Greenhouse mechanism ─────────────────────────────────────
\begin{frame}{The greenhouse mechanism}
  \begin{columns}[c]
    \column{0.42\textwidth}
    \centering
    \includegraphics[width=\textwidth]{greenhouse_one_layer}
    \source{Course figure}

    \column{0.55\textwidth}
    \begin{enumerate}
      \item Sunlight (visible, $\sim$0.5~$\mu$m) reaches the surface
      \item Surface emits thermal IR ($\sim$10--15~$\mu$m)
      \item Greenhouse gases absorb outgoing IR
      \item Absorbing layer re-emits: half up, half down
      \item Downward emission adds energy to surface
    \end{enumerate}
    \vskip6pt
    \textbf{Result:} $T_{\mathrm{surf}} > T_{\mathrm{eff}}$
  \end{columns}
\end{frame}

% ── One-layer model setup ────────────────────────────────────
\begin{frame}{One-layer greenhouse model: setup}
  Single isothermal layer, emissivity $\varepsilon$ in IR, transparent at visible:
  \vskip8pt
  \textbf{Atmospheric balance:}
  $$\varepsilon\,\sigma\,T_s^4 = 2\,\varepsilon\,\sigma\,T_a^4 \quad\Rightarrow\quad T_a^4 = \frac{T_s^4}{2}$$
  \vskip4pt
  \textbf{Surface balance:}
  $$(1-A)\frac{F_\star}{4} + \varepsilon\,\sigma\,T_a^4 = \sigma\,T_s^4$$
  \vskip4pt
  \textbf{Top-of-atmosphere:}
  $$(1-A)\frac{F_\star}{4} = \sigma\,T_{\mathrm{eff}}^4$$
\end{frame}

% ── One-layer model result ───────────────────────────────────
\begin{frame}{One-layer greenhouse model: result}
  Combining the three balance equations:
  \vskip6pt
  \keyresult{$$\boxed{T_s = T_{\mathrm{eff}}\left(\frac{2}{2-\varepsilon}\right)^{1/4}}$$}
  \vskip8pt
  \centering
  \begin{tabular}{l l}
    $\varepsilon = 0$ (no greenhouse):   & $T_s = T_{\mathrm{eff}}$ \\[4pt]
    $\varepsilon = 1$ (perfect absorber): & $T_s = 2^{1/4}\,T_{\mathrm{eff}} \approx 1.19\,T_{\mathrm{eff}}$ \\
  \end{tabular}
  \vskip8pt
  \flushleft
  For Earth: $T_s \approx 1.19 \times 255 \approx 303$~K. Reasonable first estimate.
  \vskip4pt
  \textbf{Looking further ahead:} What happens when $T_s$ rises so high that outgoing radiation saturates? $\rightarrow$ \textit{Runaway greenhouse} (Lecture~9).
\end{frame}

\section{Atmospheric escape}

% ── Jeans escape ─────────────────────────────────────────────
\begin{frame}{Thermal (Jeans) escape}
  Molecules have a Maxwell--Boltzmann velocity distribution. The high-velocity tail extends beyond escape speed.
  \vskip6pt
  The \textbf{Jeans escape parameter} at the exobase (the level where collisions become rare; defined later in this section):
  \vskip4pt
  \keyresult{$$\lambda_J = \frac{G M_p\,m}{\kB T_{\mathrm{exo}}\,r_{\mathrm{exo}}} = \frac{v_{\mathrm{esc}}^2}{v_{\mathrm{th}}^2}$$}
  \vskip6pt
  \begin{itemize}
    \item $\lambda_J \gg 1$: escape negligible (tail too depleted)
    \item $\lambda_J \lesssim 2$--3: rapid atmospheric erosion
    \item Escape flux: $\Phi_J \propto (1 + \lambda_J)\,e^{-\lambda_J}$, exponentially sensitive
  \end{itemize}
\end{frame}

\end{frame}

% ── Hydrodynamic escape ─────────────────────────────────────
\begin{frame}{Hydrodynamic escape}
  When stellar EUV heating is intense, escape transitions from molecule-by-molecule to a bulk \textbf{atmospheric wind}:
  \vskip8pt
  \begin{itemize}
    \item Most important in the first few hundred Myr (young stars: 10--100$\times$ more EUV)
    \item Can strip $\mathrm{H_2}$-rich primary atmospheres from planets up to several $\Mearth$
    \item Outflowing H can \textbf{drag along heavier species} (He, C, N, O)
    \item Leading explanation for the exoplanet \textbf{radius valley}\\
    (deficit at $\sim$1.5--2~$\Rearth$; Lecture~13)
    \item Seen directly: HD~209458~b (H), GJ~436~b (H), WASP-107~b (He)
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Vidal-Madjar et al.\ (2003); Ehrenreich et al.\ (2015); Spake et al.\ (2018)}
\end{frame}

% ── Evaporation of primary atmospheres ──────────────────
\begin{frame}{Stripping primary atmospheres}
  Young stars are highly active in EUV and X-ray; primordial H/He envelopes can evaporate.
  \vskip6pt
  \begin{itemize}
    \item EUV flux is $\sim$100$\times$ higher in the first few hundred Myr than today
    \item Mass loss rate for hydrodynamic regime: $\dot{M} \propto F_\mathrm{EUV}\,R_p^3/(GM_p)$
    \item For sub-Neptunes ($M_p \lesssim 10\,\Mearth$), atmospheres can be fully stripped in a few hundred Myr
    \item For Neptune-mass planets and above, envelopes are retained
    \item Produces the ``evaporation valley'' at $\sim$1.8~$\Rearth$ in Kepler data
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Owen \& Wu (2013); Lopez \& Fortney (2013); Fulton et al.\ (2017)}
\end{frame}

% ── Energy-limited escape ────────────────────────────────
\begin{frame}{Energy-limited escape rate}
  A convenient estimate of mass loss:
  \vskip6pt
  \keyresult{$$\dot{M}_{\mathrm{EL}} = \eta\,\frac{\pi\,F_\mathrm{EUV}\,R_p^3}{G\,M_p}$$}
  \vskip6pt
  \begin{itemize}
    \item $F_\mathrm{EUV}$: incident EUV flux
    \item $\eta$: heating efficiency, typically 0.1--0.2
    \item Scales as $R_p^3$: large, fluffy atmospheres lose mass fastest
    \item Scales inversely with $M_p$: low gravity eases escape
    \item Valid when recombination is negligible (``energy-limited regime'')
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Watson et al.\ (1981); Owen (2019)}
\end{frame}

% ── Non-thermal escape ───────────────────────────────────────
\begin{frame}{Non-thermal escape mechanisms}
  \begin{description}
    \item[Sputtering:] Solar wind ions impact upper atmosphere, eject molecules.\\
    Significant for Mars (no global magnetic field).

    \vskip4pt
    \item[Photochemical:] UV dissociates $\mathrm{H_2O} \rightarrow$ H + OH; fast H escapes.\\
    Key for Venus and Mars water loss.

    \vskip4pt
    \item[Ion pickup:] Atmospheric atoms ionised by UV, swept away by solar wind $B$-field.\\
    Effective at unmagnetised planets.

    \vskip4pt
    \item[Impact erosion:] Large impacts can eject substantial atmospheric mass in a single event.
  \end{description}
  \vskip6pt
  {\small MAVEN at Mars: the $\mathrm{O^+}$ pickup channel alone carries $\sim$100~g\,s$^{-1}$; photochemical escape dominates the oxygen budget today.}
\end{frame}

    \begin{itemize}\setlength\itemsep{1pt}
      \item Each bar is one escape channel for oxygen: photochemical escape dominates today; ion escape and sputtering are sub-dominant.
      \item Under the young, EUV-active Sun, sputtering was likely comparable or larger: the channels shift over time.
      \item Summed with hydrogen escape, the total reaches the 2--3 kg\,s$^{-1}$ MAVEN measures.
    \end{itemize}
    \vskip2pt\relax
    \keyresult{Mars leaks 2 to 3 kilograms of atmosphere every second, and the books balance channel by channel.}
  \end{columns}
\end{frame}

\section{Recent advances \& outlook}
% ════════════════════════════════════════════════════════════

% ── Recent advances ──────────────────────────────────────────
\begin{frame}{Recent advances}
  \begin{itemize}
    \item \textbf{JWST / TRAPPIST-1 b:} Thermal emission consistent with bare rock, no significant atmosphere. Inner rocky planets around active M dwarfs may be stripped.

    \vskip6pt
    \item \textbf{JWST / TRAPPIST-1 c:} Similar result, reinforcing theoretical predictions of enhanced atmospheric escape around low-mass stars.

    \vskip6pt
    \item \textbf{MAVEN at Mars:} Quantified loss rates for multiple species. Ion escape (solar wind stripping) dominates over Jeans escape. Integrated losses can account for a substantial fraction of Mars's early atmosphere.
  \end{itemize}
  \vskip6pt
  \textbf{Central theme:} a planet's atmospheric inventory is set by the balance between escape and resupply. Outgassing draws on interior heat and wanes as the interior cools (Lecture 3), so retention depends on the interior thermal evolution, not only on the present-day escape rate.
\end{frame}

% ── Impact erosion ───────────────────────────────────────
\begin{frame}{Impact erosion of atmospheres}
  Impacts erode atmospheres in two regimes (Lecture~4).
  \vskip6pt
  \begin{itemize}
    \item \textbf{Local:} an ordinary impactor expels at most the air above the tangent plane at the impact point, a fraction $\sim H/2R$ of the whole atmosphere
    \item Erosion is therefore cumulative; km-scale planetesimals are the most efficient eroders per unit mass
    \item \textbf{Global:} a giant impact shakes the whole planet, but even the Moon-forming impact removes only $\sim$20\%
    \item Key process while the bombardment lasts, the first $\sim$100~Myr; EUV-driven escape runs on a slower clock and continues for a few hundred Myr
    \item Signature: noble gas isotope ratios record impact-driven loss
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Schlichting \& Mukhopadhyay (2018); Genda \& Abe (2003, 2005)}
\end{frame}

% ── Volcanic replenishment ───────────────────────────────
\begin{frame}{Outgassing vs.\ escape: a dynamic balance}
  An atmosphere is not a static inheritance; it is a reservoir in flow.
  \vskip6pt
  \begin{itemize}\small\setlength\itemsep{1pt}
    \item \textbf{Supply:} volcanic outgassing, impact delivery, biological fluxes
    \item \textbf{Loss:} thermal escape, non-thermal escape, chemical weathering, sequestration in oceans/crust
    \item The \textbf{steady state} depends on the balance of these fluxes
    \item Earth's active volcanism continually replaces CO$_2$ on the carbonate-silicate timescale (weathering-outgassing CO$_2$ cycle; $\sim$0.5~Myr)
    \item Mars lost its active volcanism and dynamo, so escape was not replenished
    \item Venus has active (perhaps episodic) volcanism, but no oceans to absorb CO$_2$
  \end{itemize}
  \vskip4pt
  \keyresult{Atmospheric longevity is as much about \textbf{replacement} as about initial inventory.}
\end{frame}

% ── MAVEN quantitative ───────────────────────────────────
\begin{frame}{MAVEN at Mars: quantifying escape}
  NASA's MAVEN (2014--) has measured Mars' atmospheric loss in situ.
  \vskip6pt
  \begin{itemize}
    \item Present-day total escape rate: $\sim$2--3~kg/s (H and O combined)
    \item Dominant loss channel: photochemical escape of oxygen
    \item Enhanced during solar storms ($\sim$10$\times$ increase)
    \item Integrated loss over 4~Gyr: consistent with 0.5--0.8~bar of CO$_2$ lost
    \item Combined with hydrogen escape: up to $\sim$23~m global equivalent layer of water lost
    \item Young-Sun EUV may also have driven thermal escape of carbon, which blocks a dense early CO$_2$ atmosphere
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Jakosky et al.\ (2018), \textit{Icarus}; Lillis et al.\ (2017); Tian et al.\ (2009)}
\end{frame}

% ── JWST TRAPPIST results ───────────────────────────────
\begin{frame}{JWST and the fate of rocky atmospheres}
  JWST has begun to directly test whether rocky planets around M dwarfs retain atmospheres.
  \vskip6pt
  \begin{itemize}
    \item TRAPPIST-1 b: MIRI thermal emission consistent with a bare rock (dayside 503~K)
    \item TRAPPIST-1 c: MIRI shows low heat redistribution $\Rightarrow$ thin or no atmosphere
    \item LHS 3844 b: similar result using Spitzer data
    \item Implications: inner rocky planets around active M dwarfs lose their atmospheres
    \item Open question: whether \textbf{outer} TRAPPIST-1 planets retain atmospheres
    \item JWST will target TRAPPIST-1 e, f during Cycle 3--4
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Greene et al.\ (2023); Zieba et al.\ (2023), \textit{Nature}}
\end{frame}

% ── Transmission spectroscopy preview ───────────────────
\begin{frame}{Transmission spectroscopy: preview}
  During a transit, starlight passes through the planet's \textbf{limb}.
  \vskip6pt
  \begin{itemize}
    \item Absorbing molecules imprint dark features in the transmission spectrum
    \item Path length $\sim$$2\sqrt{2 R_p H}$ through a layer of scale height $H$
    \item Signal: $\Delta(R_p/R_\star)^2 \sim 2 R_p H/R_\star^2$
    \item Bigger signal for puffy atmospheres (large $H$) around small stars
    \item JWST has detected H$_2$O, CO$_2$, CH$_4$, SO$_2$ in multiple exoplanet atmospheres
    \item Full treatment in Lecture 13 (Exoplanet Atmospheres)
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Seager \& Sasselov (2000); Kreidberg et al.\ (2014)}
\end{frame}

% ── Runaway greenhouse preview ──────────────────────────
\begin{frame}{Runaway greenhouse: a look ahead}
  What happens when a water-rich atmosphere gets too hot?
  \vskip6pt
  \begin{itemize}
    \item More water vapour $\rightarrow$ stronger greenhouse $\rightarrow$ hotter surface
    \item Positive feedback: hotter surface $\rightarrow$ more evaporation $\rightarrow$ more greenhouse
    \item A water-rich atmosphere cannot radiate more than $\sim$280--310~W/m$^2$: the \textbf{Simpson-Nakajima limit}
    \item If the absorbed flux exceeds this limit, the oceans evaporate entirely and the runaway becomes unstoppable
    \item Hydrogen then escapes hydrodynamically, leaving a dry CO$_2$ atmosphere
    \item Venus' modern state is the end point of this process
  \end{itemize}
  \vskip6pt
  Derived in full in \textbf{Lecture 9: Earth \& Venus}.
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Ingersoll (1969); Nakajima et al.\ (1992)}
\end{frame}

% ── Summary ──────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}
    \item Three atmosphere types: \textbf{primary} (captured H$_2$/He), \textbf{secondary} (outgassed), \textbf{tertiary} (biologically modified)
    \item Hydrostatic equilibrium: $\dv{P}{z} = -\rho\,g$; leads to exponential pressure decay
    \item Scale height: $H = \kB T/(\mu\,m_u\,g)$; encodes thermal energy vs.\ gravity
    \item Vertical structure: troposphere, stratosphere, mesosphere, thermosphere
    \item Beer--Lambert law: $I = I_0\,e^{-\tau}$; $\tau \approx 1$ defines the emission level
    \item Energy balance: $T_{\mathrm{eff}} \propto [(1-A)L_\star/d^2]^{1/4}$; greenhouse raises $T_s$ above $T_{\mathrm{eff}}$
    \item Escape: Jeans ($\lambda_J$), hydrodynamic, non-thermal; retention requires $v_{\mathrm{esc}} \gtrsim 6\,v_{\mathrm{th}}$
  \end{enumerate}
\end{frame}
