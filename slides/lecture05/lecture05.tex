\section{Atmospheric composition}
% ════════════════════════════════════════════════════════════

% ── Primary atmospheres ──────────────────────────────────────
\begin{frame}{Primary atmospheres}
  \textbf{Captured directly from the protoplanetary disk.}
  \vskip8pt
  \begin{itemize}
    \item Dominated by $\mathrm{H_2}$ and He (solar composition)
    \item \textbf{Gas giants} keep massive $\mathrm{H_2}$/He envelopes; \textbf{ice giants} only 10--20\% of their mass
    \item Terrestrial planets are too small to retain primordial $\mathrm{H_2}$/He
  \end{itemize}
\end{frame}

\begin{frame}
  \ipsherokeyside{helled2014_core_accretion}{Growth of a giant planet by core accretion: runaway gas accretion builds the $\mathrm{H_2}$/He envelope once it outweighs the core. Reproduced from Helled et al.\ (2014), Fig.\ 1; simulation from Lissauer et al.\ (2009).}{A primary atmosphere needs a core of $\gtrsim 5$--$10\;\Mearth$ before the disk disperses in $\sim$3--10~Myr.}
\end{frame}

% ── Secondary atmospheres ────────────────────────────────────
\begin{frame}{Secondary atmospheres}
  \textbf{Produced by outgassing from the interior.}
  \vskip8pt
  \begin{itemize}
    \item Magma ocean degassing and volcanism release dissolved volatiles
    \item Speciation follows the mantle \textbf{oxygen fugacity}: oxidising gives $\mathrm{CO_2}$, $\mathrm{H_2O}$, $\mathrm{N_2}$; reducing gives $\mathrm{H_2}$, CO, $\mathrm{N_2}$
    \item \textbf{Venus:} 96.5\% $\mathrm{CO_2}$ at 92~bar; \textbf{Mars:} 95\% $\mathrm{CO_2}$ at 6~mbar; \textbf{Titan:} $\mathrm{N_2}$ (from $\mathrm{NH_3}$) at 1.5~bar
  \end{itemize}
\end{frame}

\end{frame}

\section{Hydrostatic equilibrium}
% ════════════════════════════════════════════════════════════

% ── Hydrostatic equilibrium ──────────────────────────────────
\begin{frame}{Hydrostatic equilibrium}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{hydrostatic_slab}
    \source{Course figure}

    \column{\recaptextcol}
    \small
    The net vertical force on a thin slab of area $A$ and thickness $\dd z$ vanishes:
    $$P(z)A - P(z + \dd z)A = \rho\,g\,A\,\dd z$$
    \vskip4pt
    \keyresult{$$\boxed{\dv{P}{z} = -\rho\,g}$$}
    \vskip4pt
    Each layer supports the weight of all gas above it.
  \end{columns}
\end{frame}

% ── Ideal gas law & barometric formula ───────────────────────
\begin{frame}{The ideal gas law and barometric formula}
  The ideal gas law $P = \rho\,\kB T / (\mu\,m_u)$ in $\dv{P}{z} = -\rho\,g$ gives
  $$\dv{P}{z} = -\frac{\mu\,m_u\,g}{\kB T}\,P = -\frac{P}{H}$$
  \vskip4pt
  For an \textbf{isothermal} atmosphere (constant $T$):
  \vskip4pt
  \keyresult{$$\boxed{P(z) = P_0\,\exp\!\left(-\frac{z}{H}\right)}$$}
  \vskip4pt
  with $H = \kB T/(\mu\,m_u\,g)$; the pressure drops by a factor of $e \approx 2.718$ each scale height.
\end{frame}

\begin{frame}
  \ipshero{ideal_gas_density}{Ideal gas law in atmospheric form, $\rho = P\mu m_u/(k_B T)$: density scales with pressure at fixed $T$, heavier or colder gas is denser. Course figure.}
\end{frame}

\begin{frame}
  \ipshero{barometric_profile}{The barometric formula $P/P_0 = e^{-z/H}$: a factor $e$ per scale height, 99\% of the column mass below $4.6\,H$, a straight line on a logarithmic axis. Course figure.}
\end{frame}

\section{Blackboard derivation: scale height}
% ════════════════════════════════════════════════════════════

% ── Goal and strategy ───────────────────────────────────────
\begin{frame}{Goal and strategy}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig[0.62\textheight]{board_sketch_l05}
    \column{\recaptextcolwide}
    \textbf{Goal:} derive $H = \kB T/(\mu\,m_u\,g)$ from hydrostatic equilibrium and the ideal gas law.
    \vskip8pt
    \begin{itemize}\setlength\itemsep{2pt}
      \item Balance the forces on a thin slab
      \item Replace $\rho$ by $P$ with the ideal gas law
      \item Read off the length scale, integrate, compute
    \end{itemize}
    \vskip8pt
    \keyresult{One length, $H$, sets how fast pressure falls with height.}
  \end{columns}
\end{frame}

% ── Setup ────────────────────────────────────────────────────
\begin{frame}{Deriving the scale height}
  Start from:
  $$\dv{P}{z} = -\rho\,g \qquad\text{and}\qquad P = \frac{\rho\,\kB T}{\mu\,m_u}$$
  \vskip6pt
  Express $\rho$ in terms of $P$:
  $$\rho = \frac{P\,\mu\,m_u}{\kB T}$$
  \vskip4pt
  Substitute:
  $$\dv{P}{z} = -\frac{\mu\,m_u\,g}{\kB T}\,P \equiv -\frac{P}{H}$$
\end{frame}

% ── Result ───────────────────────────────────────────────────
\begin{frame}{The atmospheric scale height}
  \keyresult{$$\boxed{H = \frac{\kB T}{\mu\,m_u\,g}}$$}
  \vskip8pt
  \begin{itemize}
    \item \textbf{Higher $T$} $\rightarrow$ larger $H$: hotter gas extends further
    \item \textbf{Heavier molecules} (larger $\mu$) $\rightarrow$ smaller $H$: harder to loft
    \item \textbf{Stronger gravity} $g$ $\rightarrow$ smaller $H$: more compressed
  \end{itemize}
  \vskip8pt
  $H$ balances thermal energy against gravitational binding; integrated over the column, the surface pressure is the column weight, $P_0 = m_{\mathrm{col}}\,g$.
\end{frame}

% ── Scale heights across the solar system ────────────────────
\begin{frame}{Scale heights across the solar system}
  \centering
  \begin{tabular}{l c c c c}
    \toprule

    \midrule

    \bottomrule
  \end{tabular}
  \vskip10pt
  \begin{itemize}
    \item Jupiter: large $H$ despite strong gravity, because $\mathrm{H_2}$ is very light ($\mu = 2.2$)
    \item Titan: large $H$ despite low $T$, thanks to very weak gravity ($g = 1.35$~m\,s$^{-2}$)
    \item Venus: hot, but heavy molecules and decent gravity yield a moderate $H$
  \end{itemize}
\end{frame}

% ── Lapse rate ───────────────────────────────────────────────
\begin{frame}{The adiabatic lapse rate}
  In the troposphere, convection sets the temperature gradient:
  \vskip6pt
  \keyresult{$$\Gamma_d = -\dv{T}{z} = \frac{g}{c_p}$$}
  \vskip6pt
  \begin{itemize}
    \item \textbf{Dry adiabatic lapse rate} for Earth: $\Gamma_d = 9.81/1004 \approx 9.8$~K\,km$^{-1}$; observed $\sim$6.5~K\,km$^{-1}$ because of latent heat
    \item Rising air expands, does work on its surroundings, and cools
    \item \textbf{Tropopause height:} $\sim$8~km (poles) to $\sim$17~km (equator)
  \end{itemize}
\end{frame}

\section{Radiative transfer basics}
% ════════════════════════════════════════════════════════════

% ── Absorption, emission, scattering ─────────────────────────
\begin{frame}{Three interactions: radiation and matter}
  \begin{itemize}\setlength\itemsep{6pt}
    \item \textbf{Absorption:} photon energy becomes internal molecular energy; key absorbers $\mathrm{CO_2}$, $\mathrm{H_2O}$, $\mathrm{CH_4}$, $\mathrm{O_3}$, $\mathrm{N_2O}$
    \item \textbf{Emission:} warm gas radiates at the absorbed wavelengths (Kirchhoff's law); only emission from $\tau \lesssim 1$ reaches space
    \item \textbf{Scattering:} photons redirected without absorption; Rayleigh ($\propto \lambda^{-4}$) and Mie scattering
  \end{itemize}
\end{frame}

\begin{frame}
  \ipshero{rayleigh_scattering}{Rayleigh scattering and the colour of the sky: short paths at noon keep the blue scattered light, long slant paths at sunset transmit only red; the cross-section scales as $\lambda^{-4}$. Course figure.}
\end{frame}

% ── Blackbody spectrum figure ────────────────────────────────
\begin{frame}
  \ipsherokey[0.65]{blackbody_spectrum}{Blackbody spectra of the Sun ($\sim$5800~K, peak $\sim$0.5~$\mu$m) and a planet ($\sim$300~K, peak $\sim$10~$\mu$m). Credit: Wikimedia, public domain.}{This \textbf{wavelength separation} is the basis of the greenhouse effect: an atmosphere can be transparent in the visible but opaque in the infrared.}
\end{frame}

% ── Optical depth & Beer-Lambert law ─────────────────────────
\begin{frame}{Optical depth and the Beer-Lambert law}
  Optical depth $\tau$ measures the cumulative absorption along a path:
  $$\tau = \int_0^s \kappa\,\rho\,\dd s'$$
  \vskip2pt
  Beam intensity decays following the \textbf{Beer-Lambert law}:
  \vskip2pt
  \keyresult{$$\boxed{I = I_0\,e^{-\tau}}$$}
  \vskip4pt
  \begin{itemize}
    \item \textbf{Optically thin} ($\tau \ll 1$): radiation passes freely through the gas
    \item \textbf{Optically thick} ($\tau \gg 1$): photons are absorbed and re-emitted many times
    \item \textbf{Photosphere} ($\tau \approx 1$): thermal radiation escapes directly to space
  \end{itemize}
\end{frame}

\begin{frame}
  \ipshero{tau_one_schematic}{The atmospheric photosphere: thermal radiation escapes to space from the $\tau \approx 1$ level, which sets the effective temperature. Course figure.}
\end{frame}

\begin{frame}
  \ipshero[0.76]{atmospheric_transmission}{Earth's atmospheric absorption spectrum from UV to IR: greenhouse gases absorb the outgoing infrared while visible sunlight and the 8--13~$\mu$m window pass. Credit: Wikimedia Commons, public domain.}
\end{frame}

\section{Energy balance \& greenhouse effect}
% ════════════════════════════════════════════════════════════

% ── Radiative-convective equilibrium ─────────────────────────
\begin{frame}
  \ipsherokey[0.61]{radiative_convective_profile}{Radiative-convective equilibrium (schematic): the radiative profile is superadiabatic near the surface, convection holds the troposphere on the dry adiabat. Course figure.}{Greenhouse opacity makes the \textbf{troposphere} convective, the optically thin \textbf{stratosphere} sits in radiative equilibrium, and the \textbf{tropopause} is where the radiative gradient becomes shallower than the adiabat.}
\end{frame}

% ── Energy balance derivation ────────────────────────────────
\begin{frame}{Planetary energy balance}
  Absorbed stellar power equals emitted thermal radiation:
  $$(1-A)\frac{S}{4} = \sigma\,T_{\mathrm{eff}}^4 \qquad\text{where}\quad S = \frac{L_\star}{4\pi d^2}$$
  \vskip4pt
  \begin{itemize}
    \item $A$: Bond albedo; $S$: stellar flux ($S_0 = 1361$~W\,m$^{-2}$ at Earth)
    \item Factor $1/4$: intercepting cross-section $\pi R_p^2$ over the emitting sphere $4\pi R_p^2$
  \end{itemize}
  \vskip6pt
  \keyresult{$$\boxed{T_{\mathrm{eff}} = \left[\frac{(1-A)\,L_\star}{16\pi\sigma\,d^2}\right]^{1/4} = \left[\frac{(1-A)\,S}{4\sigma}\right]^{1/4}}$$}
\end{frame}

\begin{frame}
  \ipshero{trenberth_energy_budget}{Earth's globally averaged energy budget in W\,m$^{-2}$: back-radiation of 333~W\,m$^{-2}$ supplies more energy to the surface than direct sunlight (160~W\,m$^{-2}$). After Trenberth et al.\ (2009).}
\end{frame}

% ── Effective vs surface T ───────────────────────────────────
\begin{frame}{Effective vs.\ surface temperature}
  \centering
  \begin{tabular}{l c c c c c}
    \toprule

    Venus   & 0.77 & 0.72 & 227 & 737  & +510 \\
    Earth   & 0.30 & 1.00 & 255 & 288  & +33  \\
    Mars    & 0.25 & 1.52 & 210 & 215  & +5 \\
    Jupiter & 0.34 & 5.20 & 110 & 165* & +55  \\
    \bottomrule
  \end{tabular}
  \vskip4pt
  {\footnotesize *Jupiter 1-bar level; excess partly from internal heat (Lecture~3).}
  \vskip8pt
  \keyresult{$\Delta T = T_{\mathrm{surf}} - T_{\mathrm{eff}}$ is the \textbf{greenhouse effect}: 510~K on Venus, 33~K on Earth, $\sim$5~K on Mars (6~mbar, no water-vapour amplifier).}
\end{frame}

\begin{frame}
  \ipshero[0.77]{venus_earth_mars_atmospheres}{Surface temperature and pressure of Venus, Earth and Mars: similar bulk compositions span a factor $10^4$ in pressure and 500~K in temperature through greenhouse warming. Course figure; data from NASA.}

% ── Greenhouse mechanism ─────────────────────────────────────
\begin{frame}
  \ipsherokeyside{greenhouse_one_layer}{The one-layer greenhouse model: sunlight in, infrared absorbed and re-emitted half up, half down. Course figure.}{Sunlight ($\sim$0.5~$\mu$m) reaches the surface, the surface emits thermal IR ($\sim$10--15~$\mu$m), greenhouse gases absorb it and re-emit half downward, and that extra energy makes $T_{\mathrm{surf}} > T_{\mathrm{eff}}$.}
\end{frame}

% ── One-layer greenhouse model ───────────────────────────────
\begin{frame}{One-layer greenhouse model}
  A single layer with infrared emissivity $\varepsilon$, transparent to visible sunlight:
  \vskip4pt
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Flux balance:}
    \begin{align*}
      \text{Atmosphere:} & \quad \varepsilon\sigma T_s^4 = 2\varepsilon\sigma T_a^4 \\
      \text{Surface:} & \quad (1-A)\frac{F_\star}{4} + \varepsilon\sigma T_a^4 = \sigma T_s^4 \\
      \text{TOA:} & \quad (1-A)\frac{F_\star}{4} = \sigma T_{\mathrm{eff}}^4
    \end{align*}

    \column{0.48\textwidth}
    \textbf{Surface temperature:}
    \keyresult{$$\boxed{T_s = T_{\mathrm{eff}}\left(\frac{2}{2-\varepsilon}\right)^{1/4}}$$}
  \end{columns}
  \vskip6pt
  \begin{itemize}
    \item $\varepsilon = 0$ (transparent): $T_s = T_{\mathrm{eff}}$
    \item $\varepsilon = 1$ (opaque): $T_s = 2^{1/4}\,T_{\mathrm{eff}} \approx 1.19\,T_{\mathrm{eff}}$ (Earth: $\approx$303~K)
  \end{itemize}
\end{frame}

\section{Atmospheric escape}
% ════════════════════════════════════════════════════════════

% ── Jeans escape ─────────────────────────────────────────────
\begin{frame}{Thermal (Jeans) escape}
  The high-velocity tail of the Maxwell-Boltzmann distribution can exceed the escape speed. The \textbf{Jeans escape parameter} at the exobase compares gravitational to thermal energy:
  \vskip4pt
  \keyresult{$$\lambda_J = \frac{G M_p\,m}{\kB T_{\mathrm{exo}}\,r_{\mathrm{exo}}} = \frac{v_{\mathrm{esc}}^2}{v_{\mathrm{th}}^2}, \qquad v_{\mathrm{th}} = \sqrt{2\kB T_{\mathrm{exo}}/m}$$}
  \vskip6pt
  \begin{itemize}
    \item $\lambda_J \gg 1$: escape is negligible because very few molecules exceed $v_{\mathrm{esc}}$
    \item $\lambda_J \lesssim 2$--3: thermal escape drives rapid atmospheric loss
    \item Escape flux $\Phi_J \propto (1 + \lambda_J)\,e^{-\lambda_J}$: exponential in $\lambda_J$
  \end{itemize}
\end{frame}

\begin{frame}
  \ipshero{escape_processes_gronoff2020}{The main atmospheric escape processes: thermal, photochemical and ion channels act on all planets, and the magnetic field selects which dominate. Gronoff et al.\ (2020).}
\end{frame}

\begin{frame}{Jeans parameter for Earth and Mars}
  \centering
  {\footnotesize Exobase temperatures: 1000~K (Earth), 270~K (Mars).}

\end{frame}

\begin{frame}
  \ipshero{exobase_definition}{The exobase: the altitude where the mean free path $\ell$ equals the scale height $H$, collisional fluid below and ballistic flight above. Course figure.}
\end{frame}

\begin{frame}

% ── Hydrodynamic escape ─────────────────────────────────────
\begin{frame}{Hydrodynamic escape}
  Intense stellar EUV heating drives bulk outflow as an \textbf{atmospheric wind}:
  \vskip8pt
  \begin{itemize}
    \item Strongest in the first few hundred Myr, when stellar EUV is 10--100$\times$ higher
    \item Strips $\mathrm{H_2}$-rich primary envelopes from planets up to several $\Mearth$; outflowing H drags heavier volatiles along
    \item The primary mechanism behind the exoplanet radius valley at $\sim$1.5--2~$\Rearth$
  \end{itemize}
  \vskip4pt
  \sourcenote{Hunten et al.\ (1987); Owen (2019)}
\end{frame}

\begin{frame}
  \ipshero{fulton2017_radius_valley}{The radius valley in the California-Kepler Survey: a deficit near 1.8~$\Rearth$ separates stripped rocky cores from envelope-retaining sub-Neptunes. After Fulton et al.\ (2017).}
\end{frame}

\begin{frame}
  \ipshero{radius_valley_owen2019}{Population synthesis after 10~Gyr of EUV-driven escape: an empty evaporation desert and a valley near 1.5 to 2~$\Rearth$. After Owen \& Wu (2013), Lopez \& Fortney (2013) and Owen (2019).}
\end{frame}

% ── Energy-limited escape ────────────────────────────────────
\begin{frame}{Energy-limited escape rate}
  Stellar EUV absorbed in the upper atmosphere drives hydrodynamic mass loss:
  \vskip6pt
  \keyresult{$$\dot{M}_{\mathrm{EL}} = \eta\,\frac{\pi\,F_\mathrm{EUV}\,R_p^3}{G\,M_p}$$}
  \vskip6pt
  \begin{itemize}
    \item Efficiency $\eta \approx 0.1$--$0.2$ converts incident EUV flux into expansion work
    \item Scales as $R_p^3$: extended, low-density atmospheres lose mass fastest
    \item Scales inversely with $M_p$: low gravity enables stronger bulk outflow
  \end{itemize}
  \vskip4pt
  \sourcenote{Watson et al.\ (1981); Owen (2019)}
\end{frame}

\begin{frame}
  \ipshero[0.76]{hot_jupiter_outflow_tripathi2015}{Hydrodynamic outflow from a hot Jupiter under stellar EUV heating: the gas accelerates outward into a night-side wake with Kelvin-Helmholtz instabilities. Reproduced from Owen (2019), Fig.\ 2; simulation from Tripathi et al.\ (2015).}
\end{frame}

% ── Non-thermal escape ───────────────────────────────────────
\begin{frame}{Non-thermal escape mechanisms}
  \begin{itemize}\setlength\itemsep{6pt}
    \item \textbf{Sputtering} and \textbf{ion pickup:} solar wind ions eject neutrals, and photoions are swept away by the interplanetary field
    \item \textbf{Photochemical:} photodissociation yields suprathermal atoms above $v_{\mathrm{esc}}$
    \item \textbf{Impact erosion:} giant impacts shock and eject atmospheric columns
  \end{itemize}
  \vskip8pt
  \keyresult{MAVEN at Mars: photochemical loss of oxygen dominates modern escape over sputtering and ion pickup.}
\end{frame}

\end{frame}

% ── Summary ──────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}\setlength\itemsep{6pt}
    \item \textbf{Atmosphere types}: primary (captured), secondary (outgassed), tertiary (biogenic); four orders of magnitude in surface pressure
    \item \textbf{Hydrostatic balance} and the \textbf{scale height} $H = \kB T/(\mu m_u g)$ give $P = P_0 e^{-z/H}$; convection sets $\Gamma_d = g/c_p$ below the 0.1~bar tropopause
    \item \textbf{Greenhouse warming} from $\tau \gtrsim 1$ raises $T_s$ above $T_{\mathrm{eff}}$; \textbf{thermal escape} is negligible when $v_{\mathrm{esc}} \gtrsim 6 v_{\mathrm{th}}$; hydrodynamic and non-thermal loss follow other rules
  \end{enumerate}

  \textbf{Next: Lecture 6}
\end{frame}
