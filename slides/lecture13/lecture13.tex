\section{Historical Context}
% ════════════════════════════════════════════════════════════

\begin{frame}{From Zero to Six Thousand in Three Decades}
  \begin{itemize}
    \item \textbf{1992:} Wolszczan + Frail. Two planets around pulsar PSR B1257+12 from pulse-timing.
    \item \textbf{1995:} Mayor + Queloz. \textbf{51 Pegasi b} via radial velocity. 0.5 $\Mjup$, 4.23-day orbit.
      \begin{itemize}
        \item Contradicted every textbook picture of giant-planet formation.
        \item $\to$ migration models reshaped formation theory.
        \item Nobel Prize 2019.
      \end{itemize}
    \item \textbf{1999:} HD 209458 b. First transit of a known RV planet.
      \begin{itemize}
        \item Combined transit depth + RV $\to$ first bulk density of an exoplanet.
        \item Confirmed hot Jupiters are gas-dominated.
      \end{itemize}
    \item \textbf{2009--2018:} Kepler + K2 transformed the field statistically.
    \item \textbf{2022--present:} JWST atmospheric characterisation era.
  \end{itemize}
\end{frame}

\begin{frame}{HD 209458 b: First Ground-Based Transit (1999)}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{hd209458b_first_transit}
    \column{\recaptextcol}
    \begin{itemize}
      \item STARE photometer, two nights in Sep 1999.
      \item Successive transits dropped flux by $\sim 1.7\%$.
      \item Independent detection by Henry et al.\ (2000).
      \item After this point: exoplanets were physical objects, not just Doppler signals.
    \end{itemize}
  \end{columns}
\end{frame}

\sectionimage{rossiter_mclaughlin}{Doppler signal of a transiting planet (Rossiter-McLaughlin). Triaud (2018)}
\section{Detection Method 1: Radial Velocity}
% ════════════════════════════════════════════════════════════

\begin{frame}{Radial Velocity: Doppler Reflex of the Star}
  \begin{itemize}
    \item Star + planet orbit a common barycentre (centre of mass).
    \item Star's reflex orbit: $a_\star = (m_p / M_\star)\,a_p$.
    \item Spectral lines blueshift / redshift periodically.
    \item For circular, edge-on orbit (semi-amplitude $K_\star$, the peak reflex velocity):
  \end{itemize}
  \[
    K_\star = \left(\frac{2\pi G}{P}\right)^{1/3}\!\frac{m_p \sin i}{M_\star^{2/3}}\,\frac{1}{\sqrt{1 - e^2}}
  \]
  Quick scales:
  \begin{itemize}
    \item Jupiter at 5.2 AU: $K_\star \approx 12.5$ m s$^{-1}$.
    \item Earth at 1 AU: $K_\star \approx 9$ cm s$^{-1}$.
  \end{itemize}
\end{frame}

\begin{frame}{Instrument Floor and Stellar Noise}
  \begin{itemize}
    \item ELODIE (1995): $\sim$10 m s$^{-1}$.
    \item HARPS (2003): $\sim$1 m s$^{-1}$.
    \item ESPRESSO (2018): 10 cm s$^{-1}$ demonstrated.
    \item Now \textbf{stellar noise}-limited: granulation, p-modes, starspots produce 10 cm s$^{-1}$ to 1 m s$^{-1}$ jitter even on quiet stars.
    \item True Earth twin: at the very edge of physical accessibility.
  \end{itemize}
\end{frame}

\begin{frame}{The $m \sin i$ Degeneracy}
  RV gives only $K_\star$, hence $m_p \sin i$.
  \begin{itemize}
    \item Face-on: $\sin i = 0$, no signal at all.
    \item Edge-on: $\sin i = 1$, true mass.
    \item RV alone $\to$ \emph{minimum} mass only.
  \end{itemize}
  \vskip6pt
  Break the degeneracy with:
  \begin{itemize}
    \item \textbf{Transit} ($\sin i \approx 1$ to a few percent),
    \item \textbf{Astrometry} (gives inclination directly),
    \item \textbf{Direct imaging} (geometric).
  \end{itemize}
  \vskip4pt
  RV biases: massive planets, short orbits, bright slowly-rotating quiet G/K stars.
\end{frame}

\sectionimage{transit_geometry}{Transit geometry. Winn (2010)}
\section{Detection Method 2: Transits}
% ════════════════════════════════════════════════════════════

\begin{frame}{Transit Geometry}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{transit_geometry}
    \column{\recaptextcol}
    \begin{itemize}
      \item Orbital plane near our line of sight $\to$ planet crosses stellar disk.
      \item Depth: $\delta = (R_p / R_\star)^2$.
      \item Probability: $\Theta \approx R_\star / a$.
      \item Strong selection: short periods + large $R_p / R_\star$.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Quick Scales}
  \begin{itemize}
    \item Jupiter ($R_J$) around Sun ($R_\odot$): $\delta \approx 1.01\%$. Easy from the ground.
    \item Earth around Sun: $\delta \approx 84$ ppm. Requires space.
    \item Earth around M dwarf (small, cool star; $R_\star = 0.15\,R_\odot$): $\delta \approx 4000$ ppm. Ground-accessible!
  \end{itemize}
  \vskip6pt
  \keyresult{Inverse scaling with stellar radius is why TRAPPIST-1 dominates atmospheric characterisation.}
\end{frame}

\begin{frame}{Light Curve Anatomy}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{transit_lightcurve_schematic}
    \column{\recaptextcol}
    \begin{itemize}
      \item Four contact times: $t_\mathrm{I}$, $t_\mathrm{II}$, $t_\mathrm{III}$, $t_\mathrm{IV}$.
      \item Duration $\to$ impact parameter $b$ (transit offset from disk centre), inclination.
      \item Ingress shape $\to$ stellar limb darkening.
      \item Depth $\to$ planet radius (given $R_\star$).
    \end{itemize}
  \end{columns}
\end{frame}

% ════════════════════════════════════════════════════════════
\sectionimage{chenkipping_mass_radius}{Empirical mass-radius relation. Chen \& Kipping (2017)}
\section{Blackboard: Transit Depth + RV $\to$ Bulk Density}
% ════════════════════════════════════════════════════════════

\begin{frame}{Why Combine Transit and RV?}
  \begin{itemize}
    \item Transit alone: $R_p$, but no mass.
    \item RV alone: $m_p \sin i$, but no radius.
    \item Combined: \textbf{both} for the same planet $\to$ bulk density.
    \item Density is the central observable that turns a detection into a physical object.
  \end{itemize}
\end{frame}

\begin{frame}{Step 1: Transit Depth}
  Star = uniform disk $R_\star$. Planet = opaque disk $R_p$.
  \[
    \frac{\Delta F}{F} = \frac{\pi R_p^2}{\pi R_\star^2} = \left(\frac{R_p}{R_\star}\right)^2
  \]
  Given $R_\star$ from spectroscopy + parallax, the depth gives $R_p$ directly.
  \vskip4pt
  Limb darkening + ingress shape $\to$ $\sim 10\%$ corrections.
\end{frame}

\begin{frame}{Step 2: RV Semi-Amplitude}
  Conservation of momentum: $M_\star a_\star = m_p a_p$, $a = a_\star + a_p$.
  \[
    v_\star = \frac{2\pi a_\star}{P} = \frac{m_p}{M_\star + m_p}\cdot\frac{2\pi a}{P}
  \]
  Use Kepler: $a^3 = G(M_\star + m_p) P^2 / (4\pi^2)$, sub in:
  \[
    K_\star = v_\star \sin i = \left(\frac{2\pi G}{P}\right)^{1/3}\!\frac{m_p \sin i}{M_\star^{2/3}}\,\frac{1}{\sqrt{1 - e^2}}
  \]
  \vskip4pt
  $K_\star \propto P^{-1/3}$: short-period planets give larger reflex.
\end{frame}

\begin{frame}{Step 3: Combine}
  A transit forces $\sin i \approx 1$ (orbit near edge-on).
  \begin{itemize}
    \item $\sin i \approx 1$ collapses the $m_p \sin i$ degeneracy.
    \item Both $R_p$ and $m_p$ measured for the same object.
    \item Bulk density follows from elementary geometry:
  \end{itemize}
  \[
    \boxed{\; \bar\rho_p = \frac{m_p}{(4/3)\pi R_p^3} = \frac{3 m_p}{4\pi R_p^3} \;}
  \]
  \vskip4pt
  \keyresult{Transit + RV is the single observational machinery that took exoplanet science from exotic claim to compositional census.}
\end{frame}

\begin{frame}{Density Maps to Composition}
  \begin{columns}[c]
    \column{\recaptextcol}
    \begin{itemize}
      \item $\bar\rho \approx 5.5$: rocky (Earth, Venus).
      \item $\bar\rho \approx 1.3$: gas giant (Jupiter).
      \item $\bar\rho \approx 0.5$: inflated H/He envelope.
      \item $\bar\rho \approx 2$--$4$: \emph{sub-Neptune}, the most common type with no solar-system analogue.
    \end{itemize}
    \column{\recapfigcol}
    \ipsrecapfig{chenkipping_mass_radius}
  \end{columns}
\end{frame}

\sectionimage{pds70_disk}{PDS 70 disk + 2 protoplanets. Haffert et al.\ (2019)}
\section{Other Detection Methods}
% ════════════════════════════════════════════════════════════

\begin{frame}{Direct Imaging: Spatial Separation of Photons}
  \begin{itemize}
    \item Conceptually simplest, technically hardest.
    \item Sun + Jupiter contrast at visible: $10^{-9}$. Sun + Earth: $10^{-10}$.
    \item Four ingredients:
    \begin{itemize}
      \item Adaptive optics (atmospheric turbulence correction)
      \item Coronagraphs (stellar light suppression)
      \item Angular + spectral differential imaging (speckle subtraction)
      \item Long integration ($\to$ averaging speckle realisations)
    \end{itemize}
    \item Strongly biased to \emph{young}, massive, wide-orbit, self-luminous giants.
  \end{itemize}
\end{frame}

\begin{frame}{HR 8799: Four Imaged Giants}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{hr8799_discovery}
    \column{\recaptextcol}
    \begin{itemize}
      \item Marois et al.\ 2008. Keck + Gemini AO; sub-panels show three discovery epochs (HK 2004, CH$_4$ 2007, JKH 2008) confirming bound co-moving companions.
      \item Three planets initially, fourth added 2010.
      \item Each $\sim 5$--$10\,\Mjup$ at $14$--$70$ AU.
      \item Host: 30 Myr A-type. Planets still self-luminous and cooling.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{$\beta$ Pictoris b: Orbital Motion in Two Epochs}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{betapic_imaging}
    \column{\recaptextcol}
    \begin{itemize}
      \item Lagrange et al.\ (2010).
      \item $\sim 8\,\Mjup$ at $\sim 9$ AU.
      \item Two epochs (2003, 2009) $\to$ orbital motion confirms bound companion.
      \item Same star: imaged debris disk (leftover dust ring) + a second giant (c).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}[plain]
  \begin{center}
    \includegraphics[width=0.95\paperwidth, height=0.85\paperheight, keepaspectratio]{pds70bc_haffert}\\
    \footnotesize PDS 70 b + c inside the disk gap in three bands: MUSE H$\alpha$ (a), SPHERE K1 (b), NACO L$'$ (c). Haffert et al.\ (2019).
  \end{center}
\end{frame}

\begin{frame}{PDS 70: Forming Planets in a Disk Gap}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{pds70_disk}
    \column{\recaptextcol}
    \begin{itemize}
      \item Host: 5 Myr K7 with cleared central cavity.
      \item SPHERE (NIR) + MUSE (H$\alpha$) detected b inside the gap (Keppler+ 2018).
      \item Second protoplanet c added later (Haffert+ 2019).
      \item Both are \emph{actively accreting} (H$\alpha$ shock emission).
      \item Cleanest direct match between planet-disk theory and observation.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Astrometry: Gaia Will Find Jupiter Analogues}
  \[
    \alpha = \frac{m_p}{M_\star}\cdot\frac{a_p}{d}
  \]
  \begin{itemize}
    \item Jupiter analogue at 10 pc: $\sim 0.5$ mas. Earth analogue: $\sim 0.3\,\mu$as.
    \item Hipparcos (1989--93): $\sim 1$ mas, no exoplanets.
    \item Gaia (2013--): $20$--$50\,\mu$as per epoch, $\sim 10\,\mu$as final.
    \item DR4 (late 2026): $10^2$--$10^3$ astrometric exoplanets, mostly Jupiter analogues.
    \item Astrometry gives \emph{inclination directly} $\to$ true mass.
  \end{itemize}
\end{frame}

\begin{frame}{Microlensing + Timing}
  \begin{itemize}
    \item \textbf{Microlensing}: foreground star magnifies background source; planet adds a small spike.
      \begin{itemize}
        \item Sensitive at kpc distances; sweet spot at 0.5--10 AU.
        \item One-shot, never repeats.
        \item Roman (2027): expected $\sim 1000$--$3000$ bound + comparable rogue planets.
      \end{itemize}
    \item \textbf{TTVs}: gravitational coupling between transiting planets perturbs transit times $\to$ dynamical masses without RV.
      \begin{itemize}
        \item Essential for faint stars: TRAPPIST-1 masses come almost entirely from TTVs.
      \end{itemize}
    \item \textbf{Pulsar timing}: extraordinarily sensitive but tiny target sample.
    \item \textbf{Eclipsing binaries}: Kepler-16 b ("Tatooine") is circumbinary (orbits both stars).
  \end{itemize}
\end{frame}

\begin{frame}{Detection Bias Summary}
  \begin{itemize}
    \item \textbf{RV}: massive, 1--10 yr orbits, bright G/K stars.
    \item \textbf{Transit}: short period, large $R_p / R_\star$, quiet bright stars.
    \item \textbf{Direct imaging}: wide ($>10$ AU), young, massive, self-luminous.
    \item \textbf{Astrometry}: wide orbits matching mission baseline.
    \item \textbf{Microlensing}: 1--10 AU, any host distance, unrepeatable.
    \item \textbf{Timing}: compact multis (TTVs) or circumbinaries.
  \end{itemize}
  \vskip6pt
  \keyresult{The shape of the observed exoplanet archive reflects the union of biases as much as the underlying physics.}
\end{frame}

\sectionimage{petigura_occurrence}{Kepler occurrence rates. Petigura et al.\ (2018)}
\section{Demographics: Kepler and TRAPPIST-1}
% ════════════════════════════════════════════════════════════

\begin{frame}{The Kepler Revolution}
  \begin{itemize}
    \item 0.95 m telescope, single field, 150{,}000 stars, 4 years, 30 ppm precision.
    \item \emph{Computable detection efficiency} $\to$ first occurrence-rate measurements (planets per star, bias-corrected).
    \item Headline: at least one planet per main-sequence star (Petigura+ 2018).
    \item Small planets ($R < 4\,\Rearth$) dominate by far.
    \item Hot Jupiters: $0.5$--$1\%$ of Sun-like stars (rare, but easy to find).
    \item $\eta_\oplus$ (Earth-size in HZ, the habitable zone): $\sim 0.1$--$0.6$, central value $\sim 0.3$ (Bryson+ 2021).
  \end{itemize}
\end{frame}

  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{petigura_occurrence}
    \column{\recaptextcol}

      \item California-Kepler Survey, after stellar parameter refinement (sub-panels label distinct period-radius regimes).
      \item Small planets common at every period.
      \item Slight fall-off at longer periods (partly observational).
      \item Long-period flattening $\to$ Kepler had only 4-yr baseline.
    \end{itemize}
  \end{columns}
\end{frame}

\end{frame}

\sectionimage{fulton_period_radius}{Fulton period-radius diagram. Fulton et al.\ (2017)}
\section{Period-Radius Diagram and the Radius Valley}
% ════════════════════════════════════════════════════════════

% Hero: Fulton period-radius diagram
\begin{frame}[plain]
  \begin{center}
    \includegraphics[width=0.95\paperwidth, height=0.85\paperheight, keepaspectratio]{fulton_period_radius}\\
    \footnotesize Period-radius distribution of small Kepler planets after stellar parameter refinement. Fulton et al.\ (2017).
  \end{center}
\end{frame}

\begin{frame}{Reading the Period-Radius Diagram}
  \begin{itemize}
    \item Hot Jupiters: tight clump at $P < 10$ d, $R > 10\,\Rearth$.
    \item Sub-Neptunes ($2$--$4\,\Rearth$): \emph{the most common class}; no solar-system analogue.
    \item Super-Earths ($1$--$1.8\,\Rearth$): rocky planets larger than Earth.
    \item Terrestrial regime ($P > 100$ d, $R < 1.5\,\Rearth$): largely unexplored. PLATO will fill.
    \item Most striking feature: the \textbf{deficit at $\sim 1.8\,\Rearth$}.
  \end{itemize}
\end{frame}

\begin{frame}{The Radius Valley (Fulton Gap)}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{fulton_gap}
    \column{\recaptextcol}
    \begin{itemize}
      \item Statistically significant deficit at $1.5$--$2\,\Rearth$.
      \item Below: super-Earths peaked at $\sim 1.3\,\Rearth$.
      \item Above: sub-Neptunes peaked at $\sim 2.4\,\Rearth$.
      \item Bimodal small-planet population. Defining demographic feature.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Photoevaporation Strips H/He Envelopes}
  Energy-limited escape rate:
  \[
    \dot M \approx \frac{\epsilon\,\pi F_\mathrm{XUV} R_p^3}{G M_p}
  \]
  \begin{itemize}
    \item Young stars saturate XUV at $\sim 10^3 \times$ present-day for $\sim 100$ Myr.
    \item Low-density, low-mass planets are \emph{easier to strip}.
    \item Cumulative loss for a $10\,\Mearth$ sub-Neptune at 0.1 AU: a few \% of an Earth mass.
    \item Comparable to a primordial H/He envelope on a rocky core.
    \item Owen \& Wu (2013): naturally produces a bimodal radius distribution.
  \end{itemize}
\end{frame}

\begin{frame}{Photoevaporation Model + Mass-Loss Tracks}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.6\textheight, keepaspectratio]{owen_evaporation_valley}\\
    \footnotesize \textbf{Left}: predicted radius valley
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.6\textheight, keepaspectratio]{owen_xuv_massloss}\\

\end{frame}

\begin{frame}{Core-Powered Mass Loss: Same Outcome, Different Driver}
  \begin{columns}[c]
    \column{\recapfigcol}

    \column{\recaptextcol}
    \begin{itemize}
      \item Ginzburg+ 2018, Gupta+ 2019.
      \item Hot rocky core retains formation heat $\to$ basal flux into envelope drives slow outflow.
      \item No XUV needed.
      \item \textbf{Bottom panel} (Rayleigh + tail): bimodal radius distribution at $\sim 1.8\,\Rearth$, mirroring photoevaporation.
      \item Both mechanisms likely operate.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Slope of the Valley with Period}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{vaneylen_radius_valley}
    \column{\recaptextcol}
    \begin{itemize}
      \item Van Eylen+ 2018, asteroseismic radii (from stellar oscillations).
      \item Valley shifts to smaller $R_p$ at longer periods.
      \item Brown: super-Earths. Blue: sub-Neptunes.
      \item Slope consistent with both photoevaporation and core-powered models.
      \item Current data cannot distinguish.
    \end{itemize}

\end{frame}

\begin{frame}{Implication: Many Super-Earths Are Stripped Cores}
  \begin{itemize}
    \item Many close-in rocky super-Earths are not "naturally" rocky.
    \item They are the bare cores of former sub-Neptunes.
    \item Whether a given super-Earth was born rocky or stripped: \emph{cannot tell from bulk density alone}.
    \item Consequence for habitability: stripped cores have violent atmospheric histories and may have very different surface/interior chemistry.
  \end{itemize}
\end{frame}

\begin{frame}{The Hot Neptune Desert}
  \begin{columns}[c]

    \column{\recaptextcol}

\begin{frame}{Two Edges, Two Mechanisms}
  \begin{itemize}
    \item \textbf{Lower edge (small mass)}: photoevaporation strips weakly-bound Neptune envelopes $\to$ leaves super-Earths.
    \item \textbf{Upper edge (large mass)}: Roche-lobe overflow (tidal gas stripping) on inflated hot Jupiters truncates the heavy population.
    \item Independent evidence that mass loss is the dominant sculptor of close-in planets.
  \end{itemize}
\end{frame}

\begin{frame}{Peas-in-a-Pod}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.55\textheight, keepaspectratio]{weiss_peas_in_pod}\\
    \footnotesize Adjacent radii: Pearson 0.65
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.55\textheight, keepaspectratio]{weiss_spacing}\\
    \footnotesize Adjacent period ratios: clustered
  \end{columns}
  \vskip4pt
  Within a Kepler multi-system, planets tend to be \emph{the same size} as each other and \emph{uniformly spaced}. Weiss et al.\ (2018).
\end{frame}

\begin{frame}{Peas-in-a-Pod: Implications}
  \begin{itemize}
    \item Implies smooth, local formation rather than stochastic giant impacts.
    \item Inner solar system (giant-impact dominated) may be unusual.
    \item TRAPPIST-1: extreme example of a peas-in-a-pod with resonant chain.
    \item Kepler-90 (8 planets), TOI-178 (6 planets in unbroken chain).
    \item Not universal: RV-discovered systems show much more architectural diversity.
  \end{itemize}
\end{frame}

\sectionimage{obliquity_pathways}{Hot Jupiter migration channels. Albrecht et al.\ (2022)}
\section{Hot Jupiters and Migration}
% ════════════════════════════════════════════════════════════

\begin{frame}{Three Migration Channels}
  Hot Jupiters at 0.05 AU \emph{must} have migrated.
  \begin{itemize}
    \item \textbf{Disk migration} (Type II): smooth, gas-driven. Preserves alignment.
    \item \textbf{High-eccentricity migration} (Kozai-Lidov): distant perturber pumps eccentricity, tides circularise. Produces misalignments.
    \item \textbf{Planet-planet scattering}: dynamical instability ejects giants, leaves survivors on eccentric orbits, tides circularise. Misalignments.
  \end{itemize}
  \vskip6pt
  Inner edge set by Roche limit + tidal circularisation timescale: pile-up at $\sim 0.04$--$0.05$ AU.
\end{frame}

\begin{frame}{Migration Channels and Stellar Obliquities}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{obliquity_pathways}
    \column{\recaptextcol}
    \begin{itemize}
      \item Albrecht+ 2022.
      \item Disk migration $\to$ aligned.
      \item Kozai-Lidov $\to$ arbitrary obliquity.
      \item Scattering $\to$ misaligned.
      \item Stellar obliquity (spin-orbit misalignment angle) is the main empirical handle.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Rossiter-McLaughlin Effect}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{rossiter_mclaughlin}
    \column{\recaptextcol}
    \begin{itemize}
      \item Transiting planet first occults blueshifted (approaching) hemisphere, then redshifted.
      \item Time-resolved line distortion traces angle $\lambda$ between projected stellar spin and orbit normal.
      \item Triaud (2018) review.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Observed Obliquity Distribution}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{obliquity_distribution}
    \column{\recaptextcol}
    \begin{itemize}
      \item $\lambda$ vs $a/R_\star$, with sub-panels per host spectral type (panel labels too small for back rows; trust the trend, not the tick labels).
      \item Cool stars ($T_\mathrm{eff} < 6250$ K): tight orbits aligned (tidal realignment).
      \item Hot stars: scattered $\lambda$.
      \item Wide scatter at large $a/R_\star$ implies a wide \emph{primordial} obliquity distribution.
      \item All three migration channels likely operate.
    \end{itemize}
  \end{columns}
\end{frame}

\sectionimage{dressing_mdwarf_occurrence}{M dwarf occurrence. Dressing \& Charbonneau (2015)}
\section{Compositions and M Dwarfs}
% ════════════════════════════════════════════════════════════

\begin{frame}{Sub-Neptunes: Water World or H/He?}
  \begin{itemize}
    \item Super-Earths (below valley): rocky, $\bar\rho \sim 4$--$8$ g cm$^{-3}$.
    \item Sub-Neptunes (above valley): $\bar\rho \sim 1$--$3$ g cm$^{-3}$, need volatiles on top of a rocky core.
    \item Volatile component degenerate: thick H/He, thick H$_2$O, or some mix.
    \item \textbf{Hycean} hypothesis (Madhusudhan+ 2023): shallow liquid ocean under thick H$_2$ atmosphere. Contested.
    \item Ultra-short-period rocky: thin atmosphere vs bare rock indistinguishable from density alone.
  \end{itemize}
\end{frame}

\begin{frame}{M Dwarfs: Easy Targets, Tough Habitability}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{dressing_mdwarf_occurrence}
    \column{\recaptextcolwide}
    \begin{itemize}
      \item $\sim 75\%$ of all main-sequence stars.
      \item M dwarfs host $\sim 2.5$ small planets each; $\sim 0.16$ in HZ.
      \item Transit + RV signals geometrically favourable.
      \item BUT: pre-main-sequence high-luminosity for hundreds of Myr.
      \item Habitable-zone planet \emph{starts} inside runaway greenhouse zone.
      \item Dressing \& Charbonneau (2015).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{M Dwarf Habitability: Two Worries}
  \begin{itemize}
    \item \textbf{Stellar activity} + early high-XUV $\to$ Luger \& Barnes (2015) showed an ocean's worth of water can be lost as O$_2$ (false-positive biosignature: a life-like signal, no biological cause).
    \item \textbf{Tidal locking}: HZ M dwarf planets in 1:1 spin-orbit resonance.
    \item Permanent dayside / nightside; climate must redistribute heat or condense volatiles on the night side.
    \item 3D GCMs (Yang+ 2013): substellar clouds can extend the inner edge.
    \item TRAPPIST-1 = central laboratory for these questions.
  \end{itemize}
\end{frame}

\sectionimage{wasp39b_prism_spectrum}{WASP-39 b PRISM spectrum. Rustamkulov et al.\ (2023)}

\begin{frame}{Transmission Spectroscopy}
  Starlight filtered through the atmosphere at transit encodes absorption.
  \vskip4pt
  Atmospheric scale height (density e-folding distance):
  \[
    H = \frac{\kB T}{\mu m_u g}
  \]
  Wavelength-dependent depth: $\delta(\lambda) = (R_p + n_H(\lambda)\, H)^2 / R_\star^2$.
  \vskip4pt
  $H$ is wavelength-independent; $n_H(\lambda)$ is the number of scale heights probed (typically a few).
  \vskip4pt
  Quick scales for hot Jupiter ($T = 1500$ K, $\mu = 2.3$, $g = 25$ m s$^{-2}$):
  \begin{itemize}
    \item $H \approx 200$ km
    \item Modulation across $\sim 5$ scale heights: $\Delta\delta / \delta \approx 2 \times 10^{-2}$
    \item Absolute modulation: $\sim$200 ppm $\to$ JWST-routine.
    \item Sub-Neptune: 10--100 ppm. Earth/M dwarf: $<$10 ppm.
  \end{itemize}
\end{frame}

\begin{frame}{The Classical Habitable Zone}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{kopparapu_hz}
    \column{\recaptextcol}
    \begin{itemize}
      \item Inner edge: Simpson-Nakajima runaway greenhouse (L9).
      \item Outer edge: maximum CO$_2$ greenhouse condenses out as ice.
      \item Kopparapu+ 2013 modernised Kasting+ 1993.
      \item Symbols: TRAPPIST-1, Tau Ceti, GJ 581/667, etc.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{HZ is a Trajectory, Not a Snapshot Line}
  \begin{itemize}
    \item History matters more than current location.
    \item Venus today sits formally inside the HZ if computed naively, yet is hellish (L9).
    \item M dwarf HZ planets traverse the runaway zone for hundreds of Myr in pre-MS phase.
    \item 1D Kasting/Kopparapu boundaries shift by 5--20\% in 3D GCMs.
    \item Treat the HZ as a first-order screening tool, not a precise boundary.
  \end{itemize}
\end{frame}

\begin{frame}{Biosignature Gases and False Positives}
  \begin{itemize}
    \item Classical: \emph{disequilibrium combinations}. On Earth: O$_2$ + CH$_4$.
    \item Catalogue: O$_2$, O$_3$, CH$_4$, N$_2$O, organosulphur, organohalogen species.
    \item No single gas in isolation is convincing.
    \item \textbf{False positives are the main challenge.}
    \begin{itemize}
      \item Wordsworth+ 2014: H$_2$O photolysis + H escape $\to$ abiotic O$_2$.
      \item CO$_2$ photolysis $\to$ abiotic O$_2$ via CO + O.
      \item Volcanic + serpentinisation $\to$ abiotic CH$_4$.
    \end{itemize}
    \item Inverse problem: must rule out \emph{all} plausible abiotic pathways.
  \end{itemize}
\end{frame}

\section{Comparative Payoff and Frontier Missions}
% ════════════════════════════════════════════════════════════

\begin{frame}{Is the Solar System Typical?}
  Honest answer: \textbf{not obviously, but not yet known to be rare}.
  \vskip4pt
  Sparsely populated corners:
  \begin{itemize}
    \item Most common host: M dwarf, not G dwarf.
    \item Most common planet: sub-Neptune ($2$--$3\,\Rearth$). \emph{Solar system has none.}
    \item Most common architecture: peas-in-a-pod, $\sim 5$--$8$ similar planets within 0.2 AU. \emph{Not the inner solar system.}
    \item No hot Jupiter or hot Neptune in our system.
    \item Wide low-eccentricity giants (vs RV-discovered eccentric giants).
  \end{itemize}
  \vskip4pt
  But: detection biases penalise solar-system analogues. Gaia DR4/DR5 + PLATO will resolve.
\end{frame}

\begin{frame}{Frontier Missions 2026--2035: Transits and Atmospheres}
  \begin{itemize}
    \item \textbf{PLATO} (ESA, end 2026): Earth analogues in HZ of bright Sun-like stars; asteroseismic stellar radii.
    \item \textbf{Ariel} (ESA, 2029): atmospheric statistics of $\sim 1000$ exoplanets at 0.5--7.8 $\mu$m.
    \item \textbf{Roman} (NASA, 2027): $1000$--$3000$ microlensing planets at 1--10 AU + comparable rogue population; coronagraph tech demo at $10^{-8}$--$10^{-9}$.
  \end{itemize}
\end{frame}

\begin{frame}{Frontier Missions 2030s--2040s: Direct Imaging}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{life_yield}
    \column{\recaptextcol}
    \begin{itemize}
      \item \textbf{HWO} (NASA, 2040s): 6 m, $10^{-10}$ contrast, $\sim 25$ Earth analogues.
      \item \textbf{LIFE} (ESA concept): nulling interferometer (starlight cancelled by interference), 4--18 $\mu$m thermal emission.
      \item Figure: yield-sensitivity panels from Quanz+ 2022; trends matter, not the small numerical labels.
      \item \textbf{ELT class} (ground, 2028+): $39$, $24.5$, $30$ m. High-contrast + high-resolution NIR.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{From Demography to Individual Characterisation}
  \begin{itemize}
    \item Today: thousands of detected planets, dozens of well-characterised atmospheres.
    \item 2040s: full atmospheric retrievals on a curated sample of Earth analogues.
    \item Open question for L14: what would convince us we have \emph{detected life}?
    \begin{itemize}
      \item Single biosignature gas?
      \item Specific combinations at specific abundances?
      \item Temporal variability (seasonal cycles)?
      \item Vegetation or photosynthetic surface features?
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Summary: Today's Course-Level Takeaways}
  \begin{enumerate}
    \item Each detection method picks out a different region of parameter space; the observed archive is the union of biases.

    \item JWST: WASP-39 b SO$_2$, TRAPPIST-1 b/c bare rock, K2-18 b biosignature controversy.

  \end{enumerate}
\end{frame}
