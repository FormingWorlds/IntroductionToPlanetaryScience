\sectionimage{rossiter_mclaughlin}{Doppler signal of a transiting planet (Rossiter-McLaughlin). Triaud (2018)}
\section{Detection Method 1: Radial Velocity}
% ════════════════════════════════════════════════════════════

% ── Radial velocity principle ───────────────────────────────
\begin{frame}{Radial Velocity: Doppler Reflex of the Star}
  \begin{itemize}\setlength\itemsep{3pt}
    \item Star and planet orbit a common barycentre: $a_\star = (m_p / M_\star)\,a_p$
    \item The periodic Doppler shift of the stellar absorption lines gives the reflex velocity
    \item Semi-amplitude $K_\star$ for inclination $i$ (edge-on: $\sin i = 1$; circular: $e = 0$):
  \end{itemize}
  \[
    K_\star = \left(\frac{2\pi G}{P}\right)^{1/3}\!\frac{m_p \sin i}{M_\star^{2/3}}\,\frac{1}{\sqrt{1 - e^2}}
  \]
  \keyresult{Jupiter at 5.2 AU: $K_\star \approx 12.5$ m s$^{-1}$; Earth at 1 AU: $K_\star \approx 9$ cm s$^{-1}$.}
\end{frame}

\begin{frame}
  \ipsherokey[0.55]{radial_velocity_signal}{(a) The reflex velocity of a Sun with a Jupiter analogue against the precisions of ELODIE, HARPS and ESPRESSO; (b) the semi-amplitudes of Jupiter, Saturn and Earth analogues. Course figure.}{An Earth analogue at 0.09 m\,s$^{-1}$ sits below the instrumental floor and within the 0.1--1 m\,s$^{-1}$ stellar jitter; RV alone gives only the minimum mass $m_p \sin i$.}
\end{frame}

\section{Detection Method 2: Transits}
% ════════════════════════════════════════════════════════════

% ── Transit geometry ────────────────────────────────────────
\begin{frame}
  \ipsherokey[0.64]{transit_geometry}{Transit geometry: a planet whose orbital plane lies near the line of sight crosses the stellar disk. Winn (2010).}{Transit depth $\delta = (R_p / R_\star)^2$ and geometric probability $p \approx R_\star / a$: a strong selection bias toward short periods and large $R_p / R_\star$.}
\end{frame}

% ── Quick scales ────────────────────────────────────────────
\begin{frame}{Quick Scales}
  Transit depth scales inversely with the square of the stellar radius, so small planets around small stars are far easier to detect.
  \vskip8pt
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Jupiter around Sun:} $\delta \approx 1.01\%$, easily detected from ground-based telescopes
    \item \textbf{Earth around Sun:} $\delta \approx 84$ ppm, requires ultra-precise space photometry
    \item \textbf{Earth around M dwarf ($0.15\,R_\odot$):} $\delta \approx 4000$ ppm, accessible from ground and space
  \end{itemize}
\end{frame}

% ── Light curve anatomy ─────────────────────────────────────
\begin{frame}
  \ipsherokey[0.65]{transit_lightcurve_schematic}{Anatomy of a transit light curve with its four contact times $t_\mathrm{I}$ to $t_\mathrm{IV}$. Winn (2010).}{The depth gives the radius ratio $R_p / R_\star$; the total duration and the ingress time give the impact parameter $b$ and the inclination. A real transit bottom curves from stellar limb darkening.}
\end{frame}

% ── JWST WASP-39 b transit hero ─────────────────────────────
\begin{frame}
  \ipshero{jwst_transit_lightcurve}{JWST/NIRSpec transit photometry of WASP-39 b: depths measured to 200--600 ppm per spectroscopic bin, below typical exoplanet absorption features. Alderson et al.\ (2023).}
\end{frame}


% ════════════════════════════════════════════════════════════
\sectionimage{chenkipping_mass_radius}{Empirical mass-radius relation. Chen \& Kipping (2017)}
\section{Blackboard: Transit Depth + RV $\to$ Bulk Density}
% ════════════════════════════════════════════════════════════

% ── Setup: goal and strategy ────────────────────────────────
\begin{frame}{Goal and strategy}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig[0.62\textheight]{board_sketch_l13}
    \column{\recaptextcolwide}
    \textbf{Goal:} a radius and a mass for the same planet, hence a density.
    \vskip8pt
    \begin{itemize}\setlength\itemsep{2pt}
      \item Transit alone: the radius $R_p$, but no mass
      \item RV alone: the minimum mass $m_p \sin i$, but no radius
      \item Combined: \textbf{both}, so the bulk density $\bar\rho_p$
    \end{itemize}
    \vskip8pt
    \keyresult{Bulk density turns an astronomical detection into a physical, geological body. To the board.}
  \end{columns}
\end{frame}

% ── Derivation result: density ──────────────────────────────
\begin{frame}{Result: Depth, Semi-Amplitude, Density}
  Transit depth $\delta = (R_p/R_\star)^2$ gives $R_p$; the reflex velocity $K_\star = (2\pi G/P)^{1/3}\, m_p \sin i\, M_\star^{-2/3}$ gives $m_p \sin i$ (board).
  \begin{itemize}\setlength\itemsep{2pt}
    \item A transit forces $\sin i \approx 1$ and collapses the $m_p \sin i$ degeneracy
    \item Both $R_p$ and $m_p$ measured for the same object
    \item Bulk density follows from elementary geometry:
  \end{itemize}
  \[
    \boxed{\; \bar\rho_p = \frac{m_p}{(4/3)\pi R_p^3} = \frac{3 m_p}{4\pi R_p^3} \;}
  \]
  \vskip4pt
  \keyresult{Transit plus RV turns a detection into a physical object: radius, mass and density from one planet.}
\end{frame}

% ── Empirical mass-radius diagram ───────────────────────────
\begin{frame}
  \ipsherokey[0.61]{chenkipping_mass_radius}{The empirical mass-radius relation of exoplanets and solar-system bodies. Chen \& Kipping (2017).}{Density maps to composition: $\bar\rho \approx 5.5\ \mathrm{g\,cm^{-3}}$ rocky (Earth, Venus), $1.3$ gas giant (Jupiter), $0.5$ an inflated envelope, and $2$--$4\ \mathrm{g\,cm^{-3}}$ the sub-Neptunes unseen in the solar system.}
\end{frame}

\begin{frame}
  \ipsherokey[0.65]{lichtenberg2025_mass_radius}{Mass against radius for small exoplanets with interior models from pure iron to volatile-rich envelopes, sorted by thermal regime. Reproduced from Lichtenberg et al.\ (2025).}{Rocky super-Earths follow the silicate-iron curves; sub-Neptunes need substantial volatile envelopes or water-rich layers to explain their larger radii.}
\end{frame}

\section{Other Detection Methods}
% ════════════════════════════════════════════════════════════

% ── Direct imaging principle ────────────────────────────────
\begin{frame}{Direct Imaging: Spatial Separation of Photons}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Contrast:} suppress the star by $10^{-9}$ (Jupiter) to $10^{-10}$ (Earth twin) at separation $\theta = a/d$
    \item \textbf{Technology:} extreme adaptive optics plus coronagraphs against diffraction and speckle noise
    \item \textbf{Bias:} young, massive, self-luminous giants on wide orbits ($a > 10$ AU)
  \end{itemize}
\end{frame}

% ── Astrometry ──────────────────────────────────────────────
\begin{frame}{Astrometry: Gaia Will Find Jupiter Analogues}
  Astrometry follows the star's reflex wobble on the sky and gives true masses.
  \[
    \alpha = \frac{m_p}{M_\star}\cdot\frac{a_p}{d}
  \]
  \begin{itemize}\setlength\itemsep{3pt}
    \item \textbf{Signal:} a Jupiter analogue at 10 pc gives $\alpha \sim 500\,\mu$as; an Earth analogue $\sim 0.3\,\mu$as
    \item \textbf{Gaia DR4 (2026):} thousands of cold giants expected; $\sim 10\,\mu$as final precision
    \item \textbf{True mass:} the inclination $i$ is measured, so no $m \sin i$ degeneracy
  \end{itemize}
\end{frame}

\begin{frame}
  \ipshero{astrometric_signal}{Astrometric reflex amplitude against distance for Jupiter and Earth analogues, with Gaia's final precision of about 10 microarcseconds as the dashed line. Course figure.}
\end{frame}

% ── Microlensing and timing ─────────────────────────────────
\begin{frame}{Microlensing, Timing, and Detection Biases}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Microlensing (Roman):} cool planets at 1--10 AU and unbound rogue worlds
    \item \textbf{Timing variations (TTVs):} dynamical masses for faint hosts such as TRAPPIST-1
    \item \textbf{Together:} transits and RV probe close-in orbits; imaging, astrometry and microlensing the wide ones
  \end{itemize}
  \vskip8pt
  \keyresult{Each method covers its own part of parameter space, so the archive carries the biases of the methods.}
\end{frame}

\begin{frame}
  \ipshero{microlensing_lightcurve}{A microlensing event with a planet: the magnification of a background source by a lens star, and the spike of hours to days that a planet around the lens adds. Course figure.}
\end{frame}

% ── Kepler occurrence ───────────────────────────────────────
\begin{frame}{The Kepler Revolution}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Planets are ubiquitous:} on average more than 1 planet per main-sequence star (Petigura et al.\ 2018)
    \item \textbf{Small planets dominate:} super-Earths and sub-Neptunes are common; hot Jupiters are rare ($0.5$--$1\%$)

% ── Photoevaporation mechanism ──────────────────────────────
\begin{frame}{Photoevaporation Strips H/He Envelopes}
  Energy-limited hydrodynamic escape driven by high stellar X-ray and EUV irradiation:
  \[
    \dot M \approx \frac{\epsilon\,\pi F_\mathrm{XUV} R_p^3}{G M_p}
  \]
  \begin{itemize}\setlength\itemsep{3pt}
    \item Young stars saturate XUV flux at $\sim 10^3\times$ modern levels for $\sim 100$ Myr
    \item Strips light primordial H/He envelopes from lower-mass cores at close-in orbits
    \item Owen \& Wu (2013): naturally carves the bimodal radius valley
  \end{itemize}
\end{frame}

\sectionimage{wasp39b_prism_spectrum}{WASP-39 b PRISM spectrum. Rustamkulov et al.\ (2023)}
\section{JWST Era: Atmospheric Characterisation}
% ════════════════════════════════════════════════════════════

% ── Transmission spectroscopy ───────────────────────────────
\begin{frame}{Transmission Spectroscopy}
  Starlight filtered through an atmosphere during transit encodes molecular absorption; with the scale height
  \[
    H = \frac{\kB T}{\mu m_u g}, \qquad \delta(\lambda) \approx \frac{(R_p + n_H(\lambda)\, H)^2}{R_\star^2}, \qquad \frac{\Delta\delta}{\delta} \approx \frac{2 n_H H}{R_p}.
  \]
  \begin{itemize}\setlength\itemsep{3pt}
    \item Hot Jupiter ($T = 1500$ K, $\mu = 2.3$, $g = 25\ \mathrm{m\,s^{-2}}$): $H \approx 200$ km, $\Delta\delta/\delta \approx 2\%$, absolute modulation $\Delta\delta \sim 200$ ppm (JWST-routine)
    \item Sub-Neptunes: 10--100 ppm; terrestrial atmospheres: $<10$ ppm
    \item High clouds or hazes flatten the spectrum (GJ 1214 b): a flat spectrum does not mean no atmosphere
  \end{itemize}
\end{frame}

\begin{frame}

% ── HZ evolution and false positives ────────────────────────
\begin{frame}{Habitable Zone Evolution and Biosignatures}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Dynamic trajectory:} HZ boundaries shift with stellar evolution; pre-MS M dwarfs subject planets to prolonged runaway desiccation
    \item \textbf{Disequilibrium pairs:} atmospheric biosignatures demand chemical disequilibrium ($\mathrm{O_2} + \mathrm{CH_4}$), not lone gas species
    \item \textbf{False positives:} abiotic $\mathrm{O_2}$ can build up by $\mathrm{H_2O}$ photolysis with hydrogen escape, or by $\mathrm{CO_2}$ photolysis
  \end{itemize}
  \vskip4pt
  \keyresult{Life detection is an inverse problem requiring exclusion of all abiotic pathways.}
\end{frame}
