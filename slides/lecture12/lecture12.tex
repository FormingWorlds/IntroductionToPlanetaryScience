\begin{frame}{Reading the Small-Bodies Inventory}
  Three great reservoirs:
  \begin{itemize}
    \item \textbf{Main belt} (2.1--3.3 AU): $\sim 10^6$ bodies $>1$ km. Total mass $\sim 4 \times 10^{-4}\,\Mearth$ (Ceres alone is $30\%$).
    \item \textbf{Trans-Neptunian region} (Kuiper Belt + scattered disk + detached): $>4{,}000$ catalogued, many more incoming with Rubin.
    \item \textbf{Oort cloud} ($\sim 2{,}000$--$50{,}000$ AU): $\sim 10^{11}$--$10^{12}$ objects inferred. \emph{Never directly observed.}
  \end{itemize}
  \vskip6pt\relax
  Plus Trojan swarms at Jupiter's L4/L5 and a growing population of \emph{interstellar} visitors.
\end{frame}

\section{Meteorite Taxonomy}
% ════════════════════════════════════════════════════════════

\begin{frame}{Why Meteorites Matter}
  \begin{itemize}
    \item $\sim$70{,}000 catalogued specimens, mostly Antarctic + desert finds.
    \item \textbf{The oldest rocks accessible to a terrestrial laboratory.}
    \item Earth's record: oldest zircons $\sim 4.0$ Gyr; Moon dominated by impact melts.
    \item Most primitive meteorites contain disk-condensation grains, some \emph{older than the Sun}.
    \item Anchor age: $4567.30 \pm 0.16$ Myr. \textbf{The most precise number in planetary science.}
  \end{itemize}
\end{frame}

\begin{frame}{The First Cut: Chondrites vs.\ Everything Else}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.6\textheight, keepaspectratio]{allende}\\
    \footnotesize Allende slab: chondrules + bright CAIs in dark matrix
    \column{0.5\textwidth}
    \begin{itemize}
      \item \textbf{Chondrites}: never melted whole. Aggregates of disk solids.
      \item Characteristic: $\sim$mm-scale silicate \emph{chondrules}.
      \item Achondrites, irons, stony irons: from \emph{differentiated} parent bodies.
      \item Differentiation needs heat $\to$ short-lived $^{26}\mathrm{Al}$.
      \item Iron meteorites $=$ cores; achondrites $=$ crusts/mantles.
    \end{itemize}
  \end{columns}
\end{frame}

\sectionimage{pb_pb_dating}{Pb-Pb isochrons for CAI 22E and chondrules. Connelly et al.\ (2012)}
\section{Radiogenic Chronometers}
% ════════════════════════════════════════════════════════════

\begin{frame}{The Decay-Clock Equation}
  Radioactive decay is the chronometer:
  \[
    \dv{N}{t} = -\lambda N, \qquad N(t) = N_0 \, e^{-\lambda t}
  \]
  Half-life: $t_{1/2} = \ln 2 / \lambda$.
  \vskip6pt
  \begin{itemize}
    \item \textbf{Long-lived chronometers} ($t_{1/2} \gg$ age of solar system): $^{238}\mathrm{U}$, $^{235}\mathrm{U}$, $^{87}\mathrm{Rb}$, $^{147}\mathrm{Sm}$. Active today.
    \item \textbf{Short-lived (extinct)}: $^{26}\mathrm{Al}\,(0.72\,\mathrm{Myr})$, $^{53}\mathrm{Mn}\,(3.7)$, $^{182}\mathrm{Hf}\,(8.9)$, $^{129}\mathrm{I}\,(16)$, $^{244}\mathrm{Pu}\,(80)$.
    \item Long-lived $\to$ absolute age. Short-lived $\to$ sharp \emph{relative} age within their decay window.
  \end{itemize}
\end{frame}

\begin{frame}{The Early Solar System Timeline}
  Anchored to the Pb-Pb age of CAIs and cross-calibrated:
  \vskip4pt
  \begin{itemize}
    \item $t = 0$: CAIs condense. $4567.30 \pm 0.16$ Myr (Connelly et al.\ 2012).
    \item $t = 1$--$4$ Myr: chondrule formation.
    \item $t = 0.5$--$4$ Myr: planetesimal differentiation, core formation.
    \item $t \gtrsim 5$ Myr: large terrestrial protoplanets.
  \end{itemize}
  \vskip6pt
  \keyresult{Three independent U/Th-Pb chronometers all give the same CAI age to $\pm 0.16$ Myr.}
\end{frame}

\sectionimage{allende}{Allende slab. James St.\ John CC BY 2.0}
\section{Blackboard Derivation: Pb-Pb Isochron of CAIs}
% ════════════════════════════════════════════════════════════

\begin{frame}{Setup: Two Parallel Chains in One Element}
  \begin{itemize}
    \item $^{238}\mathrm{U} \to {}^{206}\mathrm{Pb}$ with $\lambda_{238} = \ln 2 / (4.4683\,\mathrm{Gyr})$.
    \item $^{235}\mathrm{U} \to {}^{207}\mathrm{Pb}$ with $\lambda_{235} = \ln 2 / (0.7038\,\mathrm{Gyr})$.
    \item $^{204}\mathrm{Pb}$: stable, no significant radiogenic source. The reference isotope.
  \end{itemize}
  \vskip6pt
  Radiogenic daughter at time $t$ since closure:
  \[
    {}^{206}\mathrm{Pb}^* = {}^{238}\mathrm{U}\,[e^{\lambda_{238} t} - 1], \qquad
    {}^{207}\mathrm{Pb}^* = {}^{235}\mathrm{U}\,[e^{\lambda_{235} t} - 1]
  \]
\end{frame}

\begin{frame}{Step 1: Real Rocks Have Initial Pb}
  Total Pb $=$ initial $+$ radiogenic, so divide by $^{204}$Pb:
  \[
    \!\!\left(\frac{{}^{206}\mathrm{Pb}}{{}^{204}\mathrm{Pb}}\right)_{\!\!\text{now}}
    \!\!\!=
    \left(\frac{{}^{206}\mathrm{Pb}}{{}^{204}\mathrm{Pb}}\right)_{\!\!0}
    \!\!\!+ \frac{{}^{238}\mathrm{U}}{{}^{204}\mathrm{Pb}}\,[e^{\lambda_{238} t} - 1]
  \]
  \[
    \!\!\left(\frac{{}^{207}\mathrm{Pb}}{{}^{204}\mathrm{Pb}}\right)_{\!\!\text{now}}
    \!\!\!=
    \left(\frac{{}^{207}\mathrm{Pb}}{{}^{204}\mathrm{Pb}}\right)_{\!\!0}
    \!\!\!+ \frac{{}^{235}\mathrm{U}}{{}^{204}\mathrm{Pb}}\,[e^{\lambda_{235} t} - 1]
  \]
  \vskip4pt
  Two equations, three unknowns each ($t$, initial ratio, U/Pb of the rock). \emph{Even taken together, the system is under-constrained.}
\end{frame}

\begin{frame}{Step 2: Use Cogenetic Samples}
  \begin{itemize}
    \item Take several mineral fractions of the same rock, all formed at the same time and from the same Pb reservoir.
    \item All share the same age $t$ and the same initial Pb ratios.
    \item What \emph{differs} between fractions: the U/Pb ratio (U and Pb partition into different minerals).
    \item U $\to$ apatite, zircon. Pb $\to$ feldspar, sulphides. They split.
  \end{itemize}
\end{frame}

\begin{frame}{Step 3: Eliminate U/Pb by Taking the Ratio}
  Divide the $^{207}$Pb equation by the $^{206}$Pb equation. The $^{204}$Pb terms cancel; the U-isotope ratio simplifies to the present-day $^{235}\mathrm{U}/^{238}\mathrm{U}$:
  \[
    \!\!\!\frac{({}^{207}\mathrm{Pb}/{}^{204}\mathrm{Pb})_{\rm now} - ({}^{207}\mathrm{Pb}/{}^{204}\mathrm{Pb})_0}
              {({}^{206}\mathrm{Pb}/{}^{204}\mathrm{Pb})_{\rm now} - ({}^{206}\mathrm{Pb}/{}^{204}\mathrm{Pb})_0}
    = \frac{{}^{235}\mathrm{U}}{{}^{238}\mathrm{U}} \cdot \frac{e^{\lambda_{235} t} - 1}{e^{\lambda_{238} t} - 1}
  \]
  \vskip4pt
  \keyresult{The right-hand side depends \emph{only on time}. The U/Pb ratio of the rock has dropped out.}
\end{frame}

\begin{frame}{Step 4: The Isochron Plot}
  \begin{itemize}
    \item Plot $(^{207}\mathrm{Pb}/^{204}\mathrm{Pb})$ vs.\ $(^{206}\mathrm{Pb}/^{204}\mathrm{Pb})$ for many cogenetic mineral fractions.
    \item All points lie on a straight line: the \textbf{Pb-Pb isochron}.
    \item Slope $\to$ age $t$ (after solving the right-hand-side transcendental equation).
    \item Intercept $\to$ initial Pb composition (drops out automatically; \emph{not} an external assumption).
  \end{itemize}
  \vskip6pt
  This is why Pb-Pb is the gold standard: \emph{the double system is self-calibrating.} Single-system U-Pb requires a separate model for initial Pb.
\end{frame}

% Hero: Pb-Pb isochron plot
\begin{frame}[plain]
  \begin{center}
    \includegraphics[width=0.95\paperwidth, height=0.85\paperheight, keepaspectratio]{pb_pb_dating}\\
    \footnotesize Pb-Pb isochrons: (A) CAI 22E (Efremovka), (B) Allende chondrule C30, (C) NWA 5697 chondrule. From Connelly et al.\ (2012).
  \end{center}
\end{frame}

\begin{frame}{Step 5: Numerical Result}
  Connelly et al.\ (2012) measured CAIs from Northwest Africa 2364 (NWA 2364, CV3) after acid leaching:
  \[
    \boxed{\; t_{\text{CAI}} = 4567.30 \pm 0.16\ \mathrm{Myr} \;}
  \]
  \begin{itemize}
    \item Confirmed by independent samples + labs to the same precision (Amelin et al.\ 2010).
    \item Chondrules from the same meteorite: ages span a few Myr after CAIs.
    \item This is the \emph{absolute zero} of the solar system clock.
  \end{itemize}
  \vskip4pt
  Two parallel decay chains turned an under-constrained problem into a self-calibrating one.
\end{frame}

\sectionimage{spitzer_nc_cc_ages}{Hf-W ages of NC and CC iron meteorites. Spitzer et al.\ (2021)}
\section{The NC-CC Dichotomy}
% ════════════════════════════════════════════════════════════

\begin{frame}{Two Isotopic Reservoirs}
  \begin{itemize}
    \item Nucleosynthetic anomalies in $^{50}\mathrm{Ti}$, $^{54}\mathrm{Cr}$, $^{94}\mathrm{Mo}$, $^{100}\mathrm{Ru}$, $^{48}\mathrm{Ca}$ split meteorites \emph{cleanly} into two groups.
    \item \textbf{NC} (non-carbonaceous): ordinary + enstatite chondrites, HEDs, Mars, Earth.
    \item \textbf{CC} (carbonaceous): all carbonaceous chondrites, several iron groups, probably Jupiter and the Trojans.
    \item Temporally robust: NC and CC of overlapping age are isotopically distinct.
    \item Did not mix for $\sim 3$--$4$ Myr after CAIs.
  \end{itemize}
\end{frame}

\begin{frame}{Hf-W Age Separation}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \includegraphics[width=\linewidth, height=0.7\textheight, keepaspectratio]{spitzer_nc_cc_ages}
    \column{0.45\textwidth}
    \begin{itemize}
      \item \emph{Two different x-axes.}
      \item \emph{Top:} CC irons (blue) vs $\Delta t_{\rm CAI}$ $\to$ cluster at $\sim 3$ Myr.
      \item \emph{Bottom:} NC irons (red) vs initial $\varepsilon^{182}\mathrm{W}$ (\emph{not} age) $\to$ cluster at $\approx -3.3$.
      \item Hf-W modelling: those NC values $\to \Delta t_{\rm CAI} \approx 0.3$--$1$ Myr.
      \item NC cores form $\sim 2$ Myr before CC cores.
      \item Spitzer et al.\ (2021).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Three Live Interpretations}
  Same data, three different stories:
  \begin{enumerate}
    \item \textbf{Spatial barrier (Kruijer et al.\ 2017):} Jupiter formed early ($\lesssim 1$ Myr), opened a gap, stopped CC pebble drift inward. Spatial separation enforces compositional split.
    \item \textbf{Snow-line + pebble isolation (Lichtenberg et al.\ 2021):} disk thermal evolution alone makes two epochs of planetesimal formation; Jupiter only required to reach pebble-isolation mass.
    \item \textbf{Temporal model (Schiller et al.\ 2018; Nanne et al.\ 2019; Spitzer et al.\ 2020):} disk composition itself evolves with time as outer-disk material drifts inward. NC vs CC marks formation \emph{epoch}, not place.
  \end{enumerate}
  \vskip4pt
  \keyresult{All three match current data. The resolution is one of the major open questions in cosmochemistry.}
\end{frame}

      \item Reservoir I (red): early, dry, inner disk = NC.

      \item Bifurcation arises from disk evolution itself.

\end{frame}

\sectionimage{chelyabinsk}{Chelyabinsk meteor (2013). A.\ Ivanov CC BY 3.0}
\section{NEAs, Yarkovsky/YORP, and Planetary Defence}
% ════════════════════════════════════════════════════════════

\begin{frame}{Near-Earth Asteroids}
  \begin{itemize}
    \item NEA $=$ perihelion $q < 1.3$ AU.
    \item $\sim$38{,}000 known (early 2026); thousands of new discoveries per year.
    \item PHAs $=$ Potentially Hazardous: $D > 140$ m and Earth-MOID $< 0.05$ AU. $\sim$2{,}500 known.
    \item Typical residence time $\sim 10$ Myr before ejection or collision.
    \item Population in steady state $\to$ continuous resupply from the main belt.
  \end{itemize}
\end{frame}

\begin{frame}{Three Resupply Channels}
  \begin{itemize}
    \item Kirkwood resonances (especially 3:1, 5:2).
    \item $\nu_6$ secular resonance at the inner belt edge.
    \item \textbf{Yarkovsky drift} into one of the above.
  \end{itemize}
  \vskip6pt
  Yarkovsky: a small along-track thermal-recoil force on a rotating body $\to$ slow $\dd a/\dd t$.
  \vskip4pt
  Standard scaling for the seasonal Yarkovsky effect:
  \[
    \dv{a}{t} \propto \frac{1}{D \, \rho \, \sqrt{a}}
  \]
  Smaller, less dense, closer-in $\to$ drifts faster.
\end{frame}

\begin{frame}{Yarkovsky and YORP Detected on Real NEAs}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.55\textheight, keepaspectratio]{yarkovsky_detection_vokrouhlicky}\\
    \footnotesize \emph{Yarkovsky:} (6489) Golevka, Arecibo radar
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.55\textheight, keepaspectratio]{yorp_detection_vokrouhlicky}\\
    \footnotesize \emph{YORP:} (54509) YORP rotational acceleration
  \end{columns}
  \vskip4pt
  YORP: same thermal asymmetry, acting on \emph{spin}. Can spin small asteroids to fission or stop them.
\end{frame}

\begin{frame}{Impact Frequency, Order of Magnitude}
  \begin{itemize}
    \item \textbf{10 m} ($\sim 10$ kt): once per 5--10 years, harmless airbursts.
    \item \textbf{20 m} ($\sim 0.5$ Mt): once per decade. Chelyabinsk 2013 was 19 m.
    \item \textbf{50 m} ($\sim 10$ Mt): once per few millennia. Tunguska 1908.
    \item \textbf{140 m} ($\sim 300$ Mt): once per $\sim 30$ kyr. PHA threshold.
    \item \textbf{1 km} ($\sim 10^5$ Mt): once per $\sim 500$ kyr. Civilisation-scale.
    \item \textbf{10 km} ($\sim 10^8$ Mt): once per $\sim 100$ Myr. Mass extinction.
  \end{itemize}
\end{frame}

\begin{frame}{Two Recent Impacts Within Living Memory}
  \begin{columns}[T]
    \column{0.5\textwidth}

\sectionimage{halley}{Comet 1P/Halley nucleus, Giotto (1986). ESA/MPAe Lindau}
\section{Comets: Structure, D/H, and Origin}
% ════════════════════════════════════════════════════════════

\begin{frame}{Anatomy of an Active Comet}
  Whipple's "dirty snowball" (1950), confirmed in detail:
  \begin{itemize}
    \item \textbf{Nucleus}: $1$--$30$ km, dark ($\sim 4\%$ albedo), porous ($50$--$75\%$), $\rho \sim 0.5$ g cm$^{-3}$.
    \item Composition: H$_2$O (dominant), CO, CO$_2$, CH$_4$, NH$_3$, CH$_3$OH, HCN, dust, organics.
    \item Inside $\sim 5$ AU: water sublimation $\to$ activity:
    \begin{itemize}
      \item \textbf{Coma}: $10^4$--$10^6$ km gas+dust envelope.
      \item \textbf{Ion tail}: blue, straight, anti-sunward (solar wind).
      \item \textbf{Dust tail}: yellow, curved, lags along the orbit.
    \end{itemize}
  \end{itemize}
\end{frame}

% Hero: Halley nucleus
\begin{frame}[plain]
  \begin{center}
    \includegraphics[width=0.95\paperwidth, height=0.85\paperheight, keepaspectratio]{halley}\\
    \footnotesize Halley nucleus (15 $\times$ 8 $\times$ 8 km), Giotto Halley Multicolour Camera, 13--14 March 1986. Credit: ESA/MPAe Lindau.
  \end{center}
\end{frame}

\begin{frame}{Reading Halley + Halley from Earth}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.55\textheight, keepaspectratio]{halley2}\\
    \footnotesize \emph{Halley} from ESO, March 1986. Long tail, bright coma.
    \column{0.5\textwidth}
    \begin{itemize}
      \item Period 76 years.
      \item Nucleus $15 \times 8 \times 8$ km, irregular, very dark.
      \item Active jets confined to small surface fractions.
      \item Halley-type comet ($T_J \approx -0.6$, retrograde): dynamically Oort-cloud-derived but captured to a $76$-yr orbit.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Short-Period vs Long-Period: The Tisserand Parameter}
  \[
    T_J = \frac{a_J}{a} + 2 \sqrt{\frac{a}{a_J}\,(1 - e^2)} \cos i, \qquad a_J = 5.20\,\mathrm{AU}
  \]
  \begin{itemize}
    \item Conserved across close encounters with Jupiter (Jacobi integral).
    \item $T_J > 3$: main-belt asteroids decoupled from Jupiter.
    \item $2 < T_J < 3$: Jupiter-family comets.
    \item $T_J < 2$: long-period (Oort cloud) comets.
  \end{itemize}
  \vskip6pt
  Examples: Ceres $T_J \approx 3.3$, 67P $T_J \approx 2.7$ (JFC), Halley $T_J \approx -0.6$ (LP).
\end{frame}

\begin{frame}{D/H of Cometary Water vs Earth}
  \begin{columns}[T]
    \column{0.45\textwidth}
    \includegraphics[width=\linewidth, height=0.6\textheight, keepaspectratio]{hartley2}\\
    \footnotesize 103P/Hartley 2 (EPOXI 2010): D/H $\approx$ VSMOW.
    \column{0.55\textwidth}
    \begin{itemize}
      \item Earth oceans: D/H $= 1.56 \times 10^{-4}$ (VSMOW).
      \item Hartley 2 (JFC): $\approx$ VSMOW.
      \item 67P (JFC, Rosetta): $\approx 3 \times$ VSMOW.
      \item Long-period comets: $\approx 2 \times$ VSMOW.
      \item Comets span $3\times$ in D/H. \textbf{No clean class match for Earth.}
    \end{itemize}
  \end{columns}
  \vskip4pt
  Conclusion: Earth's water $\neq$ "delivered by comets". Carbonaceous chondrites must dominate; comets are a small tail.
\end{frame}

\section*{Summary}
% ════════════════════════════════════════════════════════════

\begin{frame}{Summary: Today's Course-Level Takeaways}
  \begin{enumerate}
    \item Small bodies are \textbf{formation fossils}: composition + dynamics still record the solar system's earliest history.
    \item Pb-Pb dates the oldest solids at $4567.30 \pm 0.16$ Myr. Two parallel U decay chains make the method self-calibrating.
    \item NC-CC dichotomy: same data, three live interpretations (spatial / dynamical / temporal). One of cosmochemistry's biggest open problems.
    \item Asteroid belt + KB orbital architecture is a fossil of giant-planet migration.
    \item Saturn's Roche limit, Jupiter's Hill sphere, the Tisserand parameter: same conservation-law toolkit applied across populations.
    \item Sample return + DART + Lucy + Psyche + Rubin + Comet Interceptor: cosmochemistry is becoming a comparative science across stars.
  \end{enumerate}
\end{frame}
