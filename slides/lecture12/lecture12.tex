\sectionimage{allende}{Allende meteorite (CV3 chondrite, fell 1969). J.\ St.\ John CC BY 2.0}
\section{Meteorite Taxonomy}
% ════════════════════════════════════════════════════════════

\begin{frame}{Why Meteorites Matter}
  Meteorites are the oldest rocks accessible in terrestrial laboratories, preserving disk-condensation grains with unparalleled chronological precision:
  \vskip8pt
  \begin{itemize}
    \item \textbf{Laboratory access:} $\sim$70{,}000 catalogued specimens, older than the oldest terrestrial zircons ($\sim 4.4$~Ga)
    \item \textbf{Presolar material:} primitive chondrites contain grains that condensed around other stars, older than the Sun
    \item \textbf{Anchor age:} CAI crystallization age of $4567.30 \pm 0.16$~Myr anchors solar system history
  \end{itemize}
\end{frame}

\begin{frame}{The First Cut: Chondrites vs.\ Everything Else}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{allende}\\
    \footnotesize Allende slab: chondrules + bright CAIs in dark matrix
    \column{\recaptextcol}
    \begin{itemize}
      \item \textbf{Chondrites}: never melted whole. Aggregates of disk solids.
      \item Characteristic: $\sim$mm-scale silicate \emph{chondrules}.
      \item Achondrites, irons, stony irons: from \emph{differentiated} parent bodies.
      \item Differentiation needs heat $\to$ short-lived $^{26}\mathrm{Al}$.
      \item Iron meteorites $=$ cores; achondrites $=$ crusts/mantles.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Chondrules in Thin Section}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{chondrite_thin}
    \column{\recaptextcol}
    \begin{itemize}
      \item Cross-polarised thin section, NWA 5930.
      \item Each chondrule: $\sim$mm bead of olivine + pyroxene.
      \item Frozen from a fully molten droplet in seconds to hours.
      \item Peak $T \sim 1500$--$1900$ K.
      \item How chondrules form remains an unsolved problem.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Iron Meteorites: Reading a Cooling Rate}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{iron_widmanstatten}
    \column{\recaptextcol}
    \begin{itemize}
      \item Widmanst\"atten pattern: kamacite $+$ taenite Fe-Ni lattice.
      \item Forms only at cooling rates $\lesssim 100$ K Myr$^{-1}$.
      \item Slow cooling possible only inside metallic cores of $\sim$tens-of-km bodies.
      \item Lattice spacing $\to$ original parent-body radius.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Pallasite: A Core-Mantle Boundary in Hand}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{pallasite}
    \column{\recaptextcol}
    \begin{itemize}
      \item Esquel pallasite: yellow-green olivine in silvery Fe-Ni metal.
      \item Texture suggests boundary between molten core and silicate mantle.
      \item Olivine crystals sank into metal during a giant impact.
      \item \textbf{A direct sample of a destroyed planetesimal's core-mantle boundary.}
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Vesta: HED Parent Body}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{vesta}
    \column{\recaptextcol}
    \small
    \begin{itemize}
      \item HED meteorites (howardites/eucrites/diogenites) $=$ $\sim 6\%$ of falls.
      \item Identical mineralogy + oxygen isotopes.
      \item Dawn (2011--2012): confirmed Vesta as parent.
      \item Rheasilvia basin (south pole): excavated and ejected the HEDs.
      \item Lunar + Martian meteorites identified the same way (oxygen isotopes + trapped Mars atmosphere).
    \end{itemize}
  \end{columns}
\end{frame}

\sectionimage{pb_pb_dating}{Pb-Pb isochrons for CAI 22E and chondrules. Connelly et al.\ (2012)}
\section{Radiogenic Chronometers}
% ════════════════════════════════════════════════════════════

\begin{frame}{Decay Clocks and Early Solar System Timeline}
  Radioactive decay $\dv{N}{t} = -\lambda N$ ($t_{1/2} = \ln 2 / \lambda$) provides absolute and relative chronometers:
  \vskip4pt
  \begin{itemize}
    \item \textbf{Long-lived:} $^{238}\mathrm{U}$, $^{235}\mathrm{U}$ yield absolute ages; CAIs condense at $t_0 = 4567.30 \pm 0.16$~Myr
    \item \textbf{Short-lived (extinct):} $^{26}\mathrm{Al}$ ($0.717$~Myr), $^{182}\mathrm{Hf}$ ($8.9$~Myr) yield relative ages
    \item \textbf{Early chronology:} Differentiation at $0.5$--$4$~Myr, chondrules at $1$--$4$~Myr after CAIs
  \end{itemize}
  \vskip4pt
  \keyresult{Long-lived chronometers anchor absolute time; extinct nuclides resolve events to $<1$~Myr.}
\end{frame}


% ════════════════════════════════════════════════════════════
\sectionimage{allende}{Allende slab. James St.\ John CC BY 2.0}
\section{Blackboard Derivation: Pb-Pb Isochron of CAIs}
% ════════════════════════════════════════════════════════════

\begin{frame}{Setup: Two Parallel Chains in One Element}
  \begin{itemize}
    \item $^{238}\mathrm{U} \to {}^{206}\mathrm{Pb}$ with $\lambda_{238} = \ln 2 / (4.4683\,\mathrm{Gyr})$.
    \item $^{235}\mathrm{U} \to {}^{207}\mathrm{Pb}$ with $\lambda_{235} = \ln 2 / (0.7038\,\mathrm{Gyr})$.
    \item $^{204}\mathrm{Pb}$: stable, no significant radiogenic source. The reference isotope.
  \end{itemize}
  \vskip6pt
  Radiogenic daughter at time $t$ since closure:
  \[
    {}^{206}\mathrm{Pb}^* = {}^{238}\mathrm{U}\,[e^{\lambda_{238} t} - 1], \qquad
    {}^{207}\mathrm{Pb}^* = {}^{235}\mathrm{U}\,[e^{\lambda_{235} t} - 1]
  \]
  \vskip6pt
  \textit{To the board.}
\end{frame}

\begin{frame}{Result: The Pb-Pb Isochron}
  Cogenetic samples with unknown initial Pb fall on a line in $({}^{206}\mathrm{Pb}/{}^{204}\mathrm{Pb},\ {}^{207}\mathrm{Pb}/{}^{204}\mathrm{Pb})$ whose slope depends only on $t$:
  \[
    \boxed{\; \frac{{}^{207}\mathrm{Pb}^*}{{}^{206}\mathrm{Pb}^*} = \frac{{}^{235}\mathrm{U}}{{}^{238}\mathrm{U}}\,\frac{e^{\lambda_{235} t} - 1}{e^{\lambda_{238} t} - 1}, \qquad \frac{{}^{235}\mathrm{U}}{{}^{238}\mathrm{U}} = \frac{1}{137.8} \;}
  \]
  \keyresult{The slope is independent of the initial Pb and of the U/Pb ratio: the age needs no assumption about the starting composition.}
\end{frame}

% Hero: Pb-Pb isochron plot
\begin{frame}[plain]
  \begin{center}
    \includegraphics[width=0.95\paperwidth, height=0.85\paperheight, keepaspectratio]{pb_pb_dating}\\
    \footnotesize Pb-Pb isochrons for CAI 22E (Efremovka) and chondrules: cogenetic mineral fractions define a linear isochron whose slope yields age $t$ independent of initial Pb. From Connelly et al.\ (2012).
  \end{center}
\end{frame}

\begin{frame}{Step 5: Numerical Result}
  Connelly et al.\ (2012), CAIs from NWA 2364 (CV3) after acid leaching:
  \[
    \boxed{\; t_{\text{CAI}} = 4567.30 \pm 0.16\ \mathrm{Myr} \;}
  \]
  \begin{itemize}
    \item Independent samples and labs agree to the same precision (Amelin et al.\ 2010)
    \item Chondrules from the same meteorite: a few Myr younger
    \item The \emph{zero point} of the solar system clock
  \end{itemize}
  \vskip4pt
  Two parallel decay chains make the problem self-calibrating.
\end{frame}

\sectionimage{spitzer_nc_cc_ages}{Hf-W ages of NC and CC iron meteorites. Spitzer et al.\ (2021)}
\section{The NC-CC Dichotomy}
% ════════════════════════════════════════════════════════════

\begin{frame}{Two Isotopic Reservoirs}
  Nucleosynthetic anomalies in Ti, Cr, Mo, Ru, and Ca cleanly divide solar system materials into two isolated isotopic reservoirs:
  \vskip8pt
  \begin{itemize}
    \item \textbf{NC reservoir (non-carbonaceous):} Ordinary and enstatite chondrites, HEDs, Earth, and Mars
    \item \textbf{CC reservoir (carbonaceous):} Carbonaceous chondrites, several iron groups, and outer bodies
    \item \textbf{Prolonged isolation:} the two reservoirs did not mix isotopically for $\sim 3$--$4$~Myr after CAIs
  \end{itemize}
\end{frame}

\begin{frame}{Hf-W Age Separation}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{spitzer_nc_cc_ages}
    \column{\recaptextcol}
    \begin{itemize}
      \item Two axes, one position: $\varepsilon^{182}\mathrm{W}$ below, the Hf-W age it gives above.
      \item NC irons (red): $-3.37$ to $-3.20$, i.e.\ $\Delta t_{\rm CAI} = 1.0$--$2.6$ Myr.
      \item CC irons (blue): $-3.16$ to $-3.10$, i.e.\ $3.0$--$3.6$ Myr.
      \item Exception: the ungrouped SBT, $\sim 1$ Myr earlier than the other CC irons.
      \item NC cores form $\sim 2$ Myr before CC cores.
      \item Spitzer et al.\ (2021).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Three Live Interpretations}
  Three competing mechanisms explain the NC-CC dichotomy from identical cosmochemical data:
  \vskip8pt
  \begin{itemize}
    \item \textbf{Jupiter barrier:} Early proto-Jupiter ($\lesssim 1$~Myr) opens a gap, blocking inward pebble drift (Kruijer et al.\ 2017)
    \item \textbf{Snow line:} Disk thermal evolution produces two successive planetesimal formation epochs (Lichtenberg et al.\ 2021)
    \item \textbf{Temporal change:} Inward disk drift alters feeding composition over time (Schiller et al.\ 2018)
  \end{itemize}
\end{frame}

\begin{frame}{Snow-Line Mechanism: Two Reservoirs Naturally}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{lichtenberg2021_fig1}
    \column{\recaptextcol}
    \begin{itemize}
      \item Disk simulation, time vs.\ orbital distance.
      \item Reservoir I (red): early, dry, inner disk = NC.
      \item Reservoir II (blue): later, ice-rich, outer disk = CC.
      \item Bifurcation arises from disk evolution itself.
      \item Lichtenberg et al.\ (2021).
    \end{itemize}
  \end{columns}
\end{frame}

    \begin{itemize}

      \item Periodic Jovian kicks add coherently.
      \item Eccentricity pumped to planet-crossing in $10^4$--$10^6$ yr.
      \item Gaps are not empty: they are leaky.
      \item Granvik et al.\ (2018).
    \end{itemize}

\begin{frame}{Asteroid Families = Disrupted Parent Bodies}
  \begin{columns}[c]
    \column{\recapfigcol}

    \column{\recaptextcol}
    \begin{itemize}
      \item Hierarchical clustering in proper $a$, $e$, $i$.
      \item Hirayama families: Themis, Eos, Koronis, Eunomia, Vesta, Flora.
      \item Each family $=$ debris of a single disruption event.
      \item Spread $\to$ original parent body size, age.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Spectral Classes and the Volatile Gradient}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{demeo_carry_belt_mass}
    \source{DeMeo \& Carry (2014)}

    \column{\recaptextcol}
    \small
    \begin{itemize}
      \item \textbf{S-type} ($\sim 15\%$ albedo): inner belt; ordinary-chondrite analogues
      \item \textbf{C-type} ($\sim 4\%$ albedo): $\sim 75\%$ of the outer belt; volatile-rich
      \item \textbf{M-type / V-type}: metallic cores and basaltic crust (Vesta family at $2.4$~AU)
    \end{itemize}
    \vskip4pt
    \keyresult{Radial sorting preserves the disk's primordial volatile gradient.}
  \end{columns}
\end{frame}

\section{NEAs, Yarkovsky/YORP, and Planetary Defence}
% ════════════════════════════════════════════════════════════

\begin{frame}{Near-Earth Asteroids and Resupply Channels}
  NEAs ($q < 1.3$~AU, residence time $\sim 10$~Myr) require steady-state resupply from the main belt:
  \vskip4pt
  \begin{itemize}
    \item \textbf{Resonances:} Kirkwood (3:1, 5:2) and $\nu_6$ secular resonance pump eccentricity
    \item \textbf{Yarkovsky drift:} Thermal recoil drives slow semimajor axis drift:
    \[
      \dv{a}{t} \propto \frac{1}{D \, \rho \, \sqrt{a}}
    \]
    \item \textbf{Hazards:} $\sim$2{,}500 Potentially Hazardous Asteroids ($D > 140$~m, MOID $< 0.05$~AU)
  \end{itemize}
\end{frame}

\section{Dwarf Planets and the Trans-Neptunian Region}
% ════════════════════════════════════════════════════════════

\begin{frame}{Ceres: Inner-Solar-System Ocean World?}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{ceres_occator}
    \column{\recaptextcol}
    \begin{itemize}
      \item $R = 470$ km, mass $\sim 1\%$ Moon.
      \item Dawn (2015--2018): hydrated, ammonia-bearing surface.
      \item Occator: hydrated $\mathrm{Na_2CO_3}$ + $\mathrm{NH_4Cl}$ from erupted brines.
      \item Implies (or had) a partially liquid water layer at depth.
      \item Possible captured outer-solar-system body (Nice Model: a dynamical instability among the giant planets).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Trans-Neptunian Region: Five Sub-Populations}
  The trans-Neptunian region divides into distinct dynamical populations whose orbits preserve the gravitational footprint of giant-planet migration:
  \vskip8pt
  \begin{itemize}
    \item \textbf{Classicals ($a = 42$--$47$~AU):} Dynamically unperturbed cold classicals ($i \lesssim 5^\circ$) and scattered hot classicals ($i \le 30^\circ$)
    \item \textbf{Resonant (e.g.\ 3:2 plutinos):} Bodies swept into mean-motion resonances during Neptune's outward migration
    \item \textbf{Scattered disk and detached:} Neptune-scattered bodies ($q \sim 30$~AU) and decoupled detached objects ($q > 40$~AU)
  \end{itemize}
\end{frame}

\begin{frame}{TNO Orbital Architecture (OSSOS)}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{bannister_ossos_orbits}
    \column{\recaptextcol}
    \begin{itemize}
      \item 1142 characterised TNOs from OSSOS.
      \item Vertical lines: Neptune resonances.
      \item Cold classicals: dense low-$i$ cluster at 42--47 AU.
      \item Bannister et al.\ (2018).
      \item Architecture is a fossil of giant-planet migration.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Anatomy of an Active Comet}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig[0.6\textheight]{comet_diagram}
    \source{Wikimedia Commons, public domain}

    \column{\recaptextcol}
    \footnotesize
    Whipple's "dirty snowball" (1950), confirmed in detail:
    \begin{itemize}
      \item \textbf{Nucleus}: $1$--$30$ km, dark ($\sim 4\%$ albedo), porous ($50$--$75\%$), $\rho \sim 0.5$ g cm$^{-3}$; 67P (Rosetta): $533$ kg m$^{-3}$, $\sim 70\%$ porosity, a contact binary

      \begin{itemize}
        \item \textbf{Coma}: $10^4$--$10^6$ km gas+dust envelope.
        \item \textbf{Ion tail}: blue, straight, anti-sunward (solar wind).
        \item \textbf{Dust tail}: yellow, curved, lags along the orbit.
      \end{itemize}
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Short-Period vs Long-Period: The Tisserand Parameter}
  \[
    T_J = \frac{a_J}{a} + 2 \sqrt{\frac{a}{a_J}\,(1 - e^2)} \cos i, \qquad a_J = 5.20\,\mathrm{AU}
  \]
  \vskip4pt
  The Tisserand parameter is a conserved dynamical invariant that cleanly separates decoupled asteroids from Jupiter-family and long-period comets:
  \vskip4pt
  \begin{itemize}
    \item \textbf{Asteroids ($T_J > 3$):} Main-belt orbits decoupled from Jupiter encounters (e.g.\ Ceres, $T_J \approx 3.3$)
    \item \textbf{Jupiter-family comets ($2 < T_J < 3$):} Kuiper-belt-derived comets controlled by Jupiter (e.g.\ 67P, $T_J \approx 2.7$)
    \item \textbf{Long-period comets ($T_J < 2$):} Oort-cloud visitors on high-inclination orbits; Halley-type comets ($T_J \approx -0.6$) are Oort-derived but captured to short orbits (76 yr)
  \end{itemize}
\end{frame}

  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.72\paperheight, keepaspectratio]{67p_nucleus}}\\
  {\footnotesize Bilobed nucleus of Jupiter-family comet 67P/Churyumov-Gerasimenko ($T_J \approx 2.7$) imaged by Rosetta OSIRIS, showing two lobes joined at a narrow neck by a gentle primordial contact-binary merger. Credit: ESA/Rosetta/MPS for OSIRIS Team.}
\end{frame}

\begin{frame}{D/H of Cometary Water vs Earth}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{hartley2}\\
    \footnotesize 103P/Hartley 2 (EPOXI 2010): D/H $\approx$ VSMOW
    \column{\recaptextcol}
    \begin{itemize}
      \item Earth oceans: D/H $= 1.56 \times 10^{-4}$ (VSMOW standard)
      \item Hartley 2: $\approx$ VSMOW; 67P: $\approx 3 \times$ VSMOW
      \item Long-period comets: $\approx 2 \times$ VSMOW
      \item Comets span $3\times$ in D/H: no class match for Earth
    \end{itemize}
  \end{columns}
  \vskip4pt
  \keyresult{Carbonaceous chondrites dominate Earth's water; comets supply at most a minor fraction.}
\end{frame}
