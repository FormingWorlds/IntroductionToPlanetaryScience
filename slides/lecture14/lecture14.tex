% ── Where solar system is atypical ──────────────────────────
\begin{frame}{Where the Solar System is Atypical}
  \begin{itemize}\setlength\itemsep{4pt}
    \item No super-Earths or sub-Neptunes: the most common exoplanet size class is absent
    \item No close-in giant planets: hot Jupiters orbit $\sim 0.5$--$1\%$ of Sun-like stars, hot Neptunes are rarer still
    \item Outer giants on wide, circular orbits, while the inner terrestrials lack compact uniformity
  \end{itemize}
\end{frame}

\section{Magma Oceans and Water Delivery}
% ════════════════════════════════════════════════════════════

\begin{frame}{Why "Delivery" Isn't the Right Frame}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Earth, Venus, and Mars received similar initial accreted volatile budgets
    \item Magma ocean partitioning sets the core, mantle, and steam atmosphere reservoirs
    \item Evolutionary controls: early hydrodynamic escape, mantle outgassing, and stellar brightening
  \end{itemize}
  \vskip8pt
  \keyresult{Terrestrial volatile inventories diverge because differentiation and evolution matter far more than initial delivery.}
\end{frame}

\begin{frame}
  \ipsherokeyside{lichtenberg2023_magma_ocean_differentiation}{Metal-silicate differentiation by pebbles, small and large embryos, and an energetic impact. Lichtenberg et al.\ (2023).}{The accretion mode sets how much of the mantle melts. Volatiles partition between liquid silicate and the steam atmosphere; the sequestered fraction depends on mantle redox, solidification time and depth, so identical initial budgets yield radically different atmospheric masses.}
\end{frame}

\begin{frame}
  \ipsherokey{hamano2013_typeI_typeII}{Two types of terrestrial planet from magma ocean evolution. \mbox{Hamano et al.\ (2013).}}{Inside the critical flux, Type II planets lose their water to hydrodynamic escape before solidifying; Type I planets further out solidify quickly and retain oceans.}
\end{frame}


% ════════════════════════════════════════════════════════════
\sectionimage{sandberg2018_drake_posterior}{Sandberg-Drexler-Ord Drake posterior (2018)}
\section{Drake Equation and Fermi Paradox}
% ════════════════════════════════════════════════════════════

\begin{frame}{The Drake Equation: A Framing Tool}
  \[
    N = R_\star \cdot f_p \cdot n_e \cdot f_l \cdot f_i \cdot f_c \cdot L
  \]
  $R_\star$ star-formation rate, $f_p$ fraction with planets, $n_e$ habitable planets per system, $f_l$ life, $f_i$ intelligence, $f_c$ communication, $L$ lifetime of a signalling civilisation.
  \begin{itemize}
    \item Astrophysical factors are now measured: $R_\star$, $f_p \sim 1$, $n_e \approx 0.4$ to $0.9$ (Bryson+ 2021)
    \item Biological and social terms ($f_l, f_i, f_c, L$) span $4$--$10$ orders of magnitude
    \item Factors co-evolve rather than acting as independent pipeline stages
  \end{itemize}
  \vskip6pt
  \keyresult{The Drake equation is a tool for framing ignorance, not a numerical estimator.}
\end{frame}

\begin{frame}
  \ipsherokeyside{sandberg2018_drake_posterior}{Monte Carlo posterior for the number $N$ of communicating civilisations, sampled over the full published parameter ranges. Sandberg, Drexler \& Ord (2018).}{The distribution is broad and heavy-tailed, with about a third of the probability below $N = 1$ in the Milky Way: the apparent cosmic silence is unsurprising once the priors are honest.}
\end{frame}

\begin{frame}{Why So Broad? Log-Sum of Broad Priors}
  Taking the logarithm turns the Drake product into a sum of broad priors:
  \[
    \log_{10} N = \log_{10}(R_\star f_p n_e) + \log_{10} f_l + \log_{10} f_i + \log_{10} f_c + \log_{10} L
  \]
  \begin{itemize}
    \item Unconstrained factors have log-uniform priors spanning many decades
    \item The log-sum of broad, skewed priors spreads $\log N$ over tens of decades; the long low tail of $f_l$ puts a third of the mass below $N = 1$
    \item The posterior spans $\sim 30$ decades, placing high probability on $N < 1$
  \end{itemize}
\end{frame}

\begin{frame}
  \ipsherokey[0.62]{fermi_filters}{The Fermi paradox as a chain of filters, with a galactic crossing of $10^6$ to $10^7$~yr against a $10^{10}$~yr galaxy age. \mbox{Course figure.}}{The silence is an intuition check on subjective priors, not physical evidence: rare abiogenesis or intelligence, a Great Filter that keeps $L$ short, or transmissions below current detection limits.}
\end{frame}

\sectionimage{kopparapu2014_hz_stellartype}{Habitable zone vs $T_\mathrm{eff}$. Kopparapu (2014)}
\section{Derivation: Habitable Zone Boundaries}
% ════════════════════════════════════════════════════════════

\begin{frame}{Goal and strategy}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig[0.62\textheight]{board_sketch_l14}
    \column{\recaptextcolwide}
    \small
    \textbf{Goal:} locate the inner and outer edges of the habitable zone from one energy balance.
    \vskip4pt
    \begin{itemize}\setlength\itemsep{2pt}
      \item Step 1: energy balance gives $T_\mathrm{eq}(d)$
      \item Step 2: a boundary as an effective flux $S_{\mathrm{eff}}$, then a distance
      \item Step 3: inner edge from the runaway limit (Lecture 9); then the outer edge
      \item Step 4: G, K and M dwarfs, limits that depend on the star, lifetimes
    \end{itemize}
    \vskip4pt
    \keyresult{The zone is the strip between $d_{\mathrm{in}}$ and $d_{\mathrm{out}}$.}
  \end{columns}
\end{frame}

% ── Setup ───────────────────────────────────────────────────
\begin{frame}{Setup: Energy In, Energy Out}
  Star of bolometric luminosity $L_\star$; planet of radius $R_p$ at distance $d$, Bond albedo $A_B$, emissivity $\epsilon$, rotating fast enough for one uniform temperature.
  \vskip4pt
  Flux at the planet's orbit (inverse-square law):
  \[
    F_\star(d) = \frac{L_\star}{4\pi d^2}
  \]
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{In:} the planet intercepts on its cross-section $\pi R_p^2$ and keeps the fraction $1 - A_B$:
    \[
      P_{\mathrm{in}} = (1 - A_B)\,F_\star\,\pi R_p^2
    \]
    \column{0.48\textwidth}
    \textbf{Out:} the planet radiates from its whole surface $4\pi R_p^2$ (Stefan-Boltzmann):
    \[
      P_{\mathrm{out}} = 4\pi R_p^2\,\epsilon\sigma T_{\mathrm{eq}}^4
    \]
  \end{columns}
  \vskip4pt
  \keyresult{The ratio of the two areas, $1/4$, is geometry, not albedo.}
\end{frame}

% ── Step 1: algebra ─────────────────────────────────────────
\begin{frame}{Step 1: Equilibrium Temperature}
  Set $P_{\mathrm{in}} = P_{\mathrm{out}}$; the planet's area cancels:
  \[
    (1 - A_B)\,F_\star\,\cancel{\pi R_p^2} = 4\,\cancel{\pi R_p^2}\,\epsilon\sigma T_{\mathrm{eq}}^4
  \]
  Solve for $T_{\mathrm{eq}}^4$ and insert $F_\star = L_\star/(4\pi d^2)$:
  \[
    T_{\mathrm{eq}}^4 = \frac{(1 - A_B)\,F_\star}{4\epsilon\sigma} = \frac{(1 - A_B)\,L_\star}{4\epsilon\sigma \cdot 4\pi d^2}
  \]
  Take the fourth root:
  \[
    \boxed{\;T_{\mathrm{eq}}(d) = \left[\frac{L_\star (1 - A_B)}{16\pi\sigma\epsilon\,d^2}\right]^{1/4}\;}
  \]
  \keyresult{The planet's size drops out: $T_{\mathrm{eq}}$ depends only on $L_\star$, $d$, $A_B$ and $\epsilon$.}
\end{frame}

% ── Step 1: Earth numbers ───────────────────────────────────
\begin{frame}{Step 1: Earth's Equilibrium Temperature}
  \small
  Earth: $A_B = 0.3$, $\epsilon = 1$, $L_\odot = 3.828 \times 10^{26}$ W, $d = 1\ \mathrm{AU} = 1.496 \times 10^{11}$ m, \mbox{$\sigma = 5.67 \times 10^{-8}$ W\,m$^{-2}$\,K$^{-4}$}
  \[
    \begin{aligned}
      (1 - A_B)\,L_\odot &= 0.7 \times 3.828 \times 10^{26}\ \mathrm{W} = 2.680 \times 10^{26}\ \mathrm{W} \\
      d^2 &= (1.496 \times 10^{11}\ \mathrm{m})^2 = 2.238 \times 10^{22}\ \mathrm{m^2} \\
      16\pi\sigma d^2 &= 16 \times 3.142 \times 5.67 \times 10^{-8} \times 2.238 \times 10^{22} = 6.379 \times 10^{16}\ \mathrm{W\,K^{-4}} \\
      T_{\mathrm{eq}}^4 &= \frac{2.680 \times 10^{26}}{6.379 \times 10^{16}}\ \mathrm{K^4} = 4.201 \times 10^{9}\ \mathrm{K^4} \\
      T_{\mathrm{eq}} &= \sqrt{\sqrt{4.201 \times 10^{9}}}\ \mathrm{K} = \sqrt{6.48 \times 10^{4}}\ \mathrm{K} = 255\ \mathrm{K}
    \end{aligned}
  \]
  Two square roots make the fourth root. The surface is at $288$ K: the greenhouse effect adds $\sim 33$ K.
  \vskip4pt
  \keyresult{$T_{\mathrm{eq}}$ is the yardstick; 1D climate models (Kopparapu et al.\ 2013) give the real boundaries as fluxes, which we convert to distances next.}
\end{frame}

% ── Step 2: effective flux ──────────────────────────────────
\begin{frame}{Step 2: Effective Stellar Flux and the Distance}
  \small
  Earth's present top-of-atmosphere flux:
  \[
    S_\oplus = \frac{L_\odot}{4\pi (1\ \mathrm{AU})^2} = \frac{3.828 \times 10^{26}\ \mathrm{W}}{2.812 \times 10^{23}\ \mathrm{m^2}} = 1361\ \mathrm{W\,m^{-2}}
  \]
  Define the effective flux as the flux at the planet over $S_\oplus$; the $4\pi$ cancels:
  \[
    S_{\mathrm{eff}} = \frac{F_\star(d)}{S_\oplus} = \frac{L_\star/(\cancel{4\pi}\,d^2)}{L_\odot/(\cancel{4\pi}\,(1\ \mathrm{AU})^2)} = \frac{L_\star}{L_\odot}\left(\frac{1\ \mathrm{AU}}{d}\right)^{2}
  \]
  Solve for $d$:
  \[
    d^2 = (1\ \mathrm{AU})^2\,\frac{L_\star/L_\odot}{S_{\mathrm{eff}}}
    \quad\Rightarrow\quad
    \boxed{\;d = 1\ \mathrm{AU}\,\sqrt{\frac{L_\star/L_\odot}{S_{\mathrm{eff}}}}\;}
  \]
  \keyresult{A boundary is a flux, not a temperature. Flux falls as $d^{-2}$, so $d$ goes as a square root.}
\end{frame}

% ── Step 3: inner edge ──────────────────────────────────────
\begin{frame}{Step 3: Inner Edge, the Runaway Greenhouse}
  \small
  \begin{itemize}\setlength\itemsep{3pt}
    \item Lecture 9: saturation clamps the photosphere temperature, so a moist atmosphere emits at most $F_{\mathrm{OLR}}^{\max} \approx 280$ W\,m$^{-2}$ (grey $253$, line by line $280$ to $310$)
    \item Absorbed flux above it: the oceans go into the atmosphere
  \end{itemize}
  Step 1 balance divided by $4\pi R_p^2$, with $F_{\mathrm{OLR}}^{\max}$ for $\epsilon\sigma T_{\mathrm{eq}}^4$ and $F_\star = S_{\mathrm{eff}}\,S_\oplus$:
  \[
    \frac{(1 - A_B)\,S_{\mathrm{in,eff}}\,S_\oplus}{4} = F_{\mathrm{OLR}}^{\max}
    \quad\Rightarrow\quad
    \boxed{\;S_{\mathrm{in,eff}} = \frac{4\,F_{\mathrm{OLR}}^{\max}}{(1 - A_B)\,S_\oplus}\;}
  \]
  Kopparapu et al.\ (2013), cloud-free steam atmosphere: $F_{\mathrm{OLR}}^{\max} = 291$ W\,m$^{-2}$, $A_B = 0.19$:
  \[
    S_{\mathrm{in,eff}} = \frac{4 \times 291}{0.81 \times 1361} = \frac{1164}{1102} = 1.06
  \]
  \keyresult{Water vapour absorbs near-infrared sunlight: the hot steam atmosphere reflects $0.19$, against $\sim 0.3$ for the same model at Earth-like temperatures.}
\end{frame}

\begin{frame}{Step 3: Inner Edge, the Distance}
  \small
  Insert $S_{\mathrm{in,eff}} = 1.06$ into the boxed distance of Step 2, with $L_\star = L_\odot$:
  \[
    d_{\mathrm{in}} = \frac{1\ \mathrm{AU}}{\sqrt{1.06}} = \frac{1\ \mathrm{AU}}{1.030} = 0.97\ \mathrm{AU}
  \]
  Albedo check: with Earth's $A_B = 0.3$ instead of $0.19$,
  \[
    S_{\mathrm{in,eff}} = \frac{4 \times 291}{0.7 \times 1361} = \frac{1164}{953} = 1.22,
    \qquad
    d_{\mathrm{in}} = \frac{1\ \mathrm{AU}}{\sqrt{1.22}} = 0.90\ \mathrm{AU}
  \]
  An albedo higher by $0.11$ moves the edge $0.07$ AU closer to the star.
  \vskip4pt
  Venus, at $0.72$ AU:
  \[
    S_{\mathrm{eff}} = \left(\frac{1\ \mathrm{AU}}{0.72\ \mathrm{AU}}\right)^2 = \frac{1}{0.518} \approx 1.9 \;>\; 1.06
  \]
  \keyresult{Venus receives about $1.8$ times the runaway flux ($1.9/1.06$): it sits well inside the inner edge.}
\end{frame}

% ── Step 3: outer edge ──────────────────────────────────────
\begin{frame}{Outer Edge: the Maximum CO$_2$ Greenhouse}
  \small
  \begin{itemize}\setlength\itemsep{3pt}
    \item More CO$_2$ warms a cold planet, but only up to a point: near $10$ bar the atmosphere is optically thick at all infrared wavelengths, so the outgoing infrared flux stops falling
    \item Rayleigh scattering by CO$_2$ continues to raise the albedo: beyond this point more CO$_2$ cools the planet
    \item The maximum greenhouse sets a minimum stellar flux that keeps the surface above $273$ K; Kopparapu et al.\ (2013), Sun-like star: $S_{\mathrm{out,eff}} \approx 0.35$
  \end{itemize}
  \[
    d_{\mathrm{out}} = \frac{1\ \mathrm{AU}}{\sqrt{0.35}} = \frac{1\ \mathrm{AU}}{0.592} = 1.69\ \mathrm{AU}
    \qquad
    \text{Mars, } 1.52\ \mathrm{AU}\text{:}\quad
    S_{\mathrm{eff}} = \left(\frac{1}{1.52}\right)^2 = \frac{1}{2.31} = 0.43 \;>\; 0.35
  \]
  \keyresult{The Sun's habitable zone runs from $0.97$ to $1.69$ AU, a width of $\sim 0.7$ AU; Mars is just inside the outer edge.}
\end{frame}

% ── Step 4: K and M roots ───────────────────────────────────
\begin{frame}{Step 4: K and M Dwarfs, Every Root}
  \small
  The same two flux limits; only $L_\star$ changes, and the zone scales as $\sqrt{L_\star}$.
  \vskip4pt
  \textbf{K dwarf} ($\epsilon$ Eri), $L_\star/L_\odot = 0.3$:
  \[
    \begin{aligned}
      d_{\mathrm{in}} &= \sqrt{0.3/1.06}\ \mathrm{AU} = \sqrt{0.283}\ \mathrm{AU} = 0.53\ \mathrm{AU} \\
      d_{\mathrm{out}} &= \sqrt{0.3/0.35}\ \mathrm{AU} = \sqrt{0.857}\ \mathrm{AU} = 0.93\ \mathrm{AU}
    \end{aligned}
  \]
  \textbf{M dwarf} (TRAPPIST-1), $L_\star/L_\odot = 5 \times 10^{-4}$:
  \[
    \begin{aligned}
      d_{\mathrm{in}} &= \sqrt{5 \times 10^{-4}/1.06}\ \mathrm{AU} = \sqrt{4.72 \times 10^{-4}}\ \mathrm{AU} = 0.022\ \mathrm{AU} \\
      d_{\mathrm{out}} &= \sqrt{5 \times 10^{-4}/0.35}\ \mathrm{AU} = \sqrt{1.43 \times 10^{-3}}\ \mathrm{AU} = 0.038\ \mathrm{AU}
    \end{aligned}
  \]
  \begin{center}
    \small
    \begin{tabular}{l c c c}
      \toprule
      Star & $L_\star/L_\odot$ & $d_{\mathrm{in}}$ [AU] & $d_{\mathrm{out}}$ [AU] \\
      \midrule
      G (Sun) & $1$ & $0.97$ & $1.69$ \\
      K ($\epsilon$ Eri) & $0.3$ & $0.53$ & $0.93$ \\
      M (TRAPPIST-1) & $5 \times 10^{-4}$ & $0.022$ & $0.038$ \\
      \bottomrule
    \end{tabular}
  \end{center}
\end{frame}

% ── Step 4: what the numbers mean ───────────────────────────
\begin{frame}{Step 4: HZ Across Stellar Types}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{G dwarf} ($\Lsun$): $d = 0.97$--$1.69$ AU, width $\sim 0.7$ AU; main-sequence lifetime $\sim 10$ Gyr; modest activity. The reference case.
    \item \textbf{K dwarf} ($0.3\,\Lsun$): $d = 0.53$--$0.93$ AU; lifetime $\sim 15$ to $35$ Gyr ($\sim 16$ Gyr for $\epsilon$ Eri at $0.82\,\Msun$); modest flares; the zone is far enough out to avoid most tidal-locking and pre-main-sequence problems. Cuntz \& Guinan (2016): perhaps the \emph{most habitable} class of host star.
    \item \textbf{M dwarf} ($5\times 10^{-4}\,\Lsun$): $d = 0.022$--$0.038$ AU with the solar limits ($0.024$ to $0.048$ AU with the M-dwarf limits, next slide), inside Mercury's orbit; lifetime trillions of years, but every zone planet is tidally locked, and a late M dwarf such as TRAPPIST-1 spends a Gyr in a bright pre-main sequence that puts any zone planet into the runaway state (Luger \& Barnes 2015). Early M dwarfs (M0 to M3): pre-main sequence of only $\sim 50$ to $150$ Myr.
  \end{itemize}
\end{frame}

% ── Step 4: limits depend on the star ───────────────────────
\begin{frame}{Step 4: The Flux Limits Depend on the Star}
  \small
  \begin{itemize}\setlength\itemsep{0pt}\footnotesize
    \item Red light: weaker Rayleigh scattering ($\propto \lambda^{-4}$), more near-infrared absorption by H$_2$O and CO$_2$, so lower $A_B$ and lower limits
    \item Kopparapu et al.\ (2013) fit for $2600$ to $7200$ K: $S_{\mathrm{eff}} = S_{\mathrm{eff},\odot} + aT_* + bT_*^2 + cT_*^3 + dT_*^4$, $T_* = T_{\mathrm{eff}} - 5780$ K (coefficients: their erratum, Table 3)
    \item TRAPPIST-1 ($2560$ K) is just below the range; at $2600$ K and $L_\star/L_\odot = 5 \times 10^{-4}$:
  \end{itemize}
  \[
    \begin{aligned}
      S_{\mathrm{in,eff}} &= 0.86 & \Rightarrow\quad  d_{\mathrm{in}} &= \sqrt{5 \times 10^{-4}/0.86}\ \mathrm{AU} = 0.024\ \mathrm{AU} \\
      S_{\mathrm{out,eff}} &= 0.22 & \Rightarrow\quad  d_{\mathrm{out}} &= \sqrt{5 \times 10^{-4}/0.22}\ \mathrm{AU} = 0.048\ \mathrm{AU}
    \end{aligned}
  \]
  \vskip-3pt
  {\centering\small
    \begin{tabular}{l c c c}
      \toprule
      TRAPPIST-1 (Gillon et al.\ 2017) & e & f & g \\
      \midrule
      $S_{\mathrm{eff}}$ received & $0.66$ & $0.38$ & $0.26$ \\
      Solar limits, $1.06$ to $0.35$ & inside & inside & outside \\
      Limits at $2600$ K, $0.86$ to $0.22$ & inside & inside & inside \\
      \bottomrule
    \end{tabular}
  \par}
  \keyresult{The spectrum moves the edges outward: TRAPPIST-1's zone holds e, f and g.}
\end{frame}

% ── Step 4: main-sequence lifetime ──────────────────────────
\begin{frame}{Step 4: How Long the Zone Lasts}
  \small
  Fuel $\propto$ mass, burn rate $=$ luminosity, $L_\star \propto M_\star^{3.5}$ (low- and intermediate-mass stars):
  \[
    t_{\mathrm{MS}} \sim \frac{E_{\mathrm{nuc}}}{L_\star} \propto \frac{M_\star}{M_\star^{3.5}} = M_\star^{-2.5}
    \quad\Rightarrow\quad
    \boxed{\;t_{\mathrm{MS}} \approx 10\ \mathrm{Gyr}\,\left(\frac{M_\star}{M_\odot}\right)^{-2.5}\;}
  \]
  {\centering\small
    \begin{tabular}{l c c c}
      \toprule
      Star & $M_\star/M_\odot$ & $(M_\star/M_\odot)^{-2.5}$ & $t_{\mathrm{MS}}$ \\
      \midrule
      $\epsilon$ Eri (K) & $0.82$ & $1.64$ & $16$ Gyr \\
      K dwarf & $0.7$ & $2.44$ & $24$ Gyr \\
      Early M dwarf & $0.5$ & $5.66$ & $57$ Gyr \\
      Late M dwarf & $0.1$ & $316$ & $3 \times 10^{3}$ Gyr $= 3 \times 10^{12}$ yr \\
      \bottomrule
    \end{tabular}
  \par}
  {\footnotesize $0.1\,M_\odot$: far longer than the age of the universe; stellar models also give trillions of years (a fully convective star mixes all its hydrogen into the core).}
  \keyresult{Cooler stars give biology more time, by a factor that grows steeply with decreasing stellar mass.}
\end{frame}

\begin{frame}
  \ipsherokey[0.62]{cuntz_guinan_xray_lyman}{Lyman-$\alpha$ and X-ray flux at the Earth-equivalent distance against stellar type and age. \mbox{Cuntz \& Guinan (2016).}}{M4-M6 dwarfs deliver about 50 to 300 times the G-dwarf flux; X-rays drop up to 200-fold over 5~Gyr but Lyman-$\alpha$ only 5 to 10-fold, so early K dwarfs offer long lifetimes without the M-dwarf XUV extremes.}
\end{frame}

\end{frame}

\begin{frame}{Tidal-Locking Timescale}
  Tidal despinning scales steeply as $a^6$, so close-in habitable planets around low-mass stars synchronise quickly:
  \[
    \tau_\mathrm{lock} \sim \frac{\omega_0\,a^6\,I_p\,Q_p}{3\,G\,M_\star^2\,k_{2,p}\,R_p^5}
  \]
  \begin{itemize}
    \item Earth at 1 AU has $\tau_\mathrm{lock} \sim 10^{11}$ yr, several times the age of the universe
    \item TRAPPIST-1 b ($a \approx 0.011$ AU) synchronises within decades to centuries, depending on $Q_p$
    \item With a system age of $\sim 8$ Gyr, essentially every known M-dwarf HZ world is locked
  \end{itemize}
\end{frame}

\begin{frame}

\begin{frame}

\end{frame}

\begin{frame}{Origin-of-Life Hypotheses}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{RNA world}: catalytic ribozymes carry the genetic code before proteins
    \item \textbf{Alkaline hydrothermal vents}: geochemical proton gradients drive early metabolism
    \item \textbf{Warm surface ponds}, where wet-dry cycles polymerise organic precursors, and \textbf{panspermia}, the transfer of viable organisms between bodies
  \end{itemize}
\end{frame}

\begin{frame}
  \ipshero{earth_eons_timeline}{Geologic eons and eras of Earth across 4.54~Gyr: the Precambrian occupies almost 90\% of the timeline, with life emerging early in the Archean. \mbox{Credit: Course-original figure; Gradstein et al.\ (2020).}}
\end{frame}

\begin{frame}{Disequilibrium is the Diagnostic}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Individual gases in isolation almost always possess plausible abiotic origins
    \item Earth pair: $\mathrm{O_2}$ (21\%) and $\mathrm{CH_4}$ (1.8 ppm) react on decade timescales
    \item Simultaneous persistence requires massive, continuous biological resupply fluxes
  \end{itemize}
  \vskip8pt
  \keyresult{Chemically incompatible gases coexisting in an atmosphere are the most convincing remote signature of active biology.}
\end{frame}

\begin{frame}

\end{frame}

\section{False Positives and Solar System Targets}
% ════════════════════════════════════════════════════════════

\begin{frame}{False Positives: Every Gas Has an Abiotic Pathway}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Abiotic $\mathrm{O_2}$: $\mathrm{H_2O}$ photolysis followed by hydrogen escape, or $\mathrm{CO_2}$ photolysis
    \item Abiotic $\mathrm{CH_4}$: hydrothermal serpentinisation or volcanic outgassing at low oxygen fugacity
    \item Abiotic $\mathrm{N_2O}$ from lightning and photochemistry; DMS has no significant abiotic Earth source, but one could exist elsewhere
  \end{itemize}
  \vskip8pt
  \keyresult{Verification is an inverse problem: confirmation requires ruling out every plausible geochemical and photochemical false positive.}
\end{frame}

\begin{frame}{Solar System Targets for Life Detection}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Mars: ancient habitable lacustrine environments; returned samples are the most direct test, but Mars Sample Return has no funding since 2026
    \item Europa and Enceladus: confirmed subsurface oceans; Enceladus's plume carries $\mathrm{H_2}$, organics and phosphates from hydrothermal activity, hypothesised for Europa
    \item Titan and Venus: complex hydrocarbon hydrology and contested cloud chemistry
  \end{itemize}
  \vskip8pt
  \keyresult{Five prime environments with liquid water reservoirs or rich organic chemistry.}
\end{frame}

DAVINCI tests Venus in situ, EnVision and VERITAS from orbit.}
\end{frame}
