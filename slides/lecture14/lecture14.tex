% ── Where solar system is atypical ──────────────────────────
\begin{frame}{Where the Solar System is Atypical}
  The architecture of the solar system diverges noticeably from typical planetary systems discovered by exoplanet surveys.
  \vskip8pt
  \begin{itemize}
    \item No super-Earths or sub-Neptunes: the most common exoplanet size class is absent

    \item Outer giants on wide, circular orbits, while inner terrestrials lack compact uniformity
  \end{itemize}
\end{frame}

\sectionimage{sandberg2018_drake_posterior}{Sandberg-Drexler-Ord Drake posterior (2018)}
\section{Drake Equation and Fermi Paradox}
% ════════════════════════════════════════════════════════════

% ── Drake equation ─────────────────────────────────────────
\begin{frame}{The Drake Equation: A Framing Tool}
  \[
    N = R_\star \cdot f_p \cdot n_e \cdot f_l \cdot f_i \cdot f_c \cdot L
  \]
  $R_\star$ star-formation rate, $f_p$ fraction with planets, $n_e$ habitable planets per system, $f_l$ life, $f_i$ intelligence, $f_c$ communication, $L$ lifetime of a signalling civilisation.
  \begin{itemize}
    \item Astrophysical factors are now measured: $R_\star$, $f_p \sim 1$, $n_e \approx 0.1$--$0.6$ (Bryson+ 2021)
    \item Biological and social terms ($f_l, f_i, f_c, L$) span $4$--$10$ orders of magnitude
    \item Factors co-evolve rather than acting as independent pipeline stages
  \end{itemize}
  \vskip6pt
  \keyresult{The Drake equation is a tool for framing ignorance, not a numerical estimator.}
\end{frame}

% ── Sandberg+ 2018 posterior ────────────────────────────────
\begin{frame}{Sandberg+ 2018: Honest Posterior Dissolves Fermi}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{sandberg2018_drake_posterior}
    \column{\recaptextcol}
    \begin{itemize}
      \item Monte Carlo sampling over full published parameter ranges.
      \item Resulting probability distribution for $N$ is \emph{bimodal}.
      \item Substantial probability mass sits below $N = 1$ in the Milky Way.
      \item Apparent cosmic silence is unsurprising once priors are honest.
    \end{itemize}
    \vskip2pt
    \sourcenote{Sandberg, Drexler \& Ord (2018)}
  \end{columns}
\end{frame}

% ── Why bimodal ─────────────────────────────────────────────
\begin{frame}{Why Bimodal? Log-Sum of Broad Priors}
  Taking the logarithm turns the Drake product into a sum of broad priors, so $\log N$ is spread over many orders of magnitude.
  \[
    \log_{10} N = \log_{10}(R_\star f_p n_e) + \log_{10} f_l + \log_{10} f_i + \log_{10} f_c + \log_{10} L
  \]
  \begin{itemize}
    \item Unconstrained factors have log-uniform priors spanning many decades
    \item Log-sum of broad, skewed priors: $\log N$ spreads over tens of decades, and the long low tail of $f_l$ makes the result bimodal
    \item Posterior width spans $\sim 30$ decades, placing high probability on $N < 1$
  \end{itemize}
\end{frame}

% ── Fermi paradox ───────────────────────────────────────────
\begin{frame}{Fermi Paradox: A Question, Not an Answer}
  The apparent silence of the cosmos acts as an intuition check on subjective priors rather than physical evidence.
  \vskip8pt
  \begin{itemize}
    \item Rare origin: abiogenesis ($f_l \ll 1$) or intelligence ($f_i \ll 1$) is exceptionally improbable
    \item Great filter: civilisations self-terminate rapidly, making lifetime $L$ short
    \item Observational threshold: transmissions exist but lie below current detection limits
  \end{itemize}
\end{frame}

\sectionimage{kopparapu2014_hz_stellartype}{Habitable zone vs $T_\mathrm{eff}$. Kopparapu (2014)}
\section{Blackboard: Habitable Zone Boundaries}
% ════════════════════════════════════════════════════════════

% ── Derivation setup ────────────────────────────────────────
\begin{frame}{Setup: Stellar Flux and Equilibrium Temperature}
  Star with bolometric luminosity $L_\star$, planet at distance $d$:
  \[
    F_\star(d) = \frac{L_\star}{4\pi d^2}
  \]
  Bond albedo $A_B$ (reflected fraction), rapidly rotating sphere, equilibrium:
  \[
    T_\mathrm{eq}(d) = \left[\frac{L_\star (1 - A_B)}{16\pi\sigma\epsilon\,d^2}\right]^{1/4}
  \]
  For Earth: $A_B = 0.3$, $\epsilon = 1$, $L_\star = \Lsun$, $d = 1$ AU $\to$ $T_\mathrm{eq} \approx 255$ K.
  \vskip4pt
  Surface is 288 K from $\sim 33$ K greenhouse warming. \textit{To the board.}
\end{frame}

\begin{frame}{Result: Habitable Zone Boundaries}
  Effective flux $S_{\mathrm{eff}} = F_\star(d)/S_\oplus$ and the boundary distances:
  \[
    d = 1\ \mathrm{AU}\,\sqrt{\frac{L_\star/\Lsun}{S_{\mathrm{eff}}}}
  \]
  \begin{itemize}
    \item Inner edge: runaway greenhouse (Simpson-Nakajima, Lecture 9), $S_{\mathrm{in}} \approx 1.06$, $\approx 0.97$ AU for the Sun
    \item Outer edge: maximum $\mathrm{CO_2}$ greenhouse, $S_{\mathrm{out}} \approx 0.35$, $\sim 1.69$ AU for the Sun
    \item Main-sequence lifetime $t_{\mathrm{MS}} \approx 10\ \mathrm{Gyr}\,(M_\star/M_\odot)^{-2.5}$ sets how long the zone lasts
  \end{itemize}
\end{frame}

% ── Derivation result ───────────────────────────────────────
\begin{frame}{HZ Across Stellar Types}
  Habitable zone distance scales with stellar luminosity: $d \propto \sqrt{L_\star}$.
  \vskip6pt
  \begin{itemize}
    \item \textbf{G dwarf} ($\Lsun$): $d = 0.97$--$1.69$ AU; main-sequence lifetime $\sim 10$ Gyr
    \item \textbf{K dwarf} ($0.3\,\Lsun$): $d = 0.53$--$0.93$ AU; lifetime $24$ Gyr (orange-dwarf sweet spot)
    \item \textbf{M dwarf} ($5\times 10^{-4}\,\Lsun$): $d = 0.022$--$0.038$ AU; lifetime $10^{12}$ yr, but tidally locked with long pre-MS runaway
  \end{itemize}
\end{frame}

% ── Derivation application figure ───────────────────────────
\begin{frame}{XUV at the HZ: M Dwarfs are Hostile}

    \column{\recapfigcol}

% ── Tidal locking timescale ─────────────────────────────────
\begin{frame}{Tidal-Locking Timescale}
  Tidal despinning scales steeply as $a^6$, ensuring that close-in habitable planets around low-mass stars synchronize their rotation rapidly.
  \[
    \tau_\mathrm{lock} \sim \frac{\omega_0\,a^6\,I_p\,Q_p}{3\,G\,M_\star^2\,k_{2,p}\,R_p^5}
  \]
  \begin{itemize}
    \item Earth at 1 AU has $\tau_\mathrm{lock} \sim 10^{12}$ yr, far exceeding universe age
    \item TRAPPIST-1 b ($a \approx 0.011$ AU) synchronises in $\lesssim 10^7$ yr
    \item With system age $\sim 8$ Gyr, all known M-dwarf HZ worlds are locked
  \end{itemize}
\end{frame}

\sectionimage{schwieterman2018_biosignature_classes}{Biosignature classes. Schwieterman et al.\ (2018)}
\section{Astrobiology and Biosignatures}
% ════════════════════════════════════════════════════════════

% ── Life and origin hypotheses ─────────────────────────────
\begin{frame}{Defining Life and Its Origin Hypotheses}
  Working definition (Lederberg 1965): a self-replicating, metabolising, evolving chemical system.
  \vskip6pt
  Four origin-of-life hypotheses:
  \begin{itemize}
    \item \textbf{RNA world}: catalytic ribozymes carry the genetic code before proteins
    \item \textbf{Alkaline hydrothermal vents}: geochemical proton gradients drive early metabolism
    \item \textbf{Warm surface ponds}: wet-dry cycles polymerise organic precursors
    \item \textbf{Panspermia}: transfer of viable organisms between bodies
  \end{itemize}
\end{frame}

  \centering

\end{frame}

% ── Biosignature classes ────────────────────────────────────
\begin{frame}{Three Classes of Biosignature}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{schwieterman2018_biosignature_classes}
    \column{\recaptextcol}
    \begin{itemize}
      \item \textbf{Gaseous}: atmospheric $\mathrm{O_2}$, $\mathrm{O_3}$, $\mathrm{CH_4}$, $\mathrm{N_2O}$, and organosulfur species.
      \item \textbf{Surface}: vegetation red edge with sharp reflectance jump near 700 nm.
      \item \textbf{Temporal}: seasonal cycles like Earth's $\mathrm{CO_2}$ Keeling oscillation.
      \item A credible detection needs several classes together.
    \end{itemize}
    \vskip2pt
    \sourcenote{Schwieterman et al.\ (2018)}
  \end{columns}
\end{frame}

% ── Biosignature gases mid-IR ───────────────────────────────
\begin{frame}{Biosignature Gases in the Mid-IR}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig[0.72\textheight]{schwieterman2018_biosignature_gases}
    \column{\recaptextcol}
    \begin{itemize}
      \item Ten candidate biosignature gases display characteristic mid-IR bands.
      \item Key features: $\mathrm{O_3}$ at $9.6\,\mu\mathrm{m}$, $\mathrm{CH_4}$ at $3.3$ and $7.7\,\mu\mathrm{m}$.
      \item $\mathrm{N_2O}$ features and DMS / DMDS window in 3--15 $\mu$m.
      \item These bands set the wavelength needs of LIFE (mid-IR interferometer) and HWO (Habitable Worlds Observatory).
    \end{itemize}
    \vskip2pt
    \sourcenote{Schwieterman et al.\ (2018)}
  \end{columns}
\end{frame}

% ── Disequilibrium is diagnostic ────────────────────────────
\begin{frame}{Disequilibrium is the Diagnostic}
  Chemically incompatible gases coexisting in an atmosphere provide the most convincing remote signature of active biological production.
  \vskip8pt
  \begin{itemize}
    \item Individual gases in isolation almost always possess plausible abiotic origins
    \item Earth pair: $\mathrm{O_2}$ (21\%) and $\mathrm{CH_4}$ (1.8 ppm) react on decade timescales
    \item Simultaneous persistence requires massive, continuous biological resupply fluxes
  \end{itemize}
\end{frame}

% ── False positives ─────────────────────────────────────────
\begin{frame}{False Positives: Every Gas Has an Abiotic Pathway}
  Biosignature verification is fundamentally an inverse problem because abiotic mechanisms can produce every individual candidate gas.
  \vskip8pt
  \begin{itemize}
    \item Abiotic $\mathrm{O_2}$: $\mathrm{H_2O}$ photolysis followed by hydrogen escape, or $\mathrm{CO_2}$ photolysis
    \item Abiotic $\mathrm{CH_4}$: hydrothermal serpentinisation or volcanic outgassing at low oxygen fugacity
    \item Abiotic $\mathrm{N_2O}$ from lightning and photochemistry; DMS has no significant abiotic Earth source, but one could exist elsewhere
    \item Confirmation requires ruling out all plausible geochemical and photochemical false positives
  \end{itemize}
\end{frame}

\end{frame}

% ── Solar system life targets ───────────────────────────────
\begin{frame}{Solar System Targets for Life Detection}
  In situ solar system exploration targets five prime environments with liquid water reservoirs or rich organic chemical syntheses.
  \vskip8pt
  \begin{itemize}
    \item Mars: ancient habitable lacustrine environments; Mars Sample Return path
    \item Europa and Enceladus: confirmed subsurface oceans; Enceladus's plume carries $\mathrm{H_2}$, organics and phosphates from hydrothermal activity, hypothesised for Europa
    \item Titan and Venus: complex hydrocarbon hydrology and contested cloud chemistry
  \end{itemize}
\end{frame}

% ── Venus phosphine case study ──────────────────────────────
\begin{frame}{Venus Phosphine: A Cautionary Case Study}
  \begin{columns}[c]
    \column{\recapfigcolwide}

    \column{\recaptextcolnarrow}
    \begin{itemize}
      \item Greaves et al.\ (2021): $\mathrm{PH_3}$ in the Venus clouds, a biosignature?
      \item Lincowski et al.\ (2021): the line is fit by mesospheric $\mathrm{SO_2}$
      \item K2-18~b, same lesson: $\mathrm{CH_4}$ ($5\sigma$) and $\mathrm{CO_2}$ ($3\sigma$) stand; DMS fell from $2.4\sigma$ to $\sim 1\sigma$
      \item DAVINCI tests this in situ, EnVision and VERITAS from orbit
    \end{itemize}
  \end{columns}
\end{frame}
