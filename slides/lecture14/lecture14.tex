\begin{frame}{Where the Solar System is Atypical}
  \begin{enumerate}
    \item \textbf{No super-Earth or sub-Neptune.} The most common single class of planet, and we have none.
    \item \textbf{Wide, low-eccentricity giant orbits.} Many exoplanet giants are eccentric, inclined.
    \item \textbf{No hot Jupiter, no hot Neptune.} Both classes occur in $\sim 0.5$--$1\%$ of systems but not in ours.
    \item \textbf{Irregular inner-system spacing.} Solar system terrestrials span $18\times$ in mass; not a peas-in-a-pod.
  \end{enumerate}
\end{frame}

\begin{frame}{Is the Solar System Rare?}
  Honest answer: \textbf{not obviously, but not yet known}.
  \vskip4pt
  \begin{itemize}
    \item Long-period detection floor set by survey duration. Kepler had only 4 yr.
    \item Earth analogues at 1 AU around G dwarfs: at the edge of current sensitivity.
    \item PLATO (2026) + Gaia DR4 (late 2026) will resolve.
    \item Confident now: inner solar system is \emph{atypical} relative to known compact systems.
    \item Open: whether this is bad luck, Jupiter's fingerprint, or the typical outcome around stars we haven't sampled yet.
  \end{itemize}
\end{frame}

\sectionimage{lichtenberg2023_nccc_timeline}{Network of coupled climate components. Lichtenberg et al.\ (2023)}
\section{Habitability as a Coupled Systems Property}
% ════════════════════════════════════════════════════════════

\begin{frame}{The Stack of Requirements}
  Habitability is not a checklist. It is a stack of necessary couplings:
  \begin{itemize}
    \item \textbf{Star}: stable luminosity over Gyr; not too active.
    \item \textbf{Orbit}: in HZ; eccentricity not too high.
    \item \textbf{Planet bulk}: massive enough to retain atmosphere + drive interior.
    \item \textbf{Atmosphere}: temperature, pressure, escape rate compatible with surface water.
    \item \textbf{Interior}: heat budget drives outgassing + dynamo (magnetic-field generation).
    \item \textbf{Surface + tectonics}: long-term volatile recycling.
    \item \textbf{Biosphere}: once present, modifies all of the above.
  \end{itemize}
\end{frame}

% Hero: True systems-level coupling stack (Lichtenberg+ 2023 NCCC)
\begin{frame}[plain]
  \begin{center}
    \includegraphics[width=0.95\paperwidth, height=0.85\paperheight, keepaspectratio]{lichtenberg2023_nccc_timeline}\\
    \footnotesize Network of coupled climate components: interior $\leftrightarrow$ atmosphere $\leftrightarrow$ surface $\leftrightarrow$ biosphere over Gyr timescales. Lichtenberg et al.\ (2023).
  \end{center}
\end{frame}

\begin{frame}{Reading the Coupling Stack}
  \begin{itemize}
    \item Interior $\leftrightarrow$ atmosphere: outgassing, volatile partitioning, magma-ocean chemistry.
    \item Atmosphere $\leftrightarrow$ surface: weathering, photochemistry, redox state of near-surface volatiles.
    \item Surface $\leftrightarrow$ interior: tectonic recycling, sequestration of carbonates and water.
    \item Star $\leftrightarrow$ atmosphere: irradiation, XUV (X-ray + extreme-UV)-driven escape, photochemical loss.
    \item Biosphere (once present) feeds back on atmosphere, surface, and ocean chemistry.
  \end{itemize}
  \vskip6pt
  \keyresult{Break any link, and the system can drift into a regime where surface liquid water is no longer thermodynamically possible.}
\end{frame}

% Hero: Wordsworth tidally-locked atmospheric circulation
\begin{frame}[plain]
  \begin{center}
    \includegraphics[width=0.95\paperwidth, height=0.85\paperheight, keepaspectratio]{wordsworth2022_tidally_locked_planet}\\
    \footnotesize Atmospheric circulation on a tidally locked (same face toward the star) rocky planet: irradiation, clouds, super-rotation, day-night heat transport. Wordsworth \& Kreidberg (2022).
  \end{center}
\end{frame}

\begin{frame}{Reading the Tidally Locked Circulation Diagram}
  \begin{itemize}
    \item Permanent dayside receives all stellar irradiation; partial reflection sets the planetary albedo.
    \item Substellar convergence region: convection lifts moisture, forms thick clouds.
    \item Equatorial super-rotation transports heat from dayside to nightside.
    \item Permanent nightside acts as a cold trap; condensable volatiles can freeze out.
    \item Atmospheric collapse possible when day-night transport is insufficient to keep volatiles vapour.
  \end{itemize}
\end{frame}

\begin{frame}{The HZ is Not a Line}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{kopparapu2014_hz_stellartype}
    \column{\recaptextcol}
    \begin{itemize}
      \item Inner edge: runaway greenhouse.
      \item Outer edge: maximum CO$_2$ greenhouse.
      \item Two planets at the same flux can have radically different climates (Earth vs Venus).
      \item Time-dependent: stars brighten over MS lifetime.
      \item HZ planet today $\neq$ habitable trajectory.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{M Dwarf HZ: Pre-MS Runaway Phase}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{lugerbarnes2015_runaway_duration}
    \column{\recaptextcol}

      \item Late M dwarfs: \emph{Gyr} of runaway before star reaches MS.

\sectionimage{lichtenberg2023_magma_ocean_differentiation}{Magma-ocean differentiation. Lichtenberg et al.\ (2023)}
\section{Magma Oceans, Water Delivery, Tectonic Thermostat}
% ════════════════════════════════════════════════════════════

\begin{frame}{Why "Delivery" Isn't the Right Frame}
  \begin{itemize}
    \item Earth, Venus, Mars likely received \emph{similar} initial volatile inventories.
    \item Yet present-day water inventories differ by orders of magnitude.
    \item Difference: \textbf{what each body did with its initial inventory}.
    \item Four levers (in order of importance):
    \begin{itemize}
      \item Magma ocean (molten mantle) partitioning (core vs mantle vs steam atmosphere)
      \item Magma ocean atmospheric escape
      \item Mantle-atmosphere exchange over Gyr
      \item Stellar evolution history
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Magma Ocean Differentiation Sets the Boundary Conditions}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{lichtenberg2023_magma_ocean_differentiation}
    \column{\recaptextcol}
    \begin{itemize}
      \item Initially molten silicate mantle separates from denser metal.
      \item Volatiles partition between mantle and steam atmosphere.
      \item Fraction sequestered vs outgassed depends on redox state, solidification timescale, depth.
      \item \emph{Identical} starting inventories $\to$ very different end states.
      \item Lichtenberg et al.\ (2023).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Tectonic Regime: The Long-Term Thermostat}
  \begin{itemize}
    \item Volcanism injects CO$_2$; rainfall + silicate weathering removes it; subduction returns carbonates to mantle.
    \item Negative feedback: warmer climate $\to$ faster weathering $\to$ cooler subsequent climate.
    \item Three conditions, all required:
    \begin{itemize}
      \item Liquid water
      \item Active volcanism
      \item Tectonic recycling of carbonates back to the mantle
    \end{itemize}
    \item Venus failure mode: surface dries $\to$ no weathering sink $\to$ CO$_2$ accumulates to 92 bar.
    \item For exoplanets: tectonic regime is \emph{unmeasurable remotely} with current/near-term tools.
  \end{itemize}
\end{frame}

\sectionimage{sandberg2018_drake_posterior}{Sandberg-Drexler-Ord Drake posterior (2018)}
\section{Drake Equation and Fermi Paradox}
% ════════════════════════════════════════════════════════════

\begin{frame}{The Drake Equation}
  \[
    N = R_\star \cdot f_p \cdot n_e \cdot f_l \cdot f_i \cdot f_c \cdot L
  \]
  \begin{itemize}
    \item Only the first three factors are now measured.
      \begin{itemize}
        \item $R_\star$: stellar formation rate. Astrophysics.
        \item $f_p$: fraction with planets. Of order unity (Kepler).
        \item $n_e$: HZ planets per system. $\sim 0.1$--$0.6$ for Sun-like (Bryson+ 2021).
      \end{itemize}
    \item $f_l$, $f_i$, $f_c$, $L$: \emph{not} constrained by orders of magnitude.
    \item Drake equation = framing tool, \textbf{not an estimator}.
  \end{itemize}
\end{frame}

\begin{frame}{Five Reasons Not to Quote a Number for $N$}
  \begin{enumerate}
    \item Factors are not independent. Biospheres co-evolve with atmospheres.
    \item Most factors span $4$--$10$ orders of magnitude in published estimates.
    \item Treats a contingent process as a steady-state pipeline.
    \item Implicitly assumes uniformity over diverse planets and stars.
    \item Anthropic selection in the only data point we have.
  \end{enumerate}
\end{frame}

\begin{frame}{Sandberg+ 2018: Honest Posterior Dissolves Fermi}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{sandberg2018_drake_posterior}
    \column{\recaptextcol}
    \begin{itemize}
      \item Monte Carlo each Drake factor over its full published range.
      \item Result is \emph{bimodal}.
      \item $\sim 1/3$ chance of being alone in the Milky Way (CDF at $N < 1$).
      \item $\sim 10\%$ chance of being alone in the entire observable universe (CDF at $N \ll 1$).
      \item Apparent silence is unsurprising once priors are honest.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Why Bimodal? Log-Sum of Broad Priors}
  Take the log:
  \[
    \log_{10} N = \log_{10}(R_\star f_p n_e) + \log_{10} f_l + \log_{10} f_i + \log_{10} f_c + \log_{10} L
  \]
  \begin{itemize}
    \item Each soft factor has log-uniform prior over many decades.
    \item Sum of log-uniform variables = quasi-Gaussian by central limit theorem.
    \item Width: $\sim 30$ decades. Tail mass below $N = 1$ is large.
    \item Lesson: \textbf{broad priors on multiplicative factors generate broad and skewed product distributions.}
  \end{itemize}
\end{frame}

\begin{frame}{Fermi Paradox: A Question, Not an Answer}
  Possible resolutions:
  \begin{itemize}
    \item \textbf{Rare-life}: $f_l$ very small.
    \item \textbf{Rare-intelligence}: $f_i$ is the bottleneck.
    \item \textbf{Great filter ahead}: $L$ short; civilisations self-terminate.
    \item \textbf{Detection threshold}: signals exist, are missed.
    \item \textbf{Zoo / quiet}: civilisations conceal themselves or use signalling we cannot recognise.
  \end{itemize}
  \vskip6pt
  Use Fermi paradox as a check on intuition about Drake-equation factors. \emph{Not} as evidence.
\end{frame}

\sectionimage{kopparapu2014_hz_stellartype}{Habitable zone vs $T_\mathrm{eff}$. Kopparapu (2014)}
\section{Blackboard: Habitable Zone Boundaries}
% ════════════════════════════════════════════════════════════

\begin{frame}{Setup: Stellar Flux and Equilibrium Temperature}
  Star with bolometric luminosity $L_\star$, planet at distance $d$:
  \[
    F_\star(d) = \frac{L_\star}{4\pi d^2}
  \]
  Bond albedo $A_B$ (reflected-light fraction), fast-rotating sphere, equilibrium:
  \[
    T_\mathrm{eq}(d) = \left[\frac{L_\star (1 - A_B)}{16\pi\sigma\epsilon\,d^2}\right]^{1/4}
  \]
  For Earth: $A_B = 0.3$, $\epsilon = 1$, $L_\star = \Lsun$, $d = 1$ AU $\to$ $T_\mathrm{eq} \approx 255$ K.
  \vskip4pt
  Surface is 288 K because of $\sim 33$ K of greenhouse warming.
\end{frame}

\begin{frame}{Effective Stellar Flux}
  Define $S_\mathrm{eff}$ = TOA flux at HZ boundary / $S_\oplus = 1361$ W m$^{-2}$.
  \[
    d = (1\,\mathrm{AU}) \sqrt{\frac{L_\star/\Lsun}{S_\mathrm{eff}}}
  \]
  \vskip4pt
  Read $S_\mathrm{eff}$ off Kopparapu+ 2013 1D radiative-convective calculation:
  \begin{itemize}
    \item Inner edge (Sun-like): $S_\mathrm{in,eff} \approx 1.06$.
    \item Outer edge (Sun-like): $S_\mathrm{out,eff} \approx 0.35$.
  \end{itemize}
\end{frame}

\begin{frame}{Inner Edge: Simpson-Nakajima Limit}
  In a moist atmosphere, OLR (outgoing longwave radiation) is bounded above:
  \[
    F_\mathrm{OLR}^\mathrm{max} \approx 280\ \mathrm{W\,m^{-2}}
  \]
  \begin{itemize}
    \item Once absorbed flux $>$ this, no radiative equilibrium possible.
    \item Oceans driven into the atmosphere $\to$ runaway greenhouse.
    \item Sun-like inner edge:
  \end{itemize}
  \[
    d_\mathrm{in} = (1\,\mathrm{AU}) / \sqrt{1.06} \approx 0.97\,\mathrm{AU}
  \]
  Venus at $0.72$ AU receives $S_\mathrm{eff} \approx 1.9$: \emph{well inside} the inner edge.
\end{frame}

\begin{frame}{Outer Edge: Maximum CO$_2$ Greenhouse}
  At low $T$ + high $p_\mathrm{CO_2}$, CO$_2$ condenses out as ice clouds.
  \begin{itemize}
    \item Maximum greenhouse warming saturates.
    \item No matter how much CO$_2$ you add, surface stays cold.
    \item Sun-like outer edge:
  \end{itemize}
  \[
    d_\mathrm{out} = (1\,\mathrm{AU}) / \sqrt{0.35} \approx 1.69\,\mathrm{AU}
  \]
  Mars at $1.52$ AU is just inside the outer edge.
\end{frame}

\begin{frame}{HZ Across Stellar Types}
  HZ width $\propto \sqrt{L_\star}$.
  \begin{itemize}
    \item \textbf{G dwarf} ($\Lsun$): $d = 0.97$--$1.69$ AU. Lifetime $\sim 10$ Gyr.
    \item \textbf{K dwarf} ($0.3\,\Lsun$): $d = 0.53$--$0.93$ AU. Lifetime $24$ Gyr. \emph{May be most habitable class} (Cuntz \& Guinan 2016).
    \item \textbf{M dwarf} ($5\times 10^{-4}\,\Lsun$, e.g.\ TRAPPIST-1): $d = 0.022$--$0.038$ AU. Inside Mercury's orbit. Lifetime $10^{12}$ yr but: tidally locked + Gyr pre-MS runaway.
  \end{itemize}
\end{frame}

\begin{frame}{Main-Sequence Lifetime: A One-Line Scaling}
  $E_\mathrm{nuc} \propto M_\star$, $L_\star \propto M_\star^{3.5}$:
  \[
    t_\mathrm{MS} \sim \frac{E_\mathrm{nuc}}{L_\star} \propto M_\star^{-2.5}
  \]
  Normalising on Sun ($t_\mathrm{MS,\odot} \approx 10$ Gyr):
  \[
    t_\mathrm{MS} \approx 10\,(M_\star/\Msun)^{-2.5}\ \mathrm{Gyr}
  \]
  \begin{itemize}
    \item $0.7\,\Msun$ K dwarf: $\approx 24$ Gyr.
    \item $0.5\,\Msun$ early M: $\approx 57$ Gyr.
    \item $0.1\,\Msun$ late M: $\approx 3000$ Gyr.
  \end{itemize}
  \vskip4pt
  Cooler stars give biology more time.\\
  \footnotesize $L \propto M^{3.5}$ is a single-power-law approximation; piecewise relations give $\sim M^{4}$ for $M \gtrsim 0.4\,\Msun$ and $\sim M^{2.3}$ below. Numbers good to factors of a few.
\end{frame}

\begin{frame}{XUV at the HZ: M Dwarfs are Hostile}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{cuntz_guinan_xray_lyman}
    \column{\recaptextcol}
    \begin{itemize}
      \item Lyman-$\alpha$ (wide bars) + X-ray (narrow bars) at the Earth-equivalent distance, by spectral type and age.
      \item At fixed age, $>100$--$500\times$ higher from G2--8 to M4--6.
      \item X-rays drop by up to $500\times$ from 0.1 to 5 Gyr; Lyman-$\alpha$ only $\sim 5$--$10\times$.
      \item Early K: the compromise. Long-lived and common, without the M dwarf extremes.
      \item Cuntz \& Guinan (2016).
    \end{itemize}
  \end{columns}
\end{frame}

\section{HZ Corrections: 3D Climate, Tidal Locking}

      \item 1D HZ underestimates M dwarf habitable region.

\end{frame}

\begin{frame}{Where M Dwarf HZ Planets Actually Sit}
  \begin{columns}[c]
    \column{\recapfigcol}

    \column{\recaptextcol}
    \begin{itemize}
      \item Shields+ 2016. Flux increases to the \emph{left}.
      \item Green/blue solid: 1D Kopparapu inner, outer edges.
      \item Dotted: 3D inner edge, red for slow and locked, black for rapid rotators.

      \item Grey dashed: tidal-locking radius. Nearly all M dwarf HZ planets are on or inside it.
    \end{itemize}
  \end{columns}

\begin{frame}{Tidal-Locking Timescale}
  \[
    \tau_\mathrm{lock} \sim \frac{\omega_0\,a^6\,I_p\,Q_p}{3\,G\,M_\star^2\,k_{2,p}\,R_p^5}
  \]
  Strong $a^6$ dependence:
  \begin{itemize}
    \item Earth at 1 AU: $\tau_\mathrm{lock} \sim 10^{12}$ yr (much longer than universe age).
    \item TRAPPIST-1 b ($a \approx 0.011$ AU): $\sim 10^7$ yr or shorter.
    \item TRAPPIST-1 system age: $\sim 8$ Gyr.
    \item Every M dwarf HZ planet has had time to lock.
  \end{itemize}
\end{frame}

\section{Astrobiology and Biosignatures}
% ════════════════════════════════════════════════════════════

\begin{frame}{What is Life?}
  Working definition (Joyce 1994, building on Lederberg 1965 + Viking-era debate):
  \begin{itemize}
    \item A self-replicating, metabolising, evolving chemical system.
    \item Operational, not philosophically deepest.
    \item Lets us decide what to look for.
  \end{itemize}
  \vskip4pt
  Chemical baseline:
  \begin{itemize}
    \item Carbon-based, water-as-solvent.
    \item Only example we have.
    \item Dominates the search because we know its spectroscopic + geochemical fingerprints.
  \end{itemize}
\end{frame}

\begin{frame}{Origin of Life: Four Live Hypotheses}
  Not solved. Present as competing, not as a settled story.
  \begin{itemize}
    \item \textbf{RNA world}: ribozyme (catalytic RNA)-driven, RNA precedes proteins. Hard prebiotic chemistry of RNA.
    \item \textbf{Metabolism-first / alkaline hydrothermal vents}: autocatalytic cycles precede information storage. Geochemistry compatible.
    \item \textbf{Surface "warm little ponds"}: wet-dry cycling concentrates and polymerises. Darwin 1871; revived by experiment.
    \item \textbf{Panspermia}: relocates the problem, doesn't solve it.
  \end{itemize}
  \vskip4pt
  Earliest direct evidence on Earth: $\sim 3.5$ Ga. Possible to $\sim 4.0$ Ga (contested).
\end{frame}

\begin{frame}{Three Classes of Biosignature}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{schwieterman2018_biosignature_classes}
    \column{\recaptextcol}
    \begin{itemize}
      \item \textbf{Gaseous}: O$_2$, O$_3$, CH$_4$, N$_2$O, organosulphur, organohalogen.
      \item \textbf{Surface}: vegetation red edge (sharp reflectance jump) near 700 nm.
      \item \textbf{Temporal}: seasonal cycles, e.g.\ Earth's CO$_2$ Keeling curve.
      \item Convincing claim should combine $\geq 2$ classes.
      \item Schwieterman+ (2018).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Biosignature Gases in the Mid-IR}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{schwieterman2018_biosignature_gases}
    \column{\recaptextcol}
    \begin{itemize}
      \item Ten potential biosignature gases.
      \item Each has a characteristic IR absorption signature.
      \item O$_3$: 9.6 $\mu$m. CH$_4$: 3.3 + 7.7 $\mu$m. N$_2$O: across mid-IR.
      \item DMS / DMDS: 3--15 $\mu$m window.
      \item Drives the LIFE / HWO wavelength designs.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Disequilibrium is the Diagnostic}
  Single gas in isolation: rarely convincing.
  \vskip4pt
  Most diagnostic: \emph{disequilibrium combinations}.
  \begin{itemize}
    \item Earth: O$_2$ (21\%) + CH$_4$ (1.8 ppm). React on $\sim$decade timescales.
    \item Joint presence requires continuous biological resupply (oxygenic photosynthesis + methanogenesis).
    \item Pre-Great Oxidation Event (before O$_2$ rose, $\sim$2.4 Ga) Earth: CH$_4$ + N$_2$O or CH$_4$ + CO$_2$.
    \item Krissansen-Totton+ 2018 formalised disequilibrium approach.
  \end{itemize}
\end{frame}

  \begin{columns}[T]

\end{frame}

\section{False Positives and Solar System Targets}
% ════════════════════════════════════════════════════════════

\begin{frame}{False Positives: Every Gas Has an Abiotic Pathway}
  \begin{itemize}
    \item \textbf{Abiotic O$_2$}: H$_2$O photolysis (light-driven splitting) + H escape (Wordsworth+ 2014, Luger \& Barnes 2015). CO$_2$ photolysis.
    \item \textbf{Abiotic CH$_4$}: serpentinisation (H$_2$-releasing reaction of olivine with water). Volcanic outgassing at low $f_{\mathrm{O_2}}$.
    \item \textbf{Abiotic N$_2$O}: lightning chemistry, photochemistry.
    \item \textbf{Abiotic DMS}: no significant Earth source known. But "not on Earth" $\neq$ "impossible elsewhere".
  \end{itemize}
  \vskip6pt
  \keyresult{Biosignature detection is an inverse problem: must rule out \emph{all} plausible abiotic pathways.}
\end{frame}

\begin{frame}{Solar System Targets for Life Detection}
  Five solar-system bodies under active astrobiological assessment:
  \begin{itemize}
    \item \textbf{Mars}: past habitability established. Mars Sample Return is the path forward.
    \item \textbf{Europa}: confirmed subsurface ocean. Clipper at Jupiter 2030.
    \item \textbf{Enceladus}: plumes deliver ocean chemistry (H$_2$ + organics + phosphates) to your spacecraft. Decadal-Survey priority.
    \item \textbf{Titan}: complete hydrocarbon hydrology + transient impact-melt water. Dragonfly arrives 2034.
    \item \textbf{Venus cloud layer}: phosphine claim contested.
  \end{itemize}
\end{frame}

\begin{frame}{Venus Phosphine: A Cautionary Case Study}
  \begin{columns}[c]
    \column{\recapfigcolwide}
    \ipsrecapfig{lincowski2021_phosphine_so2}
    \column{\recaptextcolnarrow}
    \begin{itemize}
      \item Greaves+ 2021: PH$_3$ in Venus clouds claimed as biosignature.
      \item Lincowski+ 2021: same data fits SO$_2$ in Venus mesosphere.
      \item Original feature is at the JCMT/ALMA noise floor.
      \item Direct analogue of the K2-18 b DMS controversy.
      \item DAVINCI + EnVision + VERITAS will resolve.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Venus Phosphine + K2-18 b: Same Lesson, Two Telescopes}
  \begin{itemize}
    \item Both: data at edge of instrumental sensitivity.
    \item Both: molecular identification contested.
    \item Both: plausible abiotic explanations not ruled out.
    \item Right response in either case: \emph{better data}, ideally from independent instruments.
    \item Both episodes have driven the design of next-generation instruments.
    \item Pedagogical value is high regardless of how each turns out.
  \end{itemize}
\end{frame}

\section{Exoplanet Life Detection and the Next Decade}
% ════════════════════════════════════════════════════════════

\begin{frame}{What Convincing Exoplanet Biosignature Detection Requires}
  \begin{enumerate}
    \item Multiple independent gases in disequilibrium (oxidising + reducing).
    \item Temporal variability consistent with a biological cycle.
    \item Geological context (mass, radius, host, age, escape) excluding obvious abiotic pathways.
    \item Independent confirmation by different instrument / wavelength / observatory.
  \end{enumerate}
\end{frame}

\begin{frame}{TRAPPIST-1 b: What JWST Can Already Tell Us}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{greene2023_trappist1b_eclipse}
    \column{\recaptextcol}
    \begin{itemize}
      \item MIRI 15 $\mu$m secondary eclipse.
      \item $T_d \approx 508$ K = bare-rock equilibrium.
      \item No thick atmosphere to redistribute heat.
      \item Greene+ 2023.
      \item M dwarf HZ planets often retain little atmosphere.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{LHS 475 b: Featureless Spectrum}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{lustigyaeger2023_lhs475b_spectrum}
    \column{\recaptextcol}
    \begin{itemize}
      \item NIRSpec G395H, Earth-sized M dwarf planet.
      \item Featureless transmission.
      \item H$_2$-dominated ruled out.
      \item High-mean-molecular-weight (CO$_2$-rich) or airless: still consistent.
      \item Lustig-Yaeger+ 2023.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{K2-18 b: How Tentative Biosignatures Are Tested}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{madhusudhan2023_k218b_spectrum}
    \column{\recaptextcol}
    \begin{itemize}
      \item Madhusudhan+ 2023: CH$_4$ at $\sim 5\sigma$, CO$_2$ at $\sim 3\sigma$.
      \item DMS only $2.4\sigma$, and $\sim 1\sigma$ once one instrument offset floats.
      \item Hycean (H$_2$ over global ocean) interpretation contested.
      \item Multiple groups failed to reproduce DMS.
      \item Whatever the resolution: \textbf{textbook case in real-time biosignature testing}.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{LIFE: Mid-IR Nulling Interferometer}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{quanz2022_life_yields}
    \column{\recaptextcol}
    \begin{itemize}
      \item Four free-flying spacecraft, 4--18 $\mu$m.
      \item Nulling: stellar light suppressed by destructive interference.
      \item 3.5 m aperture: $500$--$800$ planets, $\sim 25$ rocky HZ.
      \item Quanz+ 2022.
      \item Companion to HWO (visible/NIR).
    \end{itemize}
  \end{columns}
\end{frame}

\section*{Course Wrap-up}
% ════════════════════════════════════════════════════════════

\begin{frame}{The Five Biggest Things We Have Learned}
  \begin{enumerate}
    \item Planet formation is a \textbf{physical process}, not chance. Same laws everywhere.
    \item Planetary interiors are \textbf{heat engines} driving tectonics, dynamos, atmospheres on Gyr timescales.
    \item Atmospheres are \textbf{not static}. They form, evolve, escape, and feed back on surfaces and biospheres.
    \item Habitability is a \textbf{coupled systems property}, not a single condition.
    \item The solar system is \textbf{one example} in a much larger statistical population.
  \end{enumerate}
\end{frame}

\begin{frame}{The Five Biggest Open Questions}
  \begin{enumerate}
    \item How does life originate, and how often?
    \item What sets the radius valley (gap between super-Earths and sub-Neptunes)?
    \item When did Jupiter form, and was it the architect of the inner solar system?
    \item Was Mars ever inhabited?
    \item Is the solar system rare or typical?
  \end{enumerate}
\end{frame}

\begin{frame}{Final Framing}
  \begin{itemize}
    \item Planetary science is now \emph{the science of comparative climate, interior, and life-hosting trajectories}.
    \item Solar system: reference, not benchmark.
    \item Exoplanet population: statistical context.
    \item The frontier is moving fast.
    \item One systems-level view: habitability, formation, escape, weathering, biology are \emph{one coupled problem}, with one consistent set of physical principles, the same set wherever you go in the galaxy.
  \end{itemize}
  \vskip6pt
  \keyresult{That is what the course has been about.}
\end{frame}
