\section{Cloud Formation}
% ════════════════════════════════════════════════════════════

% ── Saturation ───────────────────────────────────────────────
\begin{frame}{Saturation and Condensation}
  \begin{itemize}
    \item Every vapour has a \textbf{saturation vapour pressure} $P_{\mathrm{sat}}(T)$
    \item When actual vapour pressure exceeds $P_{\mathrm{sat}}$: air is \textbf{supersaturated}
    \item $P_{\mathrm{sat}}(T)$ increases roughly exponentially with temperature
    \item \textbf{Relative humidity:}
    $$\mathrm{RH} = \frac{P_{\mathrm{vapour}}}{P_{\mathrm{sat}}(T)} \times 100\%$$
    \item $\mathrm{RH} = 100\%$: saturated; $\mathrm{RH} > 100\%$: condensation can occur
    \item Rising air cools $\rightarrow$ $P_{\mathrm{sat}}$ drops $\rightarrow$ reaches saturation $\rightarrow$ clouds
  \end{itemize}
\end{frame}

% ── Nucleation ──────────────────────────────────────────────
\begin{frame}{Nucleation: How Droplets Form}
  Even in supersaturated air, condensation needs a \textbf{seed}.
  \vskip8pt
  \begin{description}
    \item[Homogeneous nucleation:] Forming droplets from vapour alone\\
    Requires very high supersaturations ($\mathrm{RH} \gg 100\%$), extremely rare

    \vskip6pt
    \item[Heterogeneous nucleation:] Condensation onto pre-existing particles\\
    \textbf{Condensation nuclei}: dust, volcanic aerosols, sea salt, soot\\
    Occurs at modest supersaturation ($\mathrm{RH} \gtrsim 100\%$): the dominant mechanism
  \end{description}
  \vskip6pt
  \begin{itemize}
    \item Earth: oceanic and biogenic nuclei
    \item Mars: wind-lofted mineral dust
    \item Giant planets: photochemical hazes from above
  \end{itemize}
\end{frame}

% ── Lifting condensation level ──────────────────────────────
\begin{frame}{Lifting Condensation Level (LCL)}
  \begin{itemize}
    \item Rising air cools at the dry adiabatic lapse rate $\Gamma_d = g/c_p$
    \item Vapour pressure stays roughly constant during ascent
    \item But $P_{\mathrm{sat}}(T)$ decreases as temperature drops
    \item At the altitude where $P_{\mathrm{vapour}} = P_{\mathrm{sat}}(T)$: condensation begins
  \end{itemize}
  \vskip6pt
  \keyresult{The \textbf{lifting condensation level} (LCL) marks the \textbf{cloud base}. Above it, latent heat release produces the moist adiabatic lapse rate ($\sim$5--6~K\,km$^{-1}$ on Earth vs.\ $\sim$9.8~K\,km$^{-1}$ dry).}
  \vskip6pt
  Latent heat $\rightarrow$ buoyancy $\rightarrow$ vigorous convection $\rightarrow$ thunderstorms, hurricanes, cumulonimbus.
\end{frame}

% ── Cloud types across solar system ─────────────────────────
\begin{frame}{Cloud Types Across the Solar System}
  What condenses depends on temperature and composition:
  \vskip6pt
  \centering
  \small
  \begin{tabular}{l l l}
    \toprule
    \textbf{Body} & \textbf{Cloud species} & \textbf{Cloud base altitude} \\
    \midrule
    Earth    & $\mathrm{H_2O}$ (liquid \& ice)          & $\sim$1--2~km \\
    Venus    & $\mathrm{H_2SO_4}$ (sulfuric acid)        & $\sim$48~km \\
    Mars     & $\mathrm{CO_2}$ ice, $\mathrm{H_2O}$ ice & 10--100~km \\
    Titan    & $\mathrm{CH_4}$, $\mathrm{C_2H_6}$       & $\sim$8--30~km \\
    Jupiter  & $\mathrm{NH_3}$, $\mathrm{NH_4SH}$, $\mathrm{H_2O}$ & layered, 0.5--7~bar \\
    \bottomrule
  \end{tabular}
  \vskip8pt
  \flushleft
  The physics is the same everywhere: the \textbf{Clausius-Clapeyron equation} governs all of them.
\end{frame}

\section{Blackboard Derivation: Clausius-Clapeyron Equation}
% ════════════════════════════════════════════════════════════

% ── Setup ────────────────────────────────────────────────────
\begin{frame}{Deriving the Clausius-Clapeyron Equation}
  \textbf{Goal:} Derive the exponential dependence of saturation vapour pressure on temperature from thermodynamic phase equilibrium.
  \vskip10pt
  \textbf{Setup: phase equilibrium.}
  \vskip6pt
  Along the coexistence curve, liquid and vapour are in equilibrium:
  $$g_\ell(T, P) = g_v(T, P)$$
  where $g$ is the specific Gibbs free energy.
  \vskip6pt
  Moving along the coexistence curve by $(\dd T, \dd P)$:
  $$\dd g_\ell = \dd g_v$$
\end{frame}

% ── Derivation ──────────────────────────────────────────────
\begin{frame}{Derivation: Exact Clausius-Clapeyron}
  From thermodynamics: $\dd g = -s\,\dd T + v\,\dd P$
  \vskip6pt
  Setting $\dd g_\ell = \dd g_v$:
  $$(s_v - s_\ell)\,\dd T = (v_v - v_\ell)\,\dd P$$
  \vskip4pt
  $$\dv{P}{T} = \frac{\Delta s}{\Delta v} = \frac{L_v}{T(v_v - v_\ell)}$$
  \vskip6pt
  \textbf{Two approximations:}
  \begin{enumerate}
    \item $v_v \gg v_\ell$, so $v_v - v_\ell \approx v_v$
    \item Ideal gas: $v_v = R_v T / P$, where $R_v = R^*/M$
  \end{enumerate}
  \vskip4pt
  $$\dv{P_{\mathrm{sat}}}{T} = \frac{L_v\,P_{\mathrm{sat}}}{R_v\,T^2}$$
\end{frame}

% ── Integration step ───────────────────────────────────────
\begin{frame}{Integrating the Differential Form}
  Starting from
  $$\dv{P_{\mathrm{sat}}}{T} = \frac{L_v\,P_{\mathrm{sat}}}{R_v\,T^2}$$
  separate variables:
  $$\frac{\dd P_{\mathrm{sat}}}{P_{\mathrm{sat}}} = \frac{L_v}{R_v}\,\frac{\dd T}{T^2}$$
  \vskip4pt
  Integrate from a reference state $(T_{\mathrm{ref}}, P_{\mathrm{ref}})$ to $(T, P_{\mathrm{sat}})$, assuming $L_v \approx \mathrm{const.}$:
  $$\ln\!\left(\frac{P_{\mathrm{sat}}}{P_{\mathrm{ref}}}\right) = -\frac{L_v}{R_v}\left(\frac{1}{T} - \frac{1}{T_{\mathrm{ref}}}\right)$$
  \vskip4pt
  Exponentiate to get the integrated form on the next slide.
\end{frame}

% ── Result ──────────────────────────────────────────────────
\begin{frame}{The Clausius-Clapeyron Equation}
  \keyresult{$$\boxed{P_{\mathrm{sat}}(T) = P_{\mathrm{ref}} \exp\!\left[-\frac{L_v}{R_v}\left(\frac{1}{T} - \frac{1}{T_{\mathrm{ref}}}\right)\right]}$$}
  \source{Derivation follows Catling (2017), \textit{Atmospheric Evolution on Inhabited and Lifeless Worlds}, Ch.~1}
  \vskip8pt
  \begin{itemize}
    \item $P_{\mathrm{sat}}$ depends \textbf{exponentially} on temperature
    \item The sensitivity is controlled by $L_v/R_v$ (units of K)
    \item Large $L_v/R_v$ $\rightarrow$ very sensitive to temperature changes
    \item This equation applies to \textbf{any} vapour $\leftrightarrow$ condensed phase transition
  \end{itemize}
\end{frame}

% ── Worked example: water ──────────────────────────────────
\begin{frame}{Worked Example: Water on Earth}
  For water vapour:
  \vskip4pt
  \centering
  \begin{tabular}{l r}
    \toprule
    $L_v$ & $2.50 \times 10^6$~J\,kg$^{-1}$ \\
    $M$ & 0.018~kg\,mol$^{-1}$ \\
    $R_v = R^*/M$ & 462~J\,kg$^{-1}$\,K$^{-1}$ \\
    $T_{\mathrm{ref}}$ (triple point) & 273~K \\
    $P_{\mathrm{ref}}$ (triple point) & 611~Pa \\
    \midrule
    $L_v/R_v$ & $\approx 5400$~K \\
    \bottomrule
  \end{tabular}
  \vskip8pt
  \flushleft
  At $T = 293$~K (20\textdegree{}C):
  $$P_{\mathrm{sat}} = 611 \exp\!\left[-5400\left(\frac{1}{293} - \frac{1}{273}\right)\right] = 611 \times 3.86 \approx 2360 \text{~Pa}$$
  Measured value: 2300~Pa; excellent agreement.
\end{frame}

% ── Condensing species table ────────────────────────────────
\begin{frame}{Condensing Species Across the Solar System}
  \centering
  \small
  \begin{tabular}{l c c c c l}
    \toprule
    \textbf{Species} & $L_v$ (kJ\,kg$^{-1}$) & $R_v$ (J\,kg$^{-1}$\,K$^{-1}$) & $L_v/R_v$ (K) & $T_{\mathrm{cond}}$ (K) & \textbf{Where} \\
    \midrule
    $\mathrm{H_2O}$      & 2500 & 462 & 5400 & 200--280 & Earth, Mars \\
    $\mathrm{H_2SO_4}$   & 540  & 85  & 6400 & 340--380 & Venus \\

    \item Low $L_v/R_v$ for $\mathrm{CH_4}$: Titan's methane clouds extend over a wide range
  \end{itemize}
\end{frame}

% ── Mars clouds ─────────────────────────────────────────────
\begin{frame}{Mars: Dust and Ice Clouds}
  Mars's thin atmosphere ($\sim$6~mbar) supports two cloud types:
  \vskip6pt
  \begin{itemize}
    \item \textbf{$\mathrm{CO_2}$ ice clouds:} High altitude (50--100~km), frost point $\sim$148~K
    \item \textbf{$\mathrm{H_2O}$ ice clouds:} Lower altitude (10--30~km), over Tharsis region, aphelion cloud belt
    \vskip6pt

    \begin{itemize}
      \item Condensation nuclei for ice clouds
      \item Absorbs solar radiation, heats atmosphere
      \item Positive feedback: more dust $\rightarrow$ more heating $\rightarrow$ stronger winds $\rightarrow$ more dust
    \end{itemize}
  \end{itemize}

% ── Titan ────────────────────────────────────────────────────
\begin{frame}{Titan: Methane Rain}
  \begin{itemize}

    \item Surface: $T \approx 94$~K, $P \approx 1.5$~bar, near the triple point of methane
    \item Liquid methane: lakes, seas, rivers carved into the icy surface
    \item $\mathrm{CH_4}$ clouds at 8--30~km; $\mathrm{C_2H_6}$ (ethane) clouds also present
    \item Rainfall infrequent but intense
    \item Seasonal cloud migration observed by Cassini--Huygens

\section{Atmospheric Dynamics}
% ════════════════════════════════════════════════════════════

% ── Hadley cell ─────────────────────────────────────────────
\begin{frame}{The Hadley Cell}
  \begin{columns}[c]
    \column{0.42\textwidth}
    \centering
    \includegraphics[width=\linewidth,height=0.60\textheight,keepaspectratio]{hadley_cells}
    \source{Wikimedia, public domain}

    \column{0.55\textwidth}
    \begin{enumerate}
      \item Intense solar heating at equator $\rightarrow$ air rises
      \item At altitude, air flows poleward
      \item Air cools radiatively, sinks at $\sim$30\textdegree{} latitude
      \item Surface return flow $\rightarrow$ \textbf{trade winds}
    \end{enumerate}
    \vskip6pt
    This loop is the \textbf{Hadley cell}:
    \begin{itemize}
      \item Dominant circulation in the tropics
      \item Primary heat transport: equator $\rightarrow$ poles
      \item Sinking zones $\rightarrow$ desert belts on Earth
    \end{itemize}
  \end{columns}
\end{frame}

% ── Coriolis effect ─────────────────────────────────────────
\begin{frame}{The Coriolis Effect}
  On a rotating planet, moving air is \textbf{deflected}:
  \begin{itemize}
    \item \textbf{Right} in the Northern Hemisphere
    \item \textbf{Left} in the Southern Hemisphere
  \end{itemize}
  \vskip6pt
  The \textbf{Coriolis parameter}:
  $$f = 2\Omega \sin\phi$$
  \vskip4pt
  \begin{itemize}
    \item Earth: $\Omega = 7.27 \times 10^{-5}$~rad\,s$^{-1}$
    \item At mid-latitudes ($\phi = 45$\textdegree{}): $f \approx 1.03 \times 10^{-4}$~s$^{-1}$
    \item At equator ($\phi = 0$\textdegree{}): $f = 0$, so no Coriolis deflection
  \end{itemize}
  \vskip6pt
  \keyresult{The Coriolis effect explains why winds do not blow directly from high to low pressure, and why weather systems rotate.}
\end{frame}

% ── Rossby number ───────────────────────────────────────────
\begin{frame}{The Rossby Number}
  Does rotation matter for a given flow?
  \vskip8pt
  $$\mathrm{Ro} = \frac{U}{fL}$$
  \vskip6pt
  \begin{description}
    \item[$\mathrm{Ro} \ll 1$:] Rotation dominates: large-scale circulation, jet streams\\
    Earth: $U \sim 10$~m\,s$^{-1}$, $L \sim 1000$~km $\rightarrow$ $\mathrm{Ro} \sim 0.1$

    \vskip4pt
    \item[$\mathrm{Ro} \gg 1$:] Rotation unimportant: tornadoes, dust devils
  \end{description}
  \vskip8pt
  Determines the number of circulation cells on a planet.
\end{frame}

% ── Circulation cells vs rotation ───────────────────────────
\begin{frame}{Circulation Cells and Rotation Rate}
  \begin{description}
    \item[Slowly rotating] (Venus, Titan):\\
    Single Hadley cell from equator to pole in each hemisphere.\\
    Venus: 243-day period $\rightarrow$ $\mathrm{Ro} \gg 1$ for large-scale flows.

    \vskip6pt
    \item[Moderately rotating] (Earth):\\
    Hadley cell to $\sim$30\textdegree{}, then Ferrel cell (mid-latitudes), polar cell.\\
    Three cells per hemisphere.

    \vskip6pt
    \item[Rapidly rotating] (Jupiter, Saturn):\\
    Many alternating cells $\rightarrow$ \textbf{banded structure}.\\
    Jupiter: $\sim$15 alternating jet streams per hemisphere.\\
    Light zones (rising air, high clouds) and dark belts (sinking air).
  \end{description}
\end{frame}

\section{Geostrophic Balance \& Jet Streams}
% ════════════════════════════════════════════════════════════

% ── Geostrophic balance ─────────────────────────────────────
\begin{frame}{Geostrophic Balance}
  At large scales ($\mathrm{Ro} \ll 1$), steady state = Coriolis force balances pressure gradient:
  \vskip6pt
  $$f\,\hat{k} \times \mathbf{v}_g = -\frac{1}{\rho}\nabla P$$
  \vskip6pt
  In scalar form:
  $$u_g = -\frac{1}{f\rho}\pdv{P}{y}, \qquad v_g = \frac{1}{f\rho}\pdv{P}{x}$$
  \vskip6pt
  \keyresult{The geostrophic wind blows \textbf{parallel to isobars}, not from high to low pressure. In the Northern Hemisphere: low pressure to the left of the wind.}
  \vskip4pt
  This is why cyclones and anticyclones \textbf{rotate} around pressure centres.
\end{frame}

% ── Thermal wind and jet streams ─────────────────────────────
\begin{frame}{Jet Streams and the Thermal Wind}
  \textbf{Jet streams:} Narrow bands of fast winds ($\sim$30--70~m\,s$^{-1}$ on Earth).
  \vskip6pt
  The \textbf{thermal wind relation} connects vertical wind shear to horizontal temperature gradients:
  $$\pdv{\mathbf{v}_g}{z} \propto \hat{k} \times \nabla T$$
  \vskip6pt
  \begin{itemize}
    \item Where $\nabla T$ is steepest (tropics vs.\ poles boundary): wind speed increases with altitude
    \item On Earth: subtropical jet ($\sim$30\textdegree{}) and polar jet ($\sim$60\textdegree{})
    \item Jet streams \textbf{steer weather systems} across the planet
  \end{itemize}
\end{frame}

% ── Giant planet banding ────────────────────────────────────
\begin{frame}{Giant Planet Banding}
  \begin{itemize}
    \item \textbf{Zones} (light): rising air, high clouds (ammonia ice tops)
    \item \textbf{Belts} (dark): sinking air, deeper cloud layers exposed
    \item Strong \textbf{zonal jets} between adjacent bands:
    \begin{itemize}
      \item Jupiter: $\sim$150~m\,s$^{-1}$
      \item Saturn: $\sim$400~m\,s$^{-1}$
    \end{itemize}
    \item Jets remarkably stable over decades of observation
    \item Juno gravity measurements: jets extend deep into the interior
  \end{itemize}
  \vskip6pt
  \keyresult{The same geostrophic and thermal wind physics that produces Earth's jet streams creates the banded structure of the giant planets, just with many more cells due to rapid rotation.}
\end{frame}

% ── Mars dust storms ────────────────────────────────────────
\begin{frame}{Mars: Dust Storms}
  \begin{columns}[c]
    \column{0.45\textwidth}
    \centering
    \includegraphics[width=\textwidth]{mars_dust_storm}
    \source{NASA/JPL-Caltech/MSSS}

    \column{0.52\textwidth}
    \begin{itemize}

      \item Occasionally grow into \textbf{global dust storms}
      \item 2018 global storm ended \textit{Opportunity}'s 15-year mission
      \item \textbf{Positive feedback loop:}\\

% ── Venus super-rotation ────────────────────────────────────
\begin{frame}{Venus: Super-Rotation}
  \begin{itemize}
    \item Venus's surface rotates once every 243 Earth days (retrograde)
    \item Cloud-top winds: $\sim$100~m\,s$^{-1}$, circling the planet in $\sim$4 days
    \item Atmosphere rotates \textbf{60$\times$ faster than the planet}
    \item This is \textbf{atmospheric super-rotation}
  \end{itemize}
  \vskip6pt
  \begin{alertblock}{A major puzzle}
    Super-rotation requires angular momentum transport \textbf{upward and equatorward}, against friction.
    Leading explanation: thermal tides + planetary-scale waves.
    The detailed mechanism remains an active area of research.
  \end{alertblock}
\end{frame}

% ── Jupiter GRS ─────────────────────────────────────────────
\begin{frame}{Jupiter: The Great Red Spot}
  \begin{columns}[c]
    \column{0.45\textwidth}
    \centering
    \includegraphics[width=\textwidth]{jupiter_great_red_spot}
    \source{NASA/JPL-Caltech/SwRI/MSSS (Juno)}

    \column{0.52\textwidth}
    \begin{itemize}

% ── Saturn hexagon ──────────────────────────────────────────
\begin{frame}{Saturn: The Hexagonal Jet Stream}
  \begin{columns}[c]
    \column{0.42\textwidth}
    \centering
    \includegraphics[width=\textwidth]{saturn_hexagon}
    \source{NASA/JPL-Caltech/SSI (Cassini, 2013)}

    \column{0.55\textwidth}
    \begin{itemize}
      \item Persistent hexagonal pattern at $\sim$78\textdegree{}N
      \item Diameter: $\sim$30,000~km ($>$ Earth's diameter)
      \item Discovered by Voyager (1981), imaged by Cassini
      \item Explained as a stable \textbf{Rossby wave} with six-fold symmetry
      \item Resonance between jet speed, width, and Coriolis gradient
    \end{itemize}
    \vskip4pt
    Saturn also has periodic \textbf{Great White Storms} ($\sim$every 30~yr; last: 2010).
  \end{columns}
\end{frame}

% ── Neptune ─────────────────────────────────────────────────
\begin{frame}{Neptune: Extreme Weather on a Cold World}
  \begin{itemize}
    \item Receives only $\sim$1/900th of Earth's solar flux
    \item Yet has the \textbf{fastest winds} in the solar system: $\sim$580~m\,s$^{-1}$
    \item Voyager~2 (1989): Great Dark Spot (since vanished), new storms formed
    \item Storms more transient than Jupiter's
  \end{itemize}
  \vskip6pt
  \keyresult{Neptune radiates $\sim$2.6$\times$ more energy than it receives from the Sun. Internal heat (gravitational contraction) drives convection and storms even with negligible solar heating.}
  \vskip6pt
  A powerful reminder that solar flux alone does not determine weather.
\end{frame}


% ════════════════════════════════════════════════════════════
\sectionimage{mars_dust_storm}{Mars global dust storm (2018). Credit: NASA/JPL-Caltech/MSSS}
\section{Climate Evolution \& the Faint Young Sun}
% ════════════════════════════════════════════════════════════

% ── Solar luminosity evolution ──────────────────────────────
\begin{frame}{Solar Luminosity Evolution}
  The Sun brightens as H $\rightarrow$ He increases the core mean molecular weight:
  \vskip6pt
  $$\frac{L(t)}{\Lsun} \approx \left[1 + \frac{2}{5}\left(1 - \frac{t}{t_\odot}\right)\right]^{-1}$$
  \vskip6pt
  \begin{itemize}
    \item $t_\odot \approx 4.57$~Gyr (present age of the Sun)
    \item At formation ($t = 0$): $L/\Lsun \approx 0.71$
    \item The early Sun was $\sim$\textbf{30\% fainter} than today
    \item Even 4~Ga: still $\sim$25\% fainter
  \end{itemize}
\end{frame}

% ── The paradox ─────────────────────────────────────────────
\begin{frame}{The Faint Young Sun Paradox}
  A 30\% fainter Sun $\rightarrow$ $T_{\mathrm{eff}} \approx 255 \times (0.71)^{1/4} \approx 234$~K
  \vskip4pt
  With a modern greenhouse: surface well below freezing $\rightarrow$ \textbf{frozen ocean}.
  \vskip8pt
  But the geological record says otherwise:
  \vskip4pt
  \begin{itemize}
    \item \textbf{Zircon crystals} (4.4~Ga): $\delta^{18}\mathrm{O}$ ratios require liquid water
    \item \textbf{Pillow basalts and sedimentary rocks} (3.8~Ga): formed underwater
    \item \textbf{Stromatolites} (3.5~Ga): microbial mats requiring warm liquid water
  \end{itemize}
  \vskip6pt
  \keyresult{Faint Sun should have frozen Earth, but geological evidence demands liquid water for $>$4~Gyr. This is the \textbf{faint young Sun paradox} (Sagan \& Mullen, 1972).}
\end{frame}

% ── Solutions ────────────────────────────────────────────────
\begin{frame}{Possible Solutions}
  \begin{itemize}
    \item \textbf{Enhanced $\mathrm{CO_2}$ greenhouse:} 10--1000$\times$ more $\mathrm{CO_2}$ than today\\
    Plausible before the emergence of land plants

    \vskip4pt
    \item \textbf{Methane greenhouse:} Biogenic $\mathrm{CH_4}$ from early methanogens\\
    Long-lived in an anoxic (O$_2$-free) atmosphere

    \vskip4pt
    \item \textbf{$\mathrm{N_2}$ pressure broadening:} Thicker $\mathrm{N_2}$ atmosphere (2--3$\times$ present)\\
    Enhances $\mathrm{CO_2}$ and $\mathrm{H_2O}$ absorption bands

    \vskip4pt
    \item \textbf{Lower albedo:} Less land area + fewer clouds $\rightarrow$ absorb more sunlight
  \end{itemize}
  \vskip6pt
  Most likely: \textbf{combination} of elevated $\mathrm{CO_2}$ and $\mathrm{CH_4}$, regulated by the carbonate-silicate cycle.
\end{frame}

% ── Mars climate puzzle ─────────────────────────────────────
\begin{frame}{The Mars Climate Puzzle}
  Mars at 1.52~AU $\rightarrow$ less than half of Earth's solar flux. With a faint young Sun: even worse.
  \vskip6pt
  Yet the geological evidence for warm, wet conditions in the Noachian ($>$3.7~Ga) is strong:
  \begin{itemize}
    \item Extensive valley networks carved by flowing water
    \item Clay minerals from aqueous weathering
    \item Sedimentary deposits in ancient lake basins (e.g., Jezero crater)
  \end{itemize}
  \vskip6pt
  Dense $\mathrm{CO_2}$ alone is insufficient: it condenses into ice clouds that \textit{increase} albedo.
  \vskip4pt
  \textbf{Proposed solutions:} Reducing gases ($\mathrm{H_2}$, $\mathrm{CH_4}$) from volcanism and water-rock reactions. Early Mars climate remains a \textbf{major open problem}.
\end{frame}

% ── Milankovitch cycles ──────────────────────────────────
\begin{frame}{Milankovitch Cycles: Orbital Forcing}
  Periodic variations in Earth's orbit drive climate changes on $10^4$--$10^5$~yr timescales.
  \vskip6pt
  \begin{itemize}
    \item \textbf{Eccentricity:} varies between $e = 0$ and $e = 0.06$ on 100 and 400~kyr cycles
    \item \textbf{Obliquity (axial tilt):} varies between 22.1\textdegree{} and 24.5\textdegree{} on $\sim$41~kyr cycles
    \item \textbf{Precession:} the direction of the Earth's spin axis wobbles on $\sim$26~kyr
    \item These change the seasonal and latitudinal distribution of sunlight
    \item Pace the glacial-interglacial cycles of the Pleistocene
    \item Signal clearly visible in ocean sediment $\delta^{18}$O records
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Hays et al.\ (1976); Laskar et al.\ (2004), \textit{A\&A}}
\end{frame}

% ── Climate feedbacks ───────────────────────────────────────
\begin{frame}{Climate Feedbacks}
  Climate stability depends on feedback mechanisms:
  \vskip6pt
  \begin{description}
    \item[Ice-albedo feedback] (positive):\\
    Cooling $\rightarrow$ ice expands $\rightarrow$ higher albedo $\rightarrow$ less absorption $\rightarrow$ more cooling.\\
    Can trigger \textbf{snowball Earth} (Neoproterozoic, $\sim$700~Ma).

    \vskip4pt
    \item[Water vapour feedback] (positive):\\
    Warming $\rightarrow$ more evaporation $\rightarrow$ $\mathrm{H_2O}$ is a greenhouse gas $\rightarrow$ more warming.\\
    Approximately doubles CO$_2$ warming. If runaway: \textbf{Venus scenario}.

    \vskip4pt
    \item[Cloud feedback] (complex):\\
    Low clouds: reflect sunlight (cooling). High cirrus: trap IR (warming).\\
    Net effect: largest uncertainty in climate models.
  \end{description}
\end{frame}


% ════════════════════════════════════════════════════════════
\sectionimage{carbonate_silicate_cycle}{Carbonate-silicate cycle. Credit: Wikimedia, CC BY-SA 4.0}
\section{The Carbonate-Silicate Cycle}
% ════════════════════════════════════════════════════════════

% ── Urey reaction ───────────────────────────────────────────
\begin{frame}{The Urey Reaction}
  Silicate weathering by atmospheric $\mathrm{CO_2}$ dissolved in rainwater:
  \vskip8pt
  \keyresult{$$\mathrm{CaSiO_3} + \mathrm{CO_2} + \mathrm{H_2O} \longrightarrow \mathrm{CaCO_3} + \mathrm{SiO_2} + \mathrm{H_2O}$$}
  \vskip8pt
  \begin{itemize}
    \item $\mathrm{CO_2}$ dissolves in rain $\rightarrow$ weak acid
    \item Acid reacts with silicate minerals at the surface
    \item Products: limestone ($\mathrm{CaCO_3}$) and silica ($\mathrm{SiO_2}$)
    \item Transported by rivers to the ocean $\rightarrow$ deposited as sediment
    \item Net effect: \textbf{draws $\mathrm{CO_2}$ out of the atmosphere} (carbon sink)
  \end{itemize}
\end{frame}

% ── Volcanic source ─────────────────────────────────────────
\begin{frame}{Volcanic Outgassing: The Carbon Source}
  \begin{columns}[c]
    \column{0.42\textwidth}
    \centering
    \includegraphics[width=\linewidth,height=0.55\textheight,keepaspectratio]{carbonate_silicate_cycle}
    \source{Wikimedia, CC BY-SA 4.0}

    \column{0.55\textwidth}
    \begin{itemize}
      \item \textbf{Plate tectonics} closes the cycle
      \item Carbonate ocean floor is subducted into the mantle
      \item High $T$ and $P$ decompose carbonates $\rightarrow$ release $\mathrm{CO_2}$
      \item $\mathrm{CO_2}$ returned to the atmosphere via \textbf{volcanic outgassing}
      \item This is the long-term carbon \textbf{source}
    \end{itemize}
    \vskip4pt
    The cycle balances sink (weathering) against source (volcanism).
  \end{columns}
\end{frame}

% ── Negative feedback ───────────────────────────────────────
\begin{frame}{The Negative Feedback Thermostat}
  Weathering rate depends strongly on temperature (Arrhenius-type):
  \vskip6pt
  \begin{enumerate}
    \item \textbf{Planet warms:}\\
    More rainfall + faster reactions $\rightarrow$ weathering increases\\
    $\rightarrow$ more $\mathrm{CO_2}$ removed $\rightarrow$ greenhouse weakens $\rightarrow$ \textbf{cools down}

    \vskip6pt
    \item \textbf{Planet cools:}\\
    Less rainfall + slower reactions $\rightarrow$ weathering decreases\\
    $\rightarrow$ volcanic $\mathrm{CO_2}$ accumulates $\rightarrow$ greenhouse strengthens $\rightarrow$ \textbf{warms up}
  \end{enumerate}
  \vskip6pt
  \keyresult{This thermostat operates on $\sim$0.5~Myr timescales. It is the primary reason Earth has maintained habitable temperatures for $>$4~Gyr despite a 30\% increase in solar luminosity.}
\end{frame}

% ── Venus and Mars ──────────────────────────────────────────
\begin{frame}{Venus \& Mars: Failed Thermostats}
  The cycle requires \textbf{liquid water} (to weather rocks) and \textbf{active volcanism} (to recycle carbon):
  \vskip8pt
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Venus:}
    \begin{itemize}
      \item Lost surface water (runaway greenhouse + H escape)
      \item No water $\rightarrow$ no silicate weathering
      \item Volcanic $\mathrm{CO_2}$ accumulated unchecked
      \item Result: 92~bar $\mathrm{CO_2}$ atmosphere
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Mars:}
    \begin{itemize}
      \item Interior cooled, volcanism ceased ($\sim$3~Ga)
      \item No volcanic $\mathrm{CO_2}$ source
      \item Remaining $\mathrm{CO_2}$ drawn down by weathering + lost to space
      \item Result: thin 6~mbar atmosphere
    \end{itemize}
  \end{columns}
  \vskip8pt
  \keyresult{Habitability requires not just the right temperature, but the right \textbf{geological activity} to sustain it over billions of years.}
\end{frame}

\sectionimage{saturn_hexagon}{Saturn's hexagonal jet stream. Credit: NASA/JPL-Caltech/SSI}
\section{Recent Advances \& Outlook}
% ════════════════════════════════════════════════════════════

% ── Runaway greenhouse ──────────────────────────────────
\begin{frame}{The Runaway Greenhouse}
  A planet that absorbs too much stellar flux can lose its oceans entirely.
  \vskip6pt
  \begin{itemize}
    \item Water vapour is a strong greenhouse gas
    \item Warmer surface $\rightarrow$ more evaporation $\rightarrow$ more greenhouse $\rightarrow$ hotter surface
    \item Outgoing longwave radiation asymptotes to $\sim$280--350~W/m$^2$ (Simpson-Nakajima limit)
    \item If absorbed stellar flux exceeds this, the positive feedback is unstoppable
    \item Oceans fully evaporate; H$_2$O photolysed by UV; H escapes hydrodynamically
    \item End state: dry, CO$_2$-rich atmosphere (Venus today)
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Nakajima et al.\ (1992); Goldblatt et al.\ (2013)}
\end{frame}

% ── Snowball Earth ──────────────────────────────────────
\begin{frame}{Snowball Earth}
  The opposite extreme: a planet almost entirely covered by ice.
  \vskip6pt
  \begin{itemize}
    \item Earth may have gone through \textbf{two snowball states} in the Neoproterozoic ($\sim$720 and 635~Ma)
    \item Triggered by ice-albedo runaway when the ice line reaches $\sim$30\textdegree{} latitude
    \item Surface temperatures drop below $-50$\textdegree C, equatorial oceans freeze
    \item Escape: volcanic CO$_2$ accumulates slowly over $\sim$10--100~Myr
    \item Eventually greenhouse overwhelms albedo; ice melts rapidly
    \item Correlated with the rise of complex (multicellular) life shortly afterward
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Hoffman \& Schrag (2002), \textit{Terra Nova}; Pierrehumbert et al.\ (2011)}
\end{frame}

% ── Thermal tides ────────────────────────────────────────
\begin{frame}{Venus' Super-Rotation: Thermal Tides}
  How does Venus' atmosphere rotate 60 times faster than the surface?
  \vskip6pt
  \begin{itemize}
    \item Surface rotation: 243 days, retrograde
    \item Cloud-top winds: 4 days, $\sim$100~m/s
    \item Drives the atmosphere equatorward and upward from the surface
    \item \textbf{Thermal tides:} solar heating of the cloud layer creates day-night pressure gradients
    \item These generate atmospheric waves that transport angular momentum upward
    \item \textbf{Planetary-scale waves} redistribute angular momentum meridionally
    \item Combined effect: upward-equatorward flux sustains super-rotation
    \item Detailed mechanism still debated; Akatsuki measurements ongoing
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: S\'{a}nchez-Lavega et al.\ (2017)}
\end{frame}

% ── Carbonate-silicate equation detail ────────────────
\begin{frame}{Quantifying the Weathering Feedback}
  A simple model for the carbonate-silicate thermostat:
  \vskip6pt
  $$W(T) = W_0 \exp\!\left(\frac{T - T_0}{T_e}\right)$$
  \vskip6pt
  where $W$ is the weathering rate, $T_e \approx 10$~K is the $e$-folding temperature.
  \vskip6pt
  \begin{itemize}
    \item Doubling weathering requires $\sim$7~K warming
    \item Halving it requires $\sim$7~K cooling
    \item In steady state, $W = V$ (volcanism) sets atmospheric pCO$_2$
    \item Perturbation analysis: climate damping timescale $\sim$0.5~Myr
    \item Works only if liquid water and active volcanism coexist
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Walker et al.\ (1981); Pierrehumbert (2010)}
\end{frame}

% ── Neptune storms ─────────────────────────────────────
\begin{frame}{Neptune's Extreme Weather}
  Despite receiving only 1/900$^{\mathrm{th}}$ of Earth's solar flux, Neptune has the fastest winds in the solar system.
  \vskip6pt
  \begin{itemize}
    \item Wind speeds up to 580~m/s ($\sim$2100~km/h)
    \item \textbf{Great Dark Spot} discovered by Voyager 2 (1989)
    \item Disappeared by 1994 (HST observations); new dark spots have since formed
    \item Energy source: internal heat, not sunlight
    \item Neptune radiates $\sim$2.6$\times$ more power than it absorbs from the Sun
    \item Heat from slow contraction + possible compositional settling in the interior
    \item Transient storms contrast with Jupiter's long-lived GRS
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Hammel et al.\ (1989); Pearl \& Conrath (1991)}
\end{frame}

% ── Venus phosphine ─────────────────────────────────────
\begin{frame}{The Venus Phosphine Debate}
  In 2020, a team reported tentative detection of phosphine (PH$_3$) in Venus' clouds.
  \vskip6pt
  \begin{itemize}
    \item Greaves et al.\ (2020): $\sim$20~ppb PH$_3$ at 53--60~km altitude
    \item No known abiotic source in Venus' atmosphere could explain this abundance
    \item Sparked biological speculation: could PH$_3$ be produced by hypothetical Venusian microbes?
    \item Subsequent reanalyses: the signal is much weaker (Villanueva et al.\ 2021)
    \item Current status: unconfirmed, controversial
    \item Motivated new Venus missions: DAVINCI (NASA), EnVision (ESA), VERITAS (NASA)
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Greaves et al.\ (2020), \textit{Nat.\ Astron.}; Villanueva et al.\ (2021)}
\end{frame}

% ── Dragonfly ───────────────────────────────────────────
\begin{frame}{Dragonfly: A Rotorcraft for Titan}
  NASA's Dragonfly (launch 2028, arrival 2034) will explore Titan's surface.
  \vskip6pt
  \begin{itemize}
    \item Nuclear-powered rotorcraft, the first of its kind for planetary exploration
    \item Mass: $\sim$450~kg; can fly $\sim$10~km between science stops
    \item Will land in the Selk impact crater region
    \item Sampling organic-rich dunes for mass spectrometry
    \item Investigating prebiotic chemistry: does the combination of complex organics and transient impact-melt water produce biomolecules?
    \item 2.7~year primary mission, visiting 20+ sites
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Turtle et al.\ (2019)}
\end{frame}

% ── JWST exoplanet weather ─────────────────────────────
\begin{frame}{JWST: Weather on Distant Worlds}
  JWST is transforming our view of exoplanet atmospheres.
  \vskip6pt
  \begin{itemize}
    \item WASP-39 b: first clear detection of CO$_2$ in an exoplanet atmosphere (2022)
    \item WASP-18 b: sub-micron wavelength thermal emission shows dayside-to-nightside heat redistribution
    \item WASP-107 b: first firm detection of SO$_2$ $\Rightarrow$ photochemistry in action
    \item Measurements of terminator cloudiness (east-west asymmetry)
    \item Searching for weather patterns, circulation signatures
    \item First steps toward characterising habitable-zone rocky planet atmospheres
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: The JWST Transiting Exoplanet Community ERS Team (2023)}
\end{frame}

% ── Early Mars climate ─────────────────────────────────
\begin{frame}{Recent Advances: Early Mars Climate}
  \begin{itemize}
    \item Updated 3D climate models with reducing gases (H$_2$, CH$_4$) can warm early Mars
    \item Wordsworth et al.\ (2021): collisionally-induced absorption by H$_2$-CO$_2$ is a strong greenhouse effect at high CO$_2$ pressures
    \item Impact-triggered warming events: large impactors deliver heat and deflect vapour into the atmosphere
    \item Localised glaciation models: the ``icy highlands'' picture
    \item Episodic vs.\ sustained warming: still debated
    \item Perseverance (Jezero crater, 2021--) investigating the aqueous record
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Wordsworth (2016); Wordsworth et al.\ (2021)}
\end{frame}

% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}
    \item Clouds form when rising air cools past the \textbf{saturation vapour pressure} at the LCL
    \item The Clausius-Clapeyron equation gives $P_{\mathrm{sat}}(T) \propto \exp(-L_v/R_v T)$; governs all clouds
    \item Cloud species vary: $\mathrm{H_2O}$ (Earth), $\mathrm{H_2SO_4}$ (Venus), $\mathrm{NH_3}$/$\mathrm{H_2O}$ layers (Jupiter)
    \item Atmospheric circulation: Hadley cells, Coriolis deflection, jet streams
    \item Rotation rate sets number of circulation cells (1 on Venus, 3 on Earth, $\sim$15 on Jupiter)
    \item Weather: Mars dust storms, Venus super-rotation, Jupiter GRS, Saturn hexagon
    \item Faint young Sun paradox: early Sun 30\% fainter, yet Earth had liquid water
    \item Carbonate-silicate cycle: Earth's climate thermostat; failed on Venus and Mars
  \end{enumerate}
\end{frame}
