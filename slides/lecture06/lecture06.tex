\section{Cloud formation}
% ════════════════════════════════════════════════════════════

% ── Saturation ───────────────────────────────────────────────
\begin{frame}{Saturation and condensation}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Every vapour has a \textbf{saturation vapour pressure} $P_{\mathrm{sat}}(T)$ that rises roughly exponentially with temperature
    \item \textbf{Relative humidity} $\mathrm{RH} = P_{\mathrm{vapour}}/P_{\mathrm{sat}}(T) \times 100\%$: saturated at 100\%, \textbf{supersaturated} above it
    \item Rising air cools, $P_{\mathrm{sat}}$ drops, the air reaches saturation: clouds
  \end{itemize}
\end{frame}

\begin{frame}
  \ipsherokey[0.65]{psat_curves}{Saturation vapour pressure against temperature for the five main condensables. Course figure.}{A single equation, evaluated per species, predicts every cloud deck in the solar system.}
\end{frame}

\begin{frame}
  \ipsherokey[0.61]{kohler_curves}{K\"ohler curves: the equilibrium supersaturation of solution droplets against droplet radius. Course figure.}{Condensation needs a \textbf{seed}: homogeneous nucleation from vapour alone needs $\mathrm{RH} \gg 100\%$; heterogeneous nucleation on dust, sea salt or soot activates at only 0.04--0.4\% supersaturation.}
\end{frame}

% ── Lifting condensation level ──────────────────────────────

\section{Blackboard derivation: the Clausius-Clapeyron equation}
% ════════════════════════════════════════════════════════════

% ── Setup: goal and strategy ────────────────────────────────
\begin{frame}{Goal and strategy}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig[0.62\textheight]{board_sketch_l06}
    \column{\recaptextcolwide}
    \textbf{Goal:} the exponential dependence of $P_{\mathrm{sat}}$ on temperature from phase equilibrium.
    \vskip8pt
    \begin{itemize}\setlength\itemsep{2pt}
      \item Along the coexistence curve $g_\ell(T, P) = g_v(T, P)$, so a step $(\dd T, \dd P)$ keeps $\dd g_\ell = \dd g_v$
      \item Read off the slope $\dd P/\dd T$ and insert the latent heat
      \item Approximate and integrate
    \end{itemize}
    \vskip8pt
    \keyresult{\textit{To the board.}}
  \end{columns}
\end{frame}

% ── Result ──────────────────────────────────────────────────
\begin{frame}{The Clausius-Clapeyron equation}
  \keyresult{$$\boxed{P_{\mathrm{sat}}(T) = P_{\mathrm{ref}} \exp\!\left[-\frac{L_v}{R_v}\left(\frac{1}{T} - \frac{1}{T_{\mathrm{ref}}}\right)\right]}$$}
  \source{Derivation follows Catling (2017), \textit{Atmospheric Evolution on Inhabited and Lifeless Worlds}, Ch.~1}
  \vskip8pt
  \begin{itemize}\setlength\itemsep{3pt}
    \item $P_{\mathrm{sat}}$ depends \textbf{exponentially} on temperature
    \item The sensitivity is set by $L_v/R_v$ (units of K): large values mean a steep curve
    \item The equation applies to \textbf{any} vapour to condensed-phase transition
  \end{itemize}
\end{frame}

\begin{frame}
  \ipshero{water_phase_diagram}{Phase diagram of water: the liquid-vapour curve integrates Clausius-Clapeyron from the triple point to the critical point, the solid-vapour curve governs sublimation. Course figure.}
\end{frame}

% ── Worked example: water ──────────────────────────────────
\begin{frame}{Worked example: water on Earth}
  For water vapour:
  \vskip4pt
  \centering
  \begin{tabular}{l r}
    \toprule
    $L_v$ & $2.50 \times 10^6$~J\,kg$^{-1}$ \\
    $M$ & 0.018~kg\,mol$^{-1}$ \\
    $R_v = R^*/M$ & 462~J\,kg$^{-1}$\,K$^{-1}$ \\
    $T_{\mathrm{ref}}$ (triple point) & 273~K \\
    $P_{\mathrm{ref}}$ (triple point) & 611~Pa \\
    \midrule
    $L_v/R_v$ & $\approx 5400$~K \\
    \bottomrule
  \end{tabular}
  \vskip8pt
  \flushleft
  At $T = 293$~K (20\textdegree{}C):
  $$P_{\mathrm{sat}} = 611 \exp\!\left[-5400\left(\frac{1}{293} - \frac{1}{273}\right)\right] = 611 \times 3.86 \approx 2360 \text{~Pa}$$
  Measured value: 2300~Pa; excellent agreement.
\end{frame}

% ── Hadley cell ─────────────────────────────────────────────
\begin{frame}
  \ipsherokey[0.66]{hadley_observed}{The observed Hadley circulation. Course figure.}{Equatorial heating lifts air that flows poleward aloft, sinks near 30\textdegree{} latitude over the subtropical deserts and returns as the trade winds: the tropical heat transport loop.}
\end{frame}

% ── Coriolis effect ─────────────────────────────────────────
\begin{frame}{The Coriolis effect}
  Moving air is deflected right in the Northern Hemisphere, left in the Southern. The \textbf{Coriolis parameter}:
  $$f = 2\Omega \sin\phi$$
  \vskip4pt
  \begin{itemize}\setlength\itemsep{2pt}
    \item Earth: $\Omega = 7.29 \times 10^{-5}$~rad\,s$^{-1}$
    \item Mid-latitudes ($\phi = 45$\textdegree{}): $f \approx 1.03 \times 10^{-4}$~s$^{-1}$
    \item Equator: $f = 0$, no deflection
  \end{itemize}
  \vskip4pt
  \keyresult{Winds circulate around pressure centres instead of blowing from high to low.}
\end{frame}

\begin{frame}
  \ipshero{coriolis_effect}{Coriolis deflection of a polar launch seen from above the North Pole: an apparent force from the planet rotating beneath a straight path. Course figure.}
\end{frame}

% ── Rossby number ───────────────────────────────────────────
\begin{frame}{The Rossby number}
  Does planetary rotation dominate a given flow?
  \vskip6pt
  $$\mathrm{Ro} = \frac{U}{fL}$$
  \vskip4pt
  \begin{description}\setlength\itemsep{3pt}
    \item[$\mathrm{Ro} \ll 1$:] rotation dominates: large-scale circulation, jet streams (Earth: $U \sim 10$~m\,s$^{-1}$, $L \sim 1000$~km, $\mathrm{Ro} \sim 0.1$)
    \item[$\mathrm{Ro} \gg 1$:] rotation unimportant: convective plumes, tornadoes
  \end{description}
  \vskip4pt
  \keyresult{The Rossby number sets the dynamic regime and the number of circulation cells.}
\end{frame}

\begin{frame}
  \ipshero[0.76]{rossby_regimes}{Flow regimes sorted by $\mathrm{Ro} = U/(fL)$ for mid-latitude Earth: large-scale circulation where rotation dominates, dust devils and tornadoes where pressure gradients and friction govern. Course figure.}
\end{frame}

\begin{frame}

Credit: Wikimedia Commons, public domain.}{\textbf{Slow rotators} (Venus, Titan): one equator-to-pole cell per hemisphere; Earth: three cells (Hadley, Ferrel, polar); \textbf{fast rotators} (Jupiter, Saturn): many cells that appear as banded jets.}

\section{The carbonate-silicate cycle}
% ════════════════════════════════════════════════════════════

% ── Urey reaction ───────────────────────────────────────────
\begin{frame}{The Urey reaction}
  Silicate weathering draws atmospheric $\mathrm{CO_2}$ into mineral carbonates:
  \vskip6pt
  \keyresult{$$\mathrm{CaSiO_3} + \mathrm{CO_2} + \mathrm{H_2O} \longrightarrow \mathrm{CaCO_3} + \mathrm{SiO_2} + \mathrm{H_2O}$$}
  \vskip6pt
  \begin{itemize}\setlength\itemsep{3pt}
    \item $\mathrm{CO_2}$ dissolves in rain as carbonic acid ($\mathrm{H_2CO_3}$) and weathers surface silicates
    \item Rivers carry $\mathrm{Ca}^{2+}$ and $\mathrm{HCO_3^-}$ to the oceans, where carbonate ($\mathrm{CaCO_3}$) precipitates
    \item Carbon moves from atmosphere and ocean into the crust, until subduction and volcanism return it
  \end{itemize}
\end{frame}

\begin{frame}
  \ipshero{urey_reaction_pathway}{The Urey reaction pathway from rain to carbonate rock: $\mathrm{CO_2}$ drawn out of the atmosphere into the long-term carbon sink. Course schematic.}
\end{frame}

\begin{frame}
  \ipshero{foley2024_carbonate_silicate}{The carbonate-silicate cycle on the modern Earth: weathering into oceanic carbonates, subduction, and metamorphic $\mathrm{CO_2}$ returned by arc and plume volcanism. Credit: Foley (2024).}
\end{frame}

% ── Volcanic source ─────────────────────────────────────────
\begin{frame}
  \ipsherokey[0.66]{carbonate_silicate_cycle}{The carbonate-silicate cycle. Credit: Wikimedia, CC BY-SA 4.0.}{Subducted carbonate decomposes at high $T$ and $P$ and \textbf{volcanic outgassing} returns the $\mathrm{CO_2}$: the long-term steady state balances the weathering sink against the volcanic source.}
\end{frame}

% ── Weathering thermostat ──────────────────────────────────
\begin{frame}{The negative feedback thermostat}
  Silicate weathering rates rise exponentially with temperature:
  $$W(T) = W_0 \exp\!\left(\frac{T - T_0}{T_e}\right)$$
  \vskip4pt
  \begin{itemize}\setlength\itemsep{3pt}
    \item \textbf{Warming:} more rain and faster reactions draw $\mathrm{CO_2}$ down and cool the planet
    \item \textbf{Cooling:} weathering slows and volcanic $\mathrm{CO_2}$ accumulates
    \item With $T_e \approx 10$~K, doubling the weathering rate needs $\sim$7~K of warming
  \end{itemize}
  \vskip4pt
  \keyresult{Perturbations damp on $\sim$0.5~Myr timescales, maintaining liquid water for $>$4~Gyr.}
  \sourcenote{Walker et al.\ (1981); Pierrehumbert (2010)}
\end{frame}
