% ── Metal-silicate separation ──────────────────────────────
\begin{frame}
  \ipsherokey{planetary_differentiation}{Planetary differentiation: a molten body separates into a metal core and a silicate mantle. Credit: Wikimedia, CC BY-SA 4.0.}{Accretional heating and ${}^{26}\mathrm{Al}$ make a \textbf{magma ocean}; iron ($\rho \sim 7000$~kg\,m$^{-3}$) is immiscible with silicate ($\sim$3000) and its droplets sink at the Stokes speed $v_{\mathrm{Stokes}} = \frac{2}{9}\,\Delta\rho\,g\,r^2/\mu$.}
\end{frame}

\begin{frame}
  \ipsherokey[0.66]{stokes_settling}{Stokes settling of iron droplets in a magma ocean: $v \propto r^2$, 4.4~m\,s$^{-1}$ at 1~cm, about 0.5~m\,s$^{-1}$ with turbulent drag. Course figure.}{Iron rain empties a magma ocean of its metal within weeks: on geological clocks, core formation is instantaneous.}
\end{frame}

\begin{frame}{The magma ocean state, and why it is essential}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Liquid mantle} ($\mu \sim 0.1$--1~Pa\,s, $\mathrm{Ra} \gtrsim 10^{25}$): iron droplets sink at m\,s$^{-1}$, a core forms within weeks
    \item \textbf{Solid mantle} ($\mu \sim 10^{21}$~Pa\,s): a km-sized iron body needs longer than the age of the universe to reach the centre
    \item Magma ocean lifetime $10^3$--$10^8$~yr: long enough for core formation, short for planetary evolution
  \end{itemize}
  \vskip8pt
  \keyresult{Core formation happens while the planet is molten, or not at all; chondritic asteroids that never melted never formed cores.}
\end{frame}

% ── Late veneer ────────────────────────────────────────────
\begin{frame}{The late veneer}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Highly siderophile elements (Ir, Pt, Os, Au) are about 200$\times$ more abundant in Earth's mantle than full metal extraction predicts
    \item Their ratios are \textbf{chondritic}, not fractionated
    \item About 0.5\% of Earth's mass arrived \textbf{after} core formation closed; Mars and the Moon show a smaller veneer
  \end{itemize}
  \vskip8pt
  \sourcenote{Walker (2009); Day et al.\ (2016)}
\end{frame}

% ── Goldschmidt classification ─────────────────────────────
\begin{frame}
  \ipsherokey[0.62]{goldschmidt_classification}{The Goldschmidt classification of the elements by chemical affinity. Credit: Wikimedia, CC BY-SA 3.0.}{Siderophile to the metal (Fe, Ni, Pt, Au), lithophile to the silicate (Si, Mg, Al, U, Th), chalcophile to sulfide, atmophile to gas: the siderophile depletion of the mantle is \textbf{direct evidence} for core formation.}
\end{frame}

% ── Partition coefficients ────────────────────────────────
\begin{frame}{Partition coefficients, and why silicon is in the core}
  \begin{columns}[c]
    \column{0.48\textwidth}
    \centering
    $D_i^{\mathrm{met/sil}} \equiv C_i^{\mathrm{metal}} / C_i^{\mathrm{silicate}}$
    \vskip4pt
    \footnotesize
    \setlength\tabcolsep{3pt}
    \begin{tabular}{l c l}
      \toprule
      \textbf{Element} & $\boldsymbol{D^{\mathrm{met/sil}}}$ & \textbf{Behaviour} \\
      \midrule
      Ir, Os, Pt & $10^4$--$10^6$ & Highly siderophile \\
      W, Mo      & $10^2$--$10^4$ & Moderately siderophile \\
      Ni, Co     & $10^4 \to 10^2$ (1 to 50~GPa) & Moderately siderophile \\
      Fe         & $\sim$10        & Weakly siderophile \\
      Si, Mg, O  & $\ll 1$         & Lithophile \\
      \bottomrule
    \end{tabular}
    \column{0.50\textwidth}
    \small
    \begin{itemize}\setlength\itemsep{4pt}
      \item $D$ depends on pressure, temperature and \textbf{oxygen fugacity} (the effective O$_2$ partial pressure of the melt)
      \item Si is lithophile below 10~GPa, but $D_\mathrm{Si} \to 1$ at 40--60~GPa: the core holds several wt\% Si, part of its 5--10\% density deficit (Lecture 8)
    \end{itemize}
  \end{columns}
  \vskip4pt
  \keyresult{Deep magma ocean partitioning at 40--60~GPa is the standard model for Earth's core composition.}
  \sourcenote{Rubie et al.\ (2015); Fischer et al.\ (2015); Hirose et al.\ (2021); Shahar et al.\ (2026)}
\end{frame}

\begin{frame}
  \ipsherokey[0.61]{siebert2013_partition_coefficients}{Metal-silicate partition coefficients of Ni and Co against pressure; the grey band is the value the mantle's Ni and Co require. Course figure, after Siebert et al.\ (2013).}{Both curves reach the band only near 40 to 60~GPa: equilibration at $54 \pm 5$~GPa, the base of a 1000 to 1500~km deep magma ocean (Fischer et al.\ 2015). The mantle remembers the depth at which the core formed.}
\end{frame}

% ── Hf-W chronometry ──────────────────────────────────────
\begin{frame}{Hf--W chronometry: when did cores form?}
  ${}^{182}\mathrm{Hf} \rightarrow {}^{182}\mathrm{W}$, half-life 8.9~Myr. Hf is lithophile, W siderophile: a core formed while ${}^{182}\mathrm{Hf}$ was alive leaves the mantle with excess ${}^{182}\mathrm{W}$; a late core leaves it chondritic, $\varepsilon^{182}\mathrm{W} \approx 0$.
  \vskip8pt
  \centering
  \footnotesize
  \setlength\tabcolsep{4pt}
  \begin{tabular}{l l l}
    \toprule
    \textbf{Body} & $\boldsymbol{\varepsilon^{182}\mathrm{W}}$ & \textbf{Core formation timing} \\
    \midrule
    Earth & $+2$ & two-stage $\sim$30~Myr (lower limit), done by $\sim$100~Myr \\
    Mars  & $+0.4$ bulk, $+1.8$ shergottites & complete within $\sim$10 Myr \\
    Moon  & $\sim$Earth & consistent with the giant impact \\
    \bottomrule
  \end{tabular}
  \vskip8pt
  \sourcenote{Kleine et al.\ (2009); Nimmo \& Kleine (2015); Kleine \& Walker (2017)}
\end{frame}

% ── Outgassing from the magma ocean ───────────────────────
\begin{frame}{Outgassing and volatile solubility}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Speciation follows the \textbf{oxygen fugacity} $f_{\mathrm{O}_2}$: reducing gives H$_2$, CO, N$_2$, CH$_4$; oxidising gives H$_2$O, CO$_2$, N$_2$, SO$_2$
    \item Water dissolves as two OH groups per molecule, so $X_{\mathrm{H_2O}} \propto p_{\mathrm{H_2O}}^{1/2}$; CO$_2$ is about 100$\times$ less soluble
    \item A thicker steam atmosphere lets the magma keep more water: a self-limiting feedback
  \end{itemize}
  \vskip8pt
  \keyresult{A planet cannot outgas its entire water budget; the mantle redox state at solidification sets the initial atmosphere (Lectures 5--6).}
  \sourcenote{Hirschmann (2012)}
\end{frame}

\begin{frame}
  \ipsherokey[0.56]{outgassing_speciation_fO2}{Volatile partitioning between magma ocean and atmosphere against oxygen fugacity (Nicholls et al.\ 2024; bulk silicate Earth inventory of Krijt et al.\ 2023).}{The melt keeps almost all the water dissolved (over 3 oceans) while carbon fills the atmosphere at hundreds to thousands of bar: during the magma ocean stage most of the water sits in the magma.}
\end{frame}

\begin{frame}{Impact erosion and delivery}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Local:} an ordinary impactor removes at most the air above the tangent plane, $H/2R \approx 6 \times 10^{-4}$ of Earth's atmosphere per event, cumulative
    \item \textbf{Global:} even the Moon-forming impact removes only about 20\%, more with a surface ocean
    \item The same planetesimals deliver volatiles; Venus keeps about 50$\times$ more $^{36}\mathrm{Ar}$ than Earth
  \end{itemize}
  \vskip8pt
  \keyresult{Earth's air and oceans are the ledger of delivery minus erosion, closed by magma-ocean outgassing.}
  \sourcenote{Melosh \& Vickery (1989); Genda \& Abe (2003, 2005); Schlichting et al.\ (2015)}
\end{frame}

\begin{frame}
  \ipshero{roche2025_impact_loss}{Atmospheric loss in a giant impact: ejecta plumes drive near-field loss, the far-side shock breakout drives far-field loss; green particles are lost. Roche et al.\ (2025), CC BY 4.0.}
\end{frame}

% ── NC-CC dichotomy ───────────────────────────────────────
\begin{frame}{Meteoritic dichotomy: NC vs.\ CC}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Non-carbonaceous (NC)} and \textbf{carbonaceous (CC)} meteorites have distinct $^{50}\mathrm{Ti}$, $^{54}\mathrm{Cr}$ and $^{48}\mathrm{Ca}$ ratios
    \item Two reservoirs in the early nebula: \textbf{Jupiter} formed early and blocked mixing between inner and outer disk
    \item Both reservoirs fed Earth's accretion, in changing proportions over time
  \end{itemize}
  \vskip8pt
  \keyresult{The dichotomy is one of the strongest meteoritic constraints on the timing of giant planet formation.}
  \sourcenote{Kruijer et al.\ (2017); Lichtenberg et al.\ (2021)}
\end{frame}

\section{Planetary dynamos}
% ════════════════════════════════════════════════════════════

\begin{frame}
  \ipsherokey[0.55]{magnetic_shielding}{A global field as a shield: (a) a magnetised planet holds the solar wind off at the magnetopause; (b) at an unmagnetised planet sputtering and ion pickup erode the upper atmosphere. Course schematic.}{A field deflects the solar wind, limits the cosmic-ray dose, records its history in rocks and probes the interior; Mars lost its field about 4~Ga and then about 0.8~bar of atmosphere (MAVEN).}
\end{frame}

% ── Induction equation ─────────────────────────────────────
\begin{frame}{The induction equation and the three requirements for a dynamo}
  $$\pdv{\vec{B}}{t} = \underbrace{\nabla \times (\vec{v} \times \vec{B})}_{\text{advection: flow amplifies the field}} + \underbrace{\eta \nabla^2 \vec{B}}_{\text{diffusion: ohmic decay}}, \qquad \eta = \frac{1}{\mu_0 \sigma}$$
  \vskip6pt
  \begin{enumerate}\setlength\itemsep{4pt}
    \item \textbf{Electrically conducting fluid:} liquid iron alloy, or metallic hydrogen in the gas giants
    \item \textbf{Convection:} thermal (core hotter than mantle) or compositional (inner-core crystallisation)
    \item \textbf{Sufficient flow vigour:} advection beats diffusion, $\mathrm{Rm} \gtrsim 10$--100
  \end{enumerate}
\end{frame}

\begin{frame}
  \ipsherokey[0.65]{induction_rm}{(a) Advection at $\sim UB/L$ against diffusion at $\sim \eta B/L^2$; (b) $\mathrm{Rm}$ for Earth's outer core against flow speed. Course figure.}{$\mathrm{Rm} = UL/\eta$ must exceed a critical value of order 10 to 100; at $U \approx 5 \times 10^{-4}$~m\,s$^{-1}$ Earth's core reaches about 1750.}
\end{frame}

\section{Blackboard derivation: magnetic Reynolds number}
% ════════════════════════════════════════════════════════════

% ── Setup ──────────────────────────────────────────────────
\begin{frame}{Goal and strategy}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig[0.62\textheight]{board_sketch_l04}
    \column{\recaptextcolwide}
    \textbf{Goal:} does advection or diffusion win in Earth's core?
    \vskip8pt
    \begin{itemize}\setlength\itemsep{2pt}
      \item Estimate each term of the induction equation by scales
      \item Take their ratio: the field strength cancels
      \item Put in the numbers for the outer core
    \end{itemize}
    \vskip8pt
    \keyresult{A conducting shell of size $L$, flow speed $U$, diffusivity $\eta$: one dimensionless number decides.}
  \end{columns}
\end{frame}

\begin{frame}{Deriving the magnetic Reynolds number}
  Dimensional analysis of each term: advection $|\nabla \times (\vec{v} \times \vec{B})| \sim UB/L$, diffusion $|\eta \nabla^2 \vec{B}| \sim \eta B/L^2$.
  $$\mathrm{Rm} = \frac{\text{advection}}{\text{diffusion}} = \frac{UB/L}{\eta B/L^2}, \qquad B \text{ cancels:}$$
  \keyresult{$$\boxed{\mathrm{Rm} = \frac{UL}{\eta}}$$}
  \vskip4pt
  $\mathrm{Rm} \gg 1$: the field is ``frozen in'' to the flow; $\mathrm{Rm} \ll 1$: the field decays; critical $\mathrm{Rm}_c \sim 10$--100. Rm depends on the \textbf{flow}, not on the field.
\end{frame}

% ── Application: Earth ────────────────────────────────────
\begin{frame}{Application: Earth's outer core}
  \centering
  \begin{tabular}{l c r}
    \toprule
    \textbf{Parameter} & \textbf{Symbol} & \textbf{Value} \\
    \midrule
    Core radius & $L$ & $3.5 \times 10^6$~m \\
    Flow velocity & $U$ & $5 \times 10^{-4}$~m\,s$^{-1}$ \\
    Magnetic diffusivity & $\eta$ & $\sim$1~m$^2$\,s$^{-1}$ \\
    \bottomrule
  \end{tabular}
  \vskip10pt
  $$\mathrm{Rm}_\oplus = \frac{5 \times 10^{-4} \times 3.5 \times 10^6}{1} \approx 1750 \gg \mathrm{Rm}_c$$
  \vskip6pt
  \keyresult{Ohmic diffusion time $\tau_{\mathrm{ohm}} \sim L^2/\eta \approx 400{,}000$~yr $\ll$ Earth's age: the field must be \textbf{continuously regenerated}.}
\end{frame}

% ── Dynamo scaling ───────────────────────────────────────
\begin{frame}
  \ipsherokey[0.60]{christensen2006_dynamo_scaling}{Dynamo simulations collapse onto one power law that predicts Earth, Jupiter, Mercury and Ganymede. Course figure, after Christensen \& Aubert (2006).}{$B_\mathrm{dip} \propto \sqrt{\rho\mu_0}\,(F_q D)^{1/3}$ ($F_q$ buoyancy flux, $D$ shell thickness): field strength is set by the \textbf{available convective power}, not by rotation rate; Venus is absent, its dynamo off.}
\end{frame}

% ── Hf-W chronometry refinements ────────────────────────
\begin{frame}{Hf--W chronometry: beyond the first estimate}
  \begin{itemize}\setlength\itemsep{4pt}
    \item $D_W^{\mathrm{met/sil}}$ falls from about 40 at the surface to about 10 at 60~GPa: the permitted window widens to 30--100~Myr
    \item Pd--Ag and Pt--Os cross-check the timing
    \item \textbf{Disequilibrium core formation:} not all metal re-equilibrates before sinking; Earth's core formed in episodes
  \end{itemize}
  \vskip8pt
  \sourcenote{Rudge et al.\ (2010); Kleine et al.\ (2017)}
\end{frame}

% ── Archean paleomag ────────────────────────────────────
\begin{frame}{How old is Earth's magnetic field?}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Oldest confirmed field 3.4--3.5~Gyr (Archean South Africa); single zircons push it to 4.2~Gyr (Tarduno et al.\ 2015)
    \item Continuous from $\gtrsim$3.4~Ga to today: komatiites, banded iron formations
    \item Proterozoic paleointensities near modern; polar wander paths trace supercontinent cycles
  \end{itemize}
  \vskip8pt
  \keyresult{Earth's dynamo has run for at least 3.5~Gyr, possibly since the Hadean, shielding the early atmosphere while life was emerging.}
  \sourcenote{Tarduno et al.\ (2010, 2015); Biggin et al.\ (2015)}
\end{frame}

\begin{frame}{Earth's core: composition \& growth}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Inner core} (0--1220~km): solid Fe--Ni, growing at about 0.5~mm\,yr$^{-1}$, only 0.5--1~Gyr old
    \item \textbf{Outer core} (1220--3480~km): liquid Fe--Ni, 5--10\% less dense than pure Fe from light elements (S, Si, O, C, H)
    \item Inner-core crystallisation ejects light elements: \textbf{compositional convection} drives today's dynamo
  \end{itemize}
\end{frame}

\begin{frame}{Secular variation of the field}
  \begin{itemize}\setlength\itemsep{4pt}

    \item \textbf{Westward drift} at about 0.2\textdegree\,yr$^{-1}$: differential rotation between core and mantle

% ── Venus puzzle ──────────────────────────────────────────
\begin{frame}{Venus and Mercury: two puzzles}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Venus, no field:} slow rotation alone is insufficient; no inner core, and a stagnant lid that extracts core heat too slowly
    \item \textbf{Mercury, an offset dynamo:} 0.3~$\mu$T, 1\% of Earth's, offset 500~km north (MESSENGER)
    \item A thin liquid shell above a partly solid deeper core limits Mercury's convective vigour
  \end{itemize}
  \vskip8pt
  \keyresult{Venus most likely combines reduced core cooling (no plate tectonics) with no inner-core nucleation.}
  \sourcenote{Anderson et al.\ (2012) for Mercury}
\end{frame}

% ── Jupiter and Ganymede ─────────────────────────────────
\begin{frame}{Jupiter and Ganymede: two more dynamos}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Jupiter:} H$_2$ becomes \textbf{metallic} above about 100~GPa, at 0.85 Jupiter radii; the strongest field in the solar system
    \item \textbf{Ganymede:} the only moon with an intrinsic field, 0.7~$\mu$T (Galileo 1996), sustained by tidal heating or a slowly freezing core
    \item JUICE (arriving 2031) will map Ganymede's field in detail
  \end{itemize}
  \vskip8pt
  \sourcenote{Connerney et al.\ (2022); Bolton et al.\ (2017); Kivelson et al.\ (1996); Bland et al.\ (2008)}
\end{frame}

% ── Magnetopause standoff derivation ─────────────────────
\begin{frame}{Magnetopause standoff distance}
  Pressure balance at the magnetopause:
  \vskip6pt
  $$\underbrace{\frac{1}{2}\rho_{\mathrm{sw}}v_{\mathrm{sw}}^2}_{\text{solar wind ram pressure}} = \underbrace{\frac{B_{\mathrm{mp}}^2}{2\mu_0}}_{\text{magnetic pressure}}$$
  \vskip4pt
  For a dipole, $B \propto r^{-3}$, so the standoff distance scales as
  $$r_{\mathrm{mp}} \propto \left(\frac{B_0^2}{\mu_0 \rho_{\mathrm{sw}}v_{\mathrm{sw}}^2}\right)^{1/6}$$
  \vskip8pt
  \begin{itemize}\setlength\itemsep{3pt}
    \item Earth under typical solar wind: $r_{\mathrm{mp}} \approx 10\,\Rearth$; compressed to about 6~$\Rearth$ in a CME
    \item Jupiter: 50--100~$R_J$
  \end{itemize}
\end{frame}

\begin{frame}{Why aurorae happen at the poles, and Jupiter's}
  \begin{itemize}\setlength\itemsep{4pt}
    \item The dipole focuses particles toward the magnetic poles; the polar \textbf{cusps} give the solar wind direct access
    \item Electrons accelerated by tail reconnection reach 100--400~km at 1--100~keV; the lines are fingerprints of the excited species
    \item \textbf{Jupiter:} $10^{13}$--$10^{14}$~W of UV, powered by its 10~h rotation and about 1~tonne\,s$^{-1}$ of sulfur and oxygen from Io; the moons leave \textbf{auroral footprints}
  \end{itemize}
  \vskip8pt
  \keyresult{Aurorae map the power a magnetosphere dumps into its atmosphere: solar-wind coupling at Earth, rotation and Io at Jupiter.}
  \sourcenote{Connerney et al.\ (2022); Mauk et al.\ (2017)}
\end{frame}

% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}\setlength\itemsep{6pt}
    \item Magma oceans drive \textbf{metal--silicate separation}; Hf--W gives a two-stage age of about 30~Myr for Earth's core, a lower limit, with completion within about 100~Myr
    \item A dynamo needs a conducting fluid, convection and $\mathrm{Rm} \gtrsim 10$--100; Earth's $\mathrm{Rm} \approx 1750$
    \item Mars lost its dynamo between about 4.1 and 3.7~Ga and then its atmosphere; Venus has no detectable field
  \end{enumerate}
\end{frame}
