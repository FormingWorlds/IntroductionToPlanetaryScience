% ── Course description + learning outcomes (Ocasys) ─────────
\begin{frame}{What this course is about}
  \small
  A physics-based introduction to the solar system's planets and to extrasolar planetary systems: how they form and evolve, how atmospheres, surfaces, and interiors couple, and what both reveal about planetary diversity and habitability.
  \vskip4pt\relax
  \textbf{At the end of this course, you are able to:}
  {\footnotesize
  \begin{enumerate}
    \item Describe planetary-system formation and evolution, and place the solar system among exoplanets.
    \item Explain heat generation and energy transport in planetary interiors and atmospheres.
    \item Describe the structure and dynamics of atmospheres, surfaces, interiors, and magnetospheres.
    \item Apply orbital mechanics, thermodynamics, and radiative transfer to quantitative problems.
    \item Compare the major classes of solar-system bodies in formation, structure, and observables.
    \item Evaluate exoplanet detection methods and interpret demographics and characterisation data.
    \item Synthesise across these fields to assess planetary diversity and habitability.
  \end{enumerate}}
\end{frame}

% ── Three driving questions ─────────────────────────────────
\begin{frame}{The three driving questions of this course}
  \begin{enumerate}
    \item \textbf{How did our solar system form, and is it typical?}\\
      Pre-1992: zero known exoplanets. Today: $> 6{,}000$ confirmed planets in $> 4{,}500$ systems.
    \vskip4pt\relax
    \item \textbf{What determines whether a planet becomes habitable?}\\
      Venus, Earth, and Mars share a formation environment and possibly early liquid water, yet diverged across $\sim 500$ K in surface temperature and four orders of magnitude in surface pressure.
    \vskip4pt\relax
    \item \textbf{Are we alone?}\\
      JWST is now characterising atmospheres of rocky worlds, including TRAPPIST-1. The question may become \emph{partially} answerable in your careers.
  \end{enumerate}
\end{frame}

\section{What is a planet?}
% ════════════════════════════════════════════════════════════

% ── Wanderers ───────────────────────────────────────────────
\begin{frame}{Wanderers}
  \begin{columns}[T]
    \column{0.55\textwidth}
    Greek \textit{plan\=et\=es} ($\pi\lambda\alpha\nu\acute{\eta}\tau\eta\varsigma$) = ``wanderer.''
    \vskip6pt\relax
    Five naked-eye wanderers + Sun + Moon $\to$ \textbf{seven days of the week.}
    \vskip8pt\relax
    Modern additions extended the list:
    \begin{itemize}
      \item 1781: Herschel discovers \textbf{Uranus} (first telescopic planet).
      \item 1846: Galle observes \textbf{Neptune} at a position predicted by Le Verrier and Adams from Uranus's orbital perturbations.
      \item 1930: Tombaugh discovers \textbf{Pluto} at Lowell Observatory.
    \end{itemize}

    \column{0.43\textwidth}
    \textbf{Why the definition of ``planet'' came under pressure:}
    \begin{itemize}
      \item Pluto is small, icy, eccentric, inclined; crosses Neptune's orbit.
      \item 1990s--2000s: many large trans-Neptunian objects discovered.
      \item \textbf{2005: Eris} (Brown et al., comparable size to Pluto) forced the question.
      \item Either Pluto + many TNOs are planets, or Pluto is not.
    \end{itemize}
  \end{columns}
\end{frame}

% ── Pluto New Horizons ──────────────────────────────────────
\begin{frame}{New Horizons reveals Pluto}
  \begin{columns}[c]
    \column{0.50\textwidth}
    \centering
    \includegraphics[width=\linewidth, height=0.78\textheight, keepaspectratio]{pluto_new_horizons}
    \source{NASA/JHUAPL/SwRI (July 2015)}

    \column{0.47\textwidth}
    Enhanced-colour view from the 2015 flyby:
    \begin{itemize}
      \item Dark reddish \textbf{Cthulhu Macula} (left).
      \item Bright nitrogen-ice plains, western edge of \textbf{Tombaugh Regio} (the ``heart'').
      \item Water-ice mountains, glacial flows, possible cryovolcanism.
      \item Unexpectedly active for a body of Pluto's size.
    \end{itemize}
    \vskip4pt\relax
    Reignited the planet-status debate \emph{after} the 2006 IAU vote.
  \end{columns}
\end{frame}

% ── IAU definition ──────────────────────────────────────────
\begin{frame}{IAU 2006 definition of a planet}
  \begin{block}{Resolution 5A: a \textbf{planet} is a body that}
    \begin{enumerate}
      \item is in orbit around the Sun;
      \item has sufficient self-gravity to assume hydrostatic (nearly round) equilibrium;
      \item has \textbf{cleared the neighbourhood} around its orbit.
    \end{enumerate}
  \end{block}
  \vskip6pt\relax
  \begin{itemize}
    \item Satisfies (1) + (2) but not (3) $\to$ \textbf{dwarf planet.}
    \item Solar system: \textbf{8 planets}; \textbf{5 officially recognised dwarf planets} (Ceres, Pluto, Eris, Makemake, Haumea); many additional Kuiper-belt candidates.
  \end{itemize}
\end{frame}

\begin{frame}{Is the IAU definition the right one?}
  \textbf{Critique of (3) ``cleared the neighbourhood''}:
  \vskip4pt\relax
  \begin{itemize}
    \item Not precisely defined.
    \item Depends on heliocentric distance: Earth itself would \emph{not} clear its zone if placed at Neptune's orbit.
  \end{itemize}
  \vskip6pt\relax
  \textbf{Geophysical alternative} (Runyon et al.\ 2017): any body massive enough to be in hydrostatic equilibrium counts as a planet, regardless of dynamics.
  \vskip4pt\relax
  \begin{itemize}
    \item Solar system would have $> 100$ planets under this rule.
  \end{itemize}
  \vskip6pt\relax
  \keyresult{For this course: the exact classification matters less than the \emph{physics}. The solar system is a \emph{continuous spectrum} of objects from dust grains to gas giants, all governed by the same laws.}
\end{frame}

\section{Brief history of planetary science}
% ════════════════════════════════════════════════════════════

% ── Ancient ─────────────────────────────────────────────────
\begin{frame}{Ancient and pre-telescopic astronomy}
  \begin{itemize}
    \item \textbf{Babylonian astronomers ($\sim$1800 BCE):} systematic planetary-position records; arithmetical prediction of conjunctions and oppositions.
    \item \textbf{Aristotle ($\sim$350 BCE):} geocentric system of nested spheres.
    \item \textbf{Ptolemy ($\sim$150 CE):} geocentric model with \emph{epicycles}, the standard framework for $> 1{,}400$ years.
  \end{itemize}
  \vskip6pt\relax
  \textbf{The Copernican revolution (1543):}
  \begin{itemize}
    \item \textbf{Copernicus}: heliocentric model in \emph{De revolutionibus}.
    \item \textbf{Kepler} (1609--1619): three empirical laws of planetary motion; circles replaced by ellipses.
    \item \textbf{Newton} (1687): all three Kepler laws follow from a single law of universal gravitation, the first grand unification in physics.
  \end{itemize}
\end{frame}

% ── Copernican diagram ──────────────────────────────────────
\begin{frame}{Copernicus, \emph{De revolutionibus} (1543)}
  \begin{columns}[c]
    \column{0.46\textwidth}
    \centering
    \includegraphics[width=\linewidth, height=0.78\textheight, keepaspectratio]{copernican_system}
    \source{Public domain}

    \column{0.51\textwidth}
    Reading the woodcut from the centre out:
    \begin{itemize}
      \item \emph{Sol} (Sun) at the centre
      \item \emph{Mercurius}, \emph{Venus}
      \item \emph{Terra} (Earth) with the Moon as a small attached orbit
      \item \emph{Mars}, \emph{Iupiter}, \emph{Saturnus}
      \item Outer band: \emph{Stellarum Fixarum Sphaera Immobilis} (the immobile sphere of the fixed stars)
    \end{itemize}
    \vskip4pt\relax
    Conceptual shift from geocentric cosmology, dominant for over a millennium.
  \end{columns}
\end{frame}

% ── Telescopic era ──────────────────────────────────────────
\begin{frame}{The telescopic era opens (1610)}
  \begin{columns}[c]
    \column{0.36\textwidth}
    \centering
    \includegraphics[width=\linewidth, height=0.74\textheight, keepaspectratio]{sidereus_nuncius}
    \source{Galileo, \emph{Sidereus Nuncius} (1610). Public domain}

    \column{0.61\textwidth}
    Galileo's telescope turned planetary science from a \emph{mathematical} into a \emph{physical} science.
    \vskip2pt\relax
    \begin{itemize}
      \item Discovered four moons orbiting Jupiter (the \emph{Medicea Sidera}; figure rows record successive nights).
      \item Resolved the phases of Venus, confirming heliocentrism.
      \item Resolved Saturn's rings, but could not interpret them.
    \end{itemize}
    \vskip2pt\relax
    Marius (1614) proposed the names \textbf{Io, Europa, Ganymede, Callisto} (L11).
    \vskip2pt\relax
    Later centuries: \textbf{spectroscopy} (19th c.) opened remote composition; \textbf{photography} enabled systematic surveys.
  \end{columns}
\end{frame}

% ── Space age opens ─────────────────────────────────────────
\begin{frame}{The space age opens}
  \textbf{14 December 1962:} Mariner 2 flies past Venus.
  \begin{itemize}
    \item First successful spacecraft visit to another planet.
    \item Measured Venus's extreme surface temperature, overturning earlier ``habitable Venus'' speculations.
  \end{itemize}
  \vskip6pt\relax
  \textbf{The pace accelerates:}
  \begin{itemize}
    \item 1965: Mariner 4 returns the first close-up images of Mars (cratered, Moon-like).
    \item 1970: Venera 7 makes the first successful planetary landing (Venus).
    \item 1976: NASA Viking landers conduct first life-detection experiments on Mars.
    \item 1977--1989: \textbf{Voyager grand tour} of Jupiter, Saturn, Uranus, Neptune.
  \end{itemize}
\end{frame}

% ── Mariner 4 ───────────────────────────────────────────────
\begin{frame}{Mariner 4 ends the Mars-canals era (1965)}
  \begin{columns}[c]
    \column{0.42\textwidth}
    \centering
    \includegraphics[width=\linewidth, height=0.78\textheight, keepaspectratio]{mariner4_mars}
    \source{NASA/JPL (15 July 1965)}

    \column{0.55\textwidth}
    First close-up image of another planet's surface.
    \vskip6pt\relax
    \begin{itemize}
      \item 21 frames returned during the brief Mars encounter.
      \item Heavily cratered, Moon-like terrain.
      \item \textbf{Decisively ended} speculation about Martian canals and surface vegetation.

\section{Blackboard derivation: solar mass}
% ════════════════════════════════════════════════════════════

% ── Setup ───────────────────────────────────────────────────
\begin{frame}{Setup}
  Planet of mass $M_p$ in a \textbf{circular} orbit of radius $r$ around a star of mass $M_*$.
  \vskip6pt\relax
  Gravitational attraction provides the centripetal acceleration:
  \vskip4pt\relax
  \[
    \frac{G M_* M_p}{r^2} \;=\; \frac{M_p v^2}{r}
  \]
  \vskip4pt\relax
  with orbital velocity
  \[
    v = \frac{2 \pi r}{P}, \qquad P = \text{orbital period}.
  \]
\end{frame}

% ── Derivation step 1 ───────────────────────────────────────
\begin{frame}{Substitute and cancel}
  Substituting $v = 2\pi r / P$:
  \[
    \frac{G M_* M_p}{r^2} \;=\; \frac{M_p}{r}\left(\frac{2 \pi r}{P}\right)^{\!2}
  \]
  Cancelling $M_p$ on both sides:
  \[
    \frac{G M_*}{r^2} \;=\; \frac{4 \pi^2 r}{P^2}
  \]
  \vskip4pt\relax
  \textbf{Crucial:} the planet's mass cancels. The orbital period depends only on the \emph{central} mass and the orbital radius.
\end{frame}

% ── Solve ───────────────────────────────────────────────────
\begin{frame}{Newton's form of Kepler's third law}
  Solving for $M_*$:
  \vskip8pt\relax
  \[
    \boxed{\; M_* \;=\; \frac{4 \pi^2 r^3}{G P^2} \;}
  \]
  \vskip8pt\relax
  Valid in the limit $M_p \ll M_*$.
  \vskip6pt\relax
  For \emph{elliptical} orbits, replace $r$ with the semi-major axis $a$. General case requires the \textbf{vis-viva equation}, covered in L2.
\end{frame}

% ── Apply ───────────────────────────────────────────────────
\begin{frame}{Apply to the Sun-Earth system}
  \centering
  \footnotesize
  \begin{tabular}{ll}
    \toprule
    Quantity & Value \\
    \midrule
    Semi-major axis $a_\oplus$ & $1.496 \times 10^{11}$ m \\
    Orbital period $P_\oplus$  & $3.156 \times 10^{7}$ s \\
    Newton's constant $G$      & $6.674 \times 10^{-11}$ m$^3$ kg$^{-1}$ s$^{-2}$ \\
    \bottomrule
  \end{tabular}
  \vskip8pt\relax
  \normalsize
  \[
    \Msun
    \;=\; \frac{4 \pi^2 \,(1.496 \times 10^{11})^{3}}
                {6.674 \times 10^{-11}\,(3.156 \times 10^{7})^{2}}
    \;\approx\; 1.99 \times 10^{30}\ \text{kg}.
  \]
  \vskip6pt\relax
  Accepted value: $\Msun = 1.989 \times 10^{30}$ kg. \textbf{Agreement to three figures from two measurable quantities.}
\end{frame}

% ── Mass ratio ──────────────────────────────────────────────
\begin{frame}{Why the approximation $M_* + M_p \approx M_*$ is justified}
  More precisely, $P^2 = 4 \pi^2 a^3 / [\,G (M_* + M_p)\,]$.
  \vskip6pt\relax
  Test on the heaviest planet:
  \begin{itemize}
    \item Jupiter: $\Mjup \approx 318\,\Mearth \approx 1.90 \times 10^{27}$ kg
    \item $\Mjup / \Msun \approx 9.5 \times 10^{-4}$
  \end{itemize}
  All eight planets together:
  \begin{itemize}
    \item $M_{\mathrm{planets}} \approx 446\,\Mearth \approx 2.7 \times 10^{27}$ kg
    \item $M_{\mathrm{planets}} / \Msun \approx 1.3 \times 10^{-3}$
  \end{itemize}
  \vskip4pt\relax
  \keyresult{The Sun contains \textbf{99.86\%} of the solar system's mass. Jupiter alone accounts for \textbf{71\%} of the planetary mass.}
\end{frame}

% ── Motivation ──────────────────────────────────────────────
\begin{frame}{Why comparative planetology?}
  We cannot perform controlled experiments on entire worlds.
  \vskip6pt\relax

  \vskip8pt\relax

    \item planetary mass (gravity, internal heat budget)
    \item internal activity (tectonics, dynamos)

  \vskip6pt\relax

\begin{frame}{Remote-sensing methods}
  \begin{itemize}
    \item \textbf{Spectroscopy:} characteristic absorption / emission features identify materials. Reveals atmospheric composition, surface mineralogy, thermal properties.
    \item \textbf{Radar:} penetrates clouds, maps surface topography (\emph{Magellan} at Venus), detects subsurface structures (\emph{MARSIS} at Mars).

    \item \textbf{Magnetometry:} magnetic-field measurements constrain dynamo activity and interior conductivity.
  \end{itemize}
\end{frame}

% ── JWST + TRAPPIST opens ───────────────────────────────────
\begin{frame}{JWST is now characterising rocky-planet atmospheres}
  \textbf{TRAPPIST-1}: nearby ultra-cool dwarf star, $\sim 12$ pc from Earth.
  \vskip4pt\relax
  \begin{itemize}
    \item Seven roughly Earth-sized planets (b through h).
    \item Chain of \textbf{six mean-motion resonances} locking orbital-period ratios into integer relations.
    \item Three planets (e, f, g) inside conservative HZ; a fourth (d) near its inner edge.
    \item JWST is testing whether any of them retain atmospheres.
  \end{itemize}
  \vskip4pt\relax
  Greene et al.\ (2023) reported the first such measurement, for TRAPPIST-1 b.
\end{frame}

\begin{frame}{Q1. How do planetary systems form?}
  Clues we have already met:

    \item Dust gaps in HL Tau (ALMA).

  \begin{itemize}

    \item \textbf{L3}: energy budget of accretion; short-lived radionuclides.
    \item \textbf{L12}: small bodies that preserve the formation epoch (asteroids, comets, KBOs).
  \end{itemize}
\end{frame}

% ── Q2 ──────────────────────────────────────────────────────
\begin{frame}{Q2. What makes a planet habitable?}
  Venus, Earth, Mars share an origin and yet diverge by four orders of magnitude in pressure and 500 K in temperature.
  \vskip6pt\relax
  Coupled ingredients to disentangle:
  \begin{itemize}
    \item \textbf{Interiors and dynamos} (L4, L8): heat and volatile fluxes from depth.
    \item \textbf{Atmospheres and climate} (L5, L6): surface $T$, escape, available chemistry.
    \item \textbf{Surfaces} (L7): closing the volatile cycle back into the interior.
  \end{itemize}
  \vskip4pt\relax
  Case studies put numbers on each ingredient:
  \begin{itemize}
    \item \textbf{L9}: Earth, Venus.
    \item \textbf{L10}: Mercury, Mars.
  \end{itemize}
\end{frame}

% ── Q3 ──────────────────────────────────────────────────────
\begin{frame}{Q3. Does life exist beyond Earth?}
  Motivates much of the current mission queue.
  \vskip4pt\relax
  \begin{itemize}
    \item JWST atmospheric spectra of rocky M-dwarf planets (e.g.\ TRAPPIST-1 b dayside).
    \item JUICE and Europa Clipper at the icy moons.
    \item Dragonfly at Titan.
    \item ELT, GMT, TMT for direct imaging in the 2030s.
  \end{itemize}
  \vskip4pt\relax
  Mapped onto the course:
  \begin{itemize}
    \item \textbf{L11}: giant planets + ocean worlds in their moon systems.
    \item \textbf{L13}: solar system inside the exoplanet population; habitable zone revisited.
    \item \textbf{L14}: biosignatures, Drake equation, search programmes of the next two decades.
  \end{itemize}
\end{frame}

\begin{frame}{Next time: Lecture 2}
  This lecture has been a panoramic tour: bodies, history, missions, three questions.
  \vskip6pt\relax
  \textbf{L2 begins the actual physics.} We walk back to the molecular cloud and follow the collapse step by step:
  \begin{itemize}
    \item Angular momentum conservation $\to$ rotating disk rather than spherical infall.
    \item Disk temperature gradient $\to$ the \textbf{snow line} separating rocky and icy material.
    \item Dust grains $\to$ planetesimals via coagulation, settling, the streaming instability.
    \item Kepler's laws + the \textbf{vis-viva equation} + orbital resonances govern subsequent dynamics.
  \end{itemize}

  The story of how the solar system came to look the way it does begins not with the planets but with the disk that made them.

% ── Summary ─────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}
    \item \textbf{Planetary science is comparative}: studying planets, moons, and small bodies as a set of natural experiments.
    \item \textbf{Three driving questions} organise the course: how systems form, what makes them habitable, and whether we are alone.
    \item \textbf{The solar system has structure}: density falls with distance, mass is concentrated in Jupiter + Saturn, the Sun holds 99.86\% of the total.
    \item \textbf{Newton's form of Kepler's third law} ($M_* = 4\pi^2 r^3 / G P^2$) recovers $\Msun$ to three figures from Earth's $a$ and $P$.
    \item \textbf{Comparative planetology applies to exoplanets too}: $> 6{,}000$ confirmed; bimodal small-planet radius distribution; first $\mathrm{CO_2}$ detections and first rocky M-dwarf thermal measurements landing now.
  \end{enumerate}
\end{frame}
