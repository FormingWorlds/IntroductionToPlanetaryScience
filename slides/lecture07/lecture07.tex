\section{Surface processes}
% ════════════════════════════════════════════════════════════

% ── Endogenic vs exogenic ─────────────────────────────────
\begin{frame}{Endogenic vs.\ exogenic processes}
  A planet's surface is its geological record: billions of years of competing processes that create, modify, and destroy landforms.
  \vskip8pt
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Endogenic} (internal heat):
    \begin{itemize}
      \item Volcanism
      \item Tectonics (faulting, folding)
      \item More vigorous for larger, hotter bodies
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Exogenic} (external forces):
    \begin{itemize}
      \item Impact cratering
      \item Erosion (wind, water, ice)
      \item Space weathering, radiation
    \end{itemize}
  \end{columns}
  \vskip8pt
  The balance between these processes determines what we see on a surface.
\end{frame}

% ── Dominant processes table ──────────────────────────────
\begin{frame}{Dominant surface process by body}
  \centering
  \small
  \begin{tabular}{l c c}
    \toprule
    \textbf{Body} & \textbf{Dominant process} & \textbf{Key evidence} \\
    \midrule
    Moon   & Impact cratering & Saturated highland craters, $>$4~Gyr \\
    Earth  & Plate tectonics + erosion & Ocean floors $<$200~Myr; rapid weathering \\
    Mars   & Volcanism + impacts & Olympus Mons; cratered southern highlands \\
    Venus  & Volcanism & Volcanic plains cover $\sim$80\% of surface \\
    Io     & Tidal volcanism & $\sim$300--400 active volcanoes; age $<$1~Myr \\
    Europa & Ice tectonics & Lineae, chaos terrain, very few craters \\
    \bottomrule
  \end{tabular}
  \vskip6pt
  Internal heat $\rightarrow$ endogenic processes dominate.\\
  Dead interior $\rightarrow$ surface preserves ancient craters.
\end{frame}

\section{Impact cratering}
% ════════════════════════════════════════════════════════════

% ── Universality ──────────────────────────────────────────
\begin{frame}{The most universal geological process}
  \begin{itemize}
    \item \textbf{Every} solid body in the solar system bears impact craters
    \item On bodies without atmospheres or geology (Moon, Mercury), craters survive billions of years
    \item Impact velocities: 10--70~km\,s$^{-1}$ (set by orbital mechanics)
    \item Enormous energy release in a fraction of a second
  \end{itemize}
  \vskip6pt
  Three stages of crater formation:
  \vskip4pt
  \begin{enumerate}
    \item Contact and compression
    \item Excavation
    \item Modification
  \end{enumerate}
\end{frame}

% ── Stage 1: Contact and compression ─────────────────────
\begin{frame}{Stage 1: contact and compression}
  \begin{itemize}
    \item Projectile strikes at 10--70~km\,s$^{-1}$
    \item Shock wave propagates into both target and impactor
    \item Peak pressures $\sim$100~GPa (comparable to Earth's core--mantle boundary)
    \item Material near impact point is \textbf{vaporised and melted}
    \item Kinetic energy $\rightarrow$ internal energy in $< 1$~s
  \end{itemize}
  \vskip8pt
  The kinetic energy of an impactor:
  $$E_k = \frac{1}{2}mv^2$$
\end{frame}

% ── Worked energy example ─────────────────────────────────
\begin{frame}{Example: a 1~km asteroid impact}
  Rocky asteroid: $\rho \approx 3000$~kg\,m$^{-3}$, diameter = 1~km
  $$m = \frac{4}{3}\pi r^3 \rho \approx \frac{4}{3}\pi (500)^3 \times 3000 \approx 1.6 \times 10^{12}\;\text{kg}$$
  \vskip4pt
  At $v = 20$~km\,s$^{-1}$:
  $$E_k = \frac{1}{2} \times 1.6 \times 10^{12} \times (2 \times 10^4)^2 \approx 3 \times 10^{20}\;\text{J}$$
  \vskip6pt
  \keyresult{$\sim$1000$\times$ the energy of the largest nuclear weapon ever detonated (Tsar Bomba, $\sim$50~Mt $\approx 2\times10^{17}$~J), released in $<$1~s at a single point.}
\end{frame}

% ── Stages 2 and 3 ───────────────────────────────────────
\begin{frame}{Stages 2 \& 3: excavation and modification}
  \textbf{Excavation:}
  \begin{itemize}
    \item Shock wave expands hemispherically outward
    \item Rarefaction wave (decompression pulse) accelerates material upward and outward
    \item \textbf{Transient cavity} forms (depth:diameter $\approx$ 1:3)
    \item Cone-shaped \textbf{ejecta blanket} deposited around the crater
  \end{itemize}
  \vskip6pt
  \textbf{Modification:}
  \begin{itemize}
    \item Transient cavity is gravitationally unstable
    \item Small craters: walls slump inward $\rightarrow$ \textbf{simple crater} (bowl-shaped)
    \item Large craters: floor rebounds, walls collapse in terraces $\rightarrow$ \textbf{complex crater} with central peak
  \end{itemize}
\end{frame}

\section{Blackboard derivation: the crater scaling law}
% ════════════════════════════════════════════════════════════

% ── Setup ─────────────────────────────────────────────────
\begin{frame}{Setup: crater size from dimensional analysis}
  \textbf{Goal:} How does crater diameter $D$ depend on impact energy $E$, target density $\rho$, and surface gravity $g$?
  \vskip10pt
  In the \textbf{gravity regime} (craters $\gtrsim$100~m, where gravity limits size):
  $$D = C \, E^a \, \rho^b \, g^c$$
  \vskip6pt
  Dimensions ($M$, $L$, $T$):
  \vskip4pt
  \begin{tabular}{l l}
    $[D]$ & $= L$ \\
    $[E]$ & $= M L^2 T^{-2}$ \\
    $[\rho]$ & $= M L^{-3}$ \\
    $[g]$ & $= L T^{-2}$ \\
  \end{tabular}
\end{frame}

% ── Dimensional analysis ──────────────────────────────────
\begin{frame}{Solving for the exponents}
  Require $[E^a \rho^b g^c] = L$:
  \vskip6pt
  \begin{tabular}{l l}
    \textbf{Mass:} & $a + b = 0 \implies b = -a$ \\[4pt]
    \textbf{Time:} & $-2a - 2c = 0 \implies c = -a$ \\[4pt]
    \textbf{Length:} & $2a - 3b + c = 1$ \\
  \end{tabular}
  \vskip8pt
  Substituting $b = -a$ and $c = -a$:
  $$2a + 3a - a = 1 \implies 4a = 1 \implies a = \frac{1}{4}$$
  \vskip2pt
  Therefore $a = 1/4$, $b = -1/4$, $c = -1/4$:
  \keyresult{$$\boxed{D \sim \left(\frac{E}{\rho g}\right)^{1/4}}$$}
\end{frame}

% ── Interpretation ────────────────────────────────────────
\begin{frame}{Interpretation of the scaling law}
  \begin{itemize}
    \item $D$ scales as the \textbf{fourth root} of energy
    \item Doubling the energy increases $D$ by only $2^{1/4} \approx 1.19$ (19\%)
    \item This explains the relatively narrow range of crater sizes even though impactor energies span many orders of magnitude
  \end{itemize}
  \vskip6pt
  \textbf{Dimensional check:}
  $$\left[\frac{E}{\rho g}\right] = \frac{M L^2 T^{-2}}{M L^{-3} \cdot L T^{-2}} = L^4 \quad \Rightarrow \quad [D] = L$$
  \vskip6pt
  The more complete \textbf{pi-scaling framework} (Holsapple \& Schmidt, 1993) parameterises the transition between the gravity and strength regimes.
\end{frame}

% ── Strength vs gravity regime ────────────────────────────
\begin{frame}{Where the fourth-root law applies}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{holsapple1993_strength_gravity_regimes}
    \source{Holsapple (1993), Fig.~3}

    \column{\recaptextcol}
    \footnotesize
    Cratering efficiency $\pi_V = \rho V/m$ against gravity-scaled size $\pi_2 = ga/U^2$:
    \begin{itemize}
      \item \textbf{Strength regime} (left): target cohesion limits growth, so $\pi_V$ depends on impact velocity and the three curves separate
      \item \textbf{Gravity regime} (right): gravity limits growth, the curves converge on one law, $\pi_V \propto \pi_2^{-\alpha}$
      \item Laboratory impacts sit on the left, planetary craters on the right
    \end{itemize}
    \vskip6pt
    \keyresult{Our $D \sim (E/\rho g)^{1/4}$ is the gravity-regime limit. Below $\sim$100~m on the Moon, strength takes over and the law fails.}
  \end{columns}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{holsapple1993_strength_gravity_regimes}}\\
  {\footnotesize The two cratering regimes for a target with strength, at impact velocities $U = 2.5$, 10 and 40~km\,s$^{-1}$. Reproduced from Holsapple (1993), Fig.~3.}
\end{frame}

% ── Worked example: Moon ──────────────────────────────────
\begin{frame}{Worked example: a 1~km asteroid on the Moon}
  \centering
  \begin{tabular}{l c r}
    \toprule
    \textbf{Parameter} & \textbf{Symbol} & \textbf{Value} \\
    \midrule
    Impact energy   & $E$    & $3 \times 10^{20}$~J \\
    Regolith density & $\rho$ & 2500~kg\,m$^{-3}$ \\
    Surface gravity  & $g$    & 1.62~m\,s$^{-2}$ \\
    \bottomrule
  \end{tabular}
  \vskip4pt
  {\footnotesize Regolith: the loose, fragmented debris layer covering the surface.}
  \vskip6pt
  $$D \sim \left(\frac{3 \times 10^{20}}{2500 \times 1.62}\right)^{1/4} = (7.4 \times 10^{16})^{1/4} \approx 16.5\;\text{km}$$
  \vskip6pt
  Consistent with observed lunar craters from $\sim$1~km impactors.\\
  Since $D \propto L^{3/4}$ at fixed $v$ and $\rho$, the 85~km crater Tycho implies a $\sim$9~km impactor (order of magnitude only).
\end{frame}

\section{Crater morphology and chronology}
% ════════════════════════════════════════════════════════════

% ── Morphological classes ─────────────────────────────────
\begin{frame}{Crater morphology: three classes}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{crater_morphology}
    \source{NASA/GSFC/ASU}

    \column{\recaptextcol}
    \small
    \begin{enumerate}
      \item \textbf{Simple} ($D < D_t$)\\
      Bowl-shaped; depth:diameter $\approx$ 1:5\\
      Moon: $D_t \approx 15$~km
      \vskip4pt
      \item \textbf{Complex} ($D_t < D \lesssim 300$~km)\\
      Central peak, terraced walls, flat floor\\
      Gravity overcomes wall strength
      \vskip4pt
      \item \textbf{Multi-ring basins} ($D \gtrsim 300$~km)\\
      Concentric ring structures\\
      Orientale (930~km), Caloris (1550~km), Hellas (2300~km)
    \end{enumerate}
  \end{columns}
\end{frame}

% ── Caloris basin ─────────────────────────────────────────
\begin{frame}{Multi-ring basins: Caloris on Mercury}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{mercury_caloris_basin}
    \source{NASA/Johns Hopkins APL/Carnegie Institution of Washington}

    \column{\recaptextcol}
    \footnotesize
    Caloris is $\sim$1550~km across, one of the largest and best preserved multi-ring basins:
    \begin{itemize}
      \item Orange interior plains are smooth volcanic deposits emplaced \emph{after} the impact
      \item The surrounding annulus holds the ejecta and the concentric rings
      \item Chaotic ``weird terrain'' sits at the basin's antipode, focused there by the seismic shock of the same event
    \end{itemize}
    \vskip6pt
    \keyresult{A basin this size is not a scaled-up crater: the ring structure and the antipodal terrain are signatures only the largest impacts produce.}
  \end{columns}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{mercury_caloris_basin}}\\
  {\footnotesize The Caloris basin on Mercury in enhanced colour, $\sim$1550~km in diameter. Credit: NASA/Johns Hopkins APL/Carnegie Institution of Washington.}
\end{frame}

% ── Transition diameter ───────────────────────────────────
\begin{frame}{Transition diameter}
  The simple-to-complex transition scales with surface gravity:
  $$D_t \approx D_{t,\mathrm{Moon}} \cdot \frac{g_{\mathrm{Moon}}}{g}$$
  \vskip6pt
  \centering
  \begin{tabular}{l c c}
    \toprule
    \textbf{Body} & $\boldsymbol{g}$ \textbf{(m\,s$^{-2}$)} & $\boldsymbol{D_t}$ \textbf{(km)} \\
    \midrule
    Moon    & 1.62  & 15 \\
    Mars    & 3.72  & $\sim$6.5 \\
    Earth   & 9.81  & $\sim$2.5 \\
    Mercury & 3.70  & $\sim$6.6 \\
    \bottomrule
  \end{tabular}
  \vskip6pt
  On Earth, virtually all craters larger than a few km are complex.
\end{frame}

% ── Size-frequency distribution ──────────────────────────
\begin{frame}{Crater size-frequency distribution}
  The number of craters larger than a given diameter $D$ follows an approximate power law:
  \vskip6pt
  \keyresult{$$N(>D) \propto D^{-b}, \quad b \approx 2\text{--}3$$}
  \vskip6pt
  \begin{itemize}
    \item Small craters are much more numerous than large ones
    \item Slope $b$ depends on impactor size distribution and erosion
    \item A plot of $\log N$ vs.\ $\log D$ yields a straight line for a pristine surface
    \item Higher cumulative density $\Rightarrow$ older surface
    \item Deviations from the power law indicate resurfacing, saturation, or secondary craters
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Neukum et al.\ (2001), \textit{Space Sci.\ Rev.}}
\end{frame}

% ── Crater counting principle ─────────────────────────────
\begin{frame}{Crater counting: the principle}
  \textbf{Key idea:} Older surfaces have had more time to accumulate craters.
  \vskip8pt
  The \textbf{crater size-frequency distribution (SFD)} follows a power law:
  $$N(>D) = a\,D^{-b}$$
  \vskip4pt
  \begin{itemize}
    \item $N(>D)$: cumulative number of craters per unit area with diameter $> D$
    \item Production population: $b \approx 2$--3
    \item Plot $N$ vs.\ $D$ on log-log axes $\rightarrow$ straight line
    \item Higher line $=$ older surface
  \end{itemize}
  \vskip4pt
  \keyresult{Crater counting gives \textbf{relative ages}. Apollo/Luna samples provide \textbf{absolute calibration}.}
\end{frame}

% ── Lunar calibration ─────────────────────────────────────
\begin{frame}{The lunar chronology}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{crater_counting}
    \source{NASA/GSFC/ASU}

    \column{\recaptextcolwide}
    Apollo and Luna samples give radiometric ages for specific surfaces:
    \vskip4pt
    \begin{itemize}
      \item \textbf{Highlands:} saturated, $>$4~Gyr
      \item \textbf{Maria:} 3.9--3.1~Gyr
      \item Calibrated SFD $\rightarrow$ absolute ages
    \end{itemize}
    \vskip6pt
    Extrapolated to other bodies:
    \begin{itemize}
      \item Mars southern highlands: $\sim$4~Gyr
      \item Mars northern lowlands: much younger
      \item Venus: uniform crater density $\rightarrow$ $\sim$300--700~Myr
    \end{itemize}
  \end{columns}
\end{frame}

% ── The chronology function ───────────────────────────────
\begin{frame}{The chronology function}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{neukum_chronology}
    \source{Course figure, after Neukum, Ivanov and Hartmann (2001)}

    \column{\recaptextcolwide}
    \footnotesize
    $$N(1) = a\left[e^{\lambda T} - 1\right] + b\,T$$
    \begin{itemize}
      \item Linear term $bT$: the steady impact flux of the last $\sim$3~Gyr
      \item Exponential term: the early bombardment, which overtakes the linear term near 3.6~Gyr
      \item Read it backwards: a measured $N(1) = 10^{-2}$~km$^{-2}$ gives a surface age of $\sim$3.7~Gyr
    \end{itemize}
    \vskip6pt
    \keyresult{The shaded bands hold no dated sample: 1--3~Ga and older than 3.92~Ga. There the curve is interpolation and extrapolation, not calibration.}
  \end{columns}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{neukum_chronology}}\\
  {\footnotesize The lunar chronology of Neukum, Ivanov and Hartmann (2001), with the intervals that no returned sample constrains. Course figure.}
\end{frame}

% ── Key results from crater chronology ────────────────────
\begin{frame}{Key results from crater chronology}
  \begin{itemize}
    \item \textbf{Late Heavy Bombardment} ($\sim$3.9~Ga): spike in impact rate recorded on the Moon
    \vskip4pt
    \item \textbf{Mars hemispheric dichotomy:}\\
    Southern highlands heavily cratered ($\sim$4~Gyr)\\
    Northern lowlands much smoother $\rightarrow$ volcanic resurfacing
    \vskip4pt
    \item \textbf{Venus:} Remarkably uniform crater density\\
    Mean surface age $\sim$300--700~Myr\\
    Implies a \textbf{global resurfacing event}
    \vskip4pt
    \item \textbf{Europa:} Very few craters $\rightarrow$ surface age $\sim$40--90~Myr\\
    Continuously resurfaced by ice tectonics
  \end{itemize}
\end{frame}

\section{Volcanism}
% ════════════════════════════════════════════════════════════

% ── Effusive vs explosive ─────────────────────────────────
\begin{frame}{Effusive vs.\ explosive volcanism}
  The key variable is \textbf{magma viscosity}, controlled by $\mathrm{SiO_2}$ content:
  \vskip8pt
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Low viscosity (basaltic):}
    \begin{itemize}
      \item $\sim$50\% $\mathrm{SiO_2}$
      \item Flows easily
      \item Volatiles escape gradually
      \item \textbf{Effusive} eruptions
      \item Broad shield volcanoes, flood basalts (vast lava plains)
      \item Moon, Mars, Io
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{High viscosity (silicic):}
    \begin{itemize}
      \item $>$65\% $\mathrm{SiO_2}$
      \item Traps volatiles
      \item Pressure builds until failure
      \item \textbf{Explosive} eruptions
      \item Steep stratovolcanoes (layered lava-ash cones)
      \item Primarily Earth
    \end{itemize}
  \end{columns}
\end{frame}

% ── Olympus Mons ──────────────────────────────────────────
\begin{frame}{Olympus Mons: the largest volcano in the solar system}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{olympus_mons}
    \source{NASA/JPL/USGS}

    \column{\recaptextcol}
    \begin{itemize}
      \item Shield volcano on Mars
      \item Base diameter: $\sim$600~km
      \item Summit: $\sim$21.2~km above the Mars datum
      \item 2.4$\times$ Everest above sea level
      \item Basal escarpment up to 6~km high
    \end{itemize}
    \vskip6pt
    \keyresult{No plate tectonics on Mars $\rightarrow$ hotspot stays fixed $\rightarrow$ lava piles up for billions of years in one location.}
  \end{columns}
\end{frame}

% ── Olympus Mons scale comparison ─────────────────────────
\begin{frame}{How big is Olympus Mons?}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{olympus_mons_size_comparison}

    \column{\recaptextcolwide}
    \footnotesize
    Heights on one linear scale, widths schematic:
    \begin{itemize}
      \item Olympus Mons: $\sim$21.2~km above the Mars datum, $\sim$600~km base
      \item Mauna Kea: $\sim$10.2~km measured from the ocean floor
      \item Everest: $\sim$8.8~km above sea level (not a volcano, shown for scale)
    \end{itemize}
    \vskip6pt
    On Earth, plate motion carries a hotspot volcano away from its magma source within a few Myr, so no single edifice keeps growing.
    \vskip6pt
    \keyresult{Mars has no plate motion, so one plume feeds one edifice for billions of years. The size difference is a tectonic result, not a magma-supply one.}
  \end{columns}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.80\paperheight, keepaspectratio]{olympus_mons_size_comparison}}\\
  {\footnotesize Olympus Mons, Mauna Kea and Mount Everest, each rising from its own base. Heights share one linear scale; widths are schematic. Course figure.}
\end{frame}

% ── Lunar maria ───────────────────────────────────────────
\begin{frame}{Lunar maria: flood basalts}
  \begin{itemize}
    \item Vast basalt plains filling ancient impact basins on the nearside
    \item Erupted between 3.9 and 3.1~Ga (radiometric ages from Apollo samples)
    \item Cover $\sim$16\% of the lunar surface
    \item Visible from Earth as the dark patches forming the ``face'' of the Moon
    \item Record a period of residual internal heating, now extinct
  \end{itemize}
  \vskip6pt
  On Earth, plate motion carries the crust over hotspots $\rightarrow$ chains of smaller volcanoes (e.g., Hawaiian Islands) rather than single massive edifices.
\end{frame}

% ── Venus and Io volcanism ────────────────────────────────
\begin{frame}{Venus and Io: active volcanic worlds}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{io_volcanism}

\section{Tectonics}
% ════════════════════════════════════════════════════════════

% ── Plate tectonics ───────────────────────────────────────
\begin{frame}{Plate tectonics: Earth's unique regime}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{plate_tectonics}
    \source{Wikimedia, CC BY-SA 3.0}

    \column{\recaptextcol}
    Earth is the \textbf{only} body with active plate tectonics:
    \begin{itemize}
      \item $\sim$15 rigid plates, moving at 1--10~cm\,yr$^{-1}$
      \item \textbf{Divergent:} mid-ocean ridges (new crust)
      \item \textbf{Convergent:} subduction zones (recycling)
      \item \textbf{Transform:} horizontal sliding
    \end{itemize}
    \vskip4pt
    Enables the \textbf{carbonate-silicate cycle}:\\
    Carbon recycling $\rightarrow$ long-term climate regulation.
  \end{columns}
\end{frame}

% ── Stagnant lid ──────────────────────────────────────────
\begin{frame}{Stagnant-lid convection}
  \begin{itemize}
    \item All other terrestrial bodies: \textbf{single rigid lithospheric lid}
    \item Mantle convects beneath, but lid does not participate
    \item Heat escapes by conduction through the lid and occasional volcanism
    \item Lid thickens over time as interior cools $\rightarrow$ volcanism eventually shuts down
  \end{itemize}
  \vskip6pt
  \textbf{Why is Earth unique?}
  \begin{itemize}
    \item Viscosity contrast at lid base: $\sim 10^{10}$
    \item Breaking the lid requires water weakening + self-sustaining damage
    \item Likely a combination of size, water content, and thermal history
  \end{itemize}
\end{frame}

% ── Mars tectonics ────────────────────────────────────────
\begin{frame}{Mars: Tharsis and Valles Marineris}

    \column{\recaptextcol}
    \textbf{Tharsis bulge:}
    \begin{itemize}
      \item Volcanic plateau $\sim$5000~km across, $\sim$10~km high
      \item Likely uplifted by a mantle plume (rising column of hot mantle rock)
    \end{itemize}
    \vskip4pt
    \textbf{Valles Marineris:}
    \begin{itemize}
      \item Largest canyon system: $\sim$4000~km long
      \item Up to 7~km deep, 200~km wide
      \item Extensional rifting from Tharsis
      \item Lisbon to Moscow in scale
    \end{itemize}
  \end{columns}
\end{frame}

% ── Venus and Mercury tectonics ───────────────────────────
\begin{frame}{Venus and Mercury: contrasting tectonic styles}
  \textbf{Venus, episodic resurfacing:}
  \begin{itemize}
    \item Uniform crater density $\rightarrow$ mean age $\sim$300--700~Myr
    \item Hypothesis: periodic catastrophic lid foundering
    \item Entire surface resurfaced volcanically in a short interval
    \item Between episodes: tectonically quiet
  \end{itemize}
  \vskip8pt
  \textbf{Mercury, global contraction:}
  \begin{itemize}
    \item Large iron core cooled and solidified over time
    \item Planet's radius shrank by $\sim$7~km
    \item Produced \textbf{lobate scarps}: thrust faults up to several hundred km long, 1--3~km high
    \item Mapped by Mariner~10 and MESSENGER
  \end{itemize}
\end{frame}

\section{Erosion and weathering}

% ── Aeolian processes ─────────────────────────────────────
\begin{frame}{Aeolian (wind) processes}
  Wind erosion and deposition require a sufficiently dense atmosphere:
  \vskip6pt
  \begin{itemize}
    \item \textbf{Mars:} Extensive dune fields in craters and polar regions\\
    Global dust storms loft particles to 30~km altitude\\
    Active dune migration observed by Curiosity
    \vskip4pt
    \item \textbf{Titan:} Vast equatorial dune fields of organic particles (tholins)\\
    Longitudinal dunes up to 150~m tall, hundreds of km long
    \vskip4pt
    \item \textbf{Venus:} Dense atmosphere but slow surface winds ($\sim$1~m\,s$^{-1}$)\\
    Even slow winds can mobilise fine particles in thick atmosphere
  \end{itemize}
\end{frame}

% ── Glacial and chemical weathering ───────────────────────
\begin{frame}{Glacial and chemical weathering}
  \textbf{Glacial processes:}
  \begin{itemize}
    \item Earth: ice sheets covered $\sim$30\% of land at Last Glacial Maximum ($\sim$20~ka)
    \item Mars: polar $\mathrm{CO_2}$ and $\mathrm{H_2O}$ ice caps; mid-latitude features resemble rock glaciers
  \end{itemize}
  \vskip6pt
  \textbf{Chemical weathering:}
  \begin{itemize}
    \item Earth: silicate weathering by carbonic acid is the critical carbon sink
    \item Mars: orbital spectroscopy (CRISM, OMEGA) detected hydrated minerals: phyllosilicates, sulfates, carbonates
    \item Venus: high surface $T$ ($\sim$737~K) drives surface--atmosphere reactions
  \end{itemize}
\end{frame}

% ── CRISM mineral map ─────────────────────────────────────
\begin{frame}{Remote sensing of surfaces}
  We can visit only a few sites. Most surface knowledge comes from orbital remote sensing.
  \vskip8pt
  \begin{description}
    \item[Reflectance spectroscopy] (UV--IR) identifies minerals via absorption bands
    \item[Radar imaging] penetrates clouds and regolith, measures surface roughness
    \item[Laser altimetry] maps topography at sub-metre vertical precision
    \item[Gravity field mapping] constrains subsurface density anomalies
  \end{description}
  \vskip8pt
  Each technique probes a different length scale and physical property; together they build a three-dimensional picture of the surface and shallow interior.
\end{frame}

\begin{frame}{Laser altimetry and gravity mapping}
  \textbf{Laser altimetry (MOLA, LOLA, BELA):}
  \begin{itemize}
    \item Time-of-flight of laser pulses from orbit
    \item Vertical precision $\sim$1~m over global baselines
    \item MOLA produced the canonical Mars topographic map (used throughout this lecture)
  \end{itemize}
  \vskip8pt
  \textbf{Gravity field mapping (GRAIL at the Moon, GRACE/GOCE at Earth):}
  \begin{itemize}
    \item Spacecraft-to-spacecraft tracking measures tiny accelerations
    \item Gravity anomalies reveal buried mass: dense cores, thin crust, basins, mantle plumes
    \item Combined with topography, constrains crustal thickness and compensation state
  \end{itemize}
  \vskip6pt
  Altimetry and gravity together tell us what the interior \textbf{looks like} from above.
\end{frame}

\section{Cryovolcanism on icy bodies}
% ════════════════════════════════════════════════════════════

% ── Cryovolcanism overview ────────────────────────────────
\begin{frame}{Cryovolcanism: volcanism on ice worlds}
  In the outer solar system, ``volcanic'' eruptions involve volatiles rather than silicate melts:
  \vskip6pt
  \begin{itemize}
    \item Erupted materials: liquid water, ammonia--water mixtures, methane, nitrogen
    \item Energy source: \textbf{tidal heating} maintains subsurface oceans beneath icy shells
    \item ``Magma'' = low-viscosity volatile liquids
    \item Analogous to silicate volcanism, but at temperatures of 100--270~K
  \end{itemize}
\end{frame}

% ── Enceladus ─────────────────────────────────────────────
\begin{frame}{Enceladus: tiger stripes and geysers}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{enceladus_cryovolcanism}
    \source{NASA/JPL-Caltech/SSI (Cassini)}

    \column{\recaptextcol}
    \raggedright
    \begin{itemize}
      \item Radius only 252~km
      \item Water ice jets from 4 parallel ``tiger stripe'' fractures
      \item Sourced from \textbf{global subsurface ocean}
      \item Ocean in contact with rocky core
      \item Thermal emission: $\sim$15.8~GW
    \end{itemize}
    \vskip4pt
    Plume composition:
    \begin{itemize}
      \item $\mathrm{H_2}$, silica grains, organics
      \item Consistent with hydrothermal vents
    \end{itemize}
  \end{columns}
\end{frame}

% ── Europa ────────────────────────────────────────────────
\begin{frame}{Europa: a hidden ocean}
  \begin{itemize}
    \item Jupiter's moon, $R \approx 1561$~km
    \item Global ocean $\sim$100~km deep; ice shell $\sim$15--25~km thick
    \item Maintained by tidal heating (Laplace resonance with Io and Ganymede)
  \end{itemize}
  \vskip6pt
  Surface features:
  \begin{itemize}
    \item \textbf{Lineae:} cracks filled by upwelling ocean water or warm ice
    \item \textbf{Chaos terrain:} broken, rotated, refrozen ice blocks
    \item \textbf{Very few craters:} surface age $\sim$40--90~Myr
  \end{itemize}
  \vskip6pt
  \textbf{Europa Clipper} (launched 2024): dozens of close flybys to characterise the ice shell, ocean, and habitability.
\end{frame}

% ── Europa lineae and chaos ───────────────────────────────
\begin{frame}{Europa's cracked shell}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{europa_chaos}
    \source{NASA/JPL-Caltech/University of Arizona/University of Colorado}

    \column{\recaptextcol}
    \footnotesize
    Galileo imaged the region around Pwyll crater, the bright ray system at lower right:
    \begin{itemize}
      \item Dark double ridges and lineae criss-cross the whole surface
      \item Each set records one episode of fracturing as tidal stress flexed the brittle ice
      \item Chaos terrain marks where the shell broke up, rotated and refroze
    \end{itemize}
    \vskip6pt
    \keyresult{A crater density this low means a surface age of only 40--90~Myr. On a 4.5~Gyr-old moon, that is resurfacing happening now.}
  \end{columns}
\end{frame}

% ── Triton and ocean worlds ───────────────────────────────
\begin{frame}{Triton and the ocean worlds}
  \textbf{Triton} (Neptune's largest moon):
  \begin{itemize}
    \item Nitrogen geysers observed by Voyager~2 (1989)
    \item Plumes of $\mathrm{N_2}$ gas and dark dust, rising $\sim$8~km
    \item Very young surface; retrograde orbit (captured KBO)
    \item Possible subsurface ocean
  \end{itemize}
  \vskip8pt
  \keyresult{Enceladus, Europa, and Triton demonstrate that liquid water can exist far beyond the classical habitable zone, maintained by tidal heating rather than stellar radiation.}
\end{frame}

% ── Hayabusa2 / Ryugu ───────────────────────────────────
\begin{frame}{Hayabusa2: sampling Ryugu}
  JAXA's Hayabusa2 delivered 5.4~g of asteroid Ryugu to Earth in December 2020.
  \vskip6pt
  \begin{itemize}
    \item C-type carbonaceous asteroid, 900~m diameter
    \item First sample of a water-bearing asteroid returned to Earth
    \item Contains pristine organic molecules: amino acids, sugars, uracil
    \item Evidence for aqueous alteration on the parent body
    \item Noble gas and rare-earth abundances constrain its formation location
    \item Complements OSIRIS-REx by sampling a different asteroid
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Watanabe et al.\ (2019); Yada et al.\ (2022)}

% ── Perseverance ──────────────────────────────────────────
\begin{frame}{Perseverance at Jezero crater}
  \begin{itemize}
    \item Operating in Jezero crater since February 2021
    \item Crater floor: igneous rock (olivine-bearing cumulates)
    \item Subsequently altered by liquid water
    \item $>$20 sample tubes cached for Mars Sample Return
    \item Joint NASA/ESA mission: first laboratory analysis of Martian rocks
    \item Will address past habitability and possible biosignatures
  \end{itemize}
  \vskip6pt
  Jezero was chosen because it is an ancient lake crater with a preserved delta deposit, maximising the chance of finding evidence for past life.

% ── DART mission ──────────────────────────────────────────
\begin{frame}{DART: a planetary defence experiment}
  \begin{itemize}
    \item \textbf{Double Asteroid Redirection Test} (September 2022)
    \item Kinetic impactor deliberately crashed into moonlet \textbf{Dimorphos}
    \item Changed orbital period around Didymos by \textbf{33 minutes}
    \item First successful planetary defence experiment
    \item Confirmed kinetic impact as a viable deflection strategy
    \item Impact ejecta revealed properties of rubble-pile asteroid surfaces
  \end{itemize}
  \vskip6pt
  \keyresult{DART demonstrated that we can deflect a potentially hazardous asteroid using existing technology.}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Daly et al.\ (2023), \textit{Nature}; Thomas et al.\ (2023)}
\end{frame}

% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}
    \item Surfaces record the competition between \textbf{endogenic} (volcanism, tectonics) and \textbf{exogenic} (impacts, erosion) processes
    \item Impact cratering is universal; three stages: contact, excavation, modification
    \item Crater scaling law: $D \sim (E/\rho g)^{1/4}$ (fourth-root dependence on energy)
    \item Crater counting + Apollo calibration $\rightarrow$ absolute surface ages
    \item Volcanism style depends on magma viscosity; no plate tectonics $\rightarrow$ giant volcanoes
    \item Earth is unique in having plate tectonics; other bodies have stagnant lids
    \item Fluvial features on Mars: strong evidence for past liquid water
    \item Cryovolcanism on Enceladus, Europa, Triton: liquid water beyond the habitable zone
  \end{enumerate}
\end{frame}
