\section{Surface processes}
% ════════════════════════════════════════════════════════════

% ── Endogenic vs exogenic ─────────────────────────────────
\begin{frame}{Endogenic vs.\ exogenic processes}
  A surface is a geological record: processes that build landforms compete with processes that destroy them.
  \vskip8pt
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Endogenic} (internal heat): volcanism, tectonics (faulting, folding)
    \item \textbf{Exogenic} (external forces): impact cratering, erosion by wind, water and ice
  \end{itemize}
  \vskip8pt
  \keyresult{Hot interior: endogenic processes win. Dead interior: the surface keeps its ancient craters.}
\end{frame}

% ── Dominant processes table ──────────────────────────────
\begin{frame}{Dominant surface process by body}
  \centering
  \small
  \begin{tabular}{l c c}
    \toprule
    \textbf{Body} & \textbf{Dominant process} & \textbf{Key evidence} \\
    \midrule
    Moon   & Impact cratering & Saturated highland craters, $>$4~Gyr \\
    Earth  & Plate tectonics + erosion & Ocean floors $<$200~Myr; rapid weathering \\
    Mars   & Volcanism + impacts & Olympus Mons; cratered southern highlands \\
    Venus  & Volcanism & Volcanic plains cover $\sim$80\% of surface \\
    Io     & Tidal volcanism & $\sim$300--400 active volcanoes; age $<$1~Myr \\
    Europa & Ice tectonics & Lineae, chaos terrain, very few craters \\
    \bottomrule
  \end{tabular}
\end{frame}

\end{frame}

\section{Impact cratering}
% ════════════════════════════════════════════════════════════

% ── Universality ──────────────────────────────────────────
\begin{frame}{The most universal geological process}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Every} solid body in the solar system bears impact craters
    \item Without atmosphere or geology (Moon, Mercury), craters survive billions of years
    \item Impact velocities 10--70~km\,s$^{-1}$, set by orbital mechanics (Lecture~2)
  \end{itemize}
  \vskip8pt
  \keyresult{A 1~km rocky asteroid at 20~km\,s$^{-1}$ carries $E_k = \tfrac{1}{2}mv^2 \approx 3 \times 10^{20}$~J, about 1000 times the largest nuclear weapon, released in under a second.}
\end{frame}

% ── Three stages ──────────────────────────────────────────
\begin{frame}
  \ipsherokey{impact_stages}{Three stages of crater formation: contact and compression, excavation of a transient cavity, modification into a simple or complex crater. Course figure.}{Contact and compression (shock near 100~GPa, over in under 1~s), excavation of a transient cavity (depth:diameter about 1:3, ejecta blanket at the rim), modification: walls slump into a \textbf{simple crater} or the floor rebounds into a \textbf{central peak}.}
\end{frame}

\section{Blackboard derivation: the crater scaling law}
% ════════════════════════════════════════════════════════════

% ── Setup: goal and strategy ──────────────────────────────
\begin{frame}{Goal and strategy}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig[0.62\textheight]{board_sketch_l07}
    \column{\recaptextcolwide}
    \textbf{Goal:} how does crater diameter $D$ depend on impact energy $E$, target density $\rho$ and gravity $g$?
    \vskip8pt
    \begin{itemize}\setlength\itemsep{2pt}
      \item \textbf{Gravity regime} (craters $\gtrsim$100~m): $D = C\,E^a\,\rho^b\,g^c$
      \item Dimensions: $[E] = M L^2 T^{-2}$, $[\rho] = M L^{-3}$, $[g] = L T^{-2}$
      \item Three equations for $a$, $b$, $c$; then numbers
    \end{itemize}
    \vskip8pt
    \keyresult{To the board.}
  \end{columns}
\end{frame}

% ── Result and its range ──────────────────────────────────
\begin{frame}
  \ipsherokey[0.6]{holsapple1993_strength_gravity_regimes}{Crater size against impactor size in the strength and gravity regimes, curves for $U = 2.5$, 10 and 40~km\,s$^{-1}$. Holsapple (1993), Fig.~3.}{$D \sim (E/\rho g)^{1/4}$ holds in the gravity regime, above about 100~m on the Moon: doubling $E$ grows $D$ by only $2^{1/4} \approx 1.19$, and at fixed $v$ and $\rho$ a 1~km impactor makes a 16.5~km crater ($D \propto L^{3/4}$).}
\end{frame}

\end{frame}

% ── Transition diameter ───────────────────────────────────
\begin{frame}{Transition diameter}
  The simple-to-complex transition scales with surface gravity:
  $$D_t \approx D_{t,\mathrm{Moon}} \cdot \frac{g_{\mathrm{Moon}}}{g}$$
  \vskip6pt
  \centering
  \begin{tabular}{l c c}
    \toprule
    \textbf{Body} & $\boldsymbol{g}$ \textbf{(m\,s$^{-2}$)} & $\boldsymbol{D_t}$ \textbf{(km)} \\
    \midrule
    Moon    & 1.62  & 15 \\
    Mars    & 3.71  & $\sim$6.5 \\
    Earth   & 9.81  & $\sim$2.5 \\
    Mercury & 3.70  & $\sim$6.6 \\
    \bottomrule
  \end{tabular}
  \vskip6pt
  On Earth, virtually all craters larger than a few km are complex.
\end{frame}

% ── Crater counting ───────────────────────────────────────
\begin{frame}
  \ipsherokeyside{crater_counting}{Crater counting on the lunar farside: older surfaces carry more craters. Credit: NASA/GSFC/ASU; method after Neukum et al.\ (2001).}{$N(>D) = a\,D^{-b}$ with $b \approx 2$--3, a straight line on log-log axes: a higher line is an older surface, so counts give \textbf{relative} ages until \textbf{saturation}, and Apollo and Luna samples supply the \textbf{absolute} calibration.}
\end{frame}

\begin{frame}
  \ipsherokey[0.61]{neukum_chronology}{The lunar cratering chronology $N(1) = a\left[e^{\lambda T} - 1\right] + b\,T$. Course figure, after Neukum, Ivanov and Hartmann (2001).}{The linear term is the steady flux of the last 3~Gyr, the exponential term the early bombardment before 3.6~Gyr; $N(1) = 10^{-2}$~km$^{-2}$ reads as 3.7~Gyr, and the shaded bands hold no dated sample: interpolation, not calibration.}
\end{frame}

\section{Tectonics}
% ════════════════════════════════════════════════════════════

% ── Plate tectonics ───────────────────────────────────────
\begin{frame}
  \ipsherokey[0.61]{plate_tectonics}{The tectonic plates of Earth and their boundaries. Credit: Wikimedia, CC BY-SA 3.0.}{Earth is the \textbf{only} body with active plate tectonics: about 15 rigid plates at 1--10~cm\,yr$^{-1}$, \textbf{divergent} ridges that make crust, \textbf{convergent} subduction that recycles it, \textbf{transform} faults that slide; subduction closes the \textbf{carbonate-silicate cycle}.}
\end{frame}

% ── Stagnant lid ──────────────────────────────────────────
\begin{frame}{Stagnant-lid convection}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Every other terrestrial body: one rigid lithospheric lid over a convecting mantle
    \item Heat leaves by conduction and occasional volcanism; the thick lid keeps the mantle warm
    \item Melt reaches the surface only through rare plumes (Tharsis), and volcanism fades as the lid grows
  \end{itemize}
  \vskip8pt
  \keyresult{Stagnant lid is the default for temperature-dependent viscosity. Breaking the lid, as Earth does, needs water weakening and self-sustaining damage.}
\end{frame}

  \ipshero{lenardic2018_tectonic_modes}{Tectonic modes of terrestrial planets beyond the mobile-stagnant binary: active lid, sluggish lid, episodic regimes, and stagnant lid (cold or hot). Reproduced from Lenardic (2018), Fig.~2.}
\end{frame}

% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}\setlength\itemsep{6pt}
    \item Surfaces record \textbf{endogenic} (volcanism, tectonics) against \textbf{exogenic} (impacts, erosion) processes; crater counting plus Apollo calibration gives absolute ages
    \item Crater scaling law: $D \sim (E/\rho g)^{1/4}$, a fourth root of the energy
    \item No plate tectonics: giant volcanoes, one rift, a shrinking Mercury; Earth alone breaks its lid, and fluvial features on Mars are strong evidence for past liquid water
  \end{enumerate}
  \vskip8pt
  \textbf{Next:} Lecture~8 goes underneath, to the interiors that drive the endogenic half of this record.
\end{frame}
