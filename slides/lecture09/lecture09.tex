% ── Why is Earth habitable ──────────────────────────────────
\begin{frame}{What Makes Earth Habitable?}
  Four ingredients, all present, all stable for $\sim 4$ Gyr:
  \vskip6pt
  \begin{enumerate}
    \item \textbf{Liquid water} on the surface, continuously
    \item \textbf{Plate tectonics}: long-term carbon cycle, climate buffering
    \item \textbf{Magnetic field}: shields atmosphere from solar wind erosion
    \item \textbf{Stable orbit}: low eccentricity, large stabilising Moon
  \end{enumerate}
  \vskip8pt
  None of these is unique to Earth in principle.\\
  All four together, on the same planet, at the same time, may be.
  \vskip6pt
  \keyresult{The puzzle is not that Earth is special. It is to identify which features are necessary, which are sufficient, and which are accidental.}
\end{frame}

% ── TRAPPIST-1 explanation ────────────────────────────────
\begin{frame}{Why Compare to Other Systems?}
  Earth is one of $\sim 5800$ confirmed planets. The natural comparison set:
  \vskip6pt
  \begin{itemize}
    \item Inner solar system: 4 rocky planets, $0.4$--$1.5$ AU, around a G-type star
    \item TRAPPIST-1: 7 Earth-sized planets, $0.011$--$0.062$ AU, around a late M dwarf
    \item Both systems span the inner edge of the habitable zone
    \item Comparable orbital architecture, drastically different host stars
  \end{itemize}
  \vskip8pt
  Earth is the calibration point. Whether it is the typical or atypical outcome of rocky-planet evolution is exactly what we are trying to find out.
\end{frame}

% ── Plate tectonics ─────────────────────────────────────────
\begin{frame}{Plate Tectonics: The Engine}
  \begin{itemize}
    \item Earth's lithosphere (rigid outer shell) is broken into $\sim 15$ major plates moving at cm\,yr$^{-1}$
    \item Three boundary types:
    \begin{itemize}
      \item \textbf{Divergent} (mid-ocean ridges): new crust forms
      \item \textbf{Convergent} (subduction zones, one plate sinks beneath another; mountain belts): crust destroyed or thickened
      \item \textbf{Transform} (e.g.\ San Andreas): lateral slip
    \end{itemize}
    \item Driven by mantle convection, slab pull, ridge push
    \item Recycles surface volatiles into the mantle on $\sim 100$ Myr timescales
  \end{itemize}
  \vskip6pt
  \textbf{No other body in the solar system has active plate tectonics today.}
\end{frame}

% ── Long-term carbon cycle ──────────────────────────────────
\begin{frame}{The Stagnant-Lid Carbon Cycle (Venus Preview)}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{honing2021_carbon_cycle_diagram}
    \source{H\"oning et al.\ (2021)}

    \column{\recaptextcolwide}
    Three reservoirs (mantle, crust, atmosphere) coupled by:
    \begin{itemize}
      \item \textbf{Mantle degassing} (volcanism): mantle to atmosphere
      \item \textbf{Surface weathering}: atmosphere to crust
      \item \textbf{Decarbonation}: crust back to atmosphere when buried carbonate heats up
    \end{itemize}
    \vskip3pt
    \textbf{Destabilising loop} on Venus: $T_s\uparrow \Rightarrow$ shallower decarbonation $\Rightarrow$ atmospheric $\mathrm{CO_2}\uparrow \Rightarrow T_s\uparrow$. No subduction returns carbon to the mantle.
  \end{columns}
\end{frame}

% ── Geomagnetic field ───────────────────────────────────────
\begin{frame}{Earth's Magnetic Field}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{magnetosphere_anatomy_esa}
    \source{ESA, CC BY-SA 3.0 IGO}

    \column{\recaptextcolwide}
    \footnotesize
    \begin{itemize}
      \item Generated by convection in the liquid outer core (Lecture 4)
      \item Mostly dipolar, surface intensity $25$--$65$ $\mu$T
      \item Magnetopause stand-off: $\sim 10\, \Rearth$ on the dayside
      \item Reverses polarity on $10^5$--$10^6$ yr timescales (paleomagnetism)
      \item Active for at least $3.45$ Ga (Tarduno et al.\ 2010)
    \end{itemize}
  \end{columns}
  \vskip4pt
  \textbf{Key consequence:} deflects solar wind, suppresses ion-pickup escape of $\mathrm{O^+}$, $\mathrm{N_2^+}$, $\mathrm{H^+}$.\\
  Comparison to unmagnetised Mars (Lecture 10) shows the magnitude of this effect.
\end{frame}

% ── Hydrosphere & cryosphere ────────────────────────────────
\begin{frame}{Hydrosphere and Cryosphere}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Liquid water reservoirs:}
    \begin{itemize}
      \item Oceans: $1.34 \times 10^9$ km$^3$
      \item Groundwater: $\sim 2.3 \times 10^7$ km$^3$
      \item Lakes, rivers, soil moisture: small
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Ice reservoirs:}
    \begin{itemize}
      \item Antarctica: $26.5 \times 10^6$ km$^3$
      \item Greenland: $2.85 \times 10^6$ km$^3$
      \item Mountain glaciers, permafrost: smaller
    \end{itemize}
  \end{columns}
  \vskip8pt
  \begin{itemize}
    \item Total surface water $\approx 2700$ m global equivalent layer (GEL)
    \item Ocean is the largest pool, but the deep mantle holds $\geq 1$--$2$ ocean masses
    \item Ice albedo provides a strong positive feedback in cold climate excursions
  \end{itemize}
\end{frame}

% ── Climate stability ───────────────────────────────────────
\begin{frame}{Climate Stability and the Faint Young Sun}
  Earth's climate has been stable enough for liquid water for $\sim 4$ Gyr, but not constant.
  \vskip6pt
  \begin{itemize}
    \item Solar luminosity at $4$ Ga: $\sim 75\%$ of today
    \item Without a strong greenhouse, Earth would have been frozen
    \item Geological evidence (sedimentary rocks, evaporites): liquid water at $\geq 3.8$ Ga
    \item Compensating mechanism: higher $\mathrm{CO_2}$ + $\mathrm{CH_4}$, dimmer surface, weaker weathering
  \end{itemize}
  \vskip6pt
  \keyresult{The Faint Young Sun problem is solved by the carbon cycle's negative feedback, not by accidents of initial conditions.}
\end{frame}

% ── Oxygen history walkthrough ────────────────────────────
\begin{frame}{Reading the Oxygen History}
  Two great rises in atmospheric $\mathrm{O_2}$:
  \vskip6pt
  \begin{itemize}
    \item \textbf{Great Oxidation Event} (GOE) at $\sim 2.4$ Ga: trace $\rightarrow \sim 1\%$ of present atmospheric level (PAL)
    \item Long Mesoproterozoic plateau at $1\%$--$10\%$ PAL
    \item \textbf{Neoproterozoic Oxygenation Event} (NOE) at $\sim 0.6$ Ga: $\rightarrow$ near modern levels
    \item Phanerozoic fluctuations of $\sim 15$--$35\%$
  \end{itemize}
  \vskip6pt
  \textbf{Drivers:} cyanobacterial photosynthesis + organic carbon burial + decreasing reductant flux from the mantle.\\
  Co-evolution of biosphere, atmosphere, and tectonics, across more than $2$ Gyr.
\end{frame}

% ── Snowball Earth ──────────────────────────────────────────
\begin{frame}{Snowball Earth Episodes}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{snowball_earth}
    \source{Oleg Kuznetsov (3depix.com), CC BY-SA 4.0}

    \column{\recaptextcol}
    At least two global glaciation events in the Neoproterozoic:
    \begin{itemize}
      \item \textbf{Sturtian glaciation} ($\sim 717$--$660$ Ma)
      \item \textbf{Marinoan glaciation} ($\sim 650$--$635$ Ma)
      \item Glacial deposits at near-equatorial paleolatitudes
      \item Cap carbonates immediately above glacial diamictites
    \end{itemize}
  \end{columns}
  \vskip8pt
  \textbf{Mechanism:} ice albedo runaway $\Rightarrow$ frozen oceans $\Rightarrow$ weathering halts $\Rightarrow$ volcanic $\mathrm{CO_2}$ accumulates $\Rightarrow$ extreme greenhouse triggers deglaciation.
  \vskip4pt
  \sourcenote{Hoffman et al.\ (1998, 2017)}
\end{frame}

% ── Archean climate ─────────────────────────────────────────
\begin{frame}{The Archean Climate Problem}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{charnay2013_archean_temperature}
    \source{Charnay et al.\ (2013)}

    \column{\recaptextcolwide}
    \small
    \begin{itemize}
      \item Solar luminosity at $3$ Ga: $\sim 80\%$ today
      \item Yet evidence for liquid water + biosphere
      \item Required greenhouse: $\mathrm{CO_2}$, $\mathrm{CH_4}$, possibly $\mathrm{N_2}$ pressure broadening
      \item Constraints from paleosols, banded iron formations (ancient sediments)
    \end{itemize}
    \vskip6pt
    \keyresult{Archean Earth was kept warm by an order of magnitude more greenhouse gas than today, not by a closer Sun.}
  \end{columns}
\end{frame}

% ── Earth's late accretion ─────────────────────────────────
\begin{frame}{Earth's Bulk Composition: Late Accretion}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{dauphas2017_earth_accreting_material}
    \source{Dauphas (2017)}

    \column{\recaptextcol}
    \begin{itemize}
      \item Bulk Earth matches non-carbonaceous (NC, inner-disk) material in $\varepsilon^{54}$Cr (panel shown; same story in O, Ti, Ni, Mo, Ru)
      \item Most of Earth's mass came from inner-disk material
      \item Late accretion ($\lesssim$ a few \%): a small carbonaceous component, the "late veneer"
      \item Earth has a specific isotopic signature, not a generic chondrite mix
    \end{itemize}
  \end{columns}
\end{frame}

% ── Wordsworth abiotic O2 ──────────────────────────────────
\begin{frame}{Abiotic Pathways to $\mathrm{O_2}$}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{wordsworth2014_abiotic_o2_schematic}
    \source{Wordsworth \& Pierrehumbert (2014)}

    \column{\recaptextcol}
    For exoplanet biosignatures: not all $\mathrm{O_2}$ implies life.
    \vskip4pt
    \begin{itemize}
      \item $\mathrm{H_2O}$ photolysis + H escape leaves abiotic $\mathrm{O_2}$
      \item Especially at runaway-greenhouse conditions (surface too hot for liquid water)
      \item Or in $\mathrm{CO_2}$-rich atmospheres where $\mathrm{CO}$ + $\mathrm{O_2}$ recombination is slow
      \item Earth's $\mathrm{O_2}$ context-dependent biosignature
    \end{itemize}
  \end{columns}
\end{frame}

% ── Wordsworth N2 redox pump ───────────────────────────────
\begin{frame}{The $\mathrm{N_2}$ Redox Pump}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{wordsworth2016_n2_redox_pump}
    \source{Wordsworth (2016)}

    \column{\recaptextcolwide}
    \begin{itemize}
      \item Atmospheric $\mathrm{N_2}$ inventory is set by the surface + interior redox state
      \item \textbf{Earth path A$\to$C}: surface ocean condenses; ends in 1-bar $\mathrm{N_2}$ over a moderately oxidising mantle
      \item \textbf{Venus path A$\to$B$\to$D}: H escape oxidises mantle, ends in hot $\mathrm{CO_2}$+$\mathrm{N_2}$ over a highly oxidised interior
      \item Abiotic redox pump: water-loss history drives the divergence, no biology required
    \end{itemize}
  \end{columns}
\end{frame}

% ── Biosphere geological footprint ──────────────────────────
\begin{frame}{The Biosphere as Geological Force}
  Life is not a passive passenger of Earth's climate; it is a geological agent.
  \vskip6pt
  \begin{itemize}
    \item \textbf{Banded iron formations} ($\sim 3$--$1.8$ Ga): biological $\mathrm{O_2}$ + dissolved $\mathrm{Fe^{2+}}$
    \item \textbf{Continental crust composition}: shifted by oxidative weathering
    \item \textbf{Ocean redox state}: from anoxic to oxic, Snowball linked to NOE
    \item \textbf{Atmospheric $\mathrm{N_2}$ inventory}: biological denitrification + nitrification
  \end{itemize}
  \vskip6pt
  Earth's atmosphere is in extreme \textbf{thermodynamic disequilibrium} ($\mathrm{O_2 + CH_4}$ coexist) maintained continuously by biology.
\end{frame}

% ── Earth as reference summary ──────────────────────────────
\begin{frame}{Earth as Reference: Summary}
  \begin{itemize}
    \item Differentiated, magnetised, tectonically active rocky planet at $1$ AU
    \item Liquid surface water, continuously, for $\sim 4$ Gyr
    \item Long-term carbon cycle buffers climate against luminosity changes
    \item Biosphere: a major geochemical reservoir, sink/source for many volatiles
    \item Plate tectonics + magnetic field + biosphere + ocean: the four-way coincidence
  \end{itemize}
  \vskip10pt
  Now to the planet that started as Earth's apparent twin and ended as something completely different.
\end{frame}

% ── Venus mission history ──────────────────────────────────
\begin{frame}{Venus: A Long Mission History}
  \begin{itemize}

  \end{itemize}

% ── Venus terrains walkthrough ─────────────────────────────
\begin{frame}{Reading the Venus Surface Map}
  Categories highlighted in the Widemann et al.\ (2023) overlay:
  \vskip6pt
  \begin{itemize}
    \item \textbf{Plains} ($\sim 80\%$ of surface): basaltic, low elevation
    \item \textbf{Tessera}: highly deformed highlands, possibly oldest crust
    \item \textbf{Rift zones} (e.g.\ Beta Regio) and \textbf{deformed terrain}
    \item \textbf{Artemis chasma}: largest single tectonic feature
    \item \textbf{Craters} ($\sim 1000$ catalogued, $\geq 2$ km)
  \end{itemize}
  \vskip4pt
  Coronae (circular tectonic-volcanic features) are a separate generic Venus class, not a category in this overlay. Volcanic edifices: $> 85{,}000$ catalogued globally.
  \vskip4pt
  Key observation: \textbf{no spatial pattern of crustal age}. No oceanic-style mid-ocean ridges, no clear subduction belts.
\end{frame}

% ── Crater age ─────────────────────────────────────────────
\begin{frame}{Crater Statistics: A Single Surface Age}
  Venus crater statistics from Magellan:
  \vskip6pt
  \begin{itemize}
    \item $\sim 1000$ identified impact craters, $\geq 2$ km diameter
    \item Distribution is \textbf{statistically random}, no major terrain younger than another
    \item Mean crater density implies a global crater-retention age of $\sim 150$--$250$ Myr (older estimates: $300$--$1000$ Myr)
    \item No areas of clearly archaic ($> 1$ Ga) terrain
    \item No active subduction belts (no oceanic-style recycling)
  \end{itemize}
  \vskip6pt
  \keyresult{Venus' surface was either resurfaced globally and recently ($\sim 750$ Ma catastrophic-resurfacing model) or is being resurfaced steadily (equilibrium model). Crater statistics alone cannot decide between them.}
\end{frame}

% ── Topography walkthrough ────────────────────────────────
\begin{frame}{Why the Topography Differs}
  \begin{itemize}
    \item Earth: \textbf{bimodal hypsometry} (distribution of surface elevations; oceans $\sim -4$ km vs.\ continents $\sim +1$ km)
    \item Venus: \textbf{unimodal hypsometry}, single dominant elevation
    \item Earth's bimodality is a direct signature of plate tectonics + isostasy
    \item Venus has highlands (Aphrodite, Ishtar) but no continents in the Earth sense
    \item Geoid/topography ratio: deep compensation depths on Venus
  \end{itemize}

\end{frame}

\section{Venus Interior and Tectonic Regime}

% ── Venus interior ─────────────────────────────────────────
\begin{frame}{Venus' Interior: Inferred, Not Measured}
  No seismometer has ever operated on Venus.\\
  Interior structure is inferred from bulk parameters and Earth-analog models.
  \vskip6pt
  \begin{itemize}
    \item Mean density $5243$ kg\,m$^{-3}$; uncompressed $\sim 4060$ kg\,m$^{-3}$
    \item $C/MR^2 \approx 0.33$ (assumed Earth-like, not measured)
    \item Implied core radius: $R_c \approx 3000$--$3500$ km ($\sim 0.55\, R_V$)
    \item Mantle composition: probably similar to Earth's
    \item Core state: possibly partially molten, but no strong evidence either way
  \end{itemize}
  \vskip6pt
  \textbf{Key observational gap:} no $C/MR^2$, no seismicity, no high-resolution gravity.
\end{frame}

% ── Tectonic evolution schematic ──────────────────────────
\begin{frame}{Venus Tectonic Evolution: A Model Trajectory}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{smrekar2018_venus_tectonic_evolution}
    \source{Smrekar et al.\ (2018)}

    \column{\recaptextcolwide}
    Surface temperature evolution from one numerical model: successive intervals of
    \begin{itemize}
      \item \textbf{Stagnant lid}
      \item \textbf{Mobile lid}
      \item \textbf{Stagnant lid}
      \item \textbf{Episodic lid}
    \end{itemize}
    \vskip4pt
    Conceptual alternatives debated for Venus today: \textbf{steady stagnant} (localised mantle plumes) vs.\ \textbf{episodic catastrophic overturn}. Either way, no Earth-style plate tectonics.
  \end{columns}
\end{frame}

% ── No magnetic field ──────────────────────────────────────
\begin{frame}{The Missing Magnetic Field}
  \begin{itemize}
    \item Internal magnetic field $< 10^{-5}$ that of Earth (Pioneer Venus, 1978)
    \item Possible reasons:
    \begin{enumerate}
      \item Core not convecting (insufficient heat flux)
      \item Core fully solid (already crystallised)
      \item Core fully liquid but subadiabatic (temperature gradient too shallow to drive convection)
      \item Slow rotation reduces dynamo efficiency
    \end{enumerate}
    \item Most likely: \textbf{stagnant lid} suppresses core heat loss $\Rightarrow$ no convection in core
  \end{itemize}
  \vskip6pt
  Without a magnetosphere, the upper atmosphere is directly exposed to the solar wind.\\
  This is a major channel for Venus' water loss.
\end{frame}

% ── Stagnant lid vs episodic ───────────────────────────────
\begin{frame}{Why Stagnant Lid?}
  Earth-style plate tectonics needs:
  \vskip4pt
  \begin{enumerate}
    \item A weak, brittle, water-saturated lithosphere
    \item Negative thermal buoyancy at subduction zones
    \item Continuous mantle return flow
  \end{enumerate}
  \vskip6pt
  Venus likely fails on point 1: hot, dry surface $\Rightarrow$ strong, ductile lithosphere.
  \vskip6pt
  \textbf{Coupled-feedback view:} no liquid water $\Rightarrow$ dry strong lithosphere $\Rightarrow$ no plate motion $\Rightarrow$ no carbon-cycle return $\Rightarrow$ no water (one-way reinforcing loop).
  \vskip4pt
  \sourcenote{Smrekar et al.\ (2018); Gillmann et al.\ (2022)}
\end{frame}

% ── Recent volcanism ───────────────────────────────────────
\begin{frame}{Is Venus Volcanically Active Today?}
  Three independent pieces of evidence for recent or ongoing volcanism:
  \vskip6pt
  \begin{itemize}
    \item \textbf{Magellan radar reanalysis} (Herrick \& Hensley 2023): a $\sim 2$ km$^2$ vent on Maat Mons changed shape between 1991 and 1992
    \item \textbf{Venus Express thermal emissivity}: anomalously high emission from Idunn Mons, consistent with $< 250$ kyr lava flows
    \item \textbf{Atmospheric SO$_2$ variability}: order-of-magnitude changes on $\sim$decadal timescales, consistent with episodic outgassing
  \end{itemize}
  \vskip6pt
  \keyresult{Venus is almost certainly volcanically active today. The questions are: at what rate, and from how many vents?}
\end{frame}

\section{Blackboard Derivation: Simpson--Nakajima Limit}
% ════════════════════════════════════════════════════════════

% ── Setup ──────────────────────────────────────────────────
\begin{frame}{Setup: Saturated Atmosphere, Greybody}
  \textbf{Goal:} show that a water-saturated atmosphere has a maximum OLR independent of surface temperature.
  \vskip6pt
  \textbf{Assumptions:}
  \begin{itemize}
    \item Atmosphere saturated in water vapour everywhere
    \item Single grey IR opacity $\kappa$, units m$^2$\,kg$^{-1}$
    \item Hydrostatic equilibrium, surface gravity $g$
    \item Photosphere: where IR optical depth $\tau \sim 1$
  \end{itemize}
  \vskip8pt
  Clausius--Clapeyron sets the water vapour partial pressure:
  $$\dv{p_{\mathrm{sat}}}{T} = \frac{L\, p_{\mathrm{sat}}}{R_v\, T^2}$$
  with $L \approx 2.5 \times 10^6$ J\,kg$^{-1}$, $R_v = 461$ J\,kg$^{-1}$\,K$^{-1}$.
\end{frame}

% ── Step 1: photosphere column density ─────────────────────
\begin{frame}{Step 1: The Photosphere Column Density}
  An IR optical depth of unity is reached at column density:
  $$N_{\mathrm{phot}} \sim \frac{1}{\kappa}$$
  \vskip6pt
  Hydrostatic equilibrium converts column density to pressure:
  $$p_{\mathrm{phot}} = g\, N_{\mathrm{phot}} = \frac{g}{\kappa}$$
  \vskip6pt
  In a water-saturated atmosphere, the local water-vapour pressure equals saturation:
  \keyresult{$$p_{\mathrm{sat}}(T_{\mathrm{phot}}) = \frac{g}{\kappa}$$}
  \vskip4pt
  This single equation pins the photosphere temperature.
\end{frame}

% ── Step 2: solve for T_phot ───────────────────────────────
\begin{frame}{Step 2: Photosphere Temperature}
  Integrate Clausius--Clapeyron:
  $$p_{\mathrm{sat}}(T) \approx p_{\mathrm{ref}}\, \exp\!\left[-\frac{L}{R_v}\left(\frac{1}{T} - \frac{1}{T_{\mathrm{ref}}}\right)\right]$$
  \vskip4pt
  Set $p_{\mathrm{sat}} = g/\kappa$ and solve for $T$:
  \vskip4pt
  $$T_{\mathrm{phot}} \approx \frac{L/R_v}{\ln(p_{\mathrm{ref}}\, \kappa / g) + L/(R_v T_{\mathrm{ref}})}$$
  \vskip4pt
  Plug in $p_{\mathrm{ref}} = 611$ Pa, $T_{\mathrm{ref}} = 273$ K, $g = 10$ m\,s$^{-2}$, $\kappa = 5\times 10^{-2}$ m$^2$\,kg$^{-1}$:
  $$T_{\mathrm{phot}} \approx 260\, \mathrm{K}$$
  \vskip4pt
  \textbf{No surface temperature appears anywhere on the right-hand side.}
\end{frame}

% ── Step 3: OLR ────────────────────────────────────────────
\begin{frame}{Step 3: The OLR Asymptote}
  OLR is approximately blackbody emission at $T_{\mathrm{phot}}$:
  $$F_{\mathrm{OLR}} \approx \sigma\, T_{\mathrm{phot}}^4$$
  \vskip4pt
  $$F_{\mathrm{OLR}}^{\max} \approx 5.67\times 10^{-8}\times (260)^4 \approx 260\, \mathrm{W\,m^{-2}}$$
  \vskip6pt
  \keyresult{Refined line-by-line calculations: $F_{\mathrm{OLR}}^{\max} \approx 280$--$310$ W\,m$^{-2}$ (Kasting 1988; Goldblatt et al.\ 2013).}
  \vskip4pt
  As $T_s$ rises, the photosphere migrates upward but its temperature stays clamped: thermal flux saturates.
\end{frame}

\begin{frame}{Numerical Verification: Goldblatt 2013}

% ── Robustness / wrap ──────────────────────────────────────
\begin{frame}{Why the Simpson-Nakajima Limit is Robust}
  The limit depends only on:
  \vskip4pt
  \begin{itemize}
    \item Water phase diagram (Clausius--Clapeyron)
    \item Elementary radiative transfer ($\tau \sim 1$ photosphere)
  \end{itemize}
  \vskip6pt
  Insensitive to:
  \begin{itemize}
    \item Atmospheric composition (so long as water is abundant)
    \item Surface temperature (by construction)
    \item Detailed cloud microphysics (corrections at the 5--10\% level)
  \end{itemize}
  \vskip6pt
  \keyresult{Once $F_{\mathrm{abs}} > F_{\mathrm{OLR}}^{\max}$, no steady state with surface liquid water. The transition is one-way: only photolysis + H escape returns the planet to a dry state, irreversibly.}
\end{frame}

\section{Venus Water Loss and the D/H Ratio}
% ════════════════════════════════════════════════════════════

% ── When did water leave ──────────────────────────────────
\begin{frame}{When Did Venus Lose Its Water?}
  Two end-member scenarios, both physically plausible:
  \vskip6pt
  \begin{itemize}
    \item \textbf{Early loss}: runaway from the magma-ocean stage; never had condensed oceans (Hamano Type-II)
    \item \textbf{Late loss}: temperate Earth-like surface for $\sim 1$ Gyr, runaway triggered by the Sun's increasing main-sequence luminosity (Way et al.\ 2016)
  \end{itemize}
  \vskip6pt
  Distinguishing observation needed: surface mineralogy + atmospheric isotopes.\\
  None of the present datasets is conclusive.
\end{frame}

% ── Way 2016 paleo-Venus ──────────────────────────────────
\begin{frame}{Was Venus Ever Habitable? The Late-Loss Scenario}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{way2016_paleo_venus_temperature}
    \source{Way et al.\ (2016)}

    \column{\recaptextcol}
    \begin{itemize}
      \item GCM snapshot at 2.9 Gya: surface $T$ ranges from $-22^{\circ}$C polar to $+46^{\circ}$C equatorial
      \item Cloud feedback from slow rotation reflects flux back to space
      \item Across the Way et al.\ (2016) ensemble, global mean $T < 290$ K can persist for $\sim 1$ Gyr
      \item A habitable Venus is dynamically possible, not required
    \end{itemize}
  \end{columns}
\end{frame}

% ── Photolysis + H escape ──────────────────────────────────
\begin{frame}{The Loss Mechanism: Photolysis + Escape}
  Sequential loss of water in the upper atmosphere:
  \vskip4pt
  \begin{enumerate}
    \item $\mathrm{H_2O} + h\nu \rightarrow \mathrm{H} + \mathrm{OH}$ (UV photolysis)
    \item $\mathrm{H}$ escapes via Jeans + non-thermal channels (light, fast)
    \item $\mathrm{O}$ recombines, slowly oxidises crust or escapes via solar-wind pickup
  \end{enumerate}
  \vskip6pt
  Mass balance: $\sim$1 ocean equivalent of water can be lost over Gyr through this channel.
  \vskip4pt
  Without a magnetic field, the solar wind directly accesses the upper atmosphere $\Rightarrow$ enhanced O$^+$ pickup.
\end{frame}

% ── D/H ratio ──────────────────────────────────────────────
\begin{frame}{The D/H Ratio: Smoking-Gun Evidence}
  \begin{columns}[c]
    \column{0.55\textwidth}
    Measured atmospheric D/H ratios:
    \vskip6pt
    \begin{tabular}{l r}
      \toprule
      \textbf{Body} & \textbf{D/H ($\times 10^{-4}$)} \\
      \midrule
      Protosolar (H$_2$) & $\sim 0.20$ \\
      Earth (oceans) & $1.56$ \\
      Comets (Halley, 67P) & $1.5$--$5$ \\
      \textbf{Venus} & $\sim \mathbf{240}$ \\
      Mars & $\sim 8$ \\
      \bottomrule
    \end{tabular}
    \vskip6pt
    \sourcenote{Donahue et al.\ (1982); de Bergh et al.\ (1991)}

    \column{0.42\textwidth}
    \begin{itemize}
      \item Venus' D/H is \textbf{$\sim 150\times$ Earth's}
      \item Lighter $^1$H escapes more easily than heavier $^2$H
      \item D enrichment $\Rightarrow$ massive past H loss
      \item Initial water $\geq 1\%$ of Earth's ocean mass
    \end{itemize}
  \end{columns}
\end{frame}

% ── Initial water inventory ────────────────────────────────
\begin{frame}{Bounding the Initial Water Inventory}
  Rayleigh fractionation in escape: residual water becomes D-enriched as H is lost.
  \vskip4pt
  Two limiting fractionation regimes:
  \begin{itemize}
    \item \textbf{Diffusion-limited}: $\alpha \approx m_\mathrm{H}/m_\mathrm{D} \approx 0.5 \;\Rightarrow\; f \approx 150^{-2} \approx 4\times 10^{-5}$
    \item \textbf{Bare-Jeans}: $\alpha \approx \sqrt{m_\mathrm{H}/m_\mathrm{D}} \approx 0.71 \;\Rightarrow\; f \approx 150^{-3.4} \approx 4\times 10^{-8}$
  \end{itemize}
  \vskip4pt
  Multiplying $1/f$ by the present water column ($\sim$2 cm GEL) gives initial inventories of \textbf{several hundred metres} (diffusion-limited) up to unphysically large values in the bare-Jeans limit.
  \vskip4pt
  Either way: $\geq 100\times$ today's water (consistent with the previous slide's $\geq 1\%$ Earth-ocean lower bound).
\end{frame}

% ── Turbet hysteresis ────────────────────────────────────
\begin{frame}{Bistability and Hysteresis Loop}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{turbet2021_hysteresis}
    \source{Turbet et al.\ (2021), Fig.~4}

    \column{\recaptextcol}
    For some stellar fluxes, two stable states coexist:
    \begin{itemize}
      \item Wet: ocean + thin atmosphere
      \item Dry: hot, thick $\mathrm{CO_2}$ atmosphere
    \end{itemize}
    \vskip6pt
    Crossing the threshold is one-way. The dry state has no thermodynamic path back.
  \end{columns}
\end{frame}

% ── Gillmann dry/wet scenarios ────────────────────────────
\begin{frame}{Dry-Venus vs Wet-Venus Pathways}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{gillmann2022_dry_wet_venus_scenarios}
    \source{Gillmann et al.\ (2022)}

    \column{\recaptextcol}
    Coupled-model branching:
    \begin{itemize}
      \item Initial water inventory + outgassing rate determines branch
      \item Wet branch: like Earth, ocean condenses
      \item Dry branch: like Venus today
      \item Initial conditions decide everything
    \end{itemize}
  \end{columns}
\end{frame}

% ── Constantinou Venus pathways ───────────────────────────
\begin{frame}{Constraining the Trajectory}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{constantinou2024_venus_pathways}
    \source{Constantinou et al.\ (2024)}

    \column{\recaptextcolwide}
    Two end-member pathways for Venus' water history:
    \begin{itemize}
      \item \textbf{Top branch (dry):} no surface ocean; H lost early; mantle dry today
      \item \textbf{Bottom branch (wet):} ocean condenses, late loss; mantle still wet today
      \item Constantinou et al.\ (2024) argue current photochemistry favours the dry/early-loss branch (volcanic $\mathrm{H_2O} \lesssim 6\%$)
      \item DAVINCI noble-gas measurements will discriminate
    \end{itemize}
  \end{columns}
\end{frame}

\sectionimage{terrestrial_planets}{Mercury, Venus, Earth, Mars at approximate relative scale. Credit: NASA/JPL}
\section{Why Did Earth and Venus Diverge?}
% ════════════════════════════════════════════════════════════

% ── Carbonate-silicate failure ──────────────────────────────
\begin{frame}{Carbonate--Silicate Cycle: Failure Mode}
  On Earth, the carbon cycle keeps $\mathrm{CO_2}$ in the atmosphere $\sim 280$--$1000$ ppm via:
  \vskip4pt
  \begin{itemize}
    \item Volcanic outgassing (source)
    \item Silicate weathering on continents (sink, $T$-dependent)
    \item Carbonate subduction + decarbonation (return loop)
  \end{itemize}
  \vskip8pt
  On Venus this cycle has \textbf{no functional sink}:
  \begin{itemize}
    \item No liquid water $\Rightarrow$ no aqueous weathering
    \item No subduction $\Rightarrow$ no return of crustal carbonate to the mantle
    \item Volcanic outgassing accumulates indefinitely
  \end{itemize}
  \vskip4pt
  Result: $\mathrm{CO_2}$ partial pressure builds to the $90$-bar ambient observed today.
\end{frame}

% ── Lammer carbonate-silicate ─────────────────────────────
\begin{frame}{The Earth Carbonate-Silicate Schematic}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{lammer2018_carbonate_silicate}
    \source{Lammer et al.\ (2018)}

    \column{\recaptextcol}
    Earth's closed carbon loop:
    \begin{itemize}
      \item Volcanic outgassing (mantle to atmosphere)
      \item Silicate weathering on continents (atmosphere to ocean, $T$-dependent sink)
      \item Carbonate sedimentation + subduction (ocean to mantle return leg)
    \end{itemize}
    \vskip4pt
    Venus today has no liquid-water sink and no subduction return: both halves of the loop are broken (next slide).
  \end{columns}
\end{frame}

% ── Lammer accretion ─────────────────────────────────────
\begin{frame}{Initial Volatile Inventories}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{lammer2018_venus_earth_mars_accretion}
    \source{Lammer et al.\ (2018)}

    \column{\recaptextcol}
    \begin{itemize}
      \item Comparative accretion of Venus, Earth, Mars
      \item Initial water + carbon inventories overlap substantially
      \item Divergence is not from initial conditions alone
      \item Early evolution decides which branch each planet takes
    \end{itemize}
  \end{columns}
\end{frame}

% ── Why diverge: contributing causes ───────────────────────
\begin{frame}{Why Did the Twins Diverge?}
  No single cause; several contributing factors:
  \vskip4pt
  \begin{enumerate}
    \item \textbf{Distance from the Sun} ($1.91\times$ Earth's flux) $\Rightarrow$ inside the runaway-greenhouse limit
    \item \textbf{No active dynamo} $\Rightarrow$ no shielding from solar wind
    \item \textbf{Slow rotation} $\Rightarrow$ less Coriolis-driven cloud organisation
    \item \textbf{Possible early magma-ocean phase}: Hamano Type-II, volatiles never fully condensed
    \item \textbf{Stagnant lid tectonics} $\Rightarrow$ no carbon-cycle return loop
  \end{enumerate}
  \vskip6pt
  All five may have contributed; their relative weight is an active research question.
\end{frame}

% ── Habitability implications ──────────────────────────────
\begin{frame}{Implications for Habitability}
  Venus matters far beyond Venus:
  \vskip4pt
  \begin{itemize}
    \item Defines the inner edge of the classical habitable zone for any star
    \item Provides a \textbf{negative example} of what happens when key feedbacks fail
    \item Stagnant-lid + runaway combination may be the default for rocky planets, with Earth as the lucky exception
    \item Tidally-locked rocky planets near M-dwarfs are pre-conditioned to Venus-like outcomes
  \end{itemize}
  \vskip8pt
  \keyresult{Venus is the calibration target for understanding which combinations of orbital, geophysical, and tectonic parameters allow long-term liquid water.}
\end{frame}

% ── Three missions overview ────────────────────────────────
\begin{frame}{Three Missions to Venus, This Decade}
  \begin{itemize}
    \item \textbf{DAVINCI} (NASA, $\sim 2030$): atmospheric descent probe + flyby imaging
    \item \textbf{VERITAS} (NASA, $\sim 2031$): radar interferometry, surface composition mapping
    \item \textbf{EnVision} (ESA, $\sim 2031$): radar + subsurface sounder + atmospheric chemistry
  \end{itemize}
  \vskip8pt
  Combined, they will measure: noble-gas isotopes (DAVINCI), high-resolution topography (VERITAS), subsurface structure + tidal Love number $k_2$, a measure of tidal deformation (EnVision).
  \vskip6pt
  After a $\sim 30$-year gap in dedicated Venus missions, three at once.
\end{frame}

% ── DAVINCI ───────────────────────────────────────────────
\begin{frame}{DAVINCI: A Descending Atmospheric Laboratory}
  \begin{itemize}
    \item Launch: nominally 2029--2030
    \item Hour-long parachute descent through the full atmosphere
    \item Mass spectrometer: noble gases + isotopes (Ar, Kr, Xe ratios)
    \item Tunable laser spectrometer: D/H, $^{18}$O/$^{16}$O at multiple altitudes
    \item Imaging: descent images of tessera highlands at $<100$ m resolution
  \end{itemize}
  \vskip8pt
  \textbf{Will measure noble-gas isotopes that distinguish early-loss vs.\ late-loss water scenarios for the first time.}
\end{frame}

% ── VERITAS ───────────────────────────────────────────────
\begin{frame}{VERITAS: Radar at $30$ m Resolution}
  \begin{itemize}
    \item Synthetic aperture radar imagery: $\sim 30$ m/px (Magellan SAR: $\sim 100$--$300$ m/px)
    \item InSAR-derived topography: $\sim 250$ m posting, $\sim 5$ m vertical (Magellan altimetry: $\sim 10$--$20$ km horizontal)
    \item InSAR change detection: cm-scale surface deformation between flybys
    \item Near-IR emissivity: surface mineralogy through the atmosphere
    \item Gravity science: high-resolution gravity map, $C/MR^2$ measurement
  \end{itemize}
  \vskip6pt
  \textbf{Will measure $C/MR^2$ for the first time, settling the core-size question.}
\end{frame}

% ── EnVision ──────────────────────────────────────────────
\begin{frame}{EnVision: Surface, Subsurface, Atmosphere}
  \begin{itemize}
    \item ESA M5 mission, launch $\sim 2031$
    \item Synthetic aperture radar (VenSAR): selected high-resolution targets
    \item \textbf{Subsurface sounder}: layered structure of plains and tessera to $\sim 1$ km depth
    \item Spectrometers: mineralogy + atmospheric chemistry
    \item Radio science: gravity tides, $k_2$ Love number $\Rightarrow$ core liquidity
  \end{itemize}
  \vskip6pt
  \keyresult{$k_2$ measurement will distinguish a liquid from solid core, addressing the missing-dynamo question.}
\end{frame}

% ── Open questions ────────────────────────────────────────
\begin{frame}{Open Questions for the 2030s}
  \begin{enumerate}
    \item Did Venus ever have surface oceans? For how long?
    \item What is the present rate of volcanism?
    \item Is the core liquid? Why no dynamo?
    \item Does the atmosphere host disequilibrium chemistry of biological origin?
    \item Does episodic-overturn tectonics operate, and on what timescale?
  \end{enumerate}
  \vskip8pt
  Venus is moving from a puzzle with no data to a puzzle with the right data on the way.
\end{frame}


% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}
    \item Earth and Venus have nearly identical bulk parameters
    \item Their evolution diverged by $\sim 500$ K in surface temperature, $90\times$ in atmospheric mass, all surface water vs.\ none
    \item Earth: plate tectonics + magnetic dynamo + biosphere + carbon cycle, all mutually reinforcing
    \item Venus: stagnant lid + no dynamo + runaway greenhouse, all mutually reinforcing
    \item The Simpson--Nakajima limit ($\sim 280$ W\,m$^{-2}$) is the thermodynamic boundary defining the inner edge of the habitable zone
    \item Venus' D/H ratio of $\sim 150\times$ Earth's is direct evidence of past water loss
    \item DAVINCI, VERITAS, and EnVision will deliver the data to settle when, why, and how the divergence happened
  \end{enumerate}
\end{frame}
