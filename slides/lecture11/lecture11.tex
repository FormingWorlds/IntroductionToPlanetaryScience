\section{Gas Giants and Ice Giants: An Overview}
% ════════════════════════════════════════════════════════════

\begin{frame}{Two Classes, Four Planets}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{itemize}
      \item \textbf{Gas giants:} Jupiter, Saturn
      \begin{itemize}
        \item H/He envelopes dominate mass
        \item 318 and 95 $\Mearth$
        \item Together: $\sim$92\% of all planetary mass
      \end{itemize}
      \item \textbf{Ice giants:} Uranus, Neptune
      \begin{itemize}
        \item H/He only $\sim$10--20\% of mass
        \item Bulk = water, ammonia, methane fluids
        \item 14.5 and 17.1 $\Mearth$
      \end{itemize}
    \end{itemize}
    \column{0.45\textwidth}
    \begin{itemize}
      \item No solid surface; "1 bar level" is convention.
      \item Smooth gas $\to$ supercritical fluid $\to$ metallic plasma.
      \item Rapid rotation: $\sim 10$ h, oblate by 6--10\%.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Energy Budgets: Internal Heat Excess}
  \begin{itemize}
    \item Jupiter: emits $\sim$1.7$\times$ absorbed solar flux $\to$ Kelvin-Helmholtz contraction.
    \item Saturn: $\sim$1.8$\times$ $\to$ helium-rain release of gravitational energy.
    \item Uranus: $\sim$1.06$\times$ $\to$ \emph{anomalously low}; the major outlier.
    \item Neptune: $\sim$2.6$\times$ $\to$ source still uncertain.
  \end{itemize}
  \vskip6pt
  \keyresult{Three of four giants are net energy exporters; Uranus is the puzzle that drives mission concepts.}
\end{frame}

\begin{frame}{Why Rapid Rotation Matters}
  \begin{itemize}
    \item Jupiter: 9 h 56 min; Saturn: 10 h 33 min; rocky planets at 24 h or longer.
    \item Strong Coriolis $\to$ latitude-aligned bands of zonal wind.
    \item Oblateness encodes interior density via gravity harmonics $J_2, J_4, J_6, \ldots$.
    \item Juno (Jupiter) and Cassini Grand Finale (Saturn) measured $J_n$ to high order.
    \item Equatorial jet speeds counterintuitively \emph{increase} with distance from Sun.
  \end{itemize}
\end{frame}

\section{Jupiter: Interior and Atmosphere}
% ════════════════════════════════════════════════════════════

\begin{frame}{Jupiter Interior: Layered Fluid}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{itemize}
      \item Outer molecular $\mathrm{H_2}$ + He envelope
      \item Transition to metallic hydrogen at $\sim$100 GPa, $\sim$5000 K
      \item $r/R_J \approx 0.85$ for the conducting transition
      \item Dynamo source: metallic hydrogen
      \item Centre: $\sim$4000 GPa, $\sim$20\,000 K
    \end{itemize}
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.65\textheight, keepaspectratio]{jupiter_dilute_core_wahl2017}
    \footnotesize\emph{Wahl et al.\ (2017)}
  \end{columns}
\end{frame}

\begin{frame}{The Dilute Core: A Decade-Old Revolution}
  \begin{itemize}
    \item Pre-Juno: assumed compact, well-defined $\sim$10 $\Mearth$ core.
    \item Juno gravity ($J_4, J_6, J_8, J_{10}$) cannot match compact-core models.
    \item Juno data fit naturally with heavy elements distributed across the inner 30--50\% of the planet.
    \item "Dilute" / "fuzzy" core picture: Wahl+ 2017, Militzer+ 2022.
    \item Total heavy-element content: $\sim 20$--$40\,\Mearth$.
  \end{itemize}
  \vskip6pt
  \keyresult{Dilute core implies efficient mixing or non-standard accretion; rewrites textbook formation picture.}
\end{frame}

\begin{frame}{Reading the Jupiter Atmosphere Image}
  \begin{itemize}
    \item Cloud decks at 0.1--10 bar: $\mathrm{NH_3}$ (top), $\mathrm{NH_4SH}$, $\mathrm{H_2O}$ (deepest).
    \item $\sim$15 alternating zonal jets pole-to-pole; peak $\sim$180 m s$^{-1}$.
    \item Bright zones = rising air, ammonia clouds; dark belts = descending haze.
    \item Great Red Spot: anticyclonic vortex observed since at least 1830.
    \item GRS shrinking: $\sim$40\,000 km in 1900 $\to$ $\sim$14\,000 km today.
    \item Galileo probe (1995): confirmed layering but found a dry entry site.
  \end{itemize}
\end{frame}

  \begin{columns}[T]

  \vskip4pt
  Stable polygonal arrangements persist across many Juno orbits. Polar weather not seen at high resolution before 2016.
\end{frame}

\begin{frame}{Jets, Aurorae, and the Great Blue Spot}
  \begin{itemize}
    \item Kaspi+ 2018: equatorial jets penetrate to $\sim$3000 km depth (gravity asymmetries).
    \item Below: magnetic stress damps differential rotation; deep interior $\to$ solid-body.
    \item Aurorae are the strongest in the solar system.
    \item Driven by magnetosphere + Io mass loading ($\sim$1 t s$^{-1}$ of S, O).
    \item Auroral footprints of Io, Europa, Ganymede directly imaged in UV.
    \item "Great Blue Spot": isolated equatorial magnetic anomaly mapped by Juno (Connerney+ 2022).
  \end{itemize}
\end{frame}

\begin{frame}{Saturn Interior: Lower Pressures, Helium Rain}
  \begin{itemize}
    \item Mass 95 $\Mearth$; central pressure $\sim 1/3$ Jupiter's.
    \item Molecular-metallic hydrogen transition deeper, milder conditions.
    \item Cassini Grand Finale (2017): gravity field rivalling Juno's at Jupiter.
    \item Helium becomes immiscible in metallic H at 1--3 Mbar, 5--10 kK.
    \item Droplets fall, release gravitational PE $\to$ closes Saturn's energy budget.
    \item Predicts He depletion in upper envelope; consistent with Voyager.
  \end{itemize}
\end{frame}

    \item Vertical extent only $\sim$10 m: thinnest known objects in the universe (relative to extent).

\begin{frame}{Embedded Moons and Propellers}
  \begin{columns}[T]

    \begin{itemize}
      \item F ring shepherded by Prometheus + Pandora.
      \item Encke Gap held open by embedded moon Pan.
      \item "Propeller" features: $\sim$1 km moonlets too small to clear gaps.
      \item Direct measurement of orbital migration in a disk (Tiscareno+ 2013).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{The Rings Are Young}
  \begin{itemize}
    \item Total ring mass $\sim 1.5 \times 10^{19}$ kg = 40\% of Mimas (Iess+ 2019, Cassini gravity).
    \item Far too small for primordial origin: would have darkened by infall over 4.5 Gyr.
    \item Brightness + mass + composition $\to$ age $\sim$100 Myr (Crida+ 2019).
    \item "Ring rain": water + organics infall onto Saturn's atmosphere (Waite+ 2018).
    \item Ground-based H$_3^+$ ionosphere modulation confirms (O'Donoghue+ 2019).
  \end{itemize}
  \vskip6pt
  \keyresult{Remaining lifetime $\sim 10^8$ yr; Saturn's rings are a transient phase, not a permanent feature.}
\end{frame}

\begin{frame}{What Made the Rings?}
  \begin{itemize}
    \item Wisdom+ (2022): tidal disruption of an icy Mimas-class moon scattered inside the Roche limit.
    \item Could simultaneously explain ring origin and the unusual mid-sized moon orbits.
    \item Alternative: late capture of a Kuiper Belt object.
    \item Charnoz+ (2009): possibly older but with continuous mass shedding.
    \item Pedagogical message: solar-system bodies do not all date from 4.5 Gyr ago.
  \end{itemize}
\end{frame}

\section{Blackboard Derivation: The Roche Limit}
% ════════════════════════════════════════════════════════════

\begin{frame}{Setup: Why Rings Exist}
  \begin{itemize}
    \item Roche limit: distance inside which a fluid satellite's self-gravity loses to the planet's tidal stretch.
    \item Concept introduced by Roche (1849).
    \item Sets where rings can exist and where moons cannot accrete.
  \end{itemize}
  \vskip6pt
  Configuration:
  \begin{itemize}
    \item Planet: mass $M_p$, radius $R_p$, density $\rho_p$.
    \item Satellite: mass $M_s$, radius $R_s$, density $\rho_s$, at distance $d$ from planet centre.
    \item Assume $d \gg R_s$ so a Taylor expansion across the satellite is valid.
    \item Work in the satellite's rest frame.
  \end{itemize}
\end{frame}

\begin{frame}{Step 1: Tidal Acceleration Across the Satellite}
  Acceleration at the satellite's centre:
  \[
    g_{\rm centre} = \frac{G M_p}{d^2}
  \]
  At the near point (closest to planet, distance $d - R_s$):
  \[
    g_{\rm near} = \frac{G M_p}{(d-R_s)^2}
  \]
  Differential (tidal) acceleration that stretches the satellite:
  \[
    \Delta a_{\rm tidal} = g_{\rm near} - g_{\rm centre} = G M_p \left[ \frac{1}{(d-R_s)^2} - \frac{1}{d^2} \right]
  \]
\end{frame}

\begin{frame}{Step 2: Taylor Expand in $R_s/d$}
  For $R_s \ll d$:
  \[
    \frac{1}{(d-R_s)^2} = \frac{1}{d^2}\left(1 - \frac{R_s}{d}\right)^{-2}
       \approx \frac{1}{d^2}\left(1 + \frac{2R_s}{d}\right)
  \]
  So:
  \[
    \boxed{\, \Delta a_{\rm tidal} \approx \frac{2 G M_p R_s}{d^3} \,}
  \]
  \vskip6pt
  \keyresult{Tidal forces fall off as $d^{-3}$, not $d^{-2}$. This is the universal scaling.}
\end{frame}

\begin{frame}{Step 3: Self-Gravity at the Satellite's Surface}
  Uniform density $\rho_s$:
  \[
    a_{\rm self} = \frac{G M_s}{R_s^2}
        = \frac{G \cdot \tfrac{4}{3}\pi R_s^3 \rho_s}{R_s^2}
        = \frac{4}{3}\pi G \rho_s R_s
  \]
  \vskip4pt
  Setting tidal $=$ self-gravity at the disruption threshold:
  \[
    \frac{2 G M_p R_s}{d^3} = \frac{4}{3}\pi G \rho_s R_s
  \]
\end{frame}

\begin{frame}{Step 4: Solve for $d_R$}
  Cancel $G$ and $R_s$, substitute $M_p = \tfrac{4}{3}\pi R_p^3 \rho_p$:
  \[
    \frac{2 \cdot \tfrac{4}{3}\pi R_p^3 \rho_p}{d^3} = \frac{4}{3}\pi \rho_s
  \]
  Rearrange:
  \[
    d^3 = 2 R_p^3 \frac{\rho_p}{\rho_s}
  \]
  \[
    \boxed{\; d_R^{\rm rigid} = 2^{1/3} R_p \left(\frac{\rho_p}{\rho_s}\right)^{1/3}
            \approx 1.26\, R_p \left(\frac{\rho_p}{\rho_s}\right)^{1/3} \;}
  \]
\end{frame}

\begin{frame}{Fluid vs.\ Rigid Roche Limit}
  \begin{itemize}
    \item Rigid case: spherical, holds shape until disruption.
    \item Fluid case: deforms into ellipsoid first, disrupts further out.
    \item Roche's full fluid calculation gives prefactor $\approx 2.46$:
  \end{itemize}
  \[
    \boxed{\; d_R \approx 2.46\, R_p \left(\frac{\rho_p}{\rho_s}\right)^{1/3} \;}
  \]
  Real bodies with finite material strength sit between the two limits.
\end{frame}

\begin{frame}{Step 5: Apply to Saturn's Rings}
  \[
    R_p = 58{,}232~\mathrm{km}, \quad \rho_p = 687~\mathrm{kg\,m^{-3}}, \quad \rho_s \approx 1000~\mathrm{kg\,m^{-3}}
  \]
  \[
    d_R \approx 2.46 \times 58{,}232 \times (0.687)^{1/3}~\mathrm{km}
        \approx 2.46 \times 58{,}232 \times 0.883
  \]
  \[
    \boxed{\; d_R \approx 126{,}000~\mathrm{km} \;}
  \]
  \vskip6pt
  \keyresult{Within $\sim$10\% of A-ring outer edge ($\sim$137\,000 km). Discrepancy: ring-particle porosity, ice rigidity, density-wave structure.}
\end{frame}

\begin{frame}{Why Roche Matters}
  \begin{itemize}
    \item Inside $d_R$: differential pull $>$ self-gravity $\to$ no agglomerate $>10$ m holds together.
    \item Particles orbit as a collisional disk; cannot accrete into a moon.
    \item Outside $d_R$: same population can re-accrete (e.g., small inner moons of Saturn).
    \item Used: Saturn ring origin, Triton's eventual fate, comet Shoemaker-Levy 9 disruption (Jupiter, 1992).
  \end{itemize}
\end{frame}

\section{Ice Giants: Composition and Interior}
% ════════════════════════════════════════════════════════════

\begin{frame}{Ice Giants: The Exotic Twins}
  \begin{itemize}
    \item Uranus: 14.5 $\Mearth$, 4.0 $\Rearth$.
    \item Neptune: 17.1 $\Mearth$, 3.9 $\Rearth$.
    \item H/He envelope only 10--20\% of mass.
    \item Bulk: water, ammonia, methane in compressed-fluid form.
    \item "Ice" reflects formation as solid grains; current state is fluid.
    \item One spacecraft each: Voyager 2 in Jan 1986 / Aug 1989. No return mission since.
  \end{itemize}
  \vskip6pt
  \keyresult{Most under-explored major planets. Knowledge $\sim$35--40 yr out of date.}
\end{frame}

\begin{frame}{Possible Interior Structures}
  \begin{columns}[T]
    \column{0.6\textwidth}
    \includegraphics[width=\linewidth, height=0.7\textheight, keepaspectratio]{ice_giant_structures_helled2020}
    \column{0.4\textwidth}
    \footnotesize
    \begin{itemize}
      \item (a) Sharp boundaries.
      \item (b) Sharp envelope, gradual ice/rock.
      \item (c) Gradual envelope, sharp ice/rock.
      \item (d) Fully gradual gradient.
      \item Voyager $J_2,J_4$ alone cannot distinguish.
      \item Helled+ (2020).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Density Profile Degeneracy}
  \begin{columns}[T]
    \column{0.55\textwidth}

      \item Uranus (blue) and Neptune (black).
      \item Solid: empirical; dashed: 3-layer.
      \item Both fit Voyager gravity equally well.
      \item Cannot uniquely solve interior with current data.
      \item Motivates a dedicated orbiter.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Superionic Ice}
  \begin{itemize}
    \item Millot+ (2019): laser shock to 100--400 GPa, several thousand K.
    \item In-situ X-ray diffraction $\to$ rigid bcc oxygen lattice with mobile protons.
    \item Material is part solid, part liquid; \emph{ionically} conducting.
    \item Can sustain a planetary dynamo without metallic hydrogen.
    \item Soderlund+ (2020): thin-shell dynamo $\to$ multipolar, off-axis fields.
  \end{itemize}
  \vskip6pt
  \keyresult{Superionic ice is the leading dynamo source for Uranus and Neptune.}
\end{frame}

\begin{frame}{Ice Giant Magnetic Fields}
  \begin{itemize}
    \item Uranus: dipole tilt $\sim 59^\circ$ from rotation axis, offset $\sim 1/3 R_p$ from centre.
    \item Neptune: dipole tilt $\sim 47^\circ$, similarly offset.
    \item Strong quadrupole + higher-order components dominate.
    \item Very different from dipole-dominated Jupiter and Saturn.
    \item Connerney+ 1991.
    \item Best explanation: thin convecting shell of ionic / superionic fluid.
  \end{itemize}
\end{frame}

\sectionimage{europa_clipper_concept}{Europa Clipper concept. NASA/JPL-Caltech}
\section{Comparative Payoff and Exploration Frontier}
% ════════════════════════════════════════════════════════════

\begin{frame}{Why Gas Giants and Ice Giants Diverged}
  \begin{itemize}
    \item All four began as solid cores accreting disk gas (L2).
    \item For runaway gas accretion: core $\gtrsim$10 $\Mearth$ \emph{while gas is still present}.
    \item Disk lifetime 3--5 Myr is the deadline.
    \item Jupiter, Saturn: cores reached the threshold early enough.
    \item Uranus, Neptune: too late, or low-density region $\to$ stalled at 10--20 $\Mearth$.
    \item Nice model (Tsiganis+ 2005): ice giants migrated and scattered planetesimals after dispersal.
  \end{itemize}
  \vskip6pt
  \keyresult{Ice-giant masses are a natural intermediate outcome of core accretion in the outer disk.}
\end{frame}

\begin{frame}{Common Themes Across All Four Giants}
  \begin{enumerate}
    \item Internal heat excess (Uranus is the outlier).
    \item Strong zonal jets; speed \emph{increases} with distance from Sun.
    \item Global magnetic fields, but very different morphologies (dipole vs.\ multipole).
    \item Moon and ring systems, but different inventories: bright icy moons (Saturn) vs.\ captured KBOs (Triton).
    \item All four are laboratories for sub-Neptune and Neptune-mass exoplanets (L13).
  \end{enumerate}
\end{frame}

\begin{frame}{What We Still Don't Know}
  \begin{itemize}
    \item Heavy-element fraction of Uranus and Neptune?
    \item What sustains Neptune's heat flow, and why is Uranus so quiet?
    \item How and when did Saturn's rings form? How long will they last?
    \item Origin of Jupiter's dilute core?
    \item Does Callisto have a true ocean today?
    \item Deep structure of Saturn's hexagonal jet?
    \item Habitability of Enceladus, Europa, and Titan?
  \end{itemize}
\end{frame}

\begin{frame}{Two Ocean-World Missions to Jupiter}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \includegraphics[width=\linewidth, height=0.5\textheight, keepaspectratio]{europa_clipper_concept}\\
    \footnotesize Europa Clipper (NASA, launched Oct 2024)
    \column{0.5\textwidth}
    \begin{itemize}
      \item \textbf{Europa Clipper:} $\sim$50 close flybys of Europa from Jovian orbit.
      \item REASON radar, magnetometer, mass spec, thermal imager.
      \item \textbf{JUICE} (ESA, launched Apr 2023): orbits Ganymede in 2034.
      \item First-ever orbiter of any moon other than Earth's.
      \item Complementary, not redundant.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Future: Dragonfly and a Uranus Orbiter}
  \begin{itemize}
    \item Dragonfly (NASA, launch 2028, arrival 2034): first rotorcraft on another world.
    \item Selk crater on Titan: organics + transient melt-water from impact heat.
    \item Mass spec orders of magnitude better than Cassini's.
    \item 2022 Decadal Survey: \textbf{Uranus orbiter + probe} = top large flagship for the 2030s.
    \item Jupiter gravity assist $\to$ orbital insertion + atmospheric probe.
    \item Launch in early 2030s $\to$ arrival late 2030s / early 2040s.
  \end{itemize}
\end{frame}

\begin{frame}{The Voyager Legacy}
  \begin{itemize}
    \item Launched 1977; only spacecraft ever to visit Uranus (1986) and Neptune (1989).
    \item Voyager 1: directed out of ecliptic after Saturn.
    \item Both still operational in interstellar space; expected loss of contact $\sim$2030.
    \item Uranus and Neptune flyby data still yielding new science.
    \item Lesson 1: a single short flyby cannot substitute for sustained investigation.
    \item Lesson 2: well-designed missions return science long after the spacecraft is gone.
  \end{itemize}
\end{frame}

\section*{Summary}
% ════════════════════════════════════════════════════════════

\begin{frame}{Summary: Today's Course-Level Takeaways}
  \begin{enumerate}
    \item Gas giants and ice giants form a compositional continuum, set by formation timing.
    \item Both gas giants now appear to host \emph{dilute / fuzzy} cores, not compact ones.
    \item Saturn's spectacular rings are young ($\sim$100 Myr) and transient.
    \item Galilean moons and Saturnian moons span unprecedented diversity: Io, Europa, Ganymede, Callisto, Titan, Enceladus.
    \item Ice giants are the most under-explored major planets; one flyby each, $\sim$35 yr ago.
    \item Roche limit explains where rings exist and where moons cannot accrete: $\approx 2.46\, R_p (\rho_p / \rho_s)^{1/3}$.
  \end{enumerate}
\end{frame}
