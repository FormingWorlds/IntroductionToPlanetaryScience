\section{Gas Giants and Ice Giants: An Overview}
% ════════════════════════════════════════════════════════════

% ── Two classes, four planets ───────────────────────────────
\begin{frame}{Two Classes, Four Planets}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \begin{itemize}
      \item \textbf{Gas giants} (Jupiter, Saturn):
      \begin{itemize}
        \item H/He envelopes dominate mass ($318$ and $95\,\Mearth$)
        \item Together: $\sim 92\%$ of all planetary mass
      \end{itemize}
      \item \textbf{Ice giants} (Uranus, Neptune):
      \begin{itemize}
        \item H/He only $\sim 10$--$20\%$ of mass ($14.5$ and $17.1\,\Mearth$)
        \item Bulk = water, ammonia, methane fluids
      \end{itemize}
    \end{itemize}
    \column{0.48\textwidth}
    \begin{itemize}
      \item No solid surface; ``$1$~bar level'' is standard reference
      \item Smooth gas $\to$ supercritical fluid $\to$ metallic plasma
      \item Rapid rotation ($\sim 10$~h), oblate by $6$--$10\%$
    \end{itemize}
  \end{columns}
\end{frame}

% ── Energy budgets ──────────────────────────────────────────
\begin{frame}{Energy Budgets: Internal Heat Excess}
  Three of four giant planets radiate far more energy than they receive from the Sun; Uranus is the anomalously cold exception.
  \vskip8pt
  \begin{itemize}
    \item \textbf{Jupiter ($1.7\times$):} slow Kelvin-Helmholtz gravitational contraction
    \item \textbf{Saturn ($1.8\times$):} gravitational energy release from helium rain
    \item \textbf{Neptune ($2.6\times$) vs.\ Uranus ($1.06\times$):} Neptune has strong internal flux; Uranus is near solar equilibrium
  \end{itemize}
\end{frame}

% ── Rapid rotation ──────────────────────────────────────────
\begin{frame}{Why Rapid Rotation Matters}
  Rapid rotation drives planetary oblateness, latitude-aligned zonal winds, and gravitational harmonics that reveal the deep density profile.
  \vskip8pt
  \begin{itemize}
    \item \textbf{Rapid spin:} Jupiter ($9$~h $56$~min) and Saturn ($10$~h $33$~min) rotate in $\sim 10$~h
    \item \textbf{Zonal banding:} powerful Coriolis forces organise atmospheric circulation into alternating jets
    \item \textbf{Gravity harmonics $J_n$:} oblateness and deep flows encode mass distribution, measured by Juno and Cassini
  \end{itemize}
\end{frame}

\section{Jupiter: Interior and Atmosphere}
% ════════════════════════════════════════════════════════════

% ── Jupiter interior ────────────────────────────────────────
\begin{frame}{Jupiter Interior: Layered Fluid}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{jupiter_dilute_core_wahl2017}
    \par\sourcenote{Wahl et al.\ (2017)}
    \column{\recaptextcol}
    \begin{itemize}
      \item Outer molecular $\mathrm{H_2}$ + He envelope
      \item Transition to metallic hydrogen at $\sim 100$~GPa, $\sim 5000$~K
      \item $r/R_J \approx 0.85$ for the conducting transition
      \item Dynamo source (field-generating fluid motion): metallic hydrogen
      \item Centre: $\sim 4000$~GPa, $\sim 20{,}000$~K
    \end{itemize}
  \end{columns}
\end{frame}

% ── The dilute core ─────────────────────────────────────────
\begin{frame}{The Dilute Core: A Decade-Old Revolution}
  Juno gravity measurements revealed that Jupiter does not possess a compact core, but rather an extended, dilute distribution of heavy elements.
  \vskip8pt
  \begin{itemize}
    \item \textbf{Gravity harmonics:} $J_4$--$J_{10}$ cannot match compact $\sim 10\,\Mearth$ core models
    \item \textbf{Dilute core:} $20$--$40\,\Mearth$ of heavy elements distributed across inner $30$--$50\%$ of radius
    \item \textbf{Formation rewrite:} the dilute core (Wahl et al.\ 2017; Militzer et al.\ 2022) implies efficient mixing during accretion or a giant impact (Liu et al.\ 2019)
  \end{itemize}
\end{frame}

% ── Jets, aurorae, and Great Blue Spot ──────────────────────
\begin{frame}{Jets, Aurorae, and the Great Blue Spot}
  Jets reach $\sim 3000$~km deep; Io feeds the strongest aurorae in the solar system.
  \vskip8pt
  \begin{itemize}
    \item \textbf{Jet penetration:} winds extend to $\sim 3000$~km before magnetic braking enforces rigid rotation (Kaspi et al.\ 2018)
    \item \textbf{Io mass loading:} $\sim 1\ \mathrm{t\,s^{-1}}$ of sulfur and oxygen ions enter the magnetosphere
    \item \textbf{Auroral footprints:} UV spots mark the field lines to Io, Europa and Ganymede
    \item \textbf{Great Blue Spot:} isolated equatorial magnetic anomaly mapped by Juno (Connerney et al.\ 2022)
  \end{itemize}
\end{frame}

% ── Saturn interior: helium rain ────────────────────────────
\begin{frame}{Saturn Interior: Lower Pressures, Helium Rain}
  \begin{itemize}
    \item Mass $95\,\Mearth$; central pressure $\sim 1/3$ of Jupiter

\section{Blackboard Derivation: The Roche Limit}
% ════════════════════════════════════════════════════════════

% ── Setup: Why rings exist ──────────────────────────────────
\begin{frame}{Setup: Why Rings Exist}
  \begin{itemize}
    \item \textbf{Roche limit:} distance inside which tidal stretch overcomes self-gravity (Roche 1849)
    \item Sets where rings can persist and where moons cannot accrete
  \end{itemize}
  \vskip6pt
  Configuration:
  \begin{itemize}
    \item Planet: mass $M_p$, radius $R_p$, mean density $\rho_p$
    \item Satellite: mass $M_s$, radius $R_s$, mean density $\rho_s$, at orbital distance $d$
    \item Assume $d \gg R_s$ for differential tidal expansion across the satellite
  \end{itemize}
  \vskip6pt
  Balance the tidal stretch $\Delta a_{\mathrm{tidal}} \approx 2 G M_p R_s / d^3$ against the surface self-gravity $G M_s / R_s^2$. \textit{To the board.}
\end{frame}

% ── Fluid vs rigid Roche limit ──────────────────────────────
\begin{frame}{Fluid vs.\ Rigid Roche Limit}
  \begin{itemize}
    \item Rigid case: the body holds its spherical shape until disruption: $d_R^{\mathrm{rigid}} = 2^{1/3} R_p (\rho_p/\rho_s)^{1/3} \approx 1.26\, R_p (\rho_p/\rho_s)^{1/3}$
    \item Fluid case: the body deforms into an ellipsoid first and disrupts further out; Roche's calculation gives the prefactor $\approx 2.46$:
  \end{itemize}
  \[
    \boxed{\; d_R \approx 2.46\, R_p \left(\frac{\rho_p}{\rho_s}\right)^{1/3} \;}
  \]
  Real bodies with material strength sit between the rigid and fluid limits.
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.64\paperheight, keepaspectratio]{roche_geometry}}\\
  {\footnotesize Roche limit geometry: (a) a self-gravitating fluid satellite deforms into an elongated ellipsoid before it disrupts at the Roche limit $d_R$, where the differential tidal acceleration overcomes its self-gravity; (b) the differential tidal acceleration across an icy body against its surface self-gravity as a function of distance from Saturn, crossing at the rigid Roche limit, with the fluid limit accounting for the tidal elongation. Course figure.}
\end{frame}

% ── Application: Saturn's rings ─────────────────────────────
\begin{frame}{Apply to Saturn's Rings}
  \[
    R_p = 58{,}232~\mathrm{km}, \quad \rho_p = 687~\mathrm{kg\,m^{-3}}, \quad \rho_s \approx 1000~\mathrm{kg\,m^{-3}}
  \]
  \[
    d_R \approx 2.46 \times 58{,}232 \times (0.687)^{1/3}~\mathrm{km}
        \approx 2.46 \times 58{,}232 \times 0.883
  \]
  \[
    \boxed{\; d_R \approx 126{,}000~\mathrm{km} \;}
  \]
  \vskip6pt
  \keyresult{Within $\sim 10\%$ of A-ring outer edge ($\sim 137{,}000$~km). Discrepancy reflects particle porosity, ice rigidity, and resonant structures.}
\end{frame}

  \begin{itemize}
    \item Global subsurface ocean beneath a $20$--$30$~km ice shell; the 2:1 resonance with Dione forces the eccentricity that powers the tidal heating
    \item Plume $\mathrm{H_2}$ (Waite et al.\ 2017): chemical disequilibrium from hydrothermal vents
    \item Silica nanoparticles (Hsu et al.\ 2015): active rock-water reactions
    \item Phosphates (Postberg et al.\ 2023): essential biochemical building block
  \end{itemize}
  \vskip6pt
  \keyresult{Most accessible ocean world: plumes spray subsurface ocean water directly into space.}
\end{frame}

\section{Ice Giants: Composition and Interior}
% ════════════════════════════════════════════════════════════

% ── Ice giants overview ─────────────────────────────────────
\begin{frame}{Ice Giants: The Exotic Twins}
  \begin{itemize}
    \item Uranus: $14.5\,\Mearth$, $4.0\,\Rearth$; Neptune: $17.1\,\Mearth$, $3.9\,\Rearth$
    \item H/He envelope only $10$--$20\%$ of total mass
    \item Bulk composition: water, ammonia, and methane in compressed fluid state
    \item One spacecraft visit each: Voyager 2 in Jan 1986 and Aug 1989
  \end{itemize}
  \vskip6pt
  \keyresult{Most under-explored major planets; data are $35$--$40$~yr old.}
\end{frame}

% ── Superionic ice and magnetic fields ──────────────────────
\begin{frame}{Superionic Ice and Tilted Dynamos}
  \begin{itemize}
    \item \textbf{Superionic ice:} at $100$--$400$~GPa a solid oxygen lattice with mobile protons conducts (Millot et al.\ 2019)
    \item \textbf{Tilted, offset fields:} $59^\circ$ (Uranus) and $47^\circ$ (Neptune), offset $\sim 1/3\,R_p$ and $\sim 1/2\,R_p$ from the centre
    \item \textbf{Thin-shell dynamos:} convection in an outer shell explains the multipolar geometry
  \end{itemize}
  \vskip6pt
  \keyresult{Superionic ice powers dynamos without metallic hydrogen.}
\end{frame}

% ── Uranus heat budget anomaly ──────────────────────────────
\begin{frame}{Uranus: Why So Cold Inside?}
  Uranus emits almost no heat beyond the sunlight it absorbs; Neptune does.
  \vskip8pt
  \begin{itemize}
    \item \textbf{Heat anomaly:} $\sim 1.06\times$ the absorbed solar flux (Pearl et al.\ 1990); Neptune $\sim 2.6\times$
    \item \textbf{Trapped heat:} a giant impact may have left a stably stratified interior that suppresses convection
    \item \textbf{Latent heat:} deep condensation of volatiles could delay the cooling
  \end{itemize}
\end{frame}

\section{Ice Giant Rings}
% ════════════════════════════════════════════════════════════

% ── Ice giant ring systems ──────────────────────────────────
\begin{frame}{Ice Giant Rings: Narrow, Dark, and Carbonaceous}
  Ice-giant rings are narrow, dark, and carbon-rich, contrasting sharply with Saturn's broad water-ice sheet.
  \vskip8pt
  \begin{itemize}
    \item \textbf{Occultation discovery:} Uranus's rings were discovered in 1977 via stellar occultation (13 narrow rings known today)
    \item \textbf{Compositional contrast:} Saturn's rings are $>95\%$ bright water ice, whereas ice-giant rings are dark organics
    \item \textbf{Disruption origin:} sourced by collisional destruction of small inner moonlets rather than a single large icy parent
  \end{itemize}
\end{frame}

\section{Comparative Payoff and Exploration Frontier}
% ════════════════════════════════════════════════════════════

\begin{frame}{Why Gas Giants and Ice Giants Diverged}
  \begin{itemize}
    \item \textbf{Timing:} Jupiter and Saturn reached runaway gas accretion while the nebula was still massive; Uranus and Neptune grew too slowly and caught only $1$--$4\,\Mearth$ of H/He
    \item \textbf{Result:} two H/He-dominated planets with metallic hydrogen dynamos, two ice-dominated planets with superionic-ice dynamos
    \item \textbf{Common threads:} rapid rotation, internal heat (except Uranus), rings and regular moon systems, one family of physics
  \end{itemize}
\end{frame}

\section*{Summary}

% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary: Today's Course-Level Takeaways}
  \begin{enumerate}
    \item \textbf{Two classes:} gas giants (H/He-dominated, Saturn $\sim 80\%$ H/He by mass) vs.\ ice giants (H/He only $10$--$20\%$)
    \item \textbf{Dilute cores:} Juno and ring seismology reveal diffuse cores in Jupiter and Saturn
    \item \textbf{Roche limit:} $d_R \approx 2.46\,R_p(\rho_p/\rho_s)^{1/3}$ bounds rings and moon accretion
    \item \textbf{Ocean moons:} tidal heating maintains oceans within Europa and Enceladus
    \item \textbf{Under-explored ice giants:} unique tilted dynamos powered by superionic ice, and the closest analogues of the sub-Neptunes of Lecture 13
  \end{enumerate}
