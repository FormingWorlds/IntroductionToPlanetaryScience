\section{Probing planetary interiors}
% ════════════════════════════════════════════════════════════

% ── Three tools ──────────────────────────────────────────────
\begin{frame}{Three ways to see inside a planet}
  We cannot directly sample deep interiors. Three complementary tools:
  \vskip8pt
  \begin{enumerate}
    \item \textbf{Seismology:} wave speeds trace density and elastic moduli
    \item \textbf{Gravity fields:} spacecraft tracking measures mass distribution ($J_2$)
    \item \textbf{Moment of inertia:} degree of central mass concentration ($C/MR^2$)
  \end{enumerate}
  \vskip8pt
  For Earth, all three apply; for other worlds, gravity and $C/MR^2$ are often our only probes.
\end{frame}

% ── Seismic waves ────────────────────────────────────────────
\begin{frame}{Seismic waves: P-waves and S-waves}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{seismic_waves}
    \source{Wikimedia Commons, public domain}

    \column{\recaptextcol}
    \textbf{P-waves} (primary, compressional):
    \begin{itemize}
      \item Travel through solids \textit{and} liquids
      \item $v_P = \sqrt{(K + \tfrac{4}{3}\mu)/\rho}$
    \end{itemize}
    \vskip6pt
    \textbf{S-waves} (secondary, shear):
    \begin{itemize}
      \item Travel through solids \textit{only}
      \item $v_S = \sqrt{\mu/\rho}$
      \item $\mu = 0$ in a liquid $\Rightarrow$ S-waves \textbf{cannot propagate}
    \end{itemize}
  \end{columns}
\end{frame}

% ── Shadow zones hero ────────────────────────────────────────
\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.72\paperheight, keepaspectratio]{prem_shadow_zones}}\\
  {\footnotesize Rays traced through PREM, the standard seismological model of Earth's interior. The absence of S-waves beyond 103\textdegree{} proves the outer core is liquid, while P-wave refraction creates a shadow zone from 103\textdegree{} to 143\textdegree{}. Course original, from the PREM tabulation of Dziewonski \& Anderson (1981).}
\end{frame}

% ── Gravity field ────────────────────────────────────────────
\begin{frame}{Gravity field measurements}
  Planetary rotation produces an oblate shape whose gravitational quadrupole $J_2$ is measured via spacecraft Doppler tracking.
  \vskip8pt
  \begin{itemize}
    \item Rotational flattening produces a quadrupole potential term $J_2$
    \item Spacecraft radio tracking detects minute Doppler velocity shifts
    \item Juno and GRAIL constrain internal mass distribution to high precision
  \end{itemize}
\end{frame}

% ── Moment of inertia and Darwin-Radau ──────────────────────
\begin{frame}{The moment of inertia factor: $C/MR^2$}
  The moment of inertia factor measures central mass concentration:
  \vskip6pt
  \begin{columns}[c]
    \column{0.48\textwidth}
    \centering
    \footnotesize
    \begin{tabular}{l c l}
      \toprule
      \textbf{Body} & $\boldsymbol{C/MR^2}$ & \textbf{State} \\
      \midrule
      Uniform sphere & 0.400 & Homogeneous \\
      Moon           & 0.393 & Small core \\
      Mars           & 0.364 & Moderate core \\
      Mercury        & 0.346 & Large core \\
      Earth          & 0.331 & Iron core \\
      \bottomrule
    \end{tabular}
    \vskip4pt
    \sourcenote{de Pater \& Lissauer (2010)}

    \column{0.48\textwidth}
    \textbf{Darwin-Radau relation:}
    \vskip4pt\centerline{\resizebox{0.88\linewidth}{!}{$\displaystyle \frac{C}{MR^2} = \frac{2}{3}\left[1 - \frac{2}{5}\sqrt{\frac{4 - k_f}{1 + k_f}}\right]$}}\vskip4pt
    \begin{itemize}
      \item Hydrostatic bodies: from the flattening, $k_f = 3J_2/m$ with $m = \omega^2R^3/GM$
      \item Moon, Mars, Mercury: from $J_2$ and the precession or libration rate
      \item Lower value $\Rightarrow$ denser core
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.66\paperheight, keepaspectratio]{moment_of_inertia_factors}}\\
  {\footnotesize The moment of inertia factor $C/MR^2$: a uniform sphere gives 0.400; the Moon (0.393), Mars (0.364), Mercury (0.346) and Earth (0.331) fall below it in the order of their central mass concentration, from a small core to a large iron core. Values after de Pater \& Lissauer (2010). Course figure.}
\end{frame}

\section{Blackboard derivation: moment of inertia factor}
% ════════════════════════════════════════════════════════════

% ── Setup ────────────────────────────────────────────────────
\begin{frame}{Deriving $C/MR^2$}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{moi_shell_sketch}
    \source{Course original}

    \column{\recaptextcolwide}
    \textbf{Goal:} Derive the moment of inertia factor for (a) a uniform sphere and (b) a two-layer differentiated body.
    \vskip6pt
    \textbf{Starting point:} For a spherically symmetric body, the moment of inertia about the rotation axis is:
    $$C = \frac{8\pi}{3}\int_0^R \rho(r)\,r^4\,\dd r$$
    A mass element at colatitude $\theta$ contributes $r_\perp^2\,\dd m = r^2\sin^2\theta\,\dd m$; averaged over a thin shell, $\sin^2\theta \to 2/3$.
  \end{columns}
\end{frame}

% ── Uniform sphere and two-layer formulas ───────────────────
\begin{frame}{Summary of derived results}
  \textbf{Uniform sphere:} $\rho = \text{const}$
  $$C = \frac{8\pi}{15}\rho R^5, \quad M = \frac{4}{3}\pi\rho R^3 \quad \implies \quad \frac{C}{MR^2} = \frac{2}{5} = 0.400$$
  \vskip6pt
  \textbf{Two-layer body:} core ($\rho_c, R_c$) and mantle ($\rho_m, R$), with $x \equiv R_c/R$ and $f \equiv \rho_c/\rho_m$:
  \keyresult{$$\frac{C}{MR^2} = \frac{2}{5}\cdot\frac{(f-1)\,x^5 + 1}{(f-1)\,x^3 + 1}$$}
  \vskip4pt
  Since $x^5 < x^3$ for $0 < x < 1$, any dense core ($f > 1$) pulls $C/MR^2 < 0.400$.
\end{frame}

% ── Two-layer bodies visualised ──────────────────────────────
\begin{frame}{Why differentiation lowers $C/MR^2$}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{moi_two_layer}
    \source{Course original}

    \column{\recaptextcol}
    \small
    $C$ weights mass by $r^2$: mass moved inward lowers $C$ but not $M$.
    \vskip4pt
    \begin{itemize}
      \item Fixed $f = \rho_c/\rho_m = 2.5$; core radius fraction $x = R_c/R$ varies
      \item Earth ($x = 0.546$) gives $C/MR^2 \approx 0.345$; measured 0.331, the gap is self-compression
    \end{itemize}
    \vskip4pt
    \keyresult{Any dense core pulls $C/MR^2$ below 0.400.}
  \end{columns}
\end{frame}

\section{Equations of state}
% ════════════════════════════════════════════════════════════

% ── Hydrostatic equilibrium ──────────────────────────────────
\begin{frame}{Hydrostatic equilibrium in spherical symmetry}
  Pressure at each depth supports the weight of overlying material:
  \vskip8pt
  \keyresult{$$\dv{P}{r} = -\rho(r)\,g(r) = -\frac{G\,m(r)\,\rho(r)}{r^2}$$}
  \vskip6pt
  where $m(r) = 4\pi\int_0^r \rho(r')\,r'^2\,\dd r'$ is the mass enclosed within radius $r$.
  \vskip8pt
  Combined with an equation of state $P = P(\rho, T)$, this integrates inward to give the pressure-density profile.
\end{frame}

% ── Central pressure estimate ────────────────────────────────
\begin{frame}{Central pressure estimate}
  Uniform density gives an analytical central pressure; it is a lower bound for a differentiated body.
  \vskip4pt
  $$P_c \approx \frac{3\,G\,M^2}{8\pi\,R^4}$$
  \vskip4pt
  \begin{itemize}
    \item Earth: estimate $\sim$170~GPa vs.\ actual $\sim$360~GPa
    \item Mars: estimate $\sim$25~GPa vs.\ actual $\sim$40~GPa
    \item Uniform estimate is always too low because real planets are centrally concentrated
  \end{itemize}
\end{frame}

% ── Birch's law ──────────────────────────────────────────────
\begin{frame}{Birch's law}
  Birch's law provides an empirical linear bridge from measured seismic wave speeds directly to interior density.
  \vskip4pt
  $$v_P = a(\bar{M}) + b\,\rho$$
  \vskip4pt
  \begin{itemize}
    \item Linear relation between compressional velocity $v_P$ and density $\rho$
    \item Parameters depend primarily on mean atomic weight $\bar{M}$
    \item Breaks down across compositional boundaries and phase transitions
  \end{itemize}
\end{frame}

% ── Birch's law hero ─────────────────────────────────────────
\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.72\paperheight, keepaspectratio]{birchs_law}}\\
  {\footnotesize Birch's law: $v_P$ against density for mantle minerals. Compressional velocity scales linearly with density at constant mean atomic mass, while iron-rich minerals plot below the trend. Course original; trend after Birch (1952).}
\end{frame}

% ── Adams-Williamson equation ────────────────────────────────
\begin{frame}{Adams-Williamson equation}
  The Adams-Williamson equation integrates seismic wave speeds to determine density profiles under adiabatic self-compression.
  \vskip4pt
  $$\dv{\rho}{r} = -\frac{\rho(r)\,g(r)}{\phi(r)}, \quad \phi = v_P^2 - \frac{4}{3}v_S^2$$
  \vskip4pt
  \begin{itemize}
    \item Seismic parameter $\phi \equiv K_S/\rho = v_P^2 - \tfrac{4}{3}v_S^2$
    \item Assumes hydrostatic equilibrium, adiabatic gradient, and uniform composition
    \item Discontinuities mark phase changes or compositional boundaries (e.g., at the core-mantle boundary)
  \end{itemize}
\end{frame}

\section{Mantle rheology \& convection}
% ════════════════════════════════════════════════════════════

% ── Solid-state creep ────────────────────────────────────────
\begin{frame}{Solid-state creep}
  At high homologous temperatures the crystalline silicate mantle deforms by solid-state creep, behaving as a viscous fluid over geological time.
  \vskip6pt
  \begin{itemize}
    \item Solid rock flows when $T > 0.5\,T_{\mathrm{melt}}$ via atomic migration
    \item \textbf{Diffusion creep:} vacancy migration, dominant in lower mantle ($\eta \sim 10^{22}\text{--}10^{23}$~Pa\,s)
    \item \textbf{Dislocation creep:} defect glide, dominant in upper mantle ($\eta \sim 10^{20}\text{--}10^{21}$~Pa\,s)
  \end{itemize}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.66\paperheight, keepaspectratio]{creep_viscosity_ranges}}\\
  {\footnotesize Mantle viscosity by creep mechanism: dislocation creep (defect glide) in the upper mantle at $10^{20}$ to $10^{21}$ Pa\,s, diffusion creep (vacancy migration) in the lower mantle at $10^{22}$ to $10^{23}$ Pa\,s; solid rock flows once $T$ exceeds about $0.5\,T_\mathrm{melt}$. Course schematic.}
\end{frame}

% ── Rayleigh number ──────────────────────────────────────────
\begin{frame}{The Rayleigh number}
  The Rayleigh number balances thermal buoyancy against viscous resistance, determining whether a planetary mantle convects.
  \vskip4pt
  $$\mathrm{Ra} = \frac{\alpha\,\rho\,g\,\Delta T\,d^3}{\kappa\,\eta}$$
  \vskip4pt
  \begin{itemize}
    \item $\mathrm{Ra}$ measures the ratio of buoyancy driving forces to viscous dissipation
    \item Critical threshold for convection onset is $\mathrm{Ra}_c \sim 10^3$
    \item Earth's mantle has $\mathrm{Ra} \sim 10^7\text{--}10^8 \gg \mathrm{Ra}_c$, driving vigorous convection
  \end{itemize}
\end{frame}

\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.66\paperheight, keepaspectratio]{rayleigh_viscosity}}\\
  {\footnotesize The Rayleigh number against mantle viscosity for the course's Earth-mantle values: convection sets in above $\mathrm{Ra}_c \approx 10^3$, and Earth's mantle at $\eta = 10^{21}$ to $10^{22}$ Pa\,s sits at $\mathrm{Ra} \approx 10^7$ to $10^8$, four to five orders of magnitude above the threshold. Course figure.}
\end{frame}

% ── 660 km discontinuity ─────────────────────────────────────
\begin{frame}{The 660~km discontinuity}
  A negative Clapeyron slope at 660~km makes the ringwoodite breakdown endothermic, creating buoyancy that resists slab penetration.
  \vskip6pt
  \begin{itemize}
    \item Ringwoodite decomposes into bridgmanite plus ferropericlase with an $\sim$8\% density jump
    \item Negative Clapeyron slope ($\dd P/\dd T < 0$): in a cold slab the boundary sits deeper, so buoyant ringwoodite persists below 660~km
    \item Produces upward buoyancy that acts as a partial barrier to whole-mantle convection
  \end{itemize}
\end{frame}

\section{Comparative terrestrial interiors}
% ════════════════════════════════════════════════════════════

% ── The Moon ─────────────────────────────────────────────────
\begin{frame}{The Moon}
  The Moon's moment of inertia factor is 0.393, close to a uniform sphere: its metallic core is exceptionally small.
  \vskip6pt
  \begin{itemize}
    \item Mean density $\bar{\rho} = 3346$~kg\,m$^{-3}$ is close to uncompressed mantle rock
    \item Core radius fraction is only $R_c/R \sim 0.1\text{--}0.2$, containing $<2$\% of lunar mass
    \item Low core fraction supports formation via a giant impact from Earth's mantle
  \end{itemize}
\end{frame}

Credit: NASA/JPL-Caltech (PIA25282).}

% ── Mercury ──────────────────────────────────────────────────
\begin{frame}{Mercury: an iron world}
  Mercury has the highest uncompressed density of any terrestrial planet, dominated by an enormous iron core.
  \vskip6pt
  \begin{itemize}
    \item Moment of inertia factor $C/MR^2 = 0.346$ and mean density $\bar{\rho} = 5427$~kg\,m$^{-3}$
    \item Metallic core reaches $R_c \sim 2000\text{--}2050$~km ($\sim$83\% of planetary radius)
    \item Core accounts for $\sim$74\% of total mass; weak dynamo indicates a liquid outer core
  \end{itemize}
\end{frame}

% ── Saturn: helium rain ──────────────────────────────────────
\begin{frame}{Saturn: helium rain}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{saturn_interior}
    \source{NASA/JPL-Caltech}

    \column{\recaptextcol}
    \footnotesize
    Cooling inside Saturn induces helium immiscibility in metallic hydrogen, releasing gravitational energy that powers its excess infrared luminosity.
    \vskip6pt
    \begin{itemize}\setlength\itemsep{2pt}
      \item Helium separates into droplets at $P \sim 2$~Mbar and rains toward the planet centre
      \item Sinking helium releases gravitational energy; with Kelvin-Helmholtz contraction it explains why Saturn radiates $\sim$1.8$\times$ the solar flux it absorbs
      \item Ring seismology reveals a diffuse heavy-element core extending to $\sim$60\% of the radius
    \end{itemize}
  \end{columns}
\end{frame}

\section{Ice giant interiors}
% ════════════════════════════════════════════════════════════

% ── Three-layer model ────────────────────────────────────────
\begin{frame}{Ice giants: canonical layers}
  Uranus and Neptune are modelled with a hydrogen-helium envelope, a thick conducting fluid ice mantle, and a rocky core.
  \vskip6pt
  \begin{itemize}
    \item Outer H/He envelope holds $\sim$20\% of total mass
    \item Intermediate layer ($\sim$60--70\% mass): hot fluid $\mathrm{H_2O}$, $\mathrm{NH_3}$, and $\mathrm{CH_4}$ at 10--500~GPa
    \item Deep water may reach an electrically conducting superionic state
  \end{itemize}
\end{frame}

% ── Ice giant structures hero ────────────────────────────────
\begin{frame}
  \centering
  \makebox[\linewidth][c]{\includegraphics[width=0.95\paperwidth, height=0.72\paperheight, keepaspectratio]{helled2020_icegiant_structures}}\\
  {\footnotesize Four candidate internal structures of an ice giant. Gravity data alone cannot distinguish sharp layers from gradual compositional gradients, leaving their degree of differentiation open. Credit: Helled et al.\ (2020), Fig.~4.}
\end{frame}

% ── Unusual magnetic fields ──────────────────────────────────
\begin{frame}{Ice giant magnetic fields: unusual geometry}
  Ice giants generate highly tilted, non-dipolar magnetic fields in thin, shallow conducting shells.
  \vskip6pt
  \begin{itemize}
    \item Field axes are tilted (59\textdegree{} Uranus, 47\textdegree{} Neptune) and offset from centre
    \item Dynamos operate in thin outer fluid ice shells, not deep interiors
    \item Uranus emits negligible internal heat; Neptune emits $\sim$2.6$\times$ solar flux
  \end{itemize}
\end{frame}
