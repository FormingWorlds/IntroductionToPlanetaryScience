\section{Probing planetary interiors}
% ════════════════════════════════════════════════════════════

% ── Three tools ──────────────────────────────────────────────
\begin{frame}{Three ways to see inside a planet}
  We cannot directly sample deep interiors. Three complementary tools:
  \vskip8pt
  \begin{enumerate}
    \item \textbf{Seismology:} wave propagation through layers of different density and rigidity
    \item \textbf{Gravity field measurements:} mass distribution from spacecraft tracking
    \item \textbf{Moment of inertia:} degree of central mass concentration
  \end{enumerate}
  \vskip8pt
  For Earth: all three are available at high precision.\\
  For other planets: gravity and $C/MR^2$ are often all we have.
\end{frame}

% ── Seismic waves ────────────────────────────────────────────
\begin{frame}{Seismic waves: P-waves and S-waves}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{seismic_waves}
    \source{Wikimedia Commons, public domain}

    \column{\recaptextcol}
    \textbf{P-waves} (primary, compressional):
    \begin{itemize}
      \item Travel through solids \textit{and} liquids
      \item $v_P = \sqrt{(K + \tfrac{4}{3}\mu)/\rho}$
    \end{itemize}
    \vskip6pt
    \textbf{S-waves} (secondary, shear):
    \begin{itemize}
      \item Travel through solids \textit{only}
      \item $v_S = \sqrt{\mu/\rho}$
      \item $\mu = 0$ in a liquid $\Rightarrow$ S-waves \textbf{cannot propagate}
    \end{itemize}
  \end{columns}
\end{frame}

% ── Shadow zone ──────────────────────────────────────────────
\begin{frame}{The shadow zone: proof of a liquid core}
  \begin{itemize}
    \item P-waves are refracted by the liquid outer core, creating a \textbf{P-wave shadow zone} (a band with no direct arrivals) between $\sim$104\textdegree{} and $\sim$140\textdegree{} from the epicentre
    \item S-waves are \textbf{entirely absorbed} by the liquid core: shadow beyond 104\textdegree{}
    \item Discovered by Oldham (1906) and Gutenberg (1914)
  \end{itemize}
  \vskip8pt
  \keyresult{The absence of S-waves through the outer core is the definitive evidence that it is liquid.}
\end{frame}

% ── Shadow zones walkthrough ─────────────────────────────────
\begin{frame}{Where the waves go silent}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{shadow_zones_usgs}
    \source{U.S. Geological Survey, public domain}

    \column{\recaptextcolwide}
    \small
    Ray paths through Earth reveal the core:
    \begin{itemize}
      \item P-wave shadow zone at 104\textdegree{}--140\textdegree{}: refraction at the CMB, where $v_P$ drops from $\sim$13.7 to $\sim$8.1~km\,s$^{-1}$
      \item S-wave shadow beyond $\sim$104\textdegree{}: shear waves do not cross the liquid outer core at all
      \item Both zones were mapped from global earthquake records long before any drill or sample
    \end{itemize}
    \vskip6pt
    \keyresult{Two shadow zones, one conclusion: the outer core is liquid.}
  \end{columns}
\end{frame}

% ── Gravity field ────────────────────────────────────────────
\begin{frame}{Gravity field measurements}
  \begin{itemize}
    \item A perfectly spherical, homogeneous body produces a pure $1/r^2$ field
    \item Real planets are oblate due to rotation $\Rightarrow$ \textbf{gravitational quadrupole} $J_2$ (deviation from a pure $1/r^2$ field)
    \item $J_2$ depends on mass distribution between equator and poles
    \item Higher harmonics reveal lateral density variations
    \vskip6pt
    \item Measured via spacecraft radio tracking (Doppler shifts)
    \item Examples: GRAIL (Moon), Juno (Jupiter), MRO (Mars)
  \end{itemize}
  \vskip6pt
  \textbf{Precision:} Juno measures Jupiter's gravity harmonics to parts in $10^9$.
\end{frame}

% ── Moment of inertia ───────────────────────────────────────
\begin{frame}{The moment of inertia factor: $C/MR^2$}
  The \textbf{most diagnostic single number} for internal mass distribution, i.e.\ how strongly a body has separated into a dense core and light mantle:
  \vskip8pt
  \centering
  \small
  \begin{tabular}{l c l}
    \toprule
    \textbf{Body} & $\boldsymbol{C/MR^2}$ & \textbf{Interpretation} \\
    \midrule
    Uniform sphere & 0.400 & Homogeneous, undifferentiated \\
    Moon           & 0.393 & Weakly differentiated, small core \\
    Mars           & 0.364 & Moderately differentiated \\
    Mercury        & 0.346 & Large dense core \\
    Earth          & 0.331 & Strongly differentiated, iron core \\
    \bottomrule
  \end{tabular}
  \vskip6pt
  {\tiny\color{ipsText!60} Source: de Pater \& Lissauer (2010), Table~6.1}
  \vskip10pt
  \raggedright
  Determined from $J_2$ + spin-axis precession rate (or \textbf{libration} amplitude, a small periodic rocking about the mean orientation, for tidally locked bodies), combined via the \textbf{Darwin--Radau relation} (links $C/MR^2$ to $J_2$ and flattening).
\end{frame}

% ── How to measure C/MR2 ────────────────────────────────────
\begin{frame}{Measuring $C/MR^2$ in practice}
  Two observables needed:
  \vskip6pt
  \begin{enumerate}
    \item \textbf{Gravitational oblateness} $J_2$ from spacecraft tracking
    \item \textbf{Precession rate} of the spin axis (or forced libration for synchronous rotators)
  \end{enumerate}
  \vskip10pt
  These combine via the \textbf{Darwin--Radau relation}:
  $$\frac{C}{MR^2} = \frac{2}{3}\left[1 - \frac{2}{5}\sqrt{\frac{4 - k_f}{1 + k_f}}\right]$$
  where $k_f$ is the fluid Love number (a measure of the body's response to rotation).
  \vskip6pt
  \textbf{Result:} a single number that immediately distinguishes a rubble pile (0.4) from a differentiated body like Earth (0.33).
\end{frame}

% ── MoI factor walkthrough ───────────────────────────────────
\begin{frame}{One number per body}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{moi_factor}
    \source{Course original; lunar value from Williams et al.\ (2014)}

    \column{\recaptextcol}
    \footnotesize
    Measured $C/MR^2$ across the solar system:
    \begin{itemize}
      \item Every body falls below the uniform-sphere line at $2/5 = 0.400$: all are centrally condensed
      \item The Moon (0.393) is only weakly differentiated; Earth (0.331) and Ganymede (0.311) strongly so
      \item Values follow from $J_2$ plus precession or libration via the Darwin--Radau relation
    \end{itemize}
    \vskip6pt
    \keyresult{$C/MR^2$ is often the first interior constraint available for a newly characterised body.}
  \end{columns}
\end{frame}

\section{Blackboard derivation: moment of inertia factor}
% ════════════════════════════════════════════════════════════

% ── Setup ────────────────────────────────────────────────────
\begin{frame}{Deriving $C/MR^2$}
  \textbf{Goal:} Derive the moment of inertia factor for (a) a uniform sphere and (b) a two-layer differentiated body. Show how the measured value constrains core size.
  \vskip10pt
  \textbf{Starting point:} For a spherically symmetric body, the moment of inertia about the rotation axis is:
  $$C = \frac{8\pi}{3}\int_0^R \rho(r)\,r^4\,\dd r$$
  This comes from summing $\frac{2}{3}r^2\,\dd m$ over thin spherical shells.
\end{frame}

% ── Uniform sphere ───────────────────────────────────────────
\begin{frame}{Step 1: uniform sphere}
  Set $\rho = \text{const}$:
  $$C = \frac{8\pi}{3}\rho\int_0^R r^4\,\dd r = \frac{8\pi}{3}\rho\frac{R^5}{5} = \frac{8\pi}{15}\rho R^5$$
  \vskip6pt
  Total mass: $M = \frac{4}{3}\pi R^3\rho$
  \vskip6pt
  \keyresult{$$\frac{C}{MR^2} = \frac{8\pi\rho R^5/15}{(4\pi\rho R^3/3)\,R^2} = \frac{2}{5} = 0.400$$}
  \vskip6pt
  This is the \textbf{maximum} for any body with $\rho(r)$ that does not increase outward.
\end{frame}

% ── Two-layer body ───────────────────────────────────────────
\begin{frame}{Step 2: two-layer body}
  Dense core ($\rho_c$, radius $R_c$) + lighter mantle ($\rho_m$, radius $R$):
  \vskip6pt
  $$C = \frac{8\pi}{15}\left[(\rho_c - \rho_m)\,R_c^5 + \rho_m\,R^5\right]$$
  $$M = \frac{4}{3}\pi\left[(\rho_c - \rho_m)\,R_c^3 + \rho_m\,R^3\right]$$
  \vskip6pt
  Define $x \equiv R_c/R$ and $f \equiv \rho_c/\rho_m$:
  \vskip4pt
  \keyresult{$$\frac{C}{MR^2} = \frac{2}{5}\cdot\frac{(f-1)\,x^5 + 1}{(f-1)\,x^3 + 1}$$}
  \vskip6pt
  Since $x^5 < x^3$ for $0 < x < 1$, the ratio is $<1$ when $f > 1$.\\
  \textbf{More mass in the centre $\Rightarrow$ lower $C/MR^2$.}
\end{frame}

% ── Worked example: Earth ────────────────────────────────────
\begin{frame}{Worked example: Earth}
  \centering
  \begin{tabular}{l c r}
    \toprule
    \textbf{Parameter} & \textbf{Symbol} & \textbf{Value} \\
    \midrule
    Core radius fraction & $x = R_c/R$ & $3480/6371 = 0.546$ \\
    Density ratio & $f = \rho_c/\rho_m$ & $11{,}000/4{,}400 \approx 2.5$ \\
    \bottomrule
  \end{tabular}
  \vskip10pt
  $$\frac{C}{MR^2} = \frac{2}{5}\times\frac{1.5\times 0.049 + 1}{1.5\times 0.163 + 1} = 0.4\times\frac{1.073}{1.244} \approx 0.345$$
  \vskip8pt
  Measured value: $C/MR^2 = 0.3307$
  \vskip4pt
  Remaining difference: continuous density increase within each layer, especially the $\sim$30\% jump from outer to inner core.
\end{frame}

\section{Equations of state}
% ════════════════════════════════════════════════════════════

% ── Hydrostatic equilibrium ──────────────────────────────────
\begin{frame}{Hydrostatic equilibrium in spherical symmetry}
  Pressure at each depth supports the weight of overlying material:
  \vskip8pt
  \keyresult{$$\dv{P}{r} = -\rho(r)\,g(r) = -\frac{G\,m(r)\,\rho(r)}{r^2}$$}
  \vskip6pt
  where $m(r) = 4\pi\int_0^r \rho(r')\,r'^2\,\dd r'$ is the mass enclosed within radius $r$.
  \vskip10pt
  Combined with an equation of state $P = P(\rho, T)$, this can be integrated inward from the surface to determine the full pressure--density profile.
\end{frame}

% ── Central pressure estimate ────────────────────────────────
\begin{frame}{Central pressure estimate}
  For uniform density $\bar{\rho} = 3M/(4\pi R^3)$:
  \vskip6pt
  $$P_c \approx \frac{3\,G\,M^2}{8\pi\,R^4}$$
  \vskip8pt
  \centering
  \begin{tabular}{l r r}
    \toprule
    \textbf{Body} & \textbf{Estimated $P_c$} & \textbf{Actual $P_c$} \\
    \midrule
    Earth   & $\sim$170~GPa & $\sim$360~GPa \\
    Mars    & $\sim$25~GPa  & $\sim$40~GPa \\
    Jupiter & $\sim$700~GPa & $\sim$4000~GPa \\
    \bottomrule
  \end{tabular}
  \vskip8pt
  \raggedright
  The uniform-density estimate is always \textbf{too low} because real bodies have higher density at the centre.
\end{frame}

% ── Birch's law ──────────────────────────────────────────────
\begin{frame}{Birch's law}
  \textbf{Birch (1952):} empirical linear relationship between compressional wave velocity $v_P$ and density $\rho$ for silicates and oxides:
  $$v_P = a(\bar{M}) + b\,\rho$$
  where $\bar{M}$ is the mean atomic weight and $a$, $b$ are empirical constants.
  \vskip8pt
  \begin{itemize}
    \item Connects seismology directly to density
    \item Holds remarkably well for mantle minerals at high pressure
    \item Breaks down at phase transitions and compositional boundaries
  \end{itemize}
  \vskip6pt
  \keyresult{Birch's law is a cornerstone of mineral physics, linking lab measurements, seismic data, and interior models.}
\end{frame}

% ── Birch's law walkthrough ──────────────────────────────────
\begin{frame}{Birch's law in the laboratory}
  \begin{columns}[c]
    \column{\recapfigcol}
    \ipsrecapfig{birchs_law}
    \source{Course original; trend after Birch (1952)}

    \column{\recaptextcol}
    \footnotesize
    $v_P$ against density for mantle-forming minerals:
    \begin{itemize}
      \item The trend is approximately linear at fixed mean atomic mass
      \item It maps seismic velocities measured at depth to densities, provided the composition is known
      \item Fe-rich outliers such as fayalite sit below the trend: higher mean atomic mass lowers $v_P$ at a given density
    \end{itemize}
    \vskip6pt
    \keyresult{Birch's law turns velocity profiles into density profiles: the bridge from seismology to composition.}
  \end{columns}
\end{frame}

% ── Adams-Williamson equation ────────────────────────────────
\begin{frame}{Adams--Williamson equation}
  The \textbf{seismic parameter}: $\phi \equiv K_S/\rho = v_P^2 - \frac{4}{3}v_S^2$ ($K_S$: bulk modulus, resistance to compression)
  \vskip6pt
  Combining with hydrostatic equilibrium:
  \vskip4pt
  \keyresult{$$\dv{\rho}{r} = -\frac{\rho(r)\,g(r)}{\phi(r)}$$}
  \vskip8pt
  \begin{itemize}
    \item Determines density profile from measured seismic wave speeds
    \item Valid where composition is approximately uniform
    \item Breaks down at phase transitions and compositional boundaries
    \item Foundation of the PREM density model
  \end{itemize}
\end{frame}

% ── Adams-Williamson walkthrough ─────────────────────────────
\begin{frame}{Testing Adams--Williamson against PREM}
  \begin{columns}[c]
    \column{\recapfigcolnarrow}
    \ipsrecapfig{adams_williamson}
    \source{Course original, from the PREM tabulation}

    \column{\recaptextcolwide}
    \footnotesize
    The equation against the real Earth:
    \begin{itemize}
      \item In the lower mantle (771--2741~km) the integrated profile reproduces PREM: the density rise is self-compression alone
      \item At the CMB the observed density jumps from $\sim$5570 to $\sim$9900~kg\,m$^{-3}$; the extrapolation predicts a smooth, far smaller increase
      \item The deviation exposes the compositional change from silicate mantle to iron-alloy core
    \end{itemize}
    \vskip6pt
    \keyresult{Where Adams--Williamson breaks down is as informative as where it works: discontinuities mark composition changes.}
  \end{columns}
\end{frame}

\section{Mantle rheology \& convection}
% ════════════════════════════════════════════════════════════

% ── Solid-state creep ────────────────────────────────────────
\begin{frame}{Solid-state creep}
  The mantle is \textbf{solid} (transmits S-waves), yet it flows over geological timescales because its \textbf{viscosity} (resistance to flow) is finite.
  \vskip4pt
  \begin{itemize}
    \item At $T > 0.5\,T_{\mathrm{melt}}$: atoms migrate along grain boundaries and through the lattice
    \item \textbf{Diffusion creep:} viscosity depends on grain size, dominant in lower mantle
    \item \textbf{Dislocation creep:} viscosity depends on stress, dominant in upper mantle
  \end{itemize}
  \vskip4pt
  \centering
  \small
  \begin{tabular}{l r}
    \toprule
    \textbf{Material} & \textbf{Viscosity (Pa\,s)} \\
    \midrule
    Water          & $10^{-3}$ \\
    Glacial ice    & $10^{12}$ \\
    Upper mantle   & $10^{20}$--$10^{21}$ \\
    Lower mantle   & $10^{22}$--$10^{23}$ \\
    \bottomrule
  \end{tabular}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Karato (2008), \textit{Deformation of Earth Materials}}
\end{frame}

% ── Rayleigh number ──────────────────────────────────────────
\begin{frame}{The Rayleigh number}
  Governed by the \textbf{Rayleigh number} (buoyancy vs.\ viscosity):
  \vskip4pt
  $$\mathrm{Ra} = \frac{\alpha\,\rho\,g\,\Delta T\,d^3}{\kappa\,\eta}$$
  \vskip4pt
  \centering
  \small
  \begin{tabular}{l c r}
    \toprule
    \textbf{Parameter} & \textbf{Symbol} & \textbf{Earth value} \\
    \midrule
    Thermal expansivity & $\alpha$ & $3\times 10^{-5}$~K$^{-1}$ \\
    Density & $\rho$ & 4000~kg\,m$^{-3}$ \\
    Gravity & $g$ & 10~m\,s$^{-2}$ \\
    Temperature contrast & $\Delta T$ & 2500~K \\
    Mantle thickness & $d$ & $2.9\times 10^6$~m \\
    Thermal diffusivity & $\kappa$ & $10^{-6}$~m$^2$\,s$^{-1}$ \\
    Viscosity & $\eta$ & $10^{21}$~Pa\,s \\
    \bottomrule
  \end{tabular}
  \vskip8pt
  \raggedright
  $\mathrm{Ra} \sim 10^7$--$10^8 \gg \mathrm{Ra}_c \approx 10^3$ $\Rightarrow$ \textbf{vigorous convection}.
\end{frame}

% ── Convection patterns ──────────────────────────────────────
\begin{frame}{Convection patterns: an open debate}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Whole-mantle convection:}
    \begin{itemize}
      \item Material circulates from surface to CMB
      \item Slabs penetrate through 660~km
      \item \textbf{Plumes} (hot upwellings) rise from CMB to surface
      \item Supported by seismic tomography (3D imaging) of some subducting slabs
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Layered convection:}
    \begin{itemize}
      \item 660~km discontinuity impedes flow
      \item Separate upper and lower cells
      \item Some slabs stagnate at 660~km
      \item Different chemical reservoirs persist
    \end{itemize}
  \end{columns}
  \vskip10pt
  \keyresult{Reality is likely a \textbf{mixed regime}: some slabs penetrate, others stagnate. The 660~km boundary is a partial barrier, not an absolute one.}
\end{frame}

\begin{frame}{Whole-mantle or layered?}

    \begin{itemize}
      \item (a) Whole-mantle: a single circulation carries material from the CMB to the surface; the 660~km boundary only weakly impedes flow
      \item (b) Layered: separate cells above and below 660~km, decoupled by the negative Clapeyron slope

% ── 660 km discontinuity ─────────────────────────────────────
\begin{frame}{The 660~km discontinuity}
  The most important internal boundary within the mantle:
  \vskip8pt
  \begin{itemize}
    \item Ringwoodite breaks down into bridgmanite + ferropericlase
    \item Largest density jump ($\sim$8\%) within the mantle
    \item \textbf{Negative Clapeyron slope:} boundary pressure \textit{decreases} with increasing temperature
    \item A descending hot slab shifts the boundary \textit{deeper}, creating buoyancy resistance
    \item Partially impedes convective flow: key to the layered vs.\ whole-mantle debate
  \end{itemize}
  \vskip6pt
  \keyresult{The sign of the Clapeyron slope determines whether a phase transition promotes or resists convective penetration.}
\end{frame}

% ── Post-perovskite ──────────────────────────────────────────
\begin{frame}{Post-perovskite and the D" layer}
  \begin{itemize}
    \item Discovered in lab experiments in 2004 (Murakami et al.)
    \item Bridgmanite $\rightarrow$ post-perovskite (a denser crystal structure) at $\sim$125~GPa, $\sim$2500~K
    \item Occurs $\sim$200~km above the CMB (the \textbf{D" layer})
    \item May explain:
    \begin{itemize}
      \item Complex seismic velocity anomalies in D"
      \item Lateral variations in the lowermost mantle
      \item Anisotropy observed near the CMB
    \end{itemize}
  \end{itemize}
  \vskip8pt
  Not unique to Earth: any sufficiently large rocky body will experience these transitions at appropriate pressures.\\
  Mars likely has an olivine--wadsleyite transition; the Moon is too small.
\end{frame}

% ── Mass-radius relation ───────────────────────────────
\begin{frame}{The mass--radius diagram}
  A single diagnostic plot for solar system planets and exoplanets.
  \vskip6pt
  \begin{itemize}
    \item Rocky planets: $R \propto M^{1/3.7}$ (bulk composition roughly similar)
    \item Ice/water worlds: shifted to larger radius at given mass
    \item Gas-envelope worlds: even larger $R$; Neptunes and sub-Neptunes
    \item Gas giants: maximum $R$ near 1 Jupiter radius, nearly independent of mass
    \item Exoplanet surveys: thousands of measurements populate the diagram
    \item \textbf{Radius valley} at 1.5--2~$\Rearth$: gap between super-Earths and mini-Neptunes
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Zeng et al.\ (2019); Fulton et al.\ (2017)}
\end{frame}

% ── Composition inference ───────────────────────────────
\begin{frame}{Inferring composition from mass and radius}
  Density alone is highly degenerate; mass and radius together constrain composition.
  \vskip6pt
  \begin{itemize}
    \item Earth (1~$\Mearth$, 1~$\Rearth$): average $\rho = 5515$~kg/m$^3$, rock + Fe core
    \item A pure ``iron planet'' at 1~$\Mearth$ would have $R \approx 0.75\,\Rearth$
    \item A pure ``water world'' at 1~$\Mearth$ would have $R \approx 1.3\,\Rearth$
    \item Intermediate compositions trace curves between these extremes
    \item With measurement errors, compositions are often ambiguous (rock vs.\ mixed)
    \item Tidal Love number $k_2$ can break degeneracies: probes internal rigidity
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Valencia et al.\ (2007); Unterborn \& Panero (2019)}
\end{frame}

% ── InSight results ──────────────────────────────────────────
\begin{frame}{InSight: first seismology on Mars since Apollo}
  NASA InSight (2018--2022) delivered landmark results:
  \vskip6pt
  \begin{itemize}
    \item Detected hundreds of marsquakes with the SEIS instrument
    \item \textbf{Liquid iron-alloy core:} $R_c \sim 1830$~km (Stähler et al.\ 2021), revised to $\sim$1650--1675~km beneath a molten silicate layer (Khan et al.\ 2023; Samuel et al.\ 2023)
    \item \textbf{Thick lithosphere} (rigid outer shell): $\sim$400--600~km (Khan et al.\ 2021)
    \item \textbf{Crust:} 24--72~km thick (thin in north, thick in south)
    \item Confirmed that Mars is differentiated but has a proportionally larger, less dense core than expected from simple two-layer models
  \end{itemize}
  \vskip4pt
  First seismological constraints on another planet's interior since Apollo (1969--1977).
\end{frame}

% ── InSight waveform interpretation ────────────────────
\begin{frame}{Reading a marsquake waveform}
  InSight's SEIS instrument revealed Mars's interior through wave-by-wave analysis.
  \vskip2pt
  \begin{itemize}
    \item \textbf{P-wave arrival:} first; travels $\sim$8~km/s through the mantle
    \item \textbf{S-wave arrival:} second, $\sim$4~km/s; absent below the CMB (liquid core)
    \item \textbf{S-wave to P-wave delay:} measures source distance ($\sim$8~s $\rightarrow$ $\sim$1000~km epicentre)
    \item Core reflections (ScS): bounce off the CMB; constrain core radius
    \item Mantle surface waves: constrain crustal thickness and attenuation
    \item Largest marsquake: \textbf{S1222a, May 2022, M 4.7}; clearly identified core phases
  \end{itemize}
  \vskip2pt
  \keyresult{Every quake is a probe. InSight's $\sim$1300 events together built a radial model of Mars's interior.}
  \vskip2pt
  {\tiny\color{ipsText!60} Source: Giardini et al.\ (2020); Duran et al.\ (2022) on S1222a}
\end{frame}

% ── Ganymede dedicated ───────────────────────────────────
\begin{frame}{Ganymede: the largest moon}
  Ganymede ($R = 2634$~km, larger than Mercury) has a complex differentiated interior.
  \vskip6pt
  \begin{itemize}
    \item Four layers: solid iron inner core, liquid iron outer core, silicate mantle, ice shell/ocean
    \item Subsurface ocean: $\sim$100~km deep, beneath an ice shell $\sim$150~km thick
    \item Ocean sandwiched between ice layers of different phases (ice I above, ice III/V/VI below)
    \item Confirmed by Hubble UV observations of aurora (Saur et al.\ 2015)
    \item Has its own intrinsic magnetic field (from liquid iron core dynamo)
    \item JUICE will dedicate most of its mission to mapping Ganymede
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Anderson et al.\ (1996); Saur et al.\ (2015)}
\end{frame}

% ── Superionic ice ──────────────────────────────────────
\begin{frame}{Superionic ice: the ice giants' exotic interior}
  At the temperatures and pressures inside Uranus and Neptune, water ice adopts an extraordinary phase.
  \vskip6pt
  \begin{itemize}
    \item \textbf{Superionic ice:} oxygen atoms form a rigid crystalline lattice; hydrogen ions diffuse like a liquid
    \item Occurs at $P > 100$~GPa, $T > 2000$~K
    \item First confirmed experimentally by Millot et al.\ (2019) using laser-driven shocks
    \item Predicted to constitute most of the ``ice'' layer in Uranus and Neptune
    \item Electrical conductivity of superionic ice is high $\Rightarrow$ supports dynamo generation
    \item Explains the tilted, multipolar magnetic fields of the ice giants
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Millot et al.\ (2019), \textit{Nature}; Ravasio et al.\ (2021)}
\end{frame}

% ── Seismic velocity profile ───────────────────────────
\begin{frame}{PREM: seismic velocities}
  The Preliminary Reference Earth Model gives $V_p$ and $V_s$ vs.\ depth.
  \vskip6pt
  \begin{itemize}
    \item $V_p$ (P-wave): 6--14~km/s from crust to inner core
    \item $V_s$ (S-wave): 3--7~km/s in solid layers; \textbf{zero} in the outer core
    \item Discontinuities at the Moho (crust--mantle), 410~km, 660~km, and CMB
    \item Outer core S-wave gap is the key evidence for a liquid outer core
    \item Inner core S-wave velocity is low ($\sim$3.5~km/s) but nonzero $\Rightarrow$ solid
    \item Combined with density, constrains composition and phase via Birch's law
  \end{itemize}
  \vskip4pt
  {\tiny\color{ipsText!60} Source: Dziewonski \& Anderson (1981), \textit{Phys.\ Earth Planet.\ Inter.}}
\end{frame}

% ── Europa Clipper ───────────────────────────────────────────
\begin{frame}{Europa Clipper: characterising an ocean world}
  \begin{itemize}
    \item Launched October 2024, arrival at Jupiter system 2030
    \item Dozens of close Europa flybys (not an orbiter)
    \item Key instruments:
    \begin{itemize}
      \item Ice-penetrating radar (REASON): ice shell thickness
      \item Magnetometer: ocean salinity and depth
      \item Gravity science: internal mass distribution
      \item Mass spectrometer: plume/surface composition

  \keyresult{Europa Clipper will provide the most detailed constraints yet on the internal structure and habitability of an ocean world.}
\end{frame}


% ── Summary ────────────────────────────────────────────────
\begin{frame}{Summary}
  \begin{enumerate}
    \item Seismology, gravity, and $C/MR^2$ are the primary tools for probing interiors
    \item $C/MR^2 = 2/5$ for uniform sphere; lower values indicate differentiation
    \item Hydrostatic equilibrium + equations of state determine interior profiles
    \item Earth: crust, mantle (with phase transitions at 410, 660~km), liquid outer core, solid inner core
    \item Mantle is solid but convects ($\mathrm{Ra} \sim 10^7$--$10^8$), driving plate tectonics
    \item Giant planets: molecular $\rightarrow$ metallic H, dilute cores (Jupiter), helium rain (Saturn)
    \item Ice giants: superionic water, tilted non-dipolar magnetic fields
    \item Ocean worlds (Europa, Enceladus, Titan) expand the concept of habitability
  \end{enumerate}
\end{frame}
